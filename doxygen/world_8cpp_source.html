42Rgb maskColor(char32_t m, const Entity& e) {
43 if (m >= U'1' && m <= U'9') return e.pal[m - U'1'];
44 Rgb base;
45 switch (m | 0x20) { // fold to lower case
46 case U'w': base = {240, 240, 240}; break;
47 case U'k': base = {70, 70, 85}; break;
48 case U'r': base = {225, 70, 70}; break;
49 case U'g': base = {80, 195, 100}; break;
50 case U'b': base = {80, 120, 235}; break;
51 case U'c': base = {90, 215, 225}; break;
52 case U'y': base = {240, 215, 80}; break;
53 case U'm': base = {205, 100, 205}; break;
54 case U'o': base = {245, 155, 60}; break;
55 case U't': base = {240, 205, 150}; break;
56 case U'n': base = {140, 92, 52}; break;
57 case U'p': base = {245, 150, 175}; break;

76void blit(Frame& out, const Entity& e, Style style, const Light& l, int maxRow, const std::vector<int>* hideBelow = nullptr) {
77 const SpriteFrame& f = e.sprite->frame(e.mirrored, e.frame);
78 const int ox = static_cast<int>(std::floor(e.x));
79 const int oy = static_cast<int>(std::floor(e.y));
80 auto colorOf = [&](char32_t m) {
81 const Rgb c = maskColor(m, e);
82 return e.glow ? c : shade(c, l);
83 };
84 for (int r = 0; r < e.height(); ++r) {
85 for (int c = 0; c < e.width(); ++c) {
86 const int px = ox + c, py = oy + r;
87 if (!out.inBounds(px, py) || py >= maxRow) continue;
88 if (hideBelow && py >= (*hideBelow)[static_cast<size_t>(px)]) continue; // behind the mountain
89 const char32_t ch = f.art[r][c], m = f.mask[r][c];
90 Cell& cell = out.at(px, py);
91 if (ch == U' ') {
92 if (m == U' ') continue;
93 cell.ch = U' ';
94 if (m != U'_') { // solid block: keeps its color unless the style is monochrome
95 cell.bg = colorOf(m);
96 cell.defaultBg = style == Style::Mono;
97 }
98 } else {
99 cell.ch = ch;
100 cell.fg = colorOf(m);
101 cell.defaultFg = style == Style::Mono;
102 if (e.backing > 0 && style == Style::Full) cell.bg = scale(cell.bg, 1 - e.backing);
103 if (e.contrast && style == Style::Full) { // dark strokes on light backgrounds
104 const double lum = (0.2126 * cell.bg.r + 0.7152 * cell.bg.g + 0.0722 * cell.bg.b) / 255.0;
105 if (lum > 0.5) cell.fg = scale(cell.fg, 0.32);
106 }
107 }
108 }
109 }

135Cell riverCell(int x, int y, double t, double time, double flow, Rgb top, Rgb deep) {
136 Cell c;
137 c.bg = lerp(top, deep, t);
138 const double f = time * flow;
139 const double v = std::sin(x * 0.6 - f + y * 1.7) + 0.5 * std::sin(x * 0.23 - f * 0.6);
140 c.ch = v > 0.8 ? U'~' : v > 0.2 ? U'-' : (hash2(x, y) % 9 == 0 ? U'.' : U' ');
141 c.fg = scale(c.bg, 1.9);
142 return c;
143}

217void hop(Entity& e, World& w, double dt) {
218 constexpr double kAirborne = 0.4; // share of the cycle spent in the air
219 const double cycle = std::fmod(e.age + e.phase, e.hopPeriod) / e.hopPeriod;
220 double lift = 0;
221 if (cycle < kAirborne) {
222 e.x += e.vx * dt / kAirborne; // all the travel happens in the air, so the average speed is vx
223 lift = e.hopHeight * std::sin(3.14159265358979 * cycle / kAirborne);
224 }
225 if ((e.vx > 0 && e.x > w.width()) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
226 e.y = e.anchorY + w.craterRows(static_cast<int>(std::floor(e.x + e.width() / 2.0))) - lift;

295void catastrophe(Entity& e, World& w, double dt) {
296 // Where it strikes: the middle (sweep), or the front, the side it moves towards (devour).
297 auto point = [&] { return e.strikeMode == Entity::StrikeMode::Devour ? (e.vx > 0 ? e.x + e.width() : e.x) : e.x + e.width() / 2.0; };
298 if (e.strikeMode != Entity::StrikeMode::Centre && !e.hasLastPoint) {
299 e.lastPoint = point();
300 e.hasLastPoint = true;
301 }
302 e.x += e.vx * dt;
303 if ((e.vx > 0 && e.x > w.width()) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
304 if (e.strikeMode != Entity::StrikeMode::Centre) { // takes each victim as it passes it
305 const double now = point();
306 const bool devour = e.strikeMode == Entity::StrikeMode::Devour; // it only takes what is in its way
307 if (devour && e.strikeReach > 0) w.lash(e);
308 if (e.strikeAround) return; // it takes only what its tentacles reach
309 // A cell of leeway each way, so that a victim walking towards it is not stepped over.
310 const size_t taken = w.wipeOut(e.wipes, e.strikeEffect, &e, std::min(e.lastPoint, now) - 1, std::max(e.lastPoint, now) + 1,
311 devour);
312 if (devour) w.noteCrushed(taken);
313 e.lastPoint = now;
314 if (taken > 0) {
315 if (e.flashTime > 0) w.flash(e.flashTime);
316 e.struck = true;
317 }
318 return;
319 }
320 if (e.struck) return;
321 const double middle = e.x + e.width() / 2.0;
322 const double progress = e.vx > 0 ? middle / w.width() : 1.0 - middle / w.width();
323 if (progress < e.strikeAt) return;
324 e.struck = true;
325 if (e.stormTime > 0) w.startRain(e.stormTime);
326 if (!e.strikeSprite.empty()) { // a bolt from the sky to the ground right below it
327 if (Entity* bolt = w.spawnEffect(e.strikeSprite, middle, 0, e.z - 1, 0, 0)) {
328 bolt->x = middle - bolt->width() / 2.0;
329 bolt->y = w.seabedY() - bolt->height();
330 }
331 }
332 w.wipeOut(e.wipes, e.strikeEffect, &e);
333 if (e.flashTime > 0) w.flash(e.flashTime);
334}

354 e.x += (e.vx + 1.8 * std::sin(e.age * 1.3 + e.phase)) * dt;
355 if ((e.vx > 0 && e.x > w.width()) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
356 for (auto& other : w.entities()) {

358 if (&o == &e || o.dead || o.flung || !o.grounded || o.vx == 0 || o.tag == "tornado") continue;
359 if (w.isAsleep(o)) continue;

361 if (overlap <= 0 || o.y + o.height() <= e.y) continue; // not in the funnel's path
362 // Up and away: the rise is a random height up to most of the sky, the sideways kick spins it out.
363 const double rise = w.rng().range(5.0, std::max(6.0, std::min(16.0, w.height() * 0.6)));

383void ledge(Entity& e, World& w, double dt) {
384 constexpr double kAirborne = 0.4; // share of a hop cycle spent in the air
385 const double before = e.x;
386 double lift = 0;
387 if (e.hops) {
388 const double cycle = std::fmod(e.age + e.phase, e.hopPeriod) / e.hopPeriod;
389 if (cycle < kAirborne) {
390 e.x += e.vx * dt / kAirborne; // all the travel happens in the air, so the average speed is vx
391 lift = e.hopHeight * std::sin(3.14159265358979 * cycle / kAirborne);
392 }
393 } else {
394 e.x += e.vx * dt;
395 }
396 if (e.laneTarget >= 0) {
397 const double d = e.laneTarget - e.anchorY, step = 1.6 * dt;
398 if (std::fabs(d) <= step) { e.anchorY = e.laneTarget; e.laneTarget = -1; }
399 else e.anchorY += d > 0 ? step : -step;
400 } else if (w.rng().chance(0.06 * dt)) {
401 w.pickLane(e);
402 }
403 if (!w.fits(e, e.x, e.anchorY)) { // the end of its ground: turn round and look for another ledge
404 e.x = before;
405 e.vx = -e.vx;
406 e.mirrored = !e.mirrored;
407 e.laneTarget = -1;
408 if (!w.fits(e, e.x, e.anchorY)) w.placeInZone(e, true, false, e.movingRight());
409 }
410 if ((e.vx > 0 && e.x > w.width()) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
411 e.y = e.anchorY - lift;
412 e.z = w.depthFor(e.anchorY + e.height());
413}

473void animate(Entity& e, double dt) {
474 const size_t n = e.sprite->frameCount();
475 if (e.fps <= 0 || n < 2) return;
476 const double period = 1.0 / e.fps;
477 e.clock += dt;
478 while (e.clock >= period) {
479 e.clock -= period;
480 if (!e.loop && e.frame + 1 >= n) {
481 if (e.dieAfterAnim) e.dead = true;
482 continue;
483 }
484 ++e.frame;
485 }
486 e.frame = e.loop ? e.frame % n : std::min(e.frame, n - 1);
487}

592void World::populate() {
593 for (const auto& pop : scene_->population) {
594 const bool perCells = pop.kind == SceneDef::Population::Kind::PerCells;
595 const bool perCols = pop.kind == SceneDef::Population::Kind::PerCols;
596 double want;
597 if (pop.sprite == "@seaweed") {
598 if (!hasMedium_) continue;
599 want = perCols ? w_ / pop.n : pop.n;
600 int n = static_cast<int>(want);
601 if (perCols && rng_.chance(want - n)) ++n;
602 for (int i = 0; i < n; ++i) spawnSeaweed();
603 continue;
604 }
605 auto sprite = pack_->find(pop.sprite);
606 if (!sprite) continue;
607 const std::string habitat = sprite->props().str("habitat", "medium");
608 if ((habitat == "medium" && !hasMedium_) || (habitat == "river" && riverRows_ == 0) ||
609 (habitat == "rail" && railY_ < 0)) {
610 continue;
611 }
612 want = perCells ? areaFor(habitat) / pop.n : perCols ? w_ / pop.n : pop.n;
613 int n = static_cast<int>(want);
614 if (pop.kind != SceneDef::Population::Kind::Count && rng_.chance(want - n)) ++n;
615 // A `herd` sprite is spawned in groups (n counts groups, not animals).
616 const bool herded = sprite->props().has("herd");
617 for (int i = 0; i < n; ++i) {
618 if (herded) spawnHerd(sprite);
619 else spawnCreature(sprite, true);
620 }
621 }
622 for (const auto& d : scene_->decor) spawnDecor(d);
623 flushIncoming();
624}

626double World::habitatY(const Props& p, int spriteHeight) {
627 if (p.has("row")) { // pinned to a row; negative counts from the bottom
628 const double row = p.num("row", 0);
629 return row >= 0 ? row : h_ + row;
630 }
631 const std::string habitat = p.str("habitat", "medium");
632 if (habitat == "ground") return seabedY_ - spriteHeight;
633 if (habitat == "rail") return railY_ - spriteHeight; // on the elevated railway
634 if (habitat == "river") return rng_.rangeInt(riverTopY_, std::max(h_ - spriteHeight, riverTopY_));
635 if (habitat == "surface") return surfaceY_ - p.num("anchor-row", spriteHeight - 1);
636 if (habitat == "sky") {
637 if (p.has("altitude")) { // hovering within a few rows above the ground
638 return std::max(surfaceY_ - spriteHeight - rng_.rangeInt(0, static_cast<int>(p.num("altitude", 0))), 0);
639 }
640 return rng_.rangeInt(0, std::max(surfaceY_ - spriteHeight, 0));
641 }
642 const int lo = surfaceY_ + 2;
643 return rng_.rangeInt(lo, std::max(seabedY_ - spriteHeight, lo));

646void World::spawnHerd(const std::shared_ptr<const Sprite>& sprite) {
647 const Props& p = sprite->props();
648 const auto [lo, hi] = p.range("herd", {2, 4});
649 const int members = rng_.rangeInt(static_cast<int>(lo), static_cast<int>(hi));

658 spawnCreature(sprite, true, false, &where);
659 }

662void World::spawnCreature(const std::shared_ptr<const Sprite>& sprite, bool initial, bool event,
663 const Placement* where) {
664 const Props& p = sprite->props();
665 if (p.str("habitat", "medium") == "rail" && railY_ < 0) return; // no railway on this screen
666 // Rush-hour traffic: a vehicle only joins the street with the chance the traffic level gives, and
667 // otherwise tries again a few seconds later, so the street fills and empties with the day.
668 if (!event && p.str("rush", "no") == "yes" && scene_->dayNight && !rng_.chance(trafficLevel())) {
669 restock_.push_back({rng_.range(2.0, 7.0), [sprite](World& w) { w.spawnCreature(sprite, false); }});
670 return;
671 }
672 auto e = std::make_unique<Entity>();
673 e->sprite = sprite;
674 applyProps(*e, p, rng_);
675 e->isEvent = event;
676
677 const auto [vmin, vmax] = p.range("speed", {4, 8});
678 const double speed = rng_.range(vmin, vmax) * (where ? where->speedMul : 1.0);
679 const bool coin = rng_.chance(0.5);
680 const bool right = (where && where->dir != 0) ? where->dir > 0 : coin;
681 e->vx = right ? speed : -speed;
682 e->vy = std::fabs(e->vx) * p.num("slope", 0); // diagonal flyers move downwards
683 e->mirrored = right != e->facesRight;
684
685 const auto [zmin, zmax] = p.range("depth", {5, 15});
686 e->z = rng_.rangeInt(static_cast<int>(zmin), static_cast<int>(zmax));
687 e->y = e->anchorY = habitatY(p, e->height());
688
689 // Ground sprites with `lanes: N` stand N rows deep on the bank; lanes further
690 // forward are drawn in front, so crowds of trees and animals get some depth.
691 const int lanes = static_cast<int>(p.num("lanes", 1));
692 if (lanes > 1 && p.str("habitat", "medium") == "ground") {
693 const int bank = (riverRows_ > 0 ? riverTopY_ : h_) - seabedY_;
694 const int usable = std::min(lanes, std::max(bank, 1));
695 const int lane = rng_.rangeInt(0, usable - 1);
696 e->y += lane;
697 e->anchorY = e->y;
698 if (usable > 1) e->z = static_cast<int>(zmax) - lane * static_cast<int>(zmax - zmin) / (usable - 1);
699 }
700
701 // `feet-rows: a..b` puts a ground sprite's feet on those ground rows (0 = the first; negative counts
702 // from the bottom, -1 is the last), and with `traffic: yes` the vehicles keep to their side of the
703 // road: the lower rows of the range for what moves right, the upper rows for what moves left.
704 if (p.has("feet-rows") && p.str("habitat", "medium") == "ground") {
705 const int rows = std::max(h_ - seabedY_, 1);
706 const auto [flo, fhi] = p.range("feet-rows", {0, 0});
707 auto resolve = [&](double v) { return std::clamp(static_cast<int>(v) < 0 ? rows + static_cast<int>(v) : static_cast<int>(v), 0, rows - 1); };
708 int lo = resolve(flo), hi = resolve(fhi);
709 if (lo > hi) std::swap(lo, hi);
710 if (p.str("traffic", "no") == "yes" && hi > lo) {
711 const int mid = (lo + hi + 1) / 2;
712 if (right) lo = mid;
713 else hi = mid - 1;
714 }
715 const int feet = rng_.rangeInt(lo, hi);
716 e->y = e->anchorY = seabedY_ - e->height() + feet + 1;
717 e->z = static_cast<int>(zmax) - feet * static_cast<int>(zmax - zmin) / std::max(rows - 1, 1);
718 }
719
720 // Members of a herd that left come back from the same side, a little apart, still facing the same way.
721 const double rejoin = (where && !initial) ? rng_.range(0, 12) : 0.0;
722 e->x = (where && where->hasX) ? where->x
723 : initial ? rng_.range(-e->width(), w_)
724 : (right ? -e->width() - rejoin : w_ + rejoin);
725 if (e->vy > 0 && !initial) e->y = e->anchorY = -e->height() + rng_.range(0, h_ / 3.0);
726 if (sprite->frameCount() > 1 && e->fps > 0) {
727 e->frame = rng_.rangeInt(0, static_cast<int>(sprite->frameCount()) - 1);
728 }
729
730 if (e->zone != Zone::None && !ridge_.empty()) placeInZone(*e, where && where->hasX, !initial, right);
731
732 // `sleeps: 0.7` = about 70% of these animals sleep at night, each from its own sun height.
733 const double sleeps = std::clamp(p.num("sleeps", 0.0), 0.0, 1.0);
734 if (sleeps > 0 && rng_.chance(sleeps)) e->sleepBelow = rng_.range(0.05, 0.9);
735
736 if (!event) {
737 const bool herded = p.has("herd");
738 const int dir = right ? 1 : -1;
739 e->onDeath = [sprite, herded, dir](World& w) {
740 Placement again;
741 again.dir = dir;
742 if (herded) w.spawnCreature(sprite, false, false, &again);
743 else w.spawnCreature(sprite, false);
744 };
745 }
746 spawn(std::move(e));
747}

749void World::spawnDecor(const SceneDef::Decor& d) {
750 auto sprite = pack_->find(d.sprite);
751 if (!sprite) return;
752 const Props& p = sprite->props();
753 if (p.str("habitat", "ground") == "rail" && railY_ < 0) return; // no railway on this screen
754 auto e = std::make_unique<Entity>();
755 e->sprite = sprite;
756 applyProps(*e, p, rng_);
757 e->z = static_cast<int>(p.range("depth", {20, 20}).first);
758
759 if (d.align == "left") e->x = d.margin;
760 else if (d.align == "right") e->x = w_ - e->width() - d.margin;
761 else if (d.align == "center") e->x = (w_ - e->width()) / 2 + d.margin;
762 else e->x = rng_.rangeInt(0, std::max(w_ - e->width(), 0));
763 // Things that stand on the sidewalk make room for the side streets: they move aside, or are left out.
764 if (!avenues_.empty() && e->grounded && p.str("habitat", "ground") == "ground") {
765 for (int pass = 0; pass < 4; ++pass) {
766 for (const Avenue& a : avenues_) {
767 const double lo = a.cx - a.halfW - 3, hi = a.cx + a.halfW + 3;
768 if (e->x + e->width() <= lo || e->x >= hi) continue;
769 const double left = lo - e->width(), right = hi;
770 const bool leftOk = left >= 0, rightOk = right + e->width() <= w_;
771 if (!leftOk && !rightOk) return;
772 e->x = (!rightOk || (leftOk && e->x - left < right - e->x)) ? left : right;
773 }
774 }
775 for (const Avenue& a : avenues_) {
776 if (e->x + e->width() > a.cx - a.halfW - 3 && e->x < a.cx + a.halfW + 3) return; // still in the way
777 }
778 }
779
780 if (d.hasY) e->y = d.y >= 0 ? d.y : h_ + d.y;
781 else if (p.str("habitat", "ground") == "sky") e->y = 1;
782 else e->y = habitatY(p, e->height());
783 e->anchorY = e->y;
784 if (e->zone != Zone::None && d.align == "random" && !ridge_.empty()) placeInZone(*e, false, false, true);
785 e->restock = [d](World& w) { w.spawnDecor(d); }; // planted again after a catastrophe
786 spawn(std::move(e));
787}

807void World::spawnSeaweed() {
808 const int maxH = std::max(seabedY_ - surfaceY_ - 2, 2);
809 const int h = std::min(rng_.rangeInt(3, 7), maxH);
810 constexpr int kFrames = 6;
811 constexpr int kWidth = 5;
812 const double phase0 = rng_.range(0, 6.28);
813
814 std::vector<SpriteFrame> frames;
815 for (int k = 0; k < kFrames; ++k) {
816 std::vector<int> col(h);
817 for (int i = 0; i < h; ++i) { // i = 0 is the tip
818 const double amp = 1.6 * (h - 1 - i) / h;
819 col[i] = 2 + static_cast<int>(std::lround(amp * std::sin(phase0 + k * 1.047 + i * 0.8)));
820 }
821 SpriteFrame f;
822 for (int i = 0; i < h; ++i) {
823 std::u32string row(kWidth, U' ');
824 const int below = i + 1 < h ? col[i + 1] : col[i];
825 const int d = col[i] - below;
826 row[col[i]] = d > 0 ? U'/' : d < 0 ? U'\\' : (i % 2 ? U'(' : U')');
827 f.art.push_back(std::move(row));
828 }
829 frames.push_back(std::move(f));
830 }
831
832 auto e = std::make_unique<Entity>();
833 e->sprite = Sprite::fromFrames("seaweed", std::move(frames));
834 e->x = rng_.rangeInt(0, std::max(w_ - kWidth, 0));
835 e->y = seabedY_ - h;
836 e->z = 21;
837 e->fps = rng_.range(2.0, 3.0);
838 e->frame = rng_.rangeInt(0, kFrames - 1);
839 e->fg = lerp({40, 150, 70}, {150, 170, 60}, rng_.uniform());
840 e->ttl = rng_.range(8 * 60, 12 * 60); // seaweed regrows every 8-12 minutes
841 e->onDeath = [](World& w) { w.spawnSeaweed(); };
842 spawn(std::move(e));
843}

860 for (const auto& ev : scene_->rare) total += ev.weight;

910 double to, bool groundedOnly) {
911 const size_t before = killed_;
912 const bool all = wipes == "life";
913 const std::vector<std::string> names = all ? std::vector<std::string>{} : splitWs(wipes);
914 auto take = [&](Entity& o) {
915 if (&o == except || o.isRare) return;

917 if (middle < from || middle > to) return;
918 if (groundedOnly && !o.grounded) return;
919 const bool victim = all ? o.life : std::find(names.begin(), names.end(), o.sprite->name()) != names.end();
920 if (victim) claim(o, effect);
921 };
922 for (auto& other : entities_) take(*other);

924 // (spawnEffect above adds to the same list, and effects are never victims.)
925 for (size_t i = 0; i < incoming_.size(); ++i) take(*incoming_[i]);
926 return killed_ - before;

936bool World::lash(const Entity& a) {
937 if (time_ < nextLash_ || a.strikeReach <= 0) return false;
938 int out = 0;
939 for (const Tongue& t : tongues_) out += t.attacker == a.id ? 1 : 0;
940 if (out >= a.strikeLashes) return false;
941 const bool all = a.wipes == "life";
942 const std::vector<std::string> names = all ? std::vector<std::string>{} : splitWs(a.wipes);
943 const int dir = a.vx > 0 ? 1 : -1;
944 const double front = dir > 0 ? a.x + a.width() : a.x;
945 const double baseY = a.y + a.mouthRow;
946 const Entity* best = nullptr;
947 double bestD = 0, bestX = 0;
948 for (auto& other : entities_) {
949 const Entity& o = *other;
950 if (&o == &a || o.dead || o.isRare || isAsleep(o)) continue;
951 if (!(all ? o.life : std::find(names.begin(), names.end(), o.sprite->name()) != names.end())) continue;
952 bool taken = false; // another tongue is already after it
953 for (const Tongue& t : tongues_) taken |= t.victim == o.id;
954 if (taken) continue;
955 const double cx = o.x + o.width() / 2.0, cy = o.y + o.height() / 2.0;
956 double d, baseX;
957 if (a.strikeAround) { // from the underside of the body, straight above the victim if it can
958 baseX = std::clamp(cx, a.x + 3.0, std::max(a.x + a.width() - 4.0, a.x + 3.0));
959 d = std::sqrt((cx - baseX) * (cx - baseX) + 4.0 * (cy - baseY) * (cy - baseY));
960 } else { // from the mouth, at walking things ahead of it
961 if (!o.grounded) continue;
962 baseX = dir > 0 ? a.x + a.width() - 1.0 : a.x;
963 d = (cx - front) * dir;
964 if (d < 1.5) continue; // what is at its very mouth is simply swallowed
965 }
966 if (d > a.strikeReach) continue;
967 if (!best || d < bestD) { best = &o; bestD = d; bestX = baseX; }
968 }
969 if (!best) return false;
970 Tongue t;
971 t.attacker = a.id;
972 t.victim = best->id;
973 t.effect = a.strikeEffect;
974 t.wipes = a.wipes;
975 t.color = a.lashColor;
976 if (a.lashShot) { // a shot is there and gone
977 t.shot = true;
978 t.reach = 0.1;
979 t.dur = 0.3;
980 ++shots_;
981 }
982 t.ox = bestX - a.x;
983 t.x0 = bestX;
984 t.y0 = baseY;
985 t.x1 = best->x + best->width() / 2.0;
986 t.y1 = best->y + best->height() / 2.0;
987 tongues_.push_back(t);
988 nextLash_ = time_ + (a.lashShot ? 0.45 : a.strikeLashes > 1 ? 0.2 : 0.0); // tentacles lash one after another
989 return true;
990}

1002void World::stepTongues(double dt) {
1003 for (size_t i = tongues_.size(); i-- > 0;) {
1004 Tongue& t = tongues_[i];
1005 Entity* a = findEntity(t.attacker);
1006 if (!a || a->dead) { // the attacker has gone: the tongue goes with it
1007 tongues_.erase(tongues_.begin() + static_cast<std::ptrdiff_t>(i));

1009 }
1010 t.age += dt;
1011 t.x0 = a->x + t.ox;
1012 t.y0 = a->y + a->mouthRow;
1013 Entity* v = findEntity(t.victim);
1014 const bool alive = v && !v->dead;
1015 if (alive && !t.hit) { // the tip follows the victim
1016 t.x1 = v->x + v->width() / 2.0;
1017 t.y1 = v->y + v->height() / 2.0;
1018 }
1019 if (!t.hit && t.age >= t.reach) {
1020 t.hit = true;
1021 if (alive && !t.harmless) {
1022 claim(*v, t.effect);
1023 if (!t.shot) noteCrushed(1);
1024 }
1025 }
1026 if (t.age >= t.dur) {
1027 if (a->strikeLashes == 1) nextLash_ = time_ + 0.25; // a single tongue rests a moment
1028 tongues_.erase(tongues_.begin() + static_cast<std::ptrdiff_t>(i));
1029 }

1033void World::renderTongues(Frame& out, Style style, const Light& light) const {
1034 for (const Tongue& t : tongues_) {
1035 if (t.fire) { // a stream of flames: it shoots out, holds, and dies away
1036 const double reach = std::max(t.reach, 0.01);
1037 const double p = t.age < reach ? t.age / reach : t.age > t.dur - 0.3 ? std::max(0.0, (t.dur - t.age) / 0.3) : 1.0;
1038 const double dx = t.x1 - t.x0, dy = t.y1 - t.y0;
1039 const double len = std::sqrt(dx * dx + dy * dy * 4.0);
1040 const int samples = std::max(3, static_cast<int>(len * 2));
1041 static const char32_t kFlames[] = {U'*', U'#', U'%', U'&', U'@', U'*'};
1042 for (int s = 0; s <= samples; ++s) {
1043 const double u = static_cast<double>(s) / samples;
1044 if (u > p) break;
1045 const double wobble = 0.9 * std::sin(u * len * 0.7 - t.age * 30.0) * u;
1046 const int cx = static_cast<int>(std::floor(t.x0 + dx * u));
1047 const int cy = static_cast<int>(std::floor(t.y0 + dy * u + wobble));
1048 const int spread = 1 + static_cast<int>(u * 2.2); // the stream widens as it goes
1049 for (int off = -spread; off <= spread; ++off) {
1050 const uint32_t h = hash2(cx * 7 + off, static_cast<int>(t.age * 24.0) + s);
1051 if (off != 0 && h % 5 == 0) continue; // ragged edges
1052 if (!out.inBounds(cx, cy + off)) continue;
1053 Cell& cell = out.at(cx, cy + off);
1054 const double edge = static_cast<double>(std::abs(off)) / (spread + 1);
1055 cell.ch = kFlames[h % 6];
1056 cell.fg = shade(lerp(lerp(t.color, Rgb{255, 255, 255}, 0.65), t.color, edge), light);
1057 cell.bg = style == Style::Full ? lerp(cell.bg, scale(t.color, 0.55), 0.6 * (1.0 - edge)) : cell.bg;
1058 cell.bold = true;
1059 cell.defaultFg = style == Style::Mono;
1060 }
1061 }
1062 continue;
1063 }
1064 // Out to the victim in the first part, back in again once it is crushed.
1065 const double p = t.hit ? std::max(0.0, 1.0 - (t.age - t.reach) / (t.dur - t.reach)) : std::min(1.0, t.age / t.reach);
1066 const double dx = t.x1 - t.x0, dy = t.y1 - t.y0;
1067 const double len = std::sqrt(dx * dx + dy * dy * 4.0); // cells are about twice as tall as wide
1068 const int samples = std::max(2, static_cast<int>(len * 2));
1069 // The glyph follows the direction of the tongue.
1070 const double ady = std::fabs(dy) * 2.0, adx = std::fabs(dx);
1071 const char32_t glyph = ady < 0.45 * adx ? U'=' : ady > 2.2 * adx ? U'|' : (dx * dy > 0 ? U'\\' : U'/');
1072 for (int s = 0; s <= samples; ++s) {
1073 const double u = static_cast<double>(s) / samples;
1074 if (u > p) break;
1075 const double wobble = t.shot ? 0.0 : 0.8 * std::sin(u * len * 0.8 - t.age * 28.0) * u * (1.0 - p * 0.5); // it lashes about
1076 const int cx = static_cast<int>(std::floor(t.x0 + dx * u));
1077 const int cy = static_cast<int>(std::floor(t.y0 + dy * u + wobble));
1078 if (!out.inBounds(cx, cy)) continue;
1079 Cell& cell = out.at(cx, cy);
1080 const bool tip = u + 1.0 / samples > p;
1081 cell.ch = t.shot ? (tip ? U'*' : U'.') : tip ? U'@' : glyph;
1082 cell.fg = shade(tip ? scale(t.color, 0.8) : t.color, light);
1083 cell.defaultFg = style == Style::Mono;
1084 }
1085 }

1335void World::step(double dt) {
1336 time_ += dt;
1337 for (double& c : crater_) c = std::max(0.0, c - dt / scene_->dustSettle); // dust fills craters
1338 for (float& c : scar_) c = std::max(0.0f, c - static_cast<float>(dt / 40.0)); // the swath of an avalanche fades
1339 if (rainLeft_ > 0) {
1340 rainLeft_ = std::max(0.0, rainLeft_ - dt);
1341 spawnWeather(dt);
1342 }
1343 if (scene_->dayNight && timeMode_ == TimeMode::Cycle) phase_ = std::fmod(phase_ + dt / scene_->dayLength, 1.0);
1344 const double cloudTarget =
1345 (rainLeft_ > 0 || countTag("tornado") > 0 || countTag("gloom") > 0) ? 1.0 : 0.0;
1346 flashElapsed_ += dt;
1347 overcast_ += std::clamp(cloudTarget - overcast_, -0.25 * dt, 0.4 * dt);
1348 const double sun = scene_->dayNight ? light().sun : 1.0;
1349 bool eventAlive = false, rareAlive = false;
1350 for (auto& e : entities_) {
1351 if (e->dead) continue;
1352 eventAlive |= e->isEvent;
1353 rareAlive |= e->isRare;
1354 if (e->sleepBelow > 0 && sun < e->sleepBelow) continue; // asleep: frozen where it is
1355 if (e->flung) { // thrown by a tornado: flies on an arc, lands dazed and walks on afterwards
1356 e->age += dt;
1357 e->vy += scene_->gravity * dt;
1358 e->x += e->vx * dt;
1359 e->y += e->vy * dt;
1360 if (e->vy > 0 && e->y >= e->anchorY) {
1361 e->y = e->anchorY;
1362 e->vy = 0;
1363 e->vx = e->flungVx;
1364 e->flung = false;
1365 e->pauseLeft = 1.5;
1366 }
1367 continue;
1368 }
1369 e->age += dt;
1370 if (e->fleeing) { // in a panic: straight off the screen, and back (as a replacement) a while later
1371 e->pauseLeft = 0;
1372 e->x += e->vx * dt;
1373 animate(*e, dt);
1374 if ((e->vx > 0 && e->x > w_) || (e->vx < 0 && e->x + e->width() < 0)) {
1375 e->dead = true;
1376 if (e->onDeath) restock_.push_back({rng_.range(10.0, 40.0), std::move(e->onDeath)});
1377 e->onDeath = nullptr;
1378 }
1379 continue;
1380 }
1381 if (e->pauseLeft > 0) { // busy (eating): stays frozen where it is
1382 e->pauseLeft = std::max(0.0, e->pauseLeft - dt);
1383 continue;
1384 }
1385 // A cordoned-off fire: traffic and people stop short of the site (those already in it clear off).
1386 if (cordon_ && e->grounded && e->vx != 0 && e->driveState == 0 && e->sprite->name() != "firetruck") {
1387 // Stops short of the barrier with its whole body: it may not reach into the site.
1388 const double lo = cordonX_ - cordonHalf_, hi = cordonX_ + cordonHalf_;
1389 const double next = e->x + e->vx * dt;
1390 const bool inNow = e->x < hi && e->x + e->width() > lo, inNext = next < hi && next + e->width() > lo;
1391 if (!inNow && inNext) {
1392 e->pauseLeft = std::max(e->pauseLeft, 2 * dt); // waits at the edge until it is lifted
1393 continue;
1394 }
1395 }
1396 animate(*e, dt);
1397 for (BehaviorFn b : e->behaviors) b(*e, *this, dt);
1398 if (e->ttl > 0 && e->age > e->ttl) e->dead = true;
1399 }
1400
1401 stepTongues(dt);
1402 stepToxin(dt);
1403 stepFlood(dt);
1404 if (!avenues_.empty()) stepCity(dt);
1405
1406 if (!scene_->events.empty() && !eventAlive) {
1407 if (eventTimer_ < 0) eventTimer_ = rng_.range(scene_->eventInterval.first, scene_->eventInterval.second);
1408 eventTimer_ -= dt;
1409 if (eventTimer_ <= 0) {
1410 spawnEvent();
1411 eventTimer_ = -1;
1412 }
1413 }
1414
1415 if (!scene_->rare.empty() && !rareAlive) {
1416 if (rareTimer_ < 0) rareTimer_ = rng_.range(scene_->rareInterval.first, scene_->rareInterval.second);
1417 rareTimer_ -= dt;
1418 if (rareTimer_ <= 0) {
1419 spawnRare();
1420 rareTimer_ = -1;
1421 }
1422 }
1423
1424 std::vector<std::function<void(World&)>> callbacks;
1425 for (auto& r : restock_) r.left -= dt; // what a catastrophe wiped out comes back, one by one
1426 for (size_t i = restock_.size(); i-- > 0;) {
1427 if (restock_[i].left > 0) continue;
1428 callbacks.push_back(std::move(restock_[i].fn));
1429 restock_.erase(restock_.begin() + static_cast<std::ptrdiff_t>(i));
1430 }
1431 for (auto& e : entities_) {
1432 if (e->dead && e->onDeath) callbacks.push_back(std::move(e->onDeath));
1433 }
1434 std::erase_if(entities_, [](const auto& e) { return e->dead; });
1435 for (auto& cb : callbacks) cb(*this);
1436 flushIncoming();
1437}

1441void World::buildTerrain() {
1442 const SceneDef& sc = *scene_;
1443 const double H = h_;
1444 auto rowAt = [&](double climb) { return static_cast<int>(std::lround(H - 1 - climb * H)); };
1445 const int floorRow = std::max(seabedY_ - 1, 0); // lowest row the skyline may have
1446 const int minRow = std::clamp(rowAt(sc.peakHeight), 0, floorRow);
1447 const int baseRow = std::clamp(rowAt(sc.treeline + 0.05), minRow, floorRow);
1448
1449 // Peaks of different heights with their own slopes on either side; the skyline follows the
1450 // highest of them and falls to the foothills between them.
1451 struct Peak { double x, row, left, right; };
1452 const int n = std::clamp(w_ / 30, 1, 7);
1453 const int tallest = rng_.rangeInt(0, n - 1);
1454 std::vector<Peak> peaks;
1455 for (int i = 0; i < n; ++i) {
1456 const double span = static_cast<double>(w_) / n;
1457 peaks.push_back({(i + 0.5) * span + rng_.range(-0.3, 0.3) * span,
1458 i == tallest ? minRow : minRow + (baseRow - minRow) * rng_.range(0.0, 0.65),
1459 rng_.range(0.32, 0.62), rng_.range(0.32, 0.62)});
1460 }
1461 const double ph[4] = {rng_.range(0, 6.28), rng_.range(0, 6.28), rng_.range(0, 6.28), rng_.range(0, 6.28)};
1462 const double amp = std::max(0.8, H * 0.03);
1463
1464 const size_t cols = static_cast<size_t>(w_);
1465 ridge_.assign(cols, baseRow);
1466 treeRow_.assign(cols, 0);
1467 snowRow_.assign(cols, 0);
1468 side_.assign(cols, 1);
1469 horizonY_ = 0;
1470 for (int x = 0; x < w_; ++x) {
1471 double v = baseRow + 1.2 * std::sin(x * 0.07 + ph[0]);
1472 const Peak* nearest = &peaks[0];
1473 for (const Peak& p : peaks) {
1474 const double dx = x - p.x;
1475 v = std::min(v, p.row + std::fabs(dx) * (dx < 0 ? p.left : p.right));
1476 if (std::fabs(dx) < std::fabs(x - nearest->x)) nearest = &p;
1477 }
1478 int r = static_cast<int>(std::lround(v));
1479 if (r < baseRow - 1) r += static_cast<int>(hash2(x / 2, 91) % 3) - 1; // ragged rock
1480 ridge_[x] = std::clamp(r, minRow, floorRow);
1481 side_[x] = x < nearest->x ? -1 : 1;
1482 horizonY_ = std::max(horizonY_, ridge_[x]);
1483 const int tree = static_cast<int>(std::lround(rowAt(sc.treeline) + amp * (std::sin(x * 0.11 + ph[2]) + 0.6 * std::sin(x * 0.31 + ph[3]))));
1484 const int snow = static_cast<int>(std::lround(rowAt(sc.snowline) + amp * (std::sin(x * 0.09 + ph[3]) + 0.6 * std::sin(x * 0.27 + ph[2]))));
1485 treeRow_[x] = std::clamp(tree, 0, h_);
1486 snowRow_[x] = std::clamp(std::min(snow, treeRow_[x] - 2), 0, h_);
1487 }
1488 const int treeMin = *std::min_element(treeRow_.begin(), treeRow_.end());
1489 const int treeMax = *std::max_element(treeRow_.begin(), treeRow_.end());
1490 const int snowMin = *std::min_element(snowRow_.begin(), snowRow_.end());
1491 const int snowMax = *std::max_element(snowRow_.begin(), snowRow_.end());
1492
1493 // The rows each zone spans (feet rows; for the air zones the lowest row of the sprite).
1494 auto band = [&](Zone z, int lo, int hi) {
1495 lo = std::clamp(lo, 0, std::max(h_ - 1, 0));
1496 zoneLo_[static_cast<int>(z)] = lo;
1497 zoneHi_[static_cast<int>(z)] = std::clamp(hi, lo, std::max(h_ - 1, 0));
1498 };
1499 const int streamBank = std::max(riverTopY_ - 1, seabedY_);
1500 band(Zone::Meadow, seabedY_, streamBank);
1501 band(Zone::Forest, treeMin, seabedY_ - 1);
1502 band(Zone::Woods, treeMin, streamBank);
1503 band(Zone::Rock, snowMin, treeMax);
1504 band(Zone::Snow, minRow, snowMax);
1505 band(Zone::Alpine, minRow, treeMax);
1506 band(Zone::Sky, 0, snowMax);
1507 band(Zone::Air, 0, treeMax);
1508 band(Zone::LowAir, treeMin, streamBank);
1509 surfaceY_ = std::clamp((treeMin + treeMax) / 2, 0, h_); // the air above the tree line
1510 scar_.assign(cols * static_cast<size_t>(h_), 0.0f);
1511}

1569 e.y = e.anchorY = feet - e.height() + 1;
1570 e.z = depthFor(feet);
1571 return;
1572 }
1573 e.y = e.anchorY = zoneHi_[zi] - e.height() + 1;
1574 e.z = depthFor(zoneHi_[zi]);

1582void World::slideStep(Entity& e, double dt) {
1583 constexpr double kWarn = 3.0; // seconds of cracking and creaking before the slab goes
1584 constexpr double kSettle = 5.0; // seconds the cloud lingers once the slide has come to rest
1585 const double bottom = h_ - 1;
1586 if (!slide_.active) { // starts high on a snowy flank and heads downhill
1587 slide_ = Slide{};
1588 slide_.active = true;
1589 std::vector<int> snowy;
1590 for (int x = 2; x < w_ - 2; ++x) {
1591 if (ridge_[static_cast<size_t>(x)] < snowRow_[static_cast<size_t>(x)] - 1) snowy.push_back(x);
1592 }
1593 int x0 = w_ / 2;
1594 if (!snowy.empty()) {
1595 x0 = snowy[static_cast<size_t>(rng_.rangeInt(0, static_cast<int>(snowy.size()) - 1))];
1596 } else {
1597 for (int x = 0; x < w_; ++x) if (ridge_[static_cast<size_t>(x)] < ridge_[static_cast<size_t>(x0)]) x0 = x;
1598 }
1599 const int left = ridgeRow(x0 - 6), right = ridgeRow(x0 + 6); // the lower side is downhill
1600 slide_.dir = left > right ? -1 : right > left ? 1 : (rng_.chance(0.5) ? 1 : -1);
1601 slide_.x0 = x0;
1602 slide_.y0 = ridgeRow(x0) + 1;
1603 slide_.gradient = rng_.range(0.4, 0.9);
1604 const double rows = std::max(bottom - slide_.y0, 1.0);
1605 const double room = slide_.dir > 0 ? w_ - 4 - x0 : x0 - 3; // keep the swath on the screen
1606 slide_.gradient = std::clamp(room / rows, 0.0, slide_.gradient);
1607 slide_.wobble = rng_.range(0, 6.28);
1608 slide_.head = slide_.lastHead = slide_.y0;
1609 }
1610 const double rows = std::max(bottom - slide_.y0, 1.0);
1611 // The middle of the swath at a row, and its half width, which grows downhill as the snow fans out.
1612 auto middle = [&](double row) {
1613 const double prog = std::clamp((row - slide_.y0) / rows, 0.0, 1.0);
1614 return slide_.x0 + slide_.dir * slide_.gradient * (row - slide_.y0) +
1615 1.8 * std::sin((row - slide_.y0) * 0.33 + slide_.wobble) * (0.4 + prog);
1616 };
1617 auto halfWidth = [&](double row) { return 1.5 + 5.0 * std::clamp((row - slide_.y0) / rows, 0.0, 1.0); };
1618
1619 if (e.age < kWarn) { // the slab cracks: snow trickles and puffs up where it is about to let go
1620 slideLevel_ = 0.15 + 0.35 * e.age / kWarn;
1621 e.x = slide_.x0 - e.width() / 2.0;
1622 e.y = slide_.y0 - e.height() / 2.0;
1623 slide_.puffs += 6 * dt;
1624 for (; slide_.puffs >= 1; slide_.puffs -= 1) {
1625 spawnEffect("snowpuff", slide_.x0 + rng_.range(-3, 3), slide_.y0 + rng_.range(0, 2), 1,
1626 rng_.range(-1.5, 1.5), -rng_.range(0.5, 1.5));
1627 }
1628 return;
1629 }
1630 if (slide_.endedAt < 0) {
1631 const double speed = std::min(3.5 + 2.5 * (e.age - kWarn), 11.0);
1632 slide_.head = std::min(bottom, slide_.head + speed * dt);
1633 const double cx = middle(slide_.head), hw = halfWidth(slide_.head);
1634 // The swath: every row the head has passed is covered with snow and rubble.
1635 for (int row = static_cast<int>(std::floor(slide_.lastHead)); row <= static_cast<int>(std::floor(slide_.head)); ++row) {
1636 if (row < 0 || row >= h_) continue;
1637 const double c = middle(row), half = halfWidth(row);
1638 for (int x = static_cast<int>(std::lround(c - half)); x <= static_cast<int>(std::lround(c + half)); ++x) {
1639 if (x >= 0 && x < w_ && row >= ridge_[static_cast<size_t>(x)]) scar_[static_cast<size_t>(row) * w_ + x] = 1.0f;
1640 }
1641 }
1642 // It buries what stands in its way: animals, trees and bushes. Flyers get out of the way.
1643 const double top = slide_.lastHead - 1, bot = slide_.head + 1;
1644 const double x0 = cx - hw - 1, x1 = cx + hw + 1;
1645 for (auto& other : entities_) {
1646 Entity& o = *other;
1647 if (o.dead || o.isRare || !o.destructible || isAirZone(o.zone)) continue;
1648 if (!o.life && o.zone == Zone::None && !o.grounded) continue; // effects
1649 if (o.x + o.width() <= x0 || o.x >= x1 || o.y + o.height() <= top || o.y >= bot) continue;
1650 if (o.life && !isAsleep(o)) ++squeals_; // an animal under the snow cries out
1651 claim(o, "snowpuff");
1652 }
1653 // Powder billows up around the head, and rock and uprooted trees tumble out of the forest.
1654 slide_.puffs += 36 * dt;
1655 for (; slide_.puffs >= 1; slide_.puffs -= 1) {
1656 spawnEffect("snowpuff", cx + rng_.range(-hw, hw), slide_.head - rng_.range(0, 3), 1,
1657 rng_.range(-3, 3), -rng_.range(0.5, 3));
1658 }
1659 const Zone here = zoneAt(static_cast<int>(std::lround(cx)), static_cast<int>(slide_.head));
1660 if (here != Zone::Snow && rng_.chance(14 * dt)) {
1661 spawnEffect("debris", cx + rng_.range(-hw, hw), slide_.head - 1, 1, rng_.range(-5, 5), -rng_.range(1.5, 4));
1662 }
1663 slide_.lastHead = slide_.head;
1664 slideLevel_ = 1.0;
1665 e.x = cx - e.width() / 2.0;
1666 e.y = slide_.head - e.height() * 0.65;
1667 if (slide_.head >= bottom) slide_.endedAt = e.age;
1668 return;
1669 }
1670 // The slide has reached the stream and comes to rest in a billowing cloud that slowly thins.
1671 const double since = e.age - slide_.endedAt;
1672 slideLevel_ = std::max(0.0, 1.0 - since / kSettle);
1673 slide_.puffs += 14 * (1 - since / kSettle) * dt;
1674 for (; slide_.puffs >= 1; slide_.puffs -= 1) {
1675 spawnEffect("snowpuff", e.x + rng_.range(0, e.width()), bottom - rng_.range(0, 4), 1,
1676 rng_.range(-3, 3), -rng_.range(0.5, 2.5));
1677 }
1678 if (since > kSettle) {
1679 slide_.active = false;
1680 slideLevel_ = 0;
1681 e.dead = true;
1682 }
1683}

1687void World::renderBackground(Frame& out, Style style, const Light& light) const {
1688 const SceneDef& sc = *scene_;
1689 // The scene's colors are its daytime look; the light turns them into the current look.
1690 const Light skyLight = sc.airless ? Light{} : light; // an airless sky does not change with the day
1691 const Rgb skyTop = lerp(backdrop(sc.skyTop, skyLight), Rgb{110, 70, 130}, light.glow * 0.3);
1692 const Rgb skyLow = lerp(backdrop(sc.skyLow, skyLight), Rgb{255, 150, 80}, light.glow * 0.55);
1693 const Rgb mediumTop = backdrop(sc.mediumTop, light), mediumDeep = backdrop(sc.mediumDeep, light);
1694 const Rgb groundA = backdrop(sc.groundA, light), groundB = backdrop(sc.groundB, light);
1695 const Rgb surfaceFg = backdrop(sc.surfaceFg, light);
1696 const Rgb riverTop = backdrop(sc.riverTop, light), riverDeep = backdrop(sc.riverDeep, light);
1697
1698 // In a mountain scene the sky reaches down to the lowest foothill; the terrain is drawn over it
1699 // afterwards, so the sun can set behind the peaks.
1700 const int skyEnd = sc.mountains ? horizonY_ : surfaceY_;
1701 for (int y = 0; y < h_; ++y) {
1702 for (int x = 0; x < w_; ++x) {
1703 Cell c;
1704 const uint32_t h = hash2(x, y);
1705 if (sc.mountains && y >= skyEnd) {
1706 c.bg = skyLow;
1707 } else if (y < skyEnd) {
1708 c.bg = lerp(skyTop, skyLow, skyEnd > 1 ? double(y) / (skyEnd - 1) : 1.0);
1709 if (sc.starDensity > 0 && light.stars > 0.2 && h % static_cast<uint32_t>(sc.starDensity) == 0) {
1710 const bool bright = std::sin(time_ * 2.0 + (h >> 8)) > 0; // twinkling stars
1711 c.ch = bright ? U'*' : U'.';
1712 c.fg = lerp(c.bg, bright ? Rgb{250, 250, 220} : Rgb{150, 150, 170}, light.stars);
1713 }
1714 } else if (riverRows_ > 0 && y >= riverTopY_) { // river: flowing ripples
1715 c = riverCell(x, y, riverRows_ > 1 ? double(y - riverTopY_) / (riverRows_ - 1) : 0.0, time_,
1716 sc.riverFlow, riverTop, riverDeep);
1717 } else if (y >= seabedY_) {
1718 const int row = y - seabedY_;
1719 const int dug = craterRows(x);
1720 c.bg = (y == seabedY_) ? groundA : groundB;
1721 const auto& g = sc.groundGlyphs;
1722 c.ch = g.empty() ? U' ' : g[h % g.size()];
1723 c.fg = scale(c.bg, 1.35);
1724 if (row < dug) { // dug out: open sky with the crater walls drawn as slopes
1725 c.bg = skyLow;
1726 c.ch = craterRows(x - 1) <= row ? U'\\' : craterRows(x + 1) <= row ? U'/' : U' ';
1727 c.fg = scale(groundA, 0.9);
1728 } else if (row == dug && dug > 0) {
1729 c.bg = groundB; // freshly exposed, darker floor
1730 }
1731 } else {
1732 const double t = double(y - surfaceY_) / std::max(seabedY_ - surfaceY_, 1);
1733 c.bg = lerp(mediumTop, mediumDeep, std::min(t * 1.2, 1.0));
1734 if (y == surfaceY_ + 1 && hash2(x + static_cast<int>(time_ * 3), y) % 5 == 0) {
1735 c.ch = U'.';
1736 c.fg = scale(surfaceFg, 0.7);
1737 }
1738 }
1739 if (const double m = mudAt(x); m > 0.01 && y >= seabedY_) { // the flood has been here
1740 const Rgb mud = backdrop({112, 84, 52}, light);
1741 c.bg = lerp(c.bg, mud, (riverRows_ > 0 && y >= riverTopY_ ? 0.55 : 0.85) * m);
1742 c.fg = scale(c.bg, 1.25);
1743 if (y < riverTopY_ || riverRows_ == 0) c.ch = h % 4 == 0 ? U'.' : h % 7 == 0 ? U',' : U' ';
1744 }
1745 if (hasMedium_ && y == surfaceY_ && sc.surfaceGlyphs.size() == 3) { // animated waterline
1746 const double v = std::sin(x * 0.45 + time_ * 1.6) + 0.5 * std::sin(x * 0.17 - time_ * 0.7);
1747 c.ch = v > 0.5 ? sc.surfaceGlyphs[0] : v > -0.5 ? sc.surfaceGlyphs[1] : sc.surfaceGlyphs[2];
1748 c.fg = surfaceFg;
1749 c.bg = mediumTop;
1750 }
1751 if (style != Style::Full) c.defaultBg = true; // no backdrop in the terminal styles
1752 if (style == Style::Mono) c.defaultFg = true;
1753 out.at(x, y) = c;
1754 }
1755 }

1758void World::renderCelestial(Frame& out, Style style, const Light& light) const {
1759 const int skyEnd = scene_->mountains ? horizonY_ : surfaceY_;
1760 if (!scene_->dayNight || skyEnd <= 0 || overcast_ > 0.6) return;
1761 constexpr double kPi = 3.14159265358979;
1762 const bool byDay = phase_ >= 0.25 && phase_ <= 0.75;
1763 // How far along its arc the body is: 0 as it rises, 1 as it sets.
1764 const double u = byDay ? (phase_ - 0.25) / 0.5 : std::fmod(phase_ + 0.25, 1.0) / 0.5;
1765 auto sprite = pack_->find(byDay ? scene_->sunSprite : scene_->moonSprite);
1766 if (!sprite) return;
1767 Entity body;
1768 body.sprite = sprite;
1769 body.glow = true;
1770 body.fg = sprite->props().color("color", {255, 240, 180});
1771 body.x = u * (w_ - 1) - sprite->width() / 2.0;
1772 // At the horizon the whole body is just below it; it climbs to the top of the sky at mid-arc.
1773 // In a sky lower than the body it stops with its feet on the sky's last row and its top cut off.
1774 const double peak = std::min(0.0, static_cast<double>(skyEnd - sprite->height()));
1775 body.y = skyEnd - (skyEnd - peak) * std::sin(kPi * u);
1776 blit(out, body, style, light, skyEnd); // rows below the horizon stay hidden
1777}

1779void World::renderTerrain(Frame& out, Style style, const Light& light) const {
1780 const SceneDef& sc = *scene_;
1781 const Rgb groundA = backdrop(sc.groundA, light), groundB = backdrop(sc.groundB, light);
1782 const Rgb forestA = backdrop(sc.forestA, light), forestB = backdrop(sc.forestB, light);
1783 const Rgb rockA = backdrop(sc.rockA, light), rockB = backdrop(sc.rockB, light);
1784 const Rgb snowA = backdrop(sc.snowA, light), snowB = backdrop(sc.snowB, light);
1785 const Rgb riverTop = backdrop(sc.riverTop, light), riverDeep = backdrop(sc.riverDeep, light);
1786 const Rgb slideSnow = backdrop({244, 248, 252}, light), rubble = backdrop({120, 98, 78}, light);
1787 auto glyph = [](const std::u32string& set, uint32_t h) { return set.empty() ? U' ' : set[h % set.size()]; };
1788
1789 for (int x = 0; x < w_; ++x) {
1790 const size_t xi = static_cast<size_t>(x);
1791 const int top = ridge_[xi];
1792 for (int y = top; y < h_; ++y) {
1793 const uint32_t h = hash2(x, y);
1794 const Zone z = zoneAt(x, y);
1795 Cell c;
1796 if (riverRows_ > 0 && y >= riverTopY_) {
1797 c = riverCell(x, y, riverRows_ > 1 ? double(y - riverTopY_) / (riverRows_ - 1) : 0.0, time_,
1798 sc.riverFlow, riverTop, riverDeep);
1799 } else {
1800 if (z == Zone::Meadow) {
1801 c.bg = y == seabedY_ ? groundA : groundB;
1802 c.ch = glyph(sc.groundGlyphs, h);
1803 } else if (z == Zone::Forest) {
1804 c.bg = lerp(forestA, forestB, double(y - treeRow_[xi]) / std::max(seabedY_ - treeRow_[xi], 1));
1805 c.ch = glyph(sc.forestGlyphs, h);
1806 } else if (z == Zone::Rock) {
1807 c.bg = lerp(rockA, rockB, double(y - snowRow_[xi]) / std::max(treeRow_[xi] - snowRow_[xi], 1));
1808 c.ch = glyph(sc.rockGlyphs, h);
1809 } else {
1810 c.bg = lerp(snowA, snowB, double(y - top) / std::max(snowRow_[xi] - top, 1));
1811 c.ch = glyph(sc.snowGlyphs, h);
1812 }
1813 if (side_[xi] > 0 && z != Zone::Meadow) c.bg = scale(c.bg, 0.86); // the far flank is in shadow
1814 c.fg = (z == Zone::Rock || z == Zone::Snow) ? scale(c.bg, 0.74) : scale(c.bg, 1.4);
1815 if (z == Zone::Rock || z == Zone::Snow) { // gullies run down the flank from the peak
1816 const int diag = side_[xi] < 0 ? x + y : x - y;
1817 if (((diag % 8) + 8) % 8 == 0 && h % 3 != 0) c.ch = side_[xi] < 0 ? U'/' : U'\\';
1818 }
1819 if (y == top && z != Zone::Meadow) { // the skyline
1820 const int l = ridge_[static_cast<size_t>(std::max(x - 1, 0))], r = ridge_[static_cast<size_t>(std::min(x + 1, w_ - 1))];
1821 c.ch = (l > top && r > top) ? U'^' : l > r ? U'/' : r > l ? U'\\' : U'-';
1822 c.fg = scale(c.bg, z == Zone::Snow ? 0.62 : 0.55);
1823 }
1824 }
1825 if (const double s = scarAt(x, y); s > 0.01) { // an avalanche has been through
1826 const double t = std::min(1.0, s * 1.3);
1827 const uint32_t q = hash2(x * 3, y * 7);
1828 c.bg = lerp(c.bg, slideSnow, t);
1829 c.fg = lerp(c.fg, q % 5 == 0 ? rubble : scale(slideSnow, 0.72), t);
1830 c.ch = q % 5 == 0 ? U'o' : q % 3 == 0 ? U'*' : q % 2 == 0 ? U':' : U'.';
1831 }
1832 if (style != Style::Full) c.defaultBg = true;
1833 if (style == Style::Mono) c.defaultFg = true;
1834 out.at(x, y) = c;
1835 }
1836 }

1839void World::render(Frame& out, Style style) const {
1840 out.resize(w_, h_);
1841 const Light l = light();
1842 renderBackground(out, style, l);
1843 renderCelestial(out, style, l);
1844 if (scene_->mountains) renderTerrain(out, style, l);
1845
1846 std::vector<const Entity*> order;
1847 order.reserve(entities_.size());
1848 for (const auto& e : entities_) {
1849 if (visibleNow(*e, l, scene_->dayNight)) order.push_back(e.get());
1850 }
1851 std::stable_sort(order.begin(), order.end(),
1852 [](const Entity* a, const Entity* b) { return a->z > b->z; });
1853 // In a city the skyline, the railway and the street come between the far things (sky, clouds,
1854 // planes) and the near ones (people, cars, shops).
1855 // Vehicles on the side streets (z 24) are drawn between the side streets and the viaduct, so that
1856 // they pass under it.
1857 constexpr int kCityZ = 25, kStreetZ = 23;
1858 bool backDrawn = !scene_->city, streetDrawn = !scene_->city;
1859 for (const Entity* e : order) {
1860 if (!backDrawn && e->z < kCityZ) {
1861 renderCityBack(out, style, l);
1862 backDrawn = true;
1863 }
1864 if (!streetDrawn && e->z < kStreetZ) {
1865 renderStreet(out, style, l);
