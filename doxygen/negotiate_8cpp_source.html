55bool routeLayout(const Layout& layout, const DeviceReport& report, int route[kMaxChannels]) {
56 const int used = std::min(report.channels, kMaxChannels);
57 bool taken[kMaxChannels] = {};
58 for (int i = 0; i < kMaxChannels; ++i) route[i] = -1;
59 for (int i = 0; i < layout.channels(); ++i) {
60 const DevicePosition want = positionOfChannel(layout.speaker(i).channel);
61 if (want == DevicePosition::None) return false;
62 for (int d = 0; d < used; ++d) {
63 if (taken[d] || report.position[d] != want) continue;
64 route[i] = d;
65 taken[d] = true;
66 break;
67 }
68 if (route[i] < 0) return false;
69 }
70 return true;
71}

143void portsForLayout(const Layout& layout, DevicePort out[kMaxChannels]) {
144 int aux = 0;
145 for (int i = 0; i < kMaxChannels; ++i) out[i] = DevicePort{};
146 for (int i = 0; i < layout.channels() && i < kMaxChannels; ++i) {
147 DevicePosition p = DevicePosition::Aux; // what has no standard position
148 switch (layout.speaker(i).channel) {
149 case Channel::L: p = DevicePosition::FrontLeft; break;
150 case Channel::R: p = DevicePosition::FrontRight; break;
151 case Channel::C: p = DevicePosition::FrontCenter; break;
152 case Channel::LFE: p = DevicePosition::Lfe; break;
153 case Channel::Ls: p = DevicePosition::SideLeft; break;
154 case Channel::Rs: p = DevicePosition::SideRight; break;
155 case Channel::Lrs: p = DevicePosition::BackLeft; break;
156 case Channel::Rrs: p = DevicePosition::BackRight; break;
157 case Channel::Ltf: p = DevicePosition::TopFrontLeft; break;
158 case Channel::Rtf: p = DevicePosition::TopFrontRight; break;
159 case Channel::Ltr: p = DevicePosition::TopBackLeft; break;
160 case Channel::Rtr: p = DevicePosition::TopBackRight; break;
161 default: break; // Lw, Rw, Ltm, Rtm and unmapped channels
162 }
163 out[i].position = p;
164 if (p == DevicePosition::Aux) out[i].aux = aux++;
165 }

180Negotiation negotiate(const LayoutChoice& choice, const DeviceReport& report) {
181 Negotiation n;
182 const int deviceChannels = std::max(1, report.channels);
183 if (!choice.automatic) {
184 n.layout = Layout::named(choice.id);
185 if (report.hasMap && report.exact) {
186 // The device's own map is known: play the layout on the channels of its positions. Where the device has no
187 // place for one of its channels, step down to the richest layout it has (7.1.4 on a 5.1 device is 5.1), which
188 // is better than letting the library mix it into an assumed map (that plays on channels the device does not
189 // use). A plain stereo or mono device is left to the system, whose mix-down (and spatializer) is wanted there.
190 LayoutId id = choice.id;
191 for (;;) {
192 const Layout l = Layout::named(id);
193 if (routeLayout(l, report, n.route)) {
194 n.layout = l;
195 n.deviceChannels = deviceChannels; // the device stays as it is; the layout's channels are written to its own
196 n.routed = true;
197 return n;
198 }
199 LayoutId lower;
200 if (deviceChannels <= 2 || !lowerLayout(id, lower)) break;
201 id = lower;
202 }
203 }
204 n.deviceChannels = n.layout.channels();
205 n.reopen = true;
206 // More channels than the device has; or a device whose own map is known and has no speaker for one of the
207 // layout's channels (a stereo device asked for 7.1.4), so that the system has to mix it into the others.
208 n.downmixed = n.layout.channels() > deviceChannels || (report.hasMap && report.exact);
209 n.spatialized = report.spatializes && n.layout.channels() > 2 && deviceChannels <= 2; // a named multichannel layout on such a route
210 return n;
211 }
212 if (report.spatializes && deviceChannels <= 2) { // headphones whose system spatializer wants more than a pair
213 n.layout = Layout::named(LayoutId::S7_1_4);
214 n.deviceChannels = deviceChannels; // the route stays as it is; the system makes the pair from the layout
215 n.downmixed = true;
216 n.spatialized = true;
217 return n;
218 }
219 int used = std::min(deviceChannels, kMaxChannels);
220 if (report.hasMap) {
221 while (used > 1 && report.position[used - 1] == DevicePosition::None) --used; // unused channels at the end are no speakers
222 if (report.exact) { // a platform's map whose channels are exactly those of a named layout, in another order or with gaps: that layout, routed
223 int active = 0;
224 for (int i = 0; i < used; ++i) active += report.position[i] != DevicePosition::None;
225 for (int id = kLayoutCount - 1; id >= 0; --id) {
226 const Layout named = Layout::named(static_cast<LayoutId>(id));
227 if (named.channels() != active || !routeLayout(named, report, n.route)) continue;
228 bool inOrder = active == used;
229 for (int i = 0; i < named.channels(); ++i) inOrder = inOrder && n.route[i] == i;
230 n.layout = named;
231 n.deviceChannels = deviceChannels;
232 n.routed = !inOrder; // (in the device's own order it is a plain open, as for a map from the library)
233 return n;
234 }
235 }
236 DeviceChannel dc[kMaxChannels];
237 for (int i = 0; i < used; ++i) dc[i] = deviceChannelFor(report.position[i]);
238 n.layout = Layout::fromDevice(dc, used);
239 for (int id = kLayoutCount - 1; id >= 0; --id) { // the device's own order is a named layout's: use the named one
240 if (sameChannels(n.layout, static_cast<LayoutId>(id))) {
241 n.layout = Layout::named(static_cast<LayoutId>(id));
242 break;
243 }
244 }
245 n.deviceChannels = deviceChannels; // (more than the layout's, if the device has more than kMaxChannels: the rest stays silent)
246 n.reopen = false;
247 return n;
248 }
249 if (deviceChannels == 1) { // a single speaker
250 const DeviceChannel one;
251 n.layout = Layout::fromDevice(&one, 1);
252 n.deviceChannels = 1;
253 return n;
254 }
255 n.layout = Layout::named(richestFor(deviceChannels));
256 n.deviceChannels = n.layout.channels();
257 n.reopen = true; // without a map the device's order is not known: it is told the count and the map
258 return n;
259}

261void padChannels(const float* in, int layoutChannels, float* out, int deviceChannels, int frames) {
262 for (int f = 0; f < frames; ++f) {
263 for (int c = 0; c < layoutChannels; ++c) out[c] = in[c];
264 for (int c = layoutChannels; c < deviceChannels; ++c) out[c] = 0.0f;
265 in += layoutChannels;
266 out += deviceChannels;
267 }

270bool lowerLayout(LayoutId from, LayoutId& to) {
271 switch (from) {
272 case LayoutId::S9_1_6:
273 case LayoutId::S9_1_4: to = LayoutId::S7_1_4; return true;
274 case LayoutId::S7_1_4:
275 case LayoutId::S7_1_2: to = LayoutId::S7_1; return true;
276 case LayoutId::S5_1_4:
277 case LayoutId::S5_1_2: to = LayoutId::S5_1; return true;
278 case LayoutId::S7_1: to = LayoutId::S5_1; return true;
279 case LayoutId::S5_1: to = LayoutId::Stereo; return true;
280 case LayoutId::Stereo: return false;
281 }
282 return false;
283}
