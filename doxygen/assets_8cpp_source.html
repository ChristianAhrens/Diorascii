22void AssetPack::addText(const std::string& sourceName, const std::string& text) {
23 std::string err;
24 if (sourceName.size() >= 6 && sourceName.compare(sourceName.size() - 6, 6, ".scene") == 0) {
25 auto scene = SceneDef::parse(text, err);
26 if (!scene) {
27 errors_.push_back(sourceName + ": " + err);
28 return;
29 }
30 scenes_[scene->name] = std::move(scene);
31 return;
32 }
33 auto sprite = Sprite::parse(text, err);
34 if (!sprite) {
35 errors_.push_back(sourceName + ": " + err);
36 return;
37 }
38 sprites_[sprite->name()] = std::move(sprite);
39}

41size_t AssetPack::addDirectory(const std::filesystem::path& dir) {
42 namespace fs = std::filesystem;
43 size_t added = 0;
44 std::error_code ec;
45 std::vector<fs::path> files;
46 for (fs::recursive_directory_iterator it(dir, ec), end; !ec && it != end; it.increment(ec)) {
47 const auto ext = it->path().extension();
48 if (it->is_regular_file() && (ext == ".sprite" || ext == ".scene")) files.push_back(it->path());
49 }
50 if (ec) errors_.push_back(dir.string() + ": " + ec.message());
51 std::sort(files.begin(), files.end());
52 struct Revalidate { AssetPack& p; ~Revalidate() { p.validate(); } } revalidate{*this};
53 for (const auto& p : files) {
54 std::ifstream in(p, std::ios::binary);
55 std::stringstream ss;
56 ss << in.rdbuf();
57 const size_t before = errors_.size();
58 addText(p.string(), ss.str());
59 if (errors_.size() == before) ++added;
60 }
61 return added;
62}

78void AssetPack::validate() {
79 auto check = [&](const SceneDef& sc, const std::string& what, const std::string& sprite) {
80 if (sprite.empty() || sprite[0] == '@' || sprites_.count(sprite)) return;
81 const std::string msg = "scene '" + sc.name + "': " + what + " refers to unknown sprite '" + sprite + "'";
82 if (std::find(errors_.begin(), errors_.end(), msg) == errors_.end()) errors_.push_back(msg);
83 };
84 for (const auto& [name, sc] : scenes_) {
85 for (const auto& p : sc->population) check(*sc, "population", p.sprite);
86 for (const auto& d : sc->decor) check(*sc, "decor", d.sprite);
87 for (const auto& e : sc->events) check(*sc, "events", e.sprite);
88 for (const auto& e : sc->rare) check(*sc, "rare", e.sprite);
89 check(*sc, "action", sc->action.sprite);
90 check(*sc, "action", sc->action.extra);
91 check(*sc, "sun-sprite", sc->sunSprite);
92 check(*sc, "moon-sprite", sc->moonSprite);
93 }
94}
