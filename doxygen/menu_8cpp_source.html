70bool Menu::handle(Action a) {
71 if (!open_) return false;
72 if (items_.empty()) {
73 if (a == Action::Back || a == Action::Menu) close();
74 return a == Action::Back || a == Action::Menu;
75 }
76 Item& cur = items_[selected_];
77 switch (a) {
78 case Action::Up: selected_ = (selected_ + items_.size() - 1) % items_.size(); return true;
79 case Action::Down: selected_ = (selected_ + 1) % items_.size(); return true;
80 case Action::Left:
81 if (!cur.options.empty()) step(cur, -1);
82 return true;
83 case Action::Right:
84 if (!cur.options.empty()) step(cur, +1);
85 return true;
86 case Action::Select:
87 if (cur.options.empty()) events_.push_back({cur.id, 0});
88 else step(cur, +1);
89 return true;
90 case Action::Back:
91 case Action::Menu: close(); return true;
92 default: return false;
93 }
94}

102void Menu::render(Frame& f, Style style) const {
103 if (!open_ || f.width() < 1 || f.height() < 1) return;
104
105 // Row text for every item, without the selection marker.
106 int labelW = 0;
107 for (const auto& it : items_) labelW = std::max(labelW, cols(it.label));
108 int optW = 0;
109 for (const auto& it : items_) {
110 for (const auto& o : it.options) optW = std::max(optW, cols(o));
111 }
112 std::vector<std::string> rows;
113 for (const auto& it : items_) {
114 std::string r = pad(it.label, labelW);
115 if (!it.options.empty()) r += " < " + pad(it.options[it.value], optW) + " >";
116 rows.push_back(std::move(r));
117 }
118
119 const std::string hintLong = "arrows: move / change enter: select esc: close";
120 const std::string hintShort = "esc: close";
121 int innerW = std::max(cols(title_), 0);
122 for (const auto& r : rows) innerW = std::max(innerW, cols(r) + 2); // + selection marker
123 const int boxW = innerW + 4;
124 const bool longHint = f.width() >= std::max(boxW, cols(hintLong) + 4);
125 const std::string& hint = longHint ? hintLong : hintShort;
126 innerW = std::max({innerW, cols(hint)});
127 for (const auto& line : footer_) innerW = std::max(innerW, cols(line));
128 const int boxWFinal = innerW + 4;
129 const int footerRows = footer_.empty() ? 0 : static_cast<int>(footer_.size()) + 1; // a blank row, then the lines
130 const int boxH = static_cast<int>(rows.size()) + 6 + footerRows; // border, title, blank, rows, blank, hint, footer, border
131
132 const bool plain = style != Style::Full;
133 auto base = [&](Rgb fg, Rgb bg) {
134 Cell c;
135 c.fg = fg;
136 c.bg = bg;
137 c.defaultFg = c.defaultBg = plain;
138 return c;
139 };
140 const Cell panel = base({225, 232, 245}, {18, 24, 44});
141 const Cell border = base({110, 170, 220}, {18, 24, 44});
142 const Cell faint = base({140, 155, 185}, {18, 24, 44});
143 Cell titleCell = base({255, 214, 120}, {18, 24, 44});
144 titleCell.bold = true;
145 Cell selCell = base({255, 255, 255}, {40, 90, 140});
146 selCell.reverse = plain;
147
148 auto put = [&](int x, int y, char32_t ch, Cell c) {
149 if (!f.inBounds(x, y)) return;
150 c.ch = ch;
151 f.at(x, y) = c;
152 };
153 auto text = [&](int x, int y, const std::string& s, const Cell& c) {
154 for (char32_t ch : decodeUtf8(s)) put(x++, y, ch, c);
155 };
156
157 // Too small for the box: show just the selected entry on one line.
158 if (boxWFinal > f.width() || boxH > f.height()) {
159 if (items_.empty()) return;
160 std::string line = "[" + title_ + "] " + items_[selected_].label;
161 if (!items_[selected_].options.empty()) {
162 line += ": < " + items_[selected_].options[items_[selected_].value] + " >";
163 }
164 line.resize(std::min<size_t>(line.size(), static_cast<size_t>(f.width())));
165 for (int x = 0; x < f.width(); ++x) put(x, f.height() / 2, U' ', selCell);
166 text(0, f.height() / 2, line, selCell);
167 return;
168 }
169
170 const int x0 = (f.width() - boxWFinal) / 2, y0 = (f.height() - boxH) / 2;
171 for (int y = 0; y < boxH; ++y) {
172 for (int x = 0; x < boxWFinal; ++x) put(x0 + x, y0 + y, U' ', panel); // opaque panel
173 }
174 for (int x = 0; x < boxWFinal; ++x) {
175 const bool corner = x == 0 || x == boxWFinal - 1;
176 put(x0 + x, y0, corner ? U'+' : U'-', border);
177 put(x0 + x, y0 + boxH - 1, corner ? U'+' : U'-', border);
178 }
179 for (int y = 1; y < boxH - 1; ++y) {
180 put(x0, y0 + y, U'|', border);
181 put(x0 + boxWFinal - 1, y0 + y, U'|', border);
182 }
183
184 text(x0 + (boxWFinal - cols(title_)) / 2, y0 + 1, title_, titleCell);
185 for (size_t i = 0; i < rows.size(); ++i) {
186 const int y = y0 + 3 + static_cast<int>(i);
187 const bool sel = i == selected_;
188 if (sel) {
189 for (int x = 1; x < boxWFinal - 1; ++x) put(x0 + x, y, U' ', selCell);
190 }
191 text(x0 + 2, y, (sel ? "> " : " ") + rows[i], sel ? selCell : panel);
192 }
193 const int hintY = y0 + boxH - 2 - footerRows;
194 text(x0 + (boxWFinal - cols(hint)) / 2, hintY, hint, panel);
195 for (size_t i = 0; i < footer_.size(); ++i) {
196 text(x0 + (boxWFinal - cols(footer_[i])) / 2, hintY + 2 + static_cast<int>(i), footer_[i], faint);
197 }
