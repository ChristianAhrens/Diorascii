70bool AudioOutput::start() {
71 if (impl_->open) return true;
72 ma_device_config cfg = ma_device_config_init(ma_device_type_playback);
73 cfg.playback.format = ma_format_f32;
74 cfg.playback.channels = 2;
75 cfg.sampleRate = 48000;
76 cfg.dataCallback = onAudio;
77 cfg.pUserData = impl_.get();
78
79 // Sound libraries such as ALSA print their complaints to stderr, which would scribble over
80 // the scene. Silence stderr while the device opens.
81#ifndef _WIN32
82 const int savedErr = dup(STDERR_FILENO);
83 const int devNull = open("/dev/null", O_WRONLY);
84 if (devNull >= 0) dup2(devNull, STDERR_FILENO);
85#endif
86 bool ok = ma_device_init(nullptr, &cfg, &impl_->device) == MA_SUCCESS;
87 if (ok && ma_device_start(&impl_->device) != MA_SUCCESS) {
88 ma_device_uninit(&impl_->device);
89 ok = false;
90 }
91#ifndef _WIN32
92 if (savedErr >= 0) { dup2(savedErr, STDERR_FILENO); close(savedErr); }
93 if (devNull >= 0) close(devNull);
94#endif
95 impl_->open = ok;
96 return ok;
97}
