19void unitVector(float azimuth, float elevation, float v[3]) {
20 const float a = azimuth * kPi / 180.0f, e = elevation * kPi / 180.0f;
21 v[0] = std::sin(a) * std::cos(e);
22 v[1] = std::cos(a) * std::cos(e);
23 v[2] = std::sin(e);

28bool normalize(float* g, const int* idx, int n) {
29 float sum = 0;
30 for (int i = 0; i < n; ++i) sum += g[idx[i]] * g[idx[i]];
31 if (!(sum > 1e-20f)) return false;
32 const float inv = 1.0f / std::sqrt(sum);
33 for (int i = 0; i < n; ++i) g[idx[i]] *= inv;
34 return true;

39void Spatializer::gains(const Layout& layout, const Source& source, float out[kMaxChannels]) {
40 std::fill(out, out + kMaxChannels, 0.0f);
41 int idx[kMaxChannels]; // the channels the panner may use
42 int n = 0;
43 for (int i = 0; i < layout.channels(); ++i)
44 if (layout.speaker(i).isDirectional()) idx[n++] = i;
45 if (n == 0) return;
46
47 const float az = finiteOr(source.azimuth, 0), el = finiteOr(source.elevation, 0);
48 const float spread = std::clamp(finiteOr(source.spread, 0), 0.0f, 1.0f);
49
50 if (n == 1) {
51 out[idx[0]] = 1.0f;
52 return;
53 }
54 if (layout.isStereo()) {
55 // The synth's own law, driven by the azimuth: +-90 degrees is hard left or right.
56 const float pan = std::clamp(0.5f + 0.5f * std::sin(az * kPi / 180.0f), 0.0f, 1.0f);
57 out[idx[0]] = std::max(0.0f, std::cos(pan * kPi * 0.5f)); // cos(pi/2) is -4e-8 in float
58 out[idx[1]] = std::max(0.0f, std::sin(pan * kPi * 0.5f));
59 } else {
60 float s[3];
61 unitVector(az, el, s);
62 for (int k = 0; k < n; ++k) {
63 const Speaker& sp = layout.speaker(idx[k]);
64 float p[3];
65 unitVector(sp.azimuth, sp.elevation, p);
66 const float dx = s[0] - p[0], dy = s[1] - p[1], dz = s[2] - p[2];
67 const float d2 = dx * dx + dy * dy + dz * dz + spatial::kBlur * spatial::kBlur;
68 out[idx[k]] = sp.weight * std::pow(d2, -0.5f * spatial::kRolloff);
69 }
70 }
71 if (!normalize(out, idx, n)) {
72 for (int k = 0; k < n; ++k) out[idx[k]] = 1.0f; // degenerate weights: fall back to an even spread
73 }
74
75 if (spread > 0) {
76 const float even = 1.0f / std::sqrt(static_cast<float>(n));
77 for (int k = 0; k < n; ++k) out[idx[k]] = (1.0f - spread) * out[idx[k]] + spread * even;
78 }
79 normalize(out, idx, n);
80}

82SpatialVoice::SpatialVoice(int sampleRate) {

84 k_ = 1.0f - std::exp(-1.0f / (spatial::kSmoothingSeconds * rate));

88void SpatialVoice::setTarget(const Layout& layout, const Source& source, bool snap) {
89 Spatializer::gains(layout, source, target_);
90
91 // The bass send goes to the LFE channel. A layout without one folds it into the front
92 // speakers (or into all of them if none stands in front), at constant power.
93 std::fill(bassTarget_, bassTarget_ + kMaxChannels, 0.0f);
94 const float send = std::clamp(finiteOr(source.lfeSend, 0), 0.0f, 1.0f);
95 if (layout.lfeIndex() >= 0) {
96 bassTarget_[layout.lfeIndex()] = send;
97 } else {
98 int front[kMaxChannels], all[kMaxChannels];
99 int nFront = 0, nAll = 0;
100 for (int i = 0; i < layout.channels(); ++i) {
101 const Speaker& sp = layout.speaker(i);
102 if (!sp.isDirectional()) continue;
103 all[nAll++] = i;
104 if (std::fabs(sp.azimuth) <= 45.0f && sp.elevation <= 1.0f) front[nFront++] = i;
105 }
106 const int* use = nFront > 0 ? front : all;
107 const int n = nFront > 0 ? nFront : nAll;
108 for (int k = 0; k < n; ++k) bassTarget_[use[k]] = send / std::sqrt(static_cast<float>(n));
109 }
110
111 if (snap) {
112 std::copy(target_, target_ + kMaxChannels, gain_);
113 std::copy(bassTarget_, bassTarget_ + kMaxChannels, bassGain_);
114 }
115}

122void SpatialVoice::step(int channels) {
123 const int n = std::min(channels, kMaxChannels);
124 // A gain that glides towards 0 would sink into the denormal numbers, which the processor computes
125 // with many times the effort: once it is this close it has arrived.
126 auto glide = [this](float& g, float target) {
127 g += (target - g) * k_;
128 if (std::fabs(target - g) < 1e-10f) g = target;
129 };
130 for (int c = 0; c < n; ++c) {
131 glide(gain_[c], target_[c]);
132 glide(bassGain_[c], bassTarget_[c]);
133 }
134}

136void SpatialVoice::add(float x, float* frame, int channels) {
137 low_[0] += (x - low_[0]) * lowK_;
138 low_[1] += (low_[0] - low_[1]) * lowK_;
139 if (std::fabs(low_[0]) < 1e-18f) low_[0] = 0; // a silent voice's filter must not sink into the denormal numbers
140 if (std::fabs(low_[1]) < 1e-18f) low_[1] = 0;
141 const int n = std::min(channels, kMaxChannels);
142 for (int c = 0; c < n; ++c) frame[c] += x * gain_[c] + low_[1] * bassGain_[c];
143}
