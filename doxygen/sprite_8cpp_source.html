76std::shared_ptr<const Sprite> Sprite::parse(std::string_view text, std::string& err) {
77 auto sprite = std::shared_ptr<Sprite>(new Sprite());
78 enum class Section { Header, Art, Mask } section = Section::Header;
79 std::vector<SpriteFrame> frames;
80
81 size_t pos = 0;
82 int lineNo = 0;
83 while (pos <= text.size()) {
84 size_t nl = text.find('\n', pos);
85 if (nl == std::string_view::npos) nl = text.size();
86 std::string_view line = text.substr(pos, nl - pos);
87 pos = nl + 1;
88 ++lineNo;
89 if (!line.empty() && line.back() == '\r') line.remove_suffix(1);
90
91 if (line.substr(0, 3) == "---") {
92 const std::string kind = lower(trim(line.substr(3)));
93 if (kind == "art") {
94 frames.emplace_back();
95 section = Section::Art;
96 } else if (kind == "mask") {
97 if (frames.empty()) {
98 err = "line " + std::to_string(lineNo) + ": '--- mask' before any '--- art'";
99 return nullptr;
100 }
101 section = Section::Mask;
102 } else {
103 err = "line " + std::to_string(lineNo) + ": unknown section '" + kind + "'";
104 return nullptr;
105 }
106 continue;
107 }
108
109 switch (section) {
110 case Section::Header: {
111 const std::string t = trim(line);
112 if (t.empty() || t[0] == '#') break;
113 const size_t colon = t.find(':');
114 if (colon == std::string::npos) {
115 err = "line " + std::to_string(lineNo) + ": expected 'key: value'";
116 return nullptr;
117 }
118 sprite->props_.set(lower(trim(t.substr(0, colon))), trim(t.substr(colon + 1)));
119 break;
120 }
121 case Section::Art:
122 frames.back().art.push_back(decodeUtf8(line));
123 break;
124 case Section::Mask:
125 frames.back().mask.push_back(decodeUtf8(line));
126 break;
127 }
128 }
129
130 sprite->name_ = sprite->props_.str("name");
131 if (sprite->name_.empty()) {
132 err = "missing 'name:' header";
133 return nullptr;
134 }
135 if (frames.empty()) {
136 err = "no '--- art' section";
137 return nullptr;
138 }
139 sprite->frames_ = std::move(frames);
140 sprite->finish();
141 return sprite;
142}

154void Sprite::finish() {
155 // Normalize: trim trailing empty rows, give art and mask the same shape.
156 for (auto& f : frames_) {
157 trimTrailingEmpty(f.art);
158 trimTrailingEmpty(f.mask);
159 h_ = std::max<int>(h_, static_cast<int>(std::max(f.art.size(), f.mask.size())));
160 for (auto& l : f.art) w_ = std::max<int>(w_, static_cast<int>(l.size()));
161 for (auto& l : f.mask) w_ = std::max<int>(w_, static_cast<int>(l.size()));
162 }
163 for (auto& f : frames_) {
164 f.art.resize(h_);
165 f.mask.resize(h_);
166 for (auto& l : f.art) l.resize(w_, U' ');
167 for (auto& l : f.mask) l.resize(w_, U' ');
168 }
169 mirrored_.clear();
170 auto letter = [](char32_t c) { return (c >= U'A' && c <= U'Z') || (c >= U'a' && c <= U'z'); };
171 for (const auto& f : frames_) {
172 SpriteFrame m;
173 for (size_t row = 0; row < f.art.size(); ++row) {
174 std::u32string r(f.art[row].rbegin(), f.art[row].rend());
175 std::u32string k(f.mask[row].rbegin(), f.mask[row].rend());
176 for (auto& c : r) c = mirrorChar(c);
177 // Lettering (FIRE, TAXI) must stay readable: a run of letters is put back in reading order.
178 for (size_t i = 0; i < r.size();) {
179 if (!letter(r[i])) { ++i; continue; }
180 size_t j = i;
181 while (j < r.size() && letter(r[j])) ++j;
182 if (j - i >= 3) {
183 std::reverse(r.begin() + static_cast<std::ptrdiff_t>(i), r.begin() + static_cast<std::ptrdiff_t>(j));
184 std::reverse(k.begin() + static_cast<std::ptrdiff_t>(i), k.begin() + static_cast<std::ptrdiff_t>(j));
185 }
186 i = j;
187 }
188 m.art.push_back(std::move(r));
189 m.mask.push_back(std::move(k));
190 }
191 mirrored_.push_back(std::move(m));
192 }
193}
