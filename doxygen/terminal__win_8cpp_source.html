56bool Terminal::init() {
57 impl_->in = GetStdHandle(STD_INPUT_HANDLE);
58 impl_->out = GetStdHandle(STD_OUTPUT_HANDLE);
59 if (!GetConsoleMode(impl_->in, &impl_->savedIn) || !GetConsoleMode(impl_->out, &impl_->savedOut))
60 return false;
61
62 DWORD outMode = impl_->savedOut | ENABLE_VIRTUAL_TERMINAL_PROCESSING | DISABLE_NEWLINE_AUTO_RETURN;
63 if (!SetConsoleMode(impl_->out, outMode)) return false; // pre-Win10 console: no VT support
64 SetConsoleMode(impl_->in, ENABLE_WINDOW_INPUT | ENABLE_PROCESSED_INPUT);
65
66 impl_->savedCp = GetConsoleOutputCP();
67 SetConsoleOutputCP(CP_UTF8);
68 SetConsoleCtrlHandler(onCtrl, TRUE);
69
70 impl_->active = true;
71 write("\x1b[?1049h\x1b[?25l\x1b[2J");
72 colorMode_ = ColorMode::TrueColor; // Windows Terminal and Win10 1703+ conhost
73 return true;
74}

90std::vector<Action> Terminal::poll(int timeoutMs) {
91 std::vector<Action> actions;
92 if (WaitForSingleObject(impl_->in, static_cast<DWORD>(timeoutMs)) != WAIT_OBJECT_0) return actions;
93
94 INPUT_RECORD rec[32];
95 DWORD n = 0;
96 if (!ReadConsoleInputW(impl_->in, rec, 32, &n)) return actions;
97 for (DWORD i = 0; i < n; ++i) {
98 if (rec[i].EventType == WINDOW_BUFFER_SIZE_EVENT) {
99 gResized = true;
100 continue;
101 }
102 if (rec[i].EventType != KEY_EVENT || !rec[i].Event.KeyEvent.bKeyDown) continue;
103 const KEY_EVENT_RECORD& k = rec[i].Event.KeyEvent;
104 switch (k.wVirtualKeyCode) {
105 case VK_UP: actions.push_back(Action::Up); continue;
106 case VK_DOWN: actions.push_back(Action::Down); continue;
107 case VK_LEFT: actions.push_back(Action::Left); continue;
108 case VK_RIGHT: actions.push_back(Action::Right); continue;
109 case VK_RETURN: actions.push_back(Action::Select); continue;
110 case VK_ESCAPE: actions.push_back(Action::Back); continue;
111 case VK_SPACE: actions.push_back(Action::Pause); continue;
112 default: break;
113 }
114 switch (k.uChar.UnicodeChar) {
115 case L'q': case L'Q': actions.push_back(Action::Quit); break;
116 case L'p': case L'P': actions.push_back(Action::Pause); break;
117 case L'f': case L'F': actions.push_back(Action::Interact); break;
118 case L'i': case L'I': actions.push_back(Action::Info); break;
119 case L'r': case L'R': actions.push_back(Action::Redraw); break;
120 case L'm': case L'M': actions.push_back(Action::Menu); break;
121 case L'c': case L'C': actions.push_back(Action::Style); break;
122 case L't': case L'T': actions.push_back(Action::Time); break;
123 case L'x': actions.push_back(Action::Rare); break;
124 case L'X': actions.push_back(Action::RareOther); break;
125 case L'a': case L'A': actions.push_back(Action::Audio); break;
126 default: break;
127 }
128 }
129 return actions;
130}

132void Terminal::write(const std::string& bytes) {
133 DWORD written = 0;
134 size_t off = 0;
135 while (off < bytes.size()) {
136 if (!WriteFile(impl_->out, bytes.data() + off, static_cast<DWORD>(bytes.size() - off), &written, nullptr))
137 return;
138 off += written;
139 }
140}
