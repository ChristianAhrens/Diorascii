42Rgb maskColor(char32_t m, const Entity& e) {
43 if (m >= U'1' && m <= U'9') return e.pal[m - U'1'];
44 Rgb base;
45 switch (m | 0x20) { // fold to lower case
46 case U'w': base = {240, 240, 240}; break;
47 case U'k': base = {70, 70, 85}; break;
48 case U'r': base = {225, 70, 70}; break;
49 case U'g': base = {80, 195, 100}; break;
50 case U'b': base = {80, 120, 235}; break;
51 case U'c': base = {90, 215, 225}; break;
52 case U'y': base = {240, 215, 80}; break;
53 case U'm': base = {205, 100, 205}; break;
54 case U'o': base = {245, 155, 60}; break;
55 case U't': base = {240, 205, 150}; break;

76void blit(Frame& out, const Entity& e, Style style, const Light& l, int maxRow, const std::vector<int>* hideBelow = nullptr) {
77 const SpriteFrame& f = e.sprite->frame(e.mirrored, e.frame);
78 const int ox = static_cast<int>(std::floor(e.x));
79 const int oy = static_cast<int>(std::floor(e.y));
80 auto colorOf = [&](char32_t m) {
81 const Rgb c = maskColor(m, e);
82 return e.glow ? c : shade(c, l);
83 };
84 for (int r = 0; r < e.height(); ++r) {
85 for (int c = 0; c < e.width(); ++c) {
86 const int px = ox + c, py = oy + r;
87 if (!out.inBounds(px, py) || py >= maxRow) continue;
88 if (hideBelow && py >= (*hideBelow)[static_cast<size_t>(px)]) continue; // behind the mountain
89 const char32_t ch = f.art[r][c], m = f.mask[r][c];
90 Cell& cell = out.at(px, py);
91 if (ch == U' ') {
92 if (m == U' ') continue;
93 cell.ch = U' ';
94 if (m != U'_') { // solid block: keeps its color unless the style is monochrome
95 cell.bg = colorOf(m);
96 cell.defaultBg = style == Style::Mono;
97 }
98 } else {
99 cell.ch = ch;
100 cell.fg = colorOf(m);
101 cell.defaultFg = style == Style::Mono;
102 if (e.backing > 0 && style == Style::Full) cell.bg = scale(cell.bg, 1 - e.backing);
103 if (e.contrast && style == Style::Full) { // dark strokes on light backgrounds
104 const double lum = (0.2126 * cell.bg.r + 0.7152 * cell.bg.g + 0.0722 * cell.bg.b) / 255.0;
105 if (lum > 0.5) cell.fg = scale(cell.fg, 0.32);
106 }
107 }
108 }
109 }

135Cell riverCell(int x, int y, double t, double time, double flow, Rgb top, Rgb deep) {
136 Cell c;
137 c.bg = lerp(top, deep, t);
138 const double f = time * flow;
139 const double v = std::sin(x * 0.6 - f + y * 1.7) + 0.5 * std::sin(x * 0.23 - f * 0.6);
140 c.ch = v > 0.8 ? U'~' : v > 0.2 ? U'-' : (hash2(x, y) % 9 == 0 ? U'.' : U' ');
141 c.fg = scale(c.bg, 1.9);
142 return c;

217void hop(Entity& e, World& w, double dt) {
218 constexpr double kAirborne = 0.4; // share of the cycle spent in the air
219 const double cycle = std::fmod(e.age + e.phase, e.hopPeriod) / e.hopPeriod;
220 double lift = 0;
221 if (cycle < kAirborne) {
222 e.x += e.vx * dt / kAirborne; // all the travel happens in the air, so the average speed is vx
223 lift = e.hopHeight * std::sin(3.14159265358979 * cycle / kAirborne);
224 }
225 if ((e.vx > 0 && e.x > w.width()) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
226 e.y = e.anchorY + w.craterRows(static_cast<int>(std::floor(e.x + e.width() / 2.0))) - lift;

295void catastrophe(Entity& e, World& w, double dt) {
296 // Where it strikes: the middle (sweep), or the front, the side it moves towards (devour).
297 auto point = [&] { return e.strikeMode == Entity::StrikeMode::Devour ? (e.vx > 0 ? e.x + e.width() : e.x) : e.x + e.width() / 2.0; };
298 if (e.strikeMode != Entity::StrikeMode::Centre && !e.hasLastPoint) {
299 e.lastPoint = point();
300 e.hasLastPoint = true;
301 }
302 e.x += e.vx * dt;
303 if ((e.vx > 0 && e.x > w.width()) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
304 if (e.strikeMode != Entity::StrikeMode::Centre) { // takes each victim as it passes it
305 const double now = point();
306 const bool devour = e.strikeMode == Entity::StrikeMode::Devour; // it only takes what is in its way
307 if (devour && e.strikeReach > 0) w.lash(e);
308 if (e.strikeAround) return; // it takes only what its tentacles reach
309 // A cell of leeway each way, so that a victim walking towards it is not stepped over.
310 const size_t taken = w.wipeOut(e.wipes, e.strikeEffect, &e, std::min(e.lastPoint, now) - 1, std::max(e.lastPoint, now) + 1,
311 devour);
312 if (devour) w.noteCrushed(taken, now, e.y + e.height() / 2.0, e.z);
313 e.lastPoint = now;
314 if (taken > 0) {
315 if (e.flashTime > 0) w.flash(e.flashTime, now);
316 e.struck = true;
317 }
318 return;
319 }
320 if (e.struck) return;
321 const double middle = e.x + e.width() / 2.0;
322 const double progress = e.vx > 0 ? middle / w.width() : 1.0 - middle / w.width();
323 if (progress < e.strikeAt) return;
324 e.struck = true;
325 if (e.stormTime > 0) w.startRain(e.stormTime);
326 if (!e.strikeSprite.empty()) { // a bolt from the sky to the ground right below it
327 if (Entity* bolt = w.spawnEffect(e.strikeSprite, middle, 0, e.z - 1, 0, 0)) {
328 bolt->x = middle - bolt->width() / 2.0;
329 bolt->y = w.seabedY() - bolt->height();
330 }
331 }
332 w.wipeOut(e.wipes, e.strikeEffect, &e);
333 if (e.flashTime > 0) w.flash(e.flashTime, middle);
334}

383void ledge(Entity& e, World& w, double dt) {
384 constexpr double kAirborne = 0.4; // share of a hop cycle spent in the air
385 const double before = e.x;
386 double lift = 0;
387 if (e.hops) {
388 const double cycle = std::fmod(e.age + e.phase, e.hopPeriod) / e.hopPeriod;
389 if (cycle < kAirborne) {
390 e.x += e.vx * dt / kAirborne; // all the travel happens in the air, so the average speed is vx
391 lift = e.hopHeight * std::sin(3.14159265358979 * cycle / kAirborne);
392 }
393 } else {
394 e.x += e.vx * dt;
395 }
396 if (e.laneTarget >= 0) {
397 const double d = e.laneTarget - e.anchorY, step = 1.6 * dt;
398 if (std::fabs(d) <= step) { e.anchorY = e.laneTarget; e.laneTarget = -1; }
399 else e.anchorY += d > 0 ? step : -step;
400 } else if (w.rng().chance(0.06 * dt)) {
401 w.pickLane(e);
402 }
403 if (!w.fits(e, e.x, e.anchorY)) { // the end of its ground: turn round and look for another ledge
404 e.x = before;
405 e.vx = -e.vx;
406 e.mirrored = !e.mirrored;
407 e.laneTarget = -1;
408 if (!w.fits(e, e.x, e.anchorY)) w.placeInZone(e, true, false, e.movingRight());
409 }
410 if ((e.vx > 0 && e.x > w.width()) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
411 e.y = e.anchorY - lift;
412 e.z = w.depthFor(e.anchorY + e.height());
413}

479void animate(Entity& e, double dt) {
480 const size_t n = e.sprite->frameCount();
481 if (e.fps <= 0 || n < 2) return;
482 const double period = 1.0 / e.fps;
483 e.clock += dt;
484 while (e.clock >= period) {
485 e.clock -= period;
486 if (!e.loop && e.frame + 1 >= n) {
487 if (e.dieAfterAnim) e.dead = true;
488 continue;
489 }
490 ++e.frame;
491 }
492 e.frame = e.loop ? e.frame % n : std::min(e.frame, n - 1);
493}

598void World::populate() {
599 for (const auto& pop : scene_->population) {
600 const bool perCells = pop.kind == SceneDef::Population::Kind::PerCells;
601 const bool perCols = pop.kind == SceneDef::Population::Kind::PerCols;
602 double want;
603 if (pop.sprite == "@seaweed") {
604 if (!hasMedium_) continue;
605 want = perCols ? w_ / pop.n : pop.n;
606 int n = static_cast<int>(want);
607 if (perCols && rng_.chance(want - n)) ++n;
608 for (int i = 0; i < n; ++i) spawnSeaweed();
609 continue;
610 }
611 auto sprite = pack_->find(pop.sprite);
612 if (!sprite) continue;
613 const std::string habitat = sprite->props().str("habitat", "medium");
614 if ((habitat == "medium" && !hasMedium_) || (habitat == "river" && riverRows_ == 0) ||
615 (habitat == "rail" && railY_ < 0)) {
616 continue;
617 }
618 want = perCells ? areaFor(habitat) / pop.n : perCols ? w_ / pop.n : pop.n;
619 int n = static_cast<int>(want);
620 if (pop.kind != SceneDef::Population::Kind::Count && rng_.chance(want - n)) ++n;
621 // A `herd` sprite is spawned in groups (n counts groups, not animals).
622 const bool herded = sprite->props().has("herd");
623 for (int i = 0; i < n; ++i) {
624 if (herded) spawnHerd(sprite);
625 else spawnCreature(sprite, true);
626 }
627 }
628 decorAnchor_ = {};
629 for (const auto& d : scene_->decor) spawnDecor(d);
630 flushIncoming();
631 if (scene_->city) placeCrossings(); // after the station: one of them lies at its foot
632}

634double World::habitatY(const Props& p, int spriteHeight) {
635 if (p.has("row")) { // pinned to a row; negative counts from the bottom
636 const double row = p.num("row", 0);
637 return row >= 0 ? row : h_ + row;
638 }
639 const std::string habitat = p.str("habitat", "medium");
640 if (habitat == "ground") return seabedY_ - spriteHeight;
641 if (habitat == "rail") return railY_ - spriteHeight; // on the elevated railway
642 if (habitat == "river") return rng_.rangeInt(riverTopY_, std::max(h_ - spriteHeight, riverTopY_));
643 if (habitat == "surface") return surfaceY_ - p.num("anchor-row", spriteHeight - 1);
644 if (habitat == "sky") {
645 if (p.has("altitude")) { // hovering within a few rows above the ground
646 return std::max(surfaceY_ - spriteHeight - rng_.rangeInt(0, static_cast<int>(p.num("altitude", 0))), 0);
647 }
648 return rng_.rangeInt(0, std::max(surfaceY_ - spriteHeight, 0));
649 }
650 const int lo = surfaceY_ + 2;
651 return rng_.rangeInt(lo, std::max(seabedY_ - spriteHeight, lo));
652}

654void World::spawnHerd(const std::shared_ptr<const Sprite>& sprite) {
655 const Props& p = sprite->props();
656 const auto [lo, hi] = p.range("herd", {2, 4});
657 const int members = rng_.rangeInt(static_cast<int>(lo), static_cast<int>(hi));
658 const double spacing = p.num("herd-spacing", 4.0);

662 where.hasX = true;
663 for (int i = 0; i < members; ++i) {

665 where.speedMul = rng_.range(0.92, 1.08);
666 spawnCreature(sprite, true, false, &where);
667 }

670void World::spawnCreature(const std::shared_ptr<const Sprite>& sprite, bool initial, bool event,
671 const Placement* where) {
672 const Props& p = sprite->props();
673 if (p.str("habitat", "medium") == "rail" && railY_ < 0) return; // no railway on this screen
674 // Rush-hour traffic: a vehicle only joins the street with the chance the traffic level gives, and
675 // otherwise tries again a few seconds later, so the street fills and empties with the day.
676 if (!event && p.str("rush", "no") == "yes" && scene_->dayNight && !rng_.chance(trafficLevel())) {
677 restock_.push_back({rng_.range(2.0, 7.0), [sprite](World& w) { w.spawnCreature(sprite, false); }});
678 return;
679 }
680 auto e = std::make_unique<Entity>();
681 e->sprite = sprite;
682 applyProps(*e, p, rng_);
683 e->isEvent = event;
684
685 const auto [vmin, vmax] = p.range("speed", {4, 8});
686 const double speed = rng_.range(vmin, vmax) * (where ? where->speedMul : 1.0);
687 const bool coin = rng_.chance(0.5);
688 const bool right = (where && where->dir != 0) ? where->dir > 0 : coin;
689 e->vx = right ? speed : -speed;
690 e->vy = std::fabs(e->vx) * p.num("slope", 0); // diagonal flyers move downwards
691 e->mirrored = right != e->facesRight;
692
693 const auto [zmin, zmax] = p.range("depth", {5, 15});
694 e->z = rng_.rangeInt(static_cast<int>(zmin), static_cast<int>(zmax));
695 e->y = e->anchorY = habitatY(p, e->height());
696
697 // Ground sprites with `lanes: N` stand N rows deep on the bank; lanes further
698 // forward are drawn in front, so crowds of trees and animals get some depth.
699 const int lanes = static_cast<int>(p.num("lanes", 1));
700 if (lanes > 1 && p.str("habitat", "medium") == "ground") {
701 const int bank = (riverRows_ > 0 ? riverTopY_ : h_) - seabedY_;
702 const int usable = std::min(lanes, std::max(bank, 1));
703 const int lane = rng_.rangeInt(0, usable - 1);
704 e->y += lane;
705 e->anchorY = e->y;
706 if (usable > 1) e->z = static_cast<int>(zmax) - lane * static_cast<int>(zmax - zmin) / (usable - 1);
707 }
708
709 // `feet-rows: a..b` puts a ground sprite's feet on those ground rows (0 = the first; negative counts
710 // from the bottom, -1 is the last), and with `traffic: yes` the vehicles keep to their side of the
711 // road: the lower rows of the range for what moves right, the upper rows for what moves left.
712 if (p.has("feet-rows") && p.str("habitat", "medium") == "ground") {
713 const int rows = std::max(h_ - seabedY_, 1);
714 const auto [flo, fhi] = p.range("feet-rows", {0, 0});
715 auto resolve = [&](double v) { return std::clamp(static_cast<int>(v) < 0 ? rows + static_cast<int>(v) : static_cast<int>(v), 0, rows - 1); };
716 int lo = resolve(flo), hi = resolve(fhi);
717 if (lo > hi) std::swap(lo, hi);
718 if (p.str("traffic", "no") == "yes" && hi > lo) {
719 const int mid = (lo + hi + 1) / 2;
720 if (right) lo = mid;
721 else hi = mid - 1;
722 }
723 const int feet = rng_.rangeInt(lo, hi);
724 e->y = e->anchorY = seabedY_ - e->height() + feet + 1;
725 e->z = static_cast<int>(zmax) - feet * static_cast<int>(zmax - zmin) / std::max(rows - 1, 1);
726 }
727
728 // Members of a herd that left come back from the same side, a little apart, still facing the same way.
729 const double rejoin = (where && !initial) ? rng_.range(0, 12) : 0.0;
730 e->x = (where && where->hasX) ? where->x
731 : initial ? rng_.range(-e->width(), w_)
732 : (right ? -e->width() - rejoin : w_ + rejoin);
733 if (e->vy > 0 && !initial) e->y = e->anchorY = -e->height() + rng_.range(0, h_ / 3.0);
734 if (sprite->frameCount() > 1 && e->fps > 0) {
735 e->frame = rng_.rangeInt(0, static_cast<int>(sprite->frameCount()) - 1);
736 }
737
738 if (e->zone != Zone::None && !ridge_.empty()) placeInZone(*e, where && where->hasX, !initial, right);
739
740 // `sleeps: 0.7` = about 70% of these animals sleep at night, each from its own sun height.
741 const double sleeps = std::clamp(p.num("sleeps", 0.0), 0.0, 1.0);
742 if (sleeps > 0 && rng_.chance(sleeps)) e->sleepBelow = rng_.range(0.05, 0.9);
743
744 if (!event) {
745 const bool herded = p.has("herd");
746 const int dir = right ? 1 : -1;
747 e->onDeath = [sprite, herded, dir](World& w) {
748 Placement again;
749 again.dir = dir;
750 if (herded) w.spawnCreature(sprite, false, false, &again);
751 else w.spawnCreature(sprite, false);
752 };
753 }
754 spawn(std::move(e));
755}

768void World::placeDecor(const SceneDef::Decor& d, const DecorAnchor* fixed) {
769 if (d.chance < 1.0 && !rng_.chance(d.chance)) return;
770 // One of the sprites separated by `|` (trees of different sizes).
771 std::vector<std::string> names;
772 for (size_t from = 0; from <= d.sprite.size();) {
773 const size_t bar = std::min(d.sprite.find('|', from), d.sprite.size());
774 names.push_back(d.sprite.substr(from, bar - from));
775 from = bar + 1;
776 }
777 auto sprite = pack_->find(names[static_cast<size_t>(rng_.rangeInt(0, static_cast<int>(names.size()) - 1))]);
778 if (!sprite) return;
779 const Props& p = sprite->props();
780 const bool onRail = p.str("habitat", "ground") == "rail";
781 if (onRail && railY_ < 0) return; // no railway on this screen
782 auto e = std::make_unique<Entity>();
783 e->sprite = sprite;
784 applyProps(*e, p, rng_);
785 e->z = static_cast<int>(p.range("depth", {20, 20}).first);
786 e->decor = true;
787
788 // Decorations in the far distance (the pyramids, z >= 30) are a backdrop: things may stand in front of them.
789 constexpr int kBackdropZ = 30;
790 const bool onSidewalk = !avenues_.empty() && e->grounded;
791 // Whether a column is free: not in a side street (a station's foot may not be either) and, for
792 // things that stand on the ground, not on top of another decoration of any size.
793 auto blocked = [&](double x) {
794 if (!avenues_.empty() && (onSidewalk || onRail)) {
795 for (const Avenue& a : avenues_) {
796 if (x + e->width() > a.cx - a.halfW - 3 && x < a.cx + a.halfW + 3) return true;
797 }
798 }
799 if (e->grounded && e->height() > 3) {
800 auto overlaps = [&](const std::unique_ptr<Entity>& o) {
801 return !o->dead && o->decor && o->grounded && o->height() > 3 && (o->z < kBackdropZ || e->z >= kBackdropZ) && x + e->width() > o->x - 1 && x < o->x + o->width() + 1;
802 };
803 if (std::any_of(entities_.begin(), entities_.end(), overlaps) || std::any_of(incoming_.begin(), incoming_.end(), overlaps)) return true;
804 }
805 return false;
806 };
807 const DecorAnchor anchor = fixed ? *fixed : decorAnchor_;
808 const int span = std::max(w_ - e->width(), 0);
809 auto pick = [&]() {
810 double x;
811 if (d.align == "left") x = d.margin;
812 else if (d.align == "right") x = w_ - e->width() - d.margin;
813 else if (d.align == "center") x = (w_ - e->width()) / 2 + d.margin;
814 else if (d.align == "from-left" && anchor.valid) x = anchor.x + d.margin;
815 else if (d.align == "from-right" && anchor.valid) x = anchor.x + anchor.width + d.margin;
816 else if (d.align == "from-center" && anchor.valid) x = anchor.x + (anchor.width - e->width()) / 2.0 + d.margin;
817 else if (d.align.rfind("from-", 0) == 0) x = (w_ - e->width()) / 2 + d.margin; // no anchor: the middle
818 else x = rng_.rangeInt(0, span);
819 if (d.jitter > 0) x += rng_.rangeInt(-d.jitter, d.jitter);
820 return std::clamp(x, 0.0, static_cast<double>(span));
821 };
822 // Several tries: a thing that finds no free place is left out (the station is the exception, it
823 // stays and the avenues are ignored once nothing else works).
824 bool placed = false;
825 for (int attempt = 0; attempt < 24 && !placed; ++attempt) {
826 e->x = pick();
827 placed = !blocked(e->x);
828 }
829 if (!placed && !onRail) return;
830 if (!placed) e->x = pick();
831
832 if (d.hasY) e->y = d.y >= 0 ? d.y : h_ + d.y;
833 else if (p.str("habitat", "ground") == "sky") e->y = 1;
834 else e->y = habitatY(p, e->height());
835 e->anchorY = e->y;
836 if (e->zone != Zone::None && d.align == "random" && !ridge_.empty()) placeInZone(*e, false, false, true);
837 if (d.anchor) decorAnchor_ = {true, e->x, e->width()};
838 // Planted again after a catastrophe, where the thing it is aligned to stood.
839 e->restock = [d, anchor](World& w) { w.placeDecor(d, &anchor); };
840 spawn(std::move(e));
841}

861void World::spawnSeaweed() {
862 const int maxH = std::max(seabedY_ - surfaceY_ - 2, 2);
863 const int h = std::min(rng_.rangeInt(3, 7), maxH);
864 constexpr int kFrames = 6;
865 constexpr int kWidth = 5;
866 const double phase0 = rng_.range(0, 6.28);
867
868 std::vector<SpriteFrame> frames;
869 for (int k = 0; k < kFrames; ++k) {
870 std::vector<int> col(h);
871 for (int i = 0; i < h; ++i) { // i = 0 is the tip
872 const double amp = 1.6 * (h - 1 - i) / h;
873 col[i] = 2 + static_cast<int>(std::lround(amp * std::sin(phase0 + k * 1.047 + i * 0.8)));
874 }
875 SpriteFrame f;
876 for (int i = 0; i < h; ++i) {
877 std::u32string row(kWidth, U' ');
878 const int below = i + 1 < h ? col[i + 1] : col[i];
879 const int d = col[i] - below;
880 row[col[i]] = d > 0 ? U'/' : d < 0 ? U'\\' : (i % 2 ? U'(' : U')');
881 f.art.push_back(std::move(row));
882 }
883 frames.push_back(std::move(f));
884 }
885
886 auto e = std::make_unique<Entity>();
887 e->sprite = Sprite::fromFrames("seaweed", std::move(frames));
888 e->x = rng_.rangeInt(0, std::max(w_ - kWidth, 0));
889 e->y = seabedY_ - h;
890 e->z = 21;
891 e->fps = rng_.range(2.0, 3.0);
892 e->frame = rng_.rangeInt(0, kFrames - 1);
893 e->fg = lerp({40, 150, 70}, {150, 170, 60}, rng_.uniform());
894 e->ttl = rng_.range(8 * 60, 12 * 60); // seaweed regrows every 8-12 minutes
895 e->onDeath = [](World& w) { w.spawnSeaweed(); };
896 spawn(std::move(e));
897}

914 for (const auto& ev : scene_->rare) total += ev.weight;
915 if (total <= 0) return;
916 double roll = rng_.range(0, total);

969size_t World::wipeOut(const std::string& wipes, const std::string& effect, const Entity* except, double from,
970 double to, bool groundedOnly) {
971 const size_t before = killed_;
972 const bool all = wipes == "life";
973 const std::vector<std::string> names = all ? std::vector<std::string>{} : splitWs(wipes);
974 auto take = [&](Entity& o) {
975 if (&o == except || o.isRare) return;
976 const double middle = o.x + o.width() / 2.0;
977 if (middle < from || middle > to) return;
978 if (groundedOnly && !o.grounded) return;
979 const bool victim = all ? o.life : std::find(names.begin(), names.end(), o.sprite->name()) != names.end();
980 if (victim) claim(o, effect);
981 };
982 for (auto& other : entities_) take(*other);
983 // Entities spawned during this very step have not joined the world yet, but are just as dead.
984 // (spawnEffect above adds to the same list, and effects are never victims.)
985 for (size_t i = 0; i < incoming_.size(); ++i) take(*incoming_[i]);
986 return killed_ - before;
987}

996bool World::lash(const Entity& a) {
997 if (time_ < nextLash_ || a.strikeReach <= 0) return false;
998 const auto out = std::count_if(tongues_.begin(), tongues_.end(), [&](const Tongue& t) { return t.attacker == a.id; });
999 if (out >= a.strikeLashes) return false;
1000 const bool all = a.wipes == "life";
1001 const std::vector<std::string> names = all ? std::vector<std::string>{} : splitWs(a.wipes);
1002 const int dir = a.vx > 0 ? 1 : -1;
1003 const double front = dir > 0 ? a.x + a.width() : a.x;
1004 const double baseY = a.y + a.mouthRow;
1005 const Entity* best = nullptr;
1006 double bestD = 0, bestX = 0;
1007 for (auto& other : entities_) {
1008 const Entity& o = *other;
1009 if (&o == &a || o.dead || o.isRare || isAsleep(o)) continue;
1010 if (!(all ? o.life : std::find(names.begin(), names.end(), o.sprite->name()) != names.end())) continue;
1011 // another tongue is already after it
1012 if (std::any_of(tongues_.begin(), tongues_.end(), [&](const Tongue& t) { return t.victim == o.id; })) continue;
1013 const double cx = o.x + o.width() / 2.0, cy = o.y + o.height() / 2.0;
1014 double d, baseX;
1015 if (a.strikeAround) { // from the underside of the body, straight above the victim if it can
1016 baseX = std::clamp(cx, a.x + 3.0, std::max(a.x + a.width() - 4.0, a.x + 3.0));
1017 d = std::sqrt((cx - baseX) * (cx - baseX) + 4.0 * (cy - baseY) * (cy - baseY));
1018 } else { // from the mouth, at walking things ahead of it
1019 if (!o.grounded) continue;
1020 baseX = dir > 0 ? a.x + a.width() - 1.0 : a.x;
1021 d = (cx - front) * dir;
1022 if (d < 1.5) continue; // what is at its very mouth is simply swallowed
1023 }
1024 if (d > a.strikeReach) continue;
1025 if (!best || d < bestD) { best = &o; bestD = d; bestX = baseX; }
1026 }
1027 if (!best) return false;
1028 Tongue t;
1029 t.attacker = a.id;
1030 t.victim = best->id;
1031 t.effect = a.strikeEffect;
1032 t.wipes = a.wipes;
1033 t.color = a.lashColor;
1034 if (a.lashShot) { // a shot is there and gone
1035 t.shot = true;
1036 t.reach = 0.1;
1037 t.dur = 0.3;
1038 ++shots_;
1039 emit(EventKind::Shot, bestX, baseY, a.z); // from the muzzle
1040 }
1041 t.ox = bestX - a.x;
1042 t.x0 = bestX;
1043 t.y0 = baseY;
1044 t.x1 = best->x + best->width() / 2.0;
1045 t.y1 = best->y + best->height() / 2.0;
1046 tongues_.push_back(t);
1047 nextLash_ = time_ + (a.lashShot ? 0.45 : a.strikeLashes > 1 ? 0.2 : 0.0); // tentacles lash one after another
1048 return true;
1049}

1061void World::stepTongues(double dt) {
1062 for (size_t i = tongues_.size(); i-- > 0;) {
1063 Tongue& t = tongues_[i];
1064 Entity* a = findEntity(t.attacker);
1065 if (!a || a->dead) { // the attacker has gone: the tongue goes with it
1066 tongues_.erase(tongues_.begin() + static_cast<std::ptrdiff_t>(i));
1067 continue;
1068 }
1069 t.age += dt;
1070 t.x0 = a->x + t.ox;
1071 t.y0 = a->y + a->mouthRow;
1072 Entity* v = findEntity(t.victim);
1073 const bool alive = v && !v->dead;
1074 if (alive && !t.hit) { // the tip follows the victim
1075 t.x1 = v->x + v->width() / 2.0;
1076 t.y1 = v->y + v->height() / 2.0;
1077 }
1078 if (!t.hit && t.age >= t.reach) {
1079 t.hit = true;
1080 if (alive && !t.harmless) {
1081 claim(*v, t.effect);
1082 if (!t.shot) noteCrushed(1, v->x + v->width() / 2.0, v->y + v->height() / 2.0, v->z);
1083 }
1084 }
1085 if (t.age >= t.dur) {
1086 if (a->strikeLashes == 1) nextLash_ = time_ + 0.25; // a single tongue rests a moment
1087 tongues_.erase(tongues_.begin() + static_cast<std::ptrdiff_t>(i));
1088 }

1092void World::renderTongues(Frame& out, Style style, const Light& light) const {
1093 for (const Tongue& t : tongues_) {
1094 if (t.fire) { // a stream of flames: it shoots out, holds, and dies away
1095 const double reach = std::max(t.reach, 0.01);
1096 const double p = t.age < reach ? t.age / reach : t.age > t.dur - 0.3 ? std::max(0.0, (t.dur - t.age) / 0.3) : 1.0;
1097 const double dx = t.x1 - t.x0, dy = t.y1 - t.y0;
1098 const double len = std::sqrt(dx * dx + dy * dy * 4.0);
1099 const int samples = std::max(3, static_cast<int>(len * 2));
1100 static const char32_t kFlames[] = {U'*', U'#', U'%', U'&', U'@', U'*'};
1101 for (int s = 0; s <= samples; ++s) {
1102 const double u = static_cast<double>(s) / samples;
1103 if (u > p) break;
1104 const double wobble = 0.9 * std::sin(u * len * 0.7 - t.age * 30.0) * u;
1105 const int cx = static_cast<int>(std::floor(t.x0 + dx * u));
1106 const int cy = static_cast<int>(std::floor(t.y0 + dy * u + wobble));
1107 const int spread = 1 + static_cast<int>(u * 2.2); // the stream widens as it goes
1108 for (int off = -spread; off <= spread; ++off) {
1109 const uint32_t h = hash2(cx * 7 + off, static_cast<int>(t.age * 24.0) + s);
1110 if (off != 0 && h % 5 == 0) continue; // ragged edges
1111 if (!out.inBounds(cx, cy + off)) continue;
1112 Cell& cell = out.at(cx, cy + off);
1113 const double edge = static_cast<double>(std::abs(off)) / (spread + 1);
1114 cell.ch = kFlames[h % 6];
1115 cell.fg = shade(lerp(lerp(t.color, Rgb{255, 255, 255}, 0.65), t.color, edge), light);
1116 cell.bg = style == Style::Full ? lerp(cell.bg, scale(t.color, 0.55), 0.6 * (1.0 - edge)) : cell.bg;
1117 cell.bold = true;
1118 cell.defaultFg = style == Style::Mono;
1119 }
1120 }
1121 continue;
1122 }
1123 // Out to the victim in the first part, back in again once it is crushed.
1124 const double p = t.hit ? std::max(0.0, 1.0 - (t.age - t.reach) / (t.dur - t.reach)) : std::min(1.0, t.age / t.reach);
1125 const double dx = t.x1 - t.x0, dy = t.y1 - t.y0;
1126 const double len = std::sqrt(dx * dx + dy * dy * 4.0); // cells are about twice as tall as wide
1127 const int samples = std::max(2, static_cast<int>(len * 2));
1128 // The glyph follows the direction of the tongue.
1129 const double ady = std::fabs(dy) * 2.0, adx = std::fabs(dx);
1130 const char32_t glyph = ady < 0.45 * adx ? U'=' : ady > 2.2 * adx ? U'|' : (dx * dy > 0 ? U'\\' : U'/');
1131 for (int s = 0; s <= samples; ++s) {
1132 const double u = static_cast<double>(s) / samples;
1133 if (u > p) break;
1134 const double wobble = t.shot ? 0.0 : 0.8 * std::sin(u * len * 0.8 - t.age * 28.0) * u * (1.0 - p * 0.5); // it lashes about
1135 const int cx = static_cast<int>(std::floor(t.x0 + dx * u));
1136 const int cy = static_cast<int>(std::floor(t.y0 + dy * u + wobble));
1137 if (!out.inBounds(cx, cy)) continue;
1138 Cell& cell = out.at(cx, cy);
1139 const bool tip = u + 1.0 / samples > p;
1140 cell.ch = t.shot ? (tip ? U'*' : U'.') : tip ? U'@' : glyph;
1141 cell.fg = shade(tip ? scale(t.color, 0.8) : t.color, light);
1142 cell.defaultFg = style == Style::Mono;
1143 }
1144 }
1145}

1208 stormDir_ * speed, 0);
1209 flushIncoming();
1210 return 1;
1211 }

1218 spawnCreature(sprite, false, true); // a visitor: it leaves again and is not replaced

1221 }
1222 return 0;

1482void World::step(double dt) {
1483 time_ += dt;
1484 for (double& c : crater_) c = std::max(0.0, c - dt / scene_->dustSettle); // dust fills craters
1485 for (float& c : scar_) c = std::max(0.0f, c - static_cast<float>(dt / 40.0)); // the swath of an avalanche fades
1486 if (rainLeft_ > 0) {
1487 rainLeft_ = std::max(0.0, rainLeft_ - dt);
1488 spawnWeather(dt);
1489 }
1490 if (scene_->dayNight && timeMode_ == TimeMode::Cycle) {
1491 phase_ = std::fmod(phase_ + dt / scene_->dayLength, 1.0);
1492 daysElapsed_ += dt / scene_->dayLength;
1493 }
1494 const double cloudTarget =
1495 (rainLeft_ > 0 || countTag("tornado") > 0 || countTag("gloom") > 0) ? 1.0 : 0.0;
1496 flashElapsed_ += dt;
1497 overcast_ += std::clamp(cloudTarget - overcast_, -0.25 * dt, 0.4 * dt);
1498 const double sun = scene_->dayNight ? light().sun : 1.0;
1499 bool eventAlive = false, rareAlive = false;
1500 for (auto& e : entities_) {
1501 if (e->dead) continue;
1502 eventAlive |= e->isEvent;
1503 rareAlive |= e->isRare;
1504 if (e->sleepBelow > 0 && sun < e->sleepBelow) continue; // asleep: frozen where it is
1505 if (e->flung) { // thrown by a tornado: flies on an arc, lands dazed and walks on afterwards
1506 e->age += dt;
1507 e->vy += scene_->gravity * dt;
1508 e->x += e->vx * dt;
1509 e->y += e->vy * dt;
1510 if (e->vy > 0 && e->y >= e->anchorY) {
1511 e->y = e->anchorY;
1512 e->vy = 0;
1513 e->vx = e->flungVx;
1514 e->flung = false;
1515 e->pauseLeft = 1.5;
1516 }
1517 continue;
1518 }
1519 e->age += dt;
1520 if (e->fleeing) { // in a panic: straight off the screen, and back (as a replacement) a while later
1521 e->pauseLeft = 0;
1522 e->x += e->vx * dt;
1523 animate(*e, dt);
1524 if ((e->vx > 0 && e->x > w_) || (e->vx < 0 && e->x + e->width() < 0)) {
1525 e->dead = true;
1526 if (e->onDeath) restock_.push_back({rng_.range(10.0, 40.0), std::move(e->onDeath)});
1527 e->onDeath = nullptr;
1528 }
1529 continue;
1530 }
1531 if (e->pauseLeft > 0) { // busy (eating): stays frozen where it is
1532 e->pauseLeft = std::max(0.0, e->pauseLeft - dt);
1533 continue;
1534 }
1535 // A cordoned-off fire: traffic and people stop short of the site (those already in it clear off).
1536 if (cordon_ && e->grounded && e->vx != 0 && e->driveState == 0 && e->sprite->name() != "firetruck") {
1537 // Stops short of the barrier with its whole body: it may not reach into the site.
1538 const double lo = cordonX_ - cordonHalf_, hi = cordonX_ + cordonHalf_;
1539 const double next = e->x + e->vx * dt;
1540 const bool inNow = e->x < hi && e->x + e->width() > lo, inNext = next < hi && next + e->width() > lo;
1541 if (!inNow && inNext) {
1542 e->pauseLeft = std::max(e->pauseLeft, 2 * dt); // waits at the edge until it is lifted
1543 continue;
1544 }
1545 }
1546 animate(*e, dt);
1547 for (BehaviorFn b : e->behaviors) b(*e, *this, dt);
1548 if (e->ttl > 0 && e->age > e->ttl) e->dead = true;
1549 }
1550
1551 stepTongues(dt);
1552 stepToxin(dt);
1553 stepFlood(dt);
1554 if (!avenues_.empty()) stepCity(dt);
1555
1556 if (!scene_->events.empty() && !eventAlive) {
1557 if (eventTimer_ < 0) eventTimer_ = rng_.range(scene_->eventInterval.first, scene_->eventInterval.second);
1558 eventTimer_ -= dt;
1559 if (eventTimer_ <= 0) {
1560 spawnEvent();
1561 eventTimer_ = -1;
1562 }
1563 }
1564
1565 if (!scene_->rare.empty() && !rareAlive) {
1566 if (rareTimer_ < 0) rareTimer_ = rng_.range(scene_->rareInterval.first, scene_->rareInterval.second);
1567 rareTimer_ -= dt;
1568 if (rareTimer_ <= 0) {
1569 spawnRare();
1570 rareTimer_ = -1;
1571 }
1572 }
1573
1574 std::vector<std::function<void(World&)>> callbacks;
1575 for (auto& r : restock_) r.left -= dt; // what a catastrophe wiped out comes back, one by one
1576 for (size_t i = restock_.size(); i-- > 0;) {
1577 if (restock_[i].left > 0) continue;
1578 callbacks.push_back(std::move(restock_[i].fn));
1579 restock_.erase(restock_.begin() + static_cast<std::ptrdiff_t>(i));
1580 }
1581 for (auto& e : entities_) {
1582 if (e->dead && e->onDeath) callbacks.push_back(std::move(e->onDeath));
1583 }
1584 std::erase_if(entities_, [](const auto& e) { return e->dead; });
1585 for (auto& cb : callbacks) cb(*this);
1586 flushIncoming();
1587}

1591void World::buildTerrain() {
1592 const SceneDef& sc = *scene_;
1593 const double H = h_;
1594 auto rowAt = [&](double climb) { return static_cast<int>(std::lround(H - 1 - climb * H)); };
1595 const int floorRow = std::max(seabedY_ - 1, 0); // lowest row the skyline may have
1596 const int minRow = std::clamp(rowAt(sc.peakHeight), 0, floorRow);
1597 const int baseRow = std::clamp(rowAt(sc.treeline + 0.05), minRow, floorRow);
1598
1599 // Peaks of different heights with their own slopes on either side; the skyline follows the
1600 // highest of them and falls to the foothills between them.
1601 struct Peak { double x, row, left, right; };
1602 const int n = std::clamp(w_ / 30, 1, 7);
1603 const int tallest = rng_.rangeInt(0, n - 1);
1604 std::vector<Peak> peaks;
1605 for (int i = 0; i < n; ++i) {
1606 const double span = static_cast<double>(w_) / n;
1607 peaks.push_back({(i + 0.5) * span + rng_.range(-0.3, 0.3) * span,
1608 i == tallest ? minRow : minRow + (baseRow - minRow) * rng_.range(0.0, 0.65),
1609 rng_.range(0.32, 0.62), rng_.range(0.32, 0.62)});
1610 }
1611 const double ph[4] = {rng_.range(0, 6.28), rng_.range(0, 6.28), rng_.range(0, 6.28), rng_.range(0, 6.28)};
1612 const double amp = std::max(0.8, H * 0.03);
1613
1614 const size_t cols = static_cast<size_t>(w_);
1615 ridge_.assign(cols, baseRow);
1616 treeRow_.assign(cols, 0);
1617 snowRow_.assign(cols, 0);
1618 side_.assign(cols, 1);
1619 horizonY_ = 0;
1620 for (int x = 0; x < w_; ++x) {
1621 double v = baseRow + 1.2 * std::sin(x * 0.07 + ph[0]);
1622 const Peak* nearest = &peaks[0];
1623 for (const Peak& p : peaks) {
1624 const double dx = x - p.x;
1625 v = std::min(v, p.row + std::fabs(dx) * (dx < 0 ? p.left : p.right));
1626 if (std::fabs(dx) < std::fabs(x - nearest->x)) nearest = &p;
1627 }
1628 int r = static_cast<int>(std::lround(v));
1629 if (r < baseRow - 1) r += static_cast<int>(hash2(x / 2, 91) % 3) - 1; // ragged rock
1630 ridge_[x] = std::clamp(r, minRow, floorRow);
1631 side_[x] = x < nearest->x ? -1 : 1;
1632 horizonY_ = std::max(horizonY_, ridge_[x]);
1633 const int tree = static_cast<int>(std::lround(rowAt(sc.treeline) + amp * (std::sin(x * 0.11 + ph[2]) + 0.6 * std::sin(x * 0.31 + ph[3]))));
1634 const int snow = static_cast<int>(std::lround(rowAt(sc.snowline) + amp * (std::sin(x * 0.09 + ph[3]) + 0.6 * std::sin(x * 0.27 + ph[2]))));
1635 treeRow_[x] = std::clamp(tree, 0, h_);
1636 snowRow_[x] = std::clamp(std::min(snow, treeRow_[x] - 2), 0, h_);
1637 }
1638 const int treeMin = *std::min_element(treeRow_.begin(), treeRow_.end());
1639 const int treeMax = *std::max_element(treeRow_.begin(), treeRow_.end());
1640 const int snowMin = *std::min_element(snowRow_.begin(), snowRow_.end());
1641 const int snowMax = *std::max_element(snowRow_.begin(), snowRow_.end());
1642
1643 // The rows each zone spans (feet rows; for the air zones the lowest row of the sprite).
1644 auto band = [&](Zone z, int lo, int hi) {
1645 lo = std::clamp(lo, 0, std::max(h_ - 1, 0));
1646 zoneLo_[static_cast<int>(z)] = lo;
1647 zoneHi_[static_cast<int>(z)] = std::clamp(hi, lo, std::max(h_ - 1, 0));
1648 };
1649 const int streamBank = std::max(riverTopY_ - 1, seabedY_);
1650 band(Zone::Meadow, seabedY_, streamBank);
1651 band(Zone::Forest, treeMin, seabedY_ - 1);
1652 band(Zone::Woods, treeMin, streamBank);
1653 band(Zone::Rock, snowMin, treeMax);
1654 band(Zone::Snow, minRow, snowMax);
1655 band(Zone::Alpine, minRow, treeMax);
1656 band(Zone::Sky, 0, snowMax);
1657 band(Zone::Air, 0, treeMax);
1658 band(Zone::LowAir, treeMin, streamBank);
1659 surfaceY_ = std::clamp((treeMin + treeMax) / 2, 0, h_); // the air above the tree line
1660 scar_.assign(cols * static_cast<size_t>(h_), 0.0f);
1661}

1719 e.y = e.anchorY = feet - e.height() + 1;
1720 e.z = depthFor(feet);
1721 return;
1722 }
1723 e.y = e.anchorY = zoneHi_[zi] - e.height() + 1;
1724 e.z = depthFor(zoneHi_[zi]);
1725}

1732void World::slideStep(Entity& e, double dt) {
1733 constexpr double kWarn = 3.0; // seconds of cracking and creaking before the slab goes
1734 constexpr double kSettle = 5.0; // seconds the cloud lingers once the slide has come to rest
1735 const double bottom = h_ - 1;
1736 if (!slide_.active) { // starts high on a snowy flank and heads downhill
1737 slide_ = Slide{};
1738 slide_.active = true;
1739 std::vector<int> snowy;
1740 for (int x = 2; x < w_ - 2; ++x) {
1741 if (ridge_[static_cast<size_t>(x)] < snowRow_[static_cast<size_t>(x)] - 1) snowy.push_back(x);
1742 }
1743 int x0 = w_ / 2;
1744 if (!snowy.empty()) {
1745 x0 = snowy[static_cast<size_t>(rng_.rangeInt(0, static_cast<int>(snowy.size()) - 1))];
1746 } else {
1747 for (int x = 0; x < w_; ++x) if (ridge_[static_cast<size_t>(x)] < ridge_[static_cast<size_t>(x0)]) x0 = x;
1748 }
1749 const int left = ridgeRow(x0 - 6), right = ridgeRow(x0 + 6); // the lower side is downhill
1750 slide_.dir = left > right ? -1 : right > left ? 1 : (rng_.chance(0.5) ? 1 : -1);
1751 slide_.x0 = x0;
1752 slide_.y0 = ridgeRow(x0) + 1;
1753 slide_.gradient = rng_.range(0.4, 0.9);
1754 const double rows = std::max(bottom - slide_.y0, 1.0);
1755 const double room = slide_.dir > 0 ? w_ - 4 - x0 : x0 - 3; // keep the swath on the screen
1756 slide_.gradient = std::clamp(room / rows, 0.0, slide_.gradient);
1757 slide_.wobble = rng_.range(0, 6.28);
1758 slide_.head = slide_.lastHead = slide_.y0;
1759 }
1760 const double rows = std::max(bottom - slide_.y0, 1.0);
1761 // The middle of the swath at a row, and its half width, which grows downhill as the snow fans out.
1762 auto middle = [&](double row) {
1763 const double prog = std::clamp((row - slide_.y0) / rows, 0.0, 1.0);
1764 return slide_.x0 + slide_.dir * slide_.gradient * (row - slide_.y0) +
1765 1.8 * std::sin((row - slide_.y0) * 0.33 + slide_.wobble) * (0.4 + prog);
1766 };
1767 auto halfWidth = [&](double row) { return 1.5 + 5.0 * std::clamp((row - slide_.y0) / rows, 0.0, 1.0); };
1768
1769 if (e.age < kWarn) { // the slab cracks: snow trickles and puffs up where it is about to let go
1770 slideLevel_ = 0.15 + 0.35 * e.age / kWarn;
1771 e.x = slide_.x0 - e.width() / 2.0;
1772 e.y = slide_.y0 - e.height() / 2.0;
1773 slideX_ = slide_.x0;
1774 slideY_ = slide_.y0;
1775 slide_.puffs += 6 * dt;
1776 for (; slide_.puffs >= 1; slide_.puffs -= 1) {
1777 spawnEffect("snowpuff", slide_.x0 + rng_.range(-3, 3), slide_.y0 + rng_.range(0, 2), 1,
1778 rng_.range(-1.5, 1.5), -rng_.range(0.5, 1.5));
1779 }
1780 return;
1781 }
1782 if (slide_.endedAt < 0) {
1783 const double speed = std::min(3.5 + 2.5 * (e.age - kWarn), 11.0);
1784 slide_.head = std::min(bottom, slide_.head + speed * dt);
1785 const double cx = middle(slide_.head), hw = halfWidth(slide_.head);
1786 slideX_ = cx;
1787 slideY_ = slide_.head;
1788 // The swath: every row the head has passed is covered with snow and rubble.
1789 for (int row = static_cast<int>(std::floor(slide_.lastHead)); row <= static_cast<int>(std::floor(slide_.head)); ++row) {
1790 if (row < 0 || row >= h_) continue;
1791 const double c = middle(row), half = halfWidth(row);
1792 for (int x = static_cast<int>(std::lround(c - half)); x <= static_cast<int>(std::lround(c + half)); ++x) {
1793 if (x >= 0 && x < w_ && row >= ridge_[static_cast<size_t>(x)]) scar_[static_cast<size_t>(row) * w_ + x] = 1.0f;
1794 }
1795 }
1796 // It buries what stands in its way: animals, trees and bushes. Flyers get out of the way.
1797 const double top = slide_.lastHead - 1, bot = slide_.head + 1;
1798 const double x0 = cx - hw - 1, x1 = cx + hw + 1;
1799 for (auto& other : entities_) {
1800 Entity& o = *other;
1801 if (o.dead || o.isRare || !o.destructible || isAirZone(o.zone)) continue;
1802 if (!o.life && o.zone == Zone::None && !o.grounded) continue; // effects
1803 if (o.x + o.width() <= x0 || o.x >= x1 || o.y + o.height() <= top || o.y >= bot) continue;
1804 if (o.life && !isAsleep(o)) { // an animal under the snow cries out
1805 ++squeals_;
1806 emit(EventKind::Squeal, o.x + o.width() / 2.0, o.y + o.height() / 2.0, o.z);
1807 }
1808 claim(o, "snowpuff");
1809 }
1810 // Powder billows up around the head, and rock and uprooted trees tumble out of the forest.
1811 slide_.puffs += 36 * dt;
1812 for (; slide_.puffs >= 1; slide_.puffs -= 1) {
1813 spawnEffect("snowpuff", cx + rng_.range(-hw, hw), slide_.head - rng_.range(0, 3), 1,
1814 rng_.range(-3, 3), -rng_.range(0.5, 3));
1815 }
1816 const Zone here = zoneAt(static_cast<int>(std::lround(cx)), static_cast<int>(slide_.head));
1817 if (here != Zone::Snow && rng_.chance(14 * dt)) {
1818 spawnEffect("debris", cx + rng_.range(-hw, hw), slide_.head - 1, 1, rng_.range(-5, 5), -rng_.range(1.5, 4));
1819 }
1820 slide_.lastHead = slide_.head;
1821 slideLevel_ = 1.0;
1822 e.x = cx - e.width() / 2.0;
1823 e.y = slide_.head - e.height() * 0.65;
1824 if (slide_.head >= bottom) slide_.endedAt = e.age;
1825 return;
1826 }
1827 // The slide has reached the stream and comes to rest in a billowing cloud that slowly thins.
1828 const double since = e.age - slide_.endedAt;
1829 slideLevel_ = std::max(0.0, 1.0 - since / kSettle);
1830 slide_.puffs += 14 * (1 - since / kSettle) * dt;
1831 for (; slide_.puffs >= 1; slide_.puffs -= 1) {
1832 spawnEffect("snowpuff", e.x + rng_.range(0, e.width()), bottom - rng_.range(0, 4), 1,
1833 rng_.range(-3, 3), -rng_.range(0.5, 2.5));
1834 }
1835 if (since > kSettle) {
1836 slide_.active = false;
1837 slideLevel_ = 0;
1838 e.dead = true;
1839 }
1840}

1844void World::renderBackground(Frame& out, Style style, const Light& light) const {
1845 const SceneDef& sc = *scene_;
1846 // The scene's colors are its daytime look; the light turns them into the current look.
1847 const Light skyLight = sc.airless ? Light{} : light; // an airless sky does not change with the day
1848 const Rgb skyTop = lerp(backdrop(sc.skyTop, skyLight), Rgb{110, 70, 130}, light.glow * 0.3);
1849 const Rgb skyLow = lerp(backdrop(sc.skyLow, skyLight), Rgb{255, 150, 80}, light.glow * 0.55);
1850 const Rgb mediumTop = backdrop(sc.mediumTop, light), mediumDeep = backdrop(sc.mediumDeep, light);
1851 const Rgb groundA = backdrop(sc.groundA, light), groundB = backdrop(sc.groundB, light);
1852 const Rgb surfaceFg = backdrop(sc.surfaceFg, light);
1853 const Rgb riverTop = backdrop(sc.riverTop, light), riverDeep = backdrop(sc.riverDeep, light);
1854
1855 // In a mountain scene the sky reaches down to the lowest foothill; the terrain is drawn over it
1856 // afterwards, so the sun can set behind the peaks.
1857 const int skyEnd = sc.mountains ? horizonY_ : surfaceY_;
1858 for (int y = 0; y < h_; ++y) {
1859 for (int x = 0; x < w_; ++x) {
1860 Cell c;
1861 const uint32_t h = hash2(x, y);
1862 if (sc.mountains && y >= skyEnd) {
1863 c.bg = skyLow;
1864 } else if (y < skyEnd) {
1865 c.bg = lerp(skyTop, skyLow, skyEnd > 1 ? double(y) / (skyEnd - 1) : 1.0);
1866 if (sc.starDensity > 0 && light.stars > 0.2 && h % static_cast<uint32_t>(sc.starDensity) == 0) {
1867 const bool bright = std::sin(time_ * 2.0 + (h >> 8)) > 0; // twinkling stars
1868 c.ch = bright ? U'*' : U'.';
1869 c.fg = lerp(c.bg, bright ? Rgb{250, 250, 220} : Rgb{150, 150, 170}, light.stars);
1870 }
1871 } else if (riverRows_ > 0 && y >= riverTopY_) { // river: flowing ripples
1872 c = riverCell(x, y, riverRows_ > 1 ? double(y - riverTopY_) / (riverRows_ - 1) : 0.0, time_,
1873 sc.riverFlow, riverTop, riverDeep);
1874 } else if (y >= seabedY_) {
1875 const int row = y - seabedY_;
1876 const int dug = craterRows(x);
1877 c.bg = (y == seabedY_) ? groundA : groundB;

1915void World::renderCelestial(Frame& out, Style style, const Light& light) const {
1916 const int skyEnd = scene_->mountains ? horizonY_ : surfaceY_;
1917 if (!scene_->dayNight || skyEnd <= 0 || overcast_ > 0.6) return;
1918 constexpr double kPi = 3.14159265358979;
1919 const bool byDay = phase_ >= 0.25 && phase_ <= 0.75;
1920 // How far along its arc the body is: 0 as it rises, 1 as it sets.
1921 const double u = byDay ? (phase_ - 0.25) / 0.5 : std::fmod(phase_ + 0.25, 1.0) / 0.5;
1922 auto sprite = pack_->find(byDay ? scene_->sunSprite : scene_->moonSprite);
1923 if (!sprite) return;
1924 Entity body;
1925 body.sprite = sprite;
1926 body.glow = true;
1927 body.fg = sprite->props().color("color", {255, 240, 180});
1928 body.x = u * (w_ - 1) - sprite->width() / 2.0;
1929 // At the horizon the whole body is just below it; it climbs to the top of the sky at mid-arc.
1930 // In a sky lower than the body it stops with its feet on the sky's last row and its top cut off.
1931 const double peak = std::min(0.0, static_cast<double>(skyEnd - sprite->height()));
1932 body.y = skyEnd - (skyEnd - peak) * std::sin(kPi * u);
1933 blit(out, body, style, light, skyEnd); // rows below the horizon stay hidden
1934}

1936void World::renderTerrain(Frame& out, Style style, const Light& light) const {
1937 const SceneDef& sc = *scene_;
1938 const Rgb groundA = backdrop(sc.groundA, light), groundB = backdrop(sc.groundB, light);
1939 const Rgb forestA = backdrop(sc.forestA, light), forestB = backdrop(sc.forestB, light);
1940 const Rgb rockA = backdrop(sc.rockA, light), rockB = backdrop(sc.rockB, light);
1941 const Rgb snowA = backdrop(sc.snowA, light), snowB = backdrop(sc.snowB, light);
1942 const Rgb riverTop = backdrop(sc.riverTop, light), riverDeep = backdrop(sc.riverDeep, light);
1943 const Rgb slideSnow = backdrop({244, 248, 252}, light), rubble = backdrop({120, 98, 78}, light);
1944 auto glyph = [](const std::u32string& set, uint32_t h) { return set.empty() ? U' ' : set[h % set.size()]; };
1945
1946 for (int x = 0; x < w_; ++x) {
1947 const size_t xi = static_cast<size_t>(x);
1948 const int top = ridge_[xi];
1949 for (int y = top; y < h_; ++y) {
1950 const uint32_t h = hash2(x, y);
1951 const Zone z = zoneAt(x, y);
1952 Cell c;
1953 if (riverRows_ > 0 && y >= riverTopY_) {
1954 c = riverCell(x, y, riverRows_ > 1 ? double(y - riverTopY_) / (riverRows_ - 1) : 0.0, time_,
1955 sc.riverFlow, riverTop, riverDeep);
1956 } else {
1957 if (z == Zone::Meadow) {
1958 c.bg = y == seabedY_ ? groundA : groundB;
1959 c.ch = glyph(sc.groundGlyphs, h);
1960 } else if (z == Zone::Forest) {
1961 c.bg = lerp(forestA, forestB, double(y - treeRow_[xi]) / std::max(seabedY_ - treeRow_[xi], 1));
1962 c.ch = glyph(sc.forestGlyphs, h);
1963 } else if (z == Zone::Rock) {
1964 c.bg = lerp(rockA, rockB, double(y - snowRow_[xi]) / std::max(treeRow_[xi] - snowRow_[xi], 1));
1965 c.ch = glyph(sc.rockGlyphs, h);
1966 } else {
1967 c.bg = lerp(snowA, snowB, double(y - top) / std::max(snowRow_[xi] - top, 1));
1968 c.ch = glyph(sc.snowGlyphs, h);
1969 }
1970 if (side_[xi] > 0 && z != Zone::Meadow) c.bg = scale(c.bg, 0.86); // the far flank is in shadow
1971 c.fg = (z == Zone::Rock || z == Zone::Snow) ? scale(c.bg, 0.74) : scale(c.bg, 1.4);
1972 if (z == Zone::Rock || z == Zone::Snow) { // gullies run down the flank from the peak
1973 const int diag = side_[xi] < 0 ? x + y : x - y;
1974 if (((diag % 8) + 8) % 8 == 0 && h % 3 != 0) c.ch = side_[xi] < 0 ? U'/' : U'\\';
1975 }
1976 if (y == top && z != Zone::Meadow) { // the skyline
1977 const int l = ridge_[static_cast<size_t>(std::max(x - 1, 0))], r = ridge_[static_cast<size_t>(std::min(x + 1, w_ - 1))];
1978 c.ch = (l > top && r > top) ? U'^' : l > r ? U'/' : r > l ? U'\\' : U'-';
1979 c.fg = scale(c.bg, z == Zone::Snow ? 0.62 : 0.55);
1980 }
1981 }
1982 if (const double s = scarAt(x, y); s > 0.01) { // an avalanche has been through
1983 const double t = std::min(1.0, s * 1.3);
1984 const uint32_t q = hash2(x * 3, y * 7);
1985 c.bg = lerp(c.bg, slideSnow, t);
1986 c.fg = lerp(c.fg, q % 5 == 0 ? rubble : scale(slideSnow, 0.72), t);
1987 c.ch = q % 5 == 0 ? U'o' : q % 3 == 0 ? U'*' : q % 2 == 0 ? U':' : U'.';
1988 }
1989 if (style != Style::Full) c.defaultBg = true;
1990 if (style == Style::Mono) c.defaultFg = true;
1991 out.at(x, y) = c;
1992 }
1993 }
