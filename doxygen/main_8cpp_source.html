88 " --gallery print every sprite (for art review) and exit\n"
89 " --version, --help\n"

95bool parseArgs(int argc, char** argv, Options& o) {
96 for (int i = 1; i < argc; ++i) {
97 const std::string a = argv[i];
98 auto next = [&]() -> const char* { return i + 1 < argc ? argv[++i] : nullptr; };
99 if (a == "--seed") { const char* v = next(); if (!v) return false; o.seed = std::strtoull(v, nullptr, 10); o.seedSet = true; }
100 else if (a == "--assets") { const char* v = next(); if (!v) return false; o.assetsDir = v; }
101 else if (a == "--scene") { const char* v = next(); if (!v) return false; o.scene = v; }
102 else if (a == "--fps") { const char* v = next(); if (!v) return false; o.fps = std::atof(v); if (o.fps < 1) o.fps = 1; }
103 else if (a == "--dump") { const char* v = next(); if (!v) return false; o.dumpSteps = std::atoi(v); }
104 else if (a == "--rare-other") { const char* v = next(); if (!v) return false; o.dumpRare = std::atoi(v); o.dumpRareOther = true; }
105 else if (a == "--rare") { const char* v = next(); if (!v) return false; o.dumpRare = std::atoi(v); }
106 else if (a == "--act") { const char* v = next(); if (!v) return false; o.dumpAct = std::atoi(v); }
107 else if (a == "--realtime") o.realtime = true;
108 else if (a == "--hour") { const char* v = next(); if (!v) return false; o.hour = std::atof(v); }
109 else if (a == "--size") { const char* v = next(); if (!v || std::sscanf(v, "%dx%d", &o.dumpWidth, &o.dumpHeight) != 2) return false; }
110 else if (a == "--style") {
111 const char* v = next();
112 if (!v) return false;
113 const std::string n = v;
114 if (n == "full") o.style = Style::Full;
115 else if (n == "terminal") o.style = Style::Terminal;
116 else if (n == "mono") o.style = Style::Mono;
117 else return false;
118 }
119 else if (a == "--truecolor") o.forceTrueColor = true;
120 else if (a == "--256") o.force256 = true;
121 else if (a == "--console") o.forceConsole = true;
122 else if (a == "--audio") o.audio = true;
123 else if (a == "--mute") o.mute = true;
124 else if (a == "--list-scenes") o.listScenes = true;
125 else if (a == "--gallery") o.gallery = true;
126 else if (a == "--version") { std::puts("diorascii " DIORASCII_VERSION); std::exit(0); }
127 else if (a == "--help" || a == "-h") { usage(); std::exit(0); }
128 else return false;
129 }
130 return true;
131}

134void printGallery(const AssetPack& pack) {
135 for (const auto& [name, sprite] : pack.sprites()) {
136 std::printf("== %s (%dx%d, %zu frame%s)\n", name.c_str(), sprite->width(), sprite->height(),
137 sprite->frameCount(), sprite->frameCount() == 1 ? "" : "s");
138 const size_t shown = std::min<size_t>(sprite->frameCount(), 2);
139 for (size_t i = 0; i < shown; ++i) {
140 for (const auto& row : sprite->frame(false, i).art) {
141 std::string utf8;
142 for (char32_t c : row) appendUtf8(utf8, c);
143 std::printf("|%s|\n", utf8.c_str());
144 }
145 }
146 }

153int main(int argc, char** argv) {
154 Options opt;
155 if (!parseArgs(argc, argv, opt)) {
156 usage();
157 return 2;
158 }
159 if (!opt.seedSet) {
160 opt.seed = static_cast<uint64_t>(std::chrono::steady_clock::now().time_since_epoch().count());
161 }
162
163 AssetPack pack = AssetPack::embedded();
164 if (!opt.assetsDir.empty()) pack.addDirectory(opt.assetsDir);
165 for (const auto& e : pack.errors()) std::fprintf(stderr, "asset error: %s\n", e.c_str());
166
167 const auto scenes = pack.scenes();
168 if (scenes.empty()) {
169 std::fprintf(stderr, "no scenes available\n");
170 return 1;
171 }
172 if (opt.gallery) { printGallery(pack); return 0; }
173 if (opt.listScenes) {
174 for (const auto& s : scenes) std::printf("%s\t%s\n", s->name.c_str(), s->title.c_str());
175 return 0;
176 }
177
178 size_t sceneIdx = 0;
179 if (!opt.scene.empty()) {
180 bool found = false;
181 for (size_t i = 0; i < scenes.size(); ++i) {
182 if (scenes[i]->name == opt.scene) { sceneIdx = i; found = true; }
183 }
184 if (!found) {
185 std::fprintf(stderr, "unknown scene '%s' (try --list-scenes)\n", opt.scene.c_str());
186 return 2;
187 }
188 }
189
190 const double dt = 1.0 / opt.fps;
191
192 if (opt.dumpSteps >= 0) {
193 World world(pack, scenes[sceneIdx], opt.dumpWidth, opt.dumpHeight, opt.seed);
194 if (opt.hour >= 0) world.setHour(opt.hour);
195 if (opt.realtime) {
196 world.setTimeMode(World::TimeMode::Realtime);
197 world.setHour(localClockHour());
198 }
199 for (int i = 0; i < opt.dumpSteps; ++i) {
200 if (i == opt.dumpAct) world.interact();
201 if (i == opt.dumpRare) world.triggerRare(opt.dumpRareOther ? RarePick::Other : RarePick::Weighted);
202 world.step(1.0 / 20);
203 }
204 Frame frame;
205 world.render(frame);
206 std::puts(frame.toText().c_str());
207 return 0;
208 }
209
210 Terminal term;
211 if (!term.init()) {
212 std::fprintf(stderr, "diorascii needs an interactive terminal (try --dump N for headless output)\n");
213 return 1;
214 }
215 if (opt.forceTrueColor) term.setColorMode(ColorMode::TrueColor);
216 if (opt.force256) term.setColorMode(ColorMode::Ansi256);
217 if (opt.forceConsole) term.setColorMode(ColorMode::Console);
218
219 int w = 80, h = 24;
220 term.size(w, h);
221 Session::Config config;
222 config.seed = opt.seed;
223 config.scene = sceneIdx;
224 config.style = opt.style;
225 config.hour = opt.hour;
226 config.realtime = opt.realtime;
227 config.sound = !opt.mute;
228 Session session(pack, scenes, config, w, h);
229
230 using clock = std::chrono::steady_clock;
231 auto next = clock::now();
232 double measuredFps = opt.fps;
233 auto lastFrame = clock::now();
234 Frame cur, prev;
235 bool havePrev = false;
236
237 // Ambient sound is on unless muted; a machine without a sound device just stays silent. The
238 // missing device is only announced when the user asked for sound (--audio or the a key).
239 AudioOutput audio;
240 bool startup = true;
241 auto syncSound = [&]() {
242 if (session.soundWanted() && !audio.running() && !audio.start()) {
243 session.soundUnavailable(!startup || opt.audio);
244 }
245 if (!session.soundWanted() && audio.running()) audio.stop();
246 startup = false;
247 };
248 syncSound();
249
250 while (!term.quitRequested() && !session.quitRequested()) {
251 for (Action a : term.poll(5)) session.handle(a);
252 syncSound();
253 if (session.quitRequested()) break;
254 if (term.takeResized()) {
255 term.size(w, h);
256 session.resize(w, h);
257 }
258
259 const auto now = clock::now();
260 if (now < next) continue;
261 next += std::chrono::duration_cast<clock::duration>(std::chrono::duration<double>(dt));
262 if (now > next + std::chrono::milliseconds(250)) next = now; // don't spiral after a stall
263
264 const double elapsed = std::chrono::duration<double>(now - lastFrame).count();
265 lastFrame = now;
266 if (elapsed > 0) measuredFps = measuredFps * 0.9 + (1.0 / elapsed) * 0.1;
267
268 session.setMeasuredFps(measuredFps);
269 session.setClockHour(localClockHour());
270 session.step(dt);
271 if (audio.running()) audio.setParams(session.ambient());
272 session.render(cur);
273 if (session.takeRepaint()) havePrev = false;
274 // The terminal style uses the user's 16 ANSI colors so it follows their theme.
275 const Style style = session.style();
276 const ColorMode mode = style == Style::Terminal ? ColorMode::Ansi16 : term.colorMode();
277 term.write(encodeFrame(havePrev ? &prev : nullptr, cur, mode));
278 prev = cur;
279 havePrev = true;
280 }
281 return 0;
282}
