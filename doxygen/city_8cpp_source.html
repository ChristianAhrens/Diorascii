84void World::blazeStep(Entity& e, double dt) {
85 if (blazeOrigin_ < 0) { // sets one of the front skyscrapers alight, away from the edges
86 std::vector<int> choices;
87 for (int minVisible : {6, 3, 1}) { // a building whose top shows above the front row; the side streets cover their own cells
88 for (size_t i = 0; i < city_.size(); ++i) {
89 const Building& b = city_[i];
90 const int mid = b.x + b.w / 2;
91 const bool inStreet = std::any_of(avenues_.begin(), avenues_.end(), [&](const Avenue& a) {
92 return b.x < a.cx + a.halfW + 2 && b.x + b.w > a.cx - a.halfW - 2;
93 });
94 int hidden = 0; // rows of it that the front row hides
95 if (b.layer < 2) {
96 for (const Building& f : city_) {
97 if (f.layer == 2 && f.x < b.x + b.w && f.x + f.w > b.x) hidden = std::max(hidden, f.h);
98 }
99 }
100 if (b.layer >= 1 && b.h - hidden >= minVisible && !inStreet && b.state == Building::State::Normal && mid > w_ / 8 && mid < w_ - w_ / 8) {
101 choices.push_back(static_cast<int>(i));
102 }
103 }
104 if (!choices.empty()) break;
105 }
106 if (choices.empty()) {
107 e.dead = true; // nothing to burn on this screen
108 return;
109 }
110 blazeOrigin_ = choices[static_cast<size_t>(rng_.rangeInt(0, static_cast<int>(choices.size()) - 1))];
111 blazeBuildings_ = {blazeOrigin_};
112 city_[static_cast<size_t>(blazeOrigin_)].state = Building::State::Burning;
113 city_[static_cast<size_t>(blazeOrigin_)].timer = 0;
114 // Sometimes the fire engine does not win: the fire spreads to 3-5 neighbours and they all burn down.
115 blazeFails_ = forcedOutcome_ >= 0 ? forcedOutcome_ == 1 : rng_.chance(kFireFails);
116 spreadLeft_ = blazeFails_ ? rng_.rangeInt(3, 5) : 0;
117 nextSpread_ = rng_.range(7.0, 10.0);
118 blazeClock_ = 0;
119 fitCordon();
120 if (auto truck = pack_->find("firetruck"); truck && countSprite("firetruck") == 0) spawnCreature(truck, false, true);
121 }
122 blazeClock_ += dt;
123 if (spreadLeft_ > 0 && blazeClock_ >= nextSpread_) { // it jumps to the nearest building that is not burning yet
124 const Building& origin = city_[static_cast<size_t>(blazeOrigin_)];
125 const double from = origin.x + origin.w / 2.0;
126 int best = -1;
127 double bestDist = 1e9;
128 for (size_t i = 0; i < city_.size(); ++i) {
129 const Building& b = city_[i];
130 const bool inStreet = std::any_of(avenues_.begin(), avenues_.end(), [&](const Avenue& a) {
131 return b.x < a.cx + a.halfW + 2 && b.x + b.w > a.cx - a.halfW - 2;
132 });
133 const double dist = std::fabs(b.x + b.w / 2.0 - from);
134 if (b.layer >= 1 && b.state == Building::State::Normal && !inStreet && b.x + b.w > 0 && b.x < w_ && dist < bestDist) {
135 best = static_cast<int>(i);
136 bestDist = dist;
137 }
138 }
139 if (best < 0) {
140 spreadLeft_ = 0;
141 } else {
142 city_[static_cast<size_t>(best)].state = Building::State::Burning;
143 city_[static_cast<size_t>(best)].timer = 0;
144 blazeBuildings_.push_back(best);
145 --spreadLeft_;
146 nextSpread_ += rng_.range(2.5, 4.0);
147 fitCordon();
148 if (cordon_) { // the barriers move out to the new edges
149 liftBarriers();
150 raiseBarriers();
151 }
152 }
153 }
154 const bool allDone = stepBuildings(dt) && spreadLeft_ == 0;
155 if (allDone) { // everything is standing again: the site is open
156 blazeBuildings_.clear();
157 blazeOrigin_ = -1;
158 cordon_ = false;
159 liftBarriers();
160 e.dead = true;
161 }
162}

164bool World::stepBuildings(double dt) {
165 double level = 0;
166 bool allDone = true;
167 fireX_ = fireY_ = 0;
168 for (int idx : blazeBuildings_) {
169 Building& b = city_[static_cast<size_t>(idx)];
170 if (b.state == Building::State::Normal) continue;
171 allDone = false;
172 b.timer += dt;
173 if (b.state == Building::State::Burning) {
174 if (!blazeFails_ && sprayTime_ > 0) b.timer += 0.7 * dt; // the water puts a fire it can beat out sooner
175 const double now = fireAt(b.timer);
176 if (now > level) { // the sound comes from the building that burns hardest
177 fireX_ = b.x + b.w / 2.0;
178 fireY_ = seabedY_ - b.h / 2.0;
179 fireZ_ = 24 - 8 * b.layer; // far buildings are further away
180 }
181 level = std::max(level, now);
182 smokePuffs_ += 6.0 * now * dt; // smoke billows up from the roof
183 for (; smokePuffs_ >= 1; smokePuffs_ -= 1) {
184 spawnEffect("smoke", b.x + rng_.range(0, std::max(b.w - 3, 1)), seabedY_ - b.h - rng_.range(0, 3), 4,
185 rng_.range(0.5, 2.0), -rng_.range(1.2, 2.6));
186 }
187 if (b.timer >= kBurn) {
188 b.state = blazeFails_ ? Building::State::Ruin : Building::State::Scaffold;
189 b.ruined = blazeFails_;
190 b.timer = 0;
191 }
192 } else if (b.state == Building::State::Ruin) {
193 if (b.timer < 14.0 && rng_.chance(2.0 * (1.0 - b.timer / 14.0) * dt)) { // the ruin smoulders for a while
194 spawnEffect("smoke", b.x + rng_.range(0, std::max(b.w - 3, 1)), seabedY_ - std::max(2, b.h / 6) - rng_.range(0, 2), 4,
195 rng_.range(-0.5, 0.5), -rng_.range(0.8, 1.6));
196 }
197 if (b.timer >= kRuin && !b.hold) { // after a long while the rebuilding starts
198 b.state = Building::State::Scaffold;
199 b.timer = 0;
200 }
201 } else if (b.timer >= (b.ruined ? kRebuild : kScaffold)) { // renovated or rebuilt: new looks, maybe a new height
202 rollBuilding(b);
203 b.state = Building::State::Normal;
204 b.ruined = false;
205 b.timer = 0;
206 }
207 }
208 sprayTime_ = std::max(0.0, sprayTime_ - dt);
209 fireLevel_ = level;
210 return allDone;
211}

226void World::placeAvenues() {
227 avenues_.clear();
228 avenueTimer_.clear();
229 if (scene_->avenues <= 0 || seabedY_ < 12) return;
230 const int n = std::min(scene_->avenues, w_ / 55);
231 if (n <= 0) return;
232 const int halfW = 6, depth = std::clamp(seabedY_ / 6, 4, 7);
233 auto clear = [&](int c) { return c >= halfW + 4 && c <= w_ - halfW - 5; };
234 for (int i = 0; i < n; ++i) {
235 const int target = w_ * (i + 1) / (n + 1);
236 int chosen = -1;
237 for (int d = 0; d <= w_ / (2 * (n + 1)) && chosen < 0; ++d) {
238 for (int sign : {1, -1}) {
239 const int c = target + sign * d;
240 const bool apart = std::all_of(avenues_.begin(), avenues_.end(), [&](const Avenue& a) { return std::abs(a.cx - c) >= 2 * halfW + 10; });
241 if (apart && clear(c)) { chosen = c; break; }
242 }
243 }
244 if (chosen < 0) continue;
245 avenues_.push_back({chosen, halfW, depth});
246 avenueTimer_.push_back(rng_.range(2.0, 6.0));

252 const int rows = std::max(h_ - seabedY_, 1);

277void World::driveStep(Entity& e, double dt) {
278 auto smooth = [](double t) { t = std::clamp(t, 0.0, 1.0); return t * t * (3 - 2 * t); };
279 // Which way the car faces the camera: frame 0 side, 1 and 2 from behind (three-quarters, straight),
280 // 3 and 4 from the front. A sprite with fewer frames keeps its side view.
281 auto setView = [&](size_t frame) { if (e.sprite->frameCount() >= 5) e.frame = frame; };
282 if (e.driveAvenue == -2) { // decide once whether to turn off, and into which street
283 e.driveAvenue = -1;
284 if (e.driveState == 0 && !avenues_.empty() && rng_.chance(e.turnChance)) {
285 const double mid = e.x + e.width() / 2.0;
286 double best = 1e9;
287 for (size_t i = 0; i < avenues_.size(); ++i) {
288 const double ahead = (avenues_[i].cx - mid) * (e.vx > 0 ? 1 : -1);
289 if (ahead > 2 && ahead < best) { best = ahead; e.driveAvenue = static_cast<int>(i); }
290 }
291 }
292 }
293 switch (e.driveState) {
294 case 0: { // along the main road
295 e.x += e.vx * dt;
296 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) { e.dead = true; return; }
297 if (e.driveAvenue >= 0) {
298 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
299 const double mid = e.x + e.width() / 2.0;
300 if ((e.vx > 0 && mid >= a.cx - 3) || (e.vx < 0 && mid <= a.cx + 3)) {
301 e.driveState = 1;
302 e.driveT = 0;
303 e.driveX0 = e.x;
304 e.driveY0 = e.y;
305 e.driveZ = e.z;
306 e.z = 24; // further away than the traffic on the main road, and behind the viaduct
307 }
308 }
309 break;
310 }
311 case 1: { // swinging into the side street
312 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
313 e.driveT += dt;
314 const double k = smooth(e.driveT / 1.1);
315 e.x = e.driveX0 + (a.cx - e.width() / 2.0 - e.driveX0) * k;
316 e.y = e.driveY0 - 2.0 * k;
317 setView(k < 0.3 ? 0 : k < 0.7 ? 1 : 2); // side, then three-quarters from behind, then from behind
318 if (e.driveT >= 1.1) e.driveState = 2;
319 break;
320 }
321 case 2: { // up the avenue, smaller steps as it gets far away, until it is gone
322 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
323 const double feet = e.y + e.height() - 1;
324 const double progress = std::clamp((seabedY_ - 1 - feet) / a.depth, 0.0, 1.0);
325 setView(2);
326 e.y -= 3.6 * (1.0 - 0.6 * progress) * dt;
327 if (e.y + e.height() - 1 <= seabedY_ - a.depth + 1) e.dead = true;
328 break;
329 }
330 case 3: { // coming down the avenue, faster as it gets near
331 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
332 const double feet = e.y + e.height() - 1;
333 const double progress = std::clamp((seabedY_ - 1 - feet) / a.depth, 0.0, 1.0); // 1 = far away
334 e.y += (2.0 + 2.2 * (1.0 - progress)) * dt;
335 setView(e.y >= e.driveY0 - 4.0 ? 3 : 4); // facing the camera, then turning to the side
336 if (e.y >= e.driveY0 - 2.0) {
337 e.driveState = 4;
338 e.driveT = 0;
339 e.driveX0 = e.x;
340 }
341 break;
342 }
343 default: { // turning out onto the main road
344 e.driveT += dt;
345 const double k = smooth(e.driveT / 1.1);
346 setView(k < 0.45 ? 3 : 0);
347 e.vx = e.driveDir * e.driveSpeed * k;
348 e.x += e.vx * dt;
349 e.y += (e.driveY0 - e.y) * std::min(1.0, 4.0 * dt);
350 if (e.driveT >= 1.1) {
351 e.driveState = 0;
352 e.driveAvenue = -1;
353 e.vx = e.driveDir * e.driveSpeed;
354 e.y = e.anchorY = e.driveY0;
355 e.z = e.driveZ;
356 }
357 break;
358 }
359 }

362void World::paintAvenue(Frame& out, const Avenue& a, Style style, const Light& light) const {
363 const bool full = style == Style::Full;
364 const Rgb pavement = backdrop(scene_->groundA, light), asphalt = backdrop(scene_->groundB, light);
365 const Rgb haze = backdrop(scene_->skyLow, light);
366 for (int r = 0; r < a.depth; ++r) {
367 const int y = seabedY_ - 1 - r;
368 const double far = static_cast<double>(r) / a.depth;
369 const int hw = std::max(2, static_cast<int>(std::lround(a.halfW * (1.0 - 0.35 * far)))); // a little narrower further back

372 if (!c) continue;
373 const int dx = x - a.cx;
374 const bool road = std::abs(dx) <= hw;
375 c->ch = road ? ((dx == 0 && r % 2 == 1) ? U':' : U' ') : (dx < 0 ? U'/' : U'\\');
376 c->fg = road ? backdrop(Rgb{225, 200, 90}, light) : scale(pavement, 0.7);
377 c->bg = lerp(road ? asphalt : pavement, haze, 0.3 * far);
378 c->defaultBg = !full;
379 c->defaultFg = style == Style::Mono;
380 }

405void World::engineStep(Entity& e, double dt) {
406 const bool burning = blazeOrigin_ >= 0 && fireBurning();
407 if (e.driveState == 0 && e.driveSpeed == 0) e.driveSpeed = std::fabs(e.vx); // its cruising speed
408 if (e.driveState == 0) {
409 e.x += e.vx * dt; // on its way to the fire
410 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) { e.dead = true; return; }
411 if (burning) {
412 const Building& b = city_[static_cast<size_t>(blazeOrigin_)];
413 const double target = b.x + b.w / 2.0, mid = e.x + e.width() / 2.0;
414 if ((e.vx > 0 && mid >= target) || (e.vx < 0 && mid <= target)) { // arrived: park and cordon the site off
415 e.driveDir = e.vx > 0 ? 1 : -1;
416 e.vx = 0;
417 e.driveState = 1;
418 cordon_ = true;
419 fitCordon();
420 raiseBarriers();
421 }
422 }
423 } else if (e.driveState == 1) {
424 if (burning) { // fighting the fire: a jet of water arcs up to the facade
425 sprayTime_ = 0.3;
426 const double mine = e.x + e.width() / 2.0;
427 const Building* target = nullptr; // the burning building nearest to it
428 for (int idx : blazeBuildings_) {
429 const Building& cand = city_[static_cast<size_t>(idx)];
430 if (cand.state != Building::State::Burning) continue;
431 if (!target || std::fabs(cand.x + cand.w / 2.0 - mine) < std::fabs(target->x + target->w / 2.0 - mine)) target = &cand;
432 }
433 if (!target) return;
434 const Building& b = *target;
435 const double nx = e.x + e.width() / 2.0, ny = e.y + 1;
436 const double reachTime = 0.55;
437 const int drops = static_cast<int>(40 * dt) + (rng_.chance(40 * dt - static_cast<int>(40 * dt)) ? 1 : 0);
438 for (int i = 0; i < drops; ++i) {
439 const double tx = b.x + b.w / 2.0 + rng_.range(-b.w / 3.0, b.w / 3.0);
440 const double ty = seabedY_ - 1 - b.h * rng_.range(0.3, 0.8);
441 spawnEffect("water", nx, ny, 3, (tx - nx) / reachTime, (ty - ny) / reachTime);
442 }
443 } else { // the fire is out, put out or burnt out: drive on, siren off
444 e.driveState = 2;
445 e.vx = e.driveDir * e.driveSpeed;
446 e.tag.clear(); // it no longer sounds its siren
447 }
448 } else {
449 e.x += e.vx * dt;
450 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
451 }
452}

464void World::trainStep(Entity& e, double dt) {
465 if (e.driveSpeed == 0) { // first step: remember the cruising speed and the direction
466 e.driveSpeed = std::fabs(e.vx);
467 e.driveDir = e.vx > 0 ? 1 : -1;
468 }
469 const double station = stationColumn();
470 // Drawn at the stop: where the middle of the train is level with the middle of the station.
471 const double stopX = station - e.width() / 2.0;
472 const double ahead = (stopX - e.x) * e.driveDir; // cells still to go to the stop; negative = passed it
473 switch (e.stopPhase) {
474 case 0: // at full speed; brakes in good time before the station
475 if (station >= 0 && ahead > 0 && ahead <= e.driveSpeed * 3.0) {
476 e.stopPhase = 1;
477 e.driveX0 = ahead; // distance at which the braking began
478 }
479 break;
480 case 1: { // braking: slower and slower, but always a little speed left to roll in
481 const double v = std::max(1.2, e.driveSpeed * std::sqrt(std::max(ahead, 0.0) / std::max(e.driveX0, 1.0)));
482 e.vx = e.driveDir * v;
483 if (ahead <= 0.2) {
484 e.x = stopX;
485 e.vx = 0;
486 e.stopPhase = 2;
487 e.driveT = 0;
488 e.unload = rng_.rangeInt(2, 5);
489 e.driveY0 = 0; // seconds until the next passenger gets out
490 }
491 break;
492 }
493 case 2: { // standing at the platform: the doors open and the passengers get out one by one
494 e.driveT += dt;
495 e.driveY0 -= dt;
496 if (e.unload > 0 && e.driveY0 <= 0) {
497 --e.unload;
498 e.driveY0 = kUnloadGap;
499 if (auto s = pack_->find("commuter")) {
500 Placement where;
501 where.hasX = true;
502 where.x = e.x + rng_.range(3.0, std::max(e.width() - 6.0, 3.0));
503 where.dir = where.x + 1 < station ? 1 : -1;
504 const size_t before = incoming_.size();
505 spawnCreature(s, false, false, &where);
506 if (incoming_.size() > before) incoming_.back()->onDeath = nullptr; // no replacement when it leaves
507 }
508 }
509 if (e.driveT >= kDwell) {
510 e.stopPhase = 3;
511 e.vx = e.driveDir * 1.2;
512 }
513 break;
514 }
515 default: // pulling out: speeds up to cruising speed
516 e.vx = e.driveDir * std::min(e.driveSpeed, std::fabs(e.vx) + e.driveSpeed / 5.0 * dt);
517 break;
518 }
519 e.x += e.vx * dt;
520 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
521}

523World::TrainSound World::trainSound() const {
524 TrainSound s;
525 double best = 1e9; // distance of the closest train so far
526 for (const auto& e : entities_) {
527 if (e->dead || e->sprite->name() != "train") continue;
528 const double dist = std::fabs(e->x + e->width() / 2.0 - w_ / 2.0);
529 if (dist >= best) continue;
530 best = dist;
531 s.motion = e->driveSpeed > 0 ? std::clamp(std::fabs(e->vx) / e->driveSpeed, 0.0, 1.0) : 1.0;
532 s.brake = e->stopPhase == 1 ? std::sqrt(1.0 - s.motion) : 0.0;
533 s.hiss = e->stopPhase == 2 ? std::clamp(1.0 - e->driveT / 1.6, 0.0, 1.0) : 0.0;
534 }
535 return s;
536}

538void World::commuteStep(Entity& e, double dt) {
539 const double station = stationColumn();
540 e.x += e.vx * dt;
541 const double mid = e.x + e.width() / 2.0;
542 if (station < 0 || (e.vx > 0 ? mid >= station : mid <= station)) { // down the stairs
543 e.dead = true;
544 if (station < 0) return;
545 static const char* const kinds[] = {"walker_front", "walker_back", "kid"};
546 if (auto s = rng_.chance(0.8) ? pack_->find(kinds[rng_.rangeInt(0, 1)]) : pack_->find(kinds[2])) {
547 Placement where;
548 where.hasX = true;
549 where.x = station - s->width() / 2.0;
550 where.dir = rng_.chance(0.5) ? 1 : -1;
551 const size_t before = incoming_.size();
552 spawnCreature(s, false, false, &where);
553 if (incoming_.size() > before) incoming_.back()->onDeath = nullptr; // a visitor, not a resident
554 }
555 }
556}

561 const double station = stationColumn();
562 if (station >= 0) crossings_.push_back(static_cast<int>(station) - 1); // the passengers cross here

566 const bool crowded = std::any_of(crossings_.begin(), crossings_.end(), [&](int c) { return std::abs(c - at) < 10; });
567 if (!inSideStreet && !crowded) crossings_.push_back(at);
568 }

575void World::rollBuilding(Building& b) {
576 const int ground = seabedY_;
577 const int lo = std::max(3, static_cast<int>(scene_->skyline.first * ground));
578 const int hi = std::clamp(static_cast<int>(scene_->skyline.second * ground), lo, std::max(ground - 2, lo));
579 b.roof = rng_.rangeInt(0, 5);
580 b.antenna = b.roof == 1 ? rng_.rangeInt(2, 4) : (b.roof == 3 || b.roof == 4) ? 2 : 0;
581 // Heights: the far row is tall and pale, the middle row mixes low offices, mid-rise and towers,
582 // the front row keeps to low and mid-rise buildings.
583 const double span = hi - lo;
584 double share;
585 if (b.layer == 0) share = rng_.range(0.35, 1.0);
586 else if (b.layer == 1) {
587 const double r = rng_.uniform();
588 share = r < 0.3 ? rng_.range(0.0, 0.3) : r < 0.7 ? rng_.range(0.25, 0.6) : rng_.range(0.6, 1.0);
589 } else {
590 share = rng_.range(0.0, 0.55);
591 }
592 const int room = std::max(ground - 1 - b.antenna, 3);
593 b.h = std::clamp(lo + static_cast<int>(share * span), 3, room);
594 b.tint = rng_.rangeInt(0, kWallCount - 1);
595 b.facade = rng_.rangeInt(0, 3);
596 b.seed = static_cast<uint32_t>(rng_.rangeInt(1, 1 << 24));
597}

599void World::buildCity() {
600 city_.clear();
601 glimpses_.clear();
602 railY_ = -1;
603 if (seabedY_ < 8) return; // too small for a skyline
604 if (seabedY_ >= 16) railY_ = seabedY_ - 7;
605 // The front row keeps out of the side streets; this is the column where the way is free again.
606 auto gapEnd = [&](int x0, int x1) {
607 for (const Avenue& a : avenues_) {
608 if (x1 > a.cx - a.halfW - 1 && x0 < a.cx + a.halfW + 2) return a.cx + a.halfW + 2;
609 }
610 return -1;
611 };
612 for (int layer = 0; layer < 3; ++layer) {
613 int x = -rng_.rangeInt(0, 6);
614 while (x < w_) {
615 Building b;
616 b.layer = layer;
617 b.w = layer == 0 ? rng_.rangeInt(9, 20) : layer == 1 ? rng_.rangeInt(6, 15) : rng_.rangeInt(5, 12);
618 if (layer == 2) {
619 for (int end = gapEnd(x, x + b.w); end >= 0; end = gapEnd(x, x + b.w)) x = end;
620 if (x >= w_) break;
621 }
622 b.x = x;
623 rollBuilding(b);
624 city_.push_back(b);
625 // The far row overlaps so that there are no holes in the sky; the front row has gaps in places.
626 x += b.w + (layer == 0 ? rng_.rangeInt(-2, 2) : layer == 1 ? rng_.rangeInt(1, 4) : (rng_.chance(0.4) ? rng_.rangeInt(6, 12) : rng_.rangeInt(2, 6)));
627 }
628 }
629 // Where the front row has a gap, another side street can be seen running back behind it.
630 std::vector<bool> covered(static_cast<size_t>(w_), false);
631 for (const Building& b : city_) {
632 if (b.layer != 2) continue;
633 const int from = std::max(b.x, 0), to = std::min(b.x + b.w, w_);
634 if (from < to) std::fill(covered.begin() + from, covered.begin() + to, true);
635 }
636 const int most = std::max(1, w_ / 55);
637 for (int x = 0; x < w_ && static_cast<int>(glimpses_.size()) < most;) {
638 if (covered[static_cast<size_t>(x)]) { ++x; continue; }
639 int end = x;
640 while (end < w_ && !covered[static_cast<size_t>(end)]) ++end;
641 const int width = end - x, cx = (x + end) / 2;
642 const bool near = std::any_of(avenues_.begin(), avenues_.end(), [&](const Avenue& a) { return std::abs(a.cx - cx) < a.halfW + 8; }) ||
643 std::any_of(glimpses_.begin(), glimpses_.end(), [&](const Avenue& g) { return std::abs(g.cx - cx) < 14; });
644 if (width >= 6 && !near && x > 0 && end < w_) glimpses_.push_back({cx, std::min(3, width / 2 - 1), std::clamp(seabedY_ / 4, 4, 9)});
645 x = end + 1;
646 }
647}

654void World::renderSkyline(Frame& out, Style style, const Light& light) const {
655 const int base = seabedY_ - 1; // the lowest sky row: where the buildings stand
656 const bool full = style == Style::Full;
657 // How many windows are lit: none by day, more and more as it gets dark.
658 const double dark = std::clamp((0.8 - light.sun) / 0.6, 0.0, 1.0);
659 const double lightsOn = 0.58 * dark;
660 const int bucket = static_cast<int>(time_ / 14.0); // a few windows switch on and off now and then
661
662 bool glimpsed = false;
663 for (const Building& b : city_) {
664 if (!glimpsed && b.layer == 2) { // streets seen between the front buildings are behind them
665 for (const Avenue& g : glimpses_) paintAvenue(out, g, style, light);
666 glimpsed = true;
667 }
668 Rgb wall = kWalls[b.tint];
669 if (b.layer == 0) wall = lerp(wall, scene_->skyLow, 0.45); // far away: pale
670 else if (b.layer == 1) wall = lerp(wall, scene_->skyLow, 0.18); // in the middle: a little hazy
671 const Rgb wallLit = backdrop(wall, light);
672 const Rgb glass = backdrop(lerp(Rgb{128, 176, 214}, wall, b.layer == 0 ? 0.5 : 0.3), light);
673 // A building that burns down falls in; one that is rebuilt from the rubble grows back up.
674 const int rubble = rubbleRows(b.h);
675 int shown = b.h;
676 if (b.state == Building::State::Ruin) shown = rubble + static_cast<int>((b.h - rubble) * std::max(0.0, 1.0 - b.timer / kCollapse));
677 else if (b.state == Building::State::Scaffold && b.ruined) shown = rubble + static_cast<int>((b.h - rubble) * std::min(1.0, b.timer / kRebuild));
678 const int top = base - shown + 1;
679 if (b.state == Building::State::Ruin) { // charred rubble with embers that fade
680 const double embers = std::max(0.0, 1.0 - b.timer / 24.0);
681 for (int y = top; y <= base; ++y) {
682 for (int x = b.x; x < b.x + b.w; ++x) {
683 Cell* c = cellAt(out, x, y);
684 if (!c) continue;
685 const uint32_t hv = hash2(x + static_cast<int>(b.seed % 97), y);
686 const bool jagged = y == top && hv % 3 == 0; // a ragged top edge
687 if (jagged) continue;
688 const bool glow = (hv >> 9) % 100 < 22 * embers;
689 c->ch = full ? (hv % 4 == 0 ? U'%' : U'#') : U'#';
690 c->fg = glow ? Rgb{255, 120, 40} : backdrop(Rgb{96, 80, 70}, light);
691 c->defaultFg = style == Style::Mono;
692 if (full) {
693 c->bg = glow ? lerp(Rgb{70, 30, 20}, Rgb{150, 50, 20}, ((hv >> 4) % 10) / 10.0) : backdrop(Rgb{40, 36, 34}, light);
694 c->defaultBg = false;
695 } else {
696 c->defaultBg = true;
697 }
698 }
699 }
700 continue;
701 }
702 for (int y = top; y <= base; ++y) {
703 // A stepped roof is narrower in its top two rows.
704 const int inset = (b.roof == 2 && y < top + 2) ? 2 : 0;
705 for (int x = b.x + inset; x < b.x + b.w - inset; ++x) {
706 Cell* c = cellAt(out, x, y);
707 if (!c) continue;
708 const int cx = x - b.x, cy = base - y; // column in the building, rows above the street
709 const bool edge = x == b.x + inset || x == b.x + b.w - 1 - inset;
710 const bool roofLine = y == top;
711 bool window = false;
712 if (!edge && !roofLine && cy >= 1 && y > top + 1) {
713 switch (b.facade) {
714 case 0: window = cx % 2 == 1 && cy % 2 == 1; break; // a grid of windows
715 case 1: window = cx % 3 != 0 && cy % 4 != 0; break; // a glass curtain wall
716 case 2: window = cy % 3 == 1; break; // ribbons of windows
717 default: window = cx % 2 == 0 && cy % 2 == 1; break; // small windows
718 }
719 }
720 bool lit = false;
721 if (window) {
722 const uint32_t h = hash2(x + static_cast<int>(b.seed % 997), y);
723 lit = (h % 1000) < lightsOn * 1000.0;
724 if (lightsOn > 0.05 && (hash2(x, y + bucket * 31) % 100) < 3) lit = !lit; // someone turns a light on or off
725 }
726 if (full) {
727 Rgb bg = wallLit;
728 if (x >= b.x + b.w - 2 - inset) bg = scale(bg, 0.84); // the far side in shade
729 if (roofLine) bg = scale(wallLit, 0.7);
730 if (window) bg = lit ? kLit : glass;
731 c->bg = bg;
732 c->ch = U' ';
733 c->defaultBg = false;
734 } else {
735 c->defaultBg = true;
736 c->ch = roofLine ? U'_' : edge ? U'|' : window ? (lit ? U'o' : U'.') : U' ';
737 c->fg = window && lit ? kLit : scale(wall, 1.1);
738 c->defaultFg = style == Style::Mono;
739 }
740 }
741 }
742 if (b.state == Building::State::Burning) {
743 const double level = fireAt(b.timer);
744 const int flameRows = std::max(2, static_cast<int>(b.h * 0.6 * level));
745 const int tick = static_cast<int>(time_ * 9);
746 for (int y = top - 2; y < top + flameRows; ++y) {
747 for (int x = b.x; x < b.x + b.w; ++x) {
748 Cell* c = cellAt(out, x, y);
749 if (!c || y > base) continue;
750 const uint32_t hv = hash2(x + tick * 13, y);
751 const bool above = y < top; // flames reaching over the roof
752 const double chance = above ? 0.55 * level * (1.0 - 0.4 * (top - y)) : (0.75 - 0.4 * (y - top) / flameRows) * level;
753 if ((hv % 100) >= chance * 100) continue;
754 const double heat = ((hv >> 8) % 100) / 100.0;
755 static const char32_t kFlame[] = {U'^', U'*', U'"', U'\'', U'('};
756 c->ch = kFlame[(hv >> 16) % 5];
757 c->fg = lerp(Rgb{255, 150, 40}, Rgb{255, 235, 110}, heat);
758 c->defaultFg = style == Style::Mono;
759 if (full && !above) {
760 c->bg = lerp(Rgb{130, 24, 10}, Rgb{250, 120, 30}, heat);
761 c->defaultBg = false;
762 }
763 }
764 }
765 if (full) { // the windows below glow red
766 for (int y = top + flameRows; y < top + static_cast<int>(b.h * 0.85); ++y) {
767 for (int x = b.x + 1; x < b.x + b.w - 1; ++x) {
768 Cell* c = cellAt(out, x, y);
769 if (c && (x - b.x) % 2 == 1 && (base - y) % 2 == 1 && (hash2(x, y + tick) % 3) == 0) c->bg = Rgb{235, 70, 35};
770 }
771 }
772 }
773 } else if (b.state == Building::State::Scaffold) {
774 const Rgb pole = backdrop(Rgb{168, 124, 72}, light);
775 for (int y = top; y <= base; ++y) {
776 for (int x = b.x; x < b.x + b.w; ++x) {
777 Cell* c = cellAt(out, x, y);
778 if (!c) continue;
779 if (full) c->bg = scale(c->bg, 0.7);
780 if (y % 3 == 0 || (x - b.x) % 4 == 0) {
781 c->ch = y % 3 == 0 ? U'=' : U'|';
782 c->fg = pole;
783 c->defaultFg = style == Style::Mono;
784 }
785 }
786 }
787 const int cx = b.x + b.w / 2; // a crane on the roof
788 for (int k = 1; k <= 3; ++k) {
789 if (Cell* c = cellAt(out, cx, top - k)) { c->ch = U'|'; c->fg = pole; c->defaultFg = style == Style::Mono; }
790 }
791 for (int k = -5; k <= 3; ++k) {
792 if (Cell* c = cellAt(out, cx + k, top - 3)) { c->ch = k == 3 ? U'v' : U'-'; c->fg = pole; c->defaultFg = style == Style::Mono; }
793 }
794 }
795 // What stands on the roof: an antenna with a red light that blinks at night, a crown, a water
796 // tower or a helipad.
797 if (b.roof == 1 && b.antenna > 0) {
798 const int ax = b.x + b.w / 2;
799 for (int k = 1; k <= b.antenna; ++k) {
800 Cell* c = cellAt(out, ax, top - k);
801 if (!c) continue;
802 const bool tip = k == b.antenna;
803 c->ch = tip ? U'*' : U'|';
804 const bool blink = std::sin(time_ * 3.0) > 0;
805 c->fg = tip ? (blink || light.sun > 0.5 ? Rgb{235, 60, 50} : Rgb{120, 40, 40}) : backdrop(Rgb{150, 152, 160}, light);
806 c->defaultFg = style == Style::Mono;
807 }
808 } else if (b.roof == 3 || b.roof == 4) {
809 const Rgb trim = scale(wallLit, 0.65);
810 auto block = [&](int x0, int x1, int y, char32_t ch) {
811 for (int x = x0; x < x1; ++x) {
812 Cell* c = cellAt(out, x, y);
813 if (!c) continue;
814 c->ch = full ? U' ' : ch;
815 c->bg = trim;
816 c->defaultBg = !full;
817 c->fg = scale(wall, 1.1);
818 c->defaultFg = style == Style::Mono;
819 }
820 };
821 if (b.roof == 3) { // a crown of two narrowing steps
822 block(b.x + 2, b.x + b.w - 2, top - 1, U'_');
823 block(b.x + b.w / 2 - 1, b.x + b.w / 2 + 1, top - 2, U'^');
824 } else { // a water tower on legs
825 const int tx = b.x + std::max(1, b.w / 2 - 2);
826 block(tx, tx + 4, top - 2, U'_');
827 block(tx, tx + 4, top - 1, U'#');
828 }
829 } else if (b.roof == 5 && b.w >= 6) { // a helipad
830 if (Cell* c = cellAt(out, b.x + b.w / 2, top)) {
831 c->ch = U'H';
832 c->fg = backdrop(Rgb{245, 245, 245}, light);
833 c->defaultFg = style == Style::Mono;
834 }
835 }
836 }
837}

839void World::renderStreet(Frame& out, Style style, const Light& light) const {
840 const bool full = style == Style::Full;
841 const int rows = h_ - seabedY_;
842 const Rgb pavement = backdrop(scene_->groundA, light), asphalt = backdrop(scene_->groundB, light);
843
844 // The elevated railway: catenary masts and a wire above the train, a dark deck with two
845 // bright rails, a concrete beam under it, and heavy pillars down to the street.
846 if (railY_ >= 0) {
847 const Rgb steel = backdrop(Rgb{215, 218, 228}, light);
848 const Rgb deck = backdrop(Rgb{58, 60, 70}, light);
849 const Rgb beam = backdrop(Rgb{134, 132, 126}, light);
850 const Rgb mast = backdrop(Rgb{92, 94, 104}, light);

861 under.bg = beam;
862 under.defaultFg = style == Style::Mono;
863 under.defaultBg = !full;

866 wire.ch = U'-';
867 wire.fg = mast;
868 wire.defaultFg = style == Style::Mono;
869 }
870 if (x % 14 == 3) { // a mast
871 for (int y = std::max(railY_ - 7, 0); y < railY_; ++y) {
872 Cell& m = out.at(x, y);
873 m.ch = U'|';
874 m.fg = mast;
875 m.defaultFg = style == Style::Mono;
876 }
877 }
878 if (x % 12 < 2) { // a pillar, two columns wide
879 for (int y = railY_ + 2; y < seabedY_; ++y) {
880 Cell& p = out.at(x, y);
881 p.ch = full ? U' ' : U'#';
882 p.bg = scale(beam, y % 2 ? 0.9 : 1.0);
883 p.fg = scale(beam, 0.7);
884 p.defaultBg = !full;
885 p.defaultFg = style == Style::Mono;
886 }
887 }
888 }
889 }
890
891 // The street: a sidewalk behind, the road with its lane line, a sidewalk in front.
892 for (int g = 0; g < rows; ++g) {
893 const int y = seabedY_ + g;
894 const bool walk = g == 0 || g == rows - 1;
895 for (int x = 0; x < w_; ++x) {
896 Cell c;
897 const uint32_t h = hash2(x, y);
898 if (walk) {
899 c.bg = pavement;
900 c.ch = (h % 7 == 0) ? scene_->groundGlyphs.empty() ? U' ' : scene_->groundGlyphs[h % scene_->groundGlyphs.size()] : (x % 5 == 0 ? U'_' : U' ');

905 c.fg = scale(asphalt, 1.5);

909 c.ch = rows <= 4 ? U'_' : U'-';
910 c.fg = backdrop(Rgb{225, 200, 90}, light);
911 }
912 }
913 if (!full) c.defaultBg = true;
914 if (style == Style::Mono) c.defaultFg = true;
915 out.at(x, y) = c;
916 }
917 }
918 // Where a side street meets the main street: the sidewalk gives way to the road, and a zebra
919 // crossing runs across the side street.
920 for (const Avenue& a : avenues_) {
921 for (int x = a.cx - a.halfW - 1; x <= a.cx + a.halfW + 1; ++x) {
922 Cell* c = cellAt(out, x, seabedY_);
923 if (!c) continue;
924 const bool edge = std::abs(x - a.cx) > a.halfW;
925 c->ch = edge ? (x < a.cx ? U'/' : U'\\') : U' ';
926 c->fg = scale(pavement, 0.7);
927 c->bg = edge ? pavement : asphalt;
928 c->defaultBg = !full;
929 c->defaultFg = style == Style::Mono;
930 }

932 Cell* c = cellAt(out, x, seabedY_ - 1);
933 if (!c) continue;
934 const bool stripe = (x - a.cx) % 2 == 0;
935 c->ch = stripe ? U'=' : U' ';
936 c->fg = backdrop(Rgb{232, 232, 236}, light);
937 c->bg = stripe ? scale(backdrop(Rgb{210, 210, 216}, light), 0.55) : asphalt;
938 c->defaultBg = !full;
939 c->defaultFg = style == Style::Mono;
940 }

942 // Zebra crossings: at the foot of the station and here and there (see placeCrossings).
943 for (const int at : crossings_) {
944 for (int g = 1; g < rows - 1; ++g) {

946 if (at + k < 0 || at + k >= w_) continue;

949 c.fg = backdrop(Rgb{210, 210, 215}, light);
950 c.defaultFg = style == Style::Mono;
951 }
952 }
953 }
954
955 // Street lamps along the back sidewalk, lit from dusk on, with a warm pool of light below.
956 const double on = std::clamp((0.55 - light.sun) / 0.3, 0.0, 1.0);
957 for (int lx = kLampFirst; lx < w_; lx += kLampGap) {
958 // no lamp stands in the mouth of a side street
959 if (std::any_of(avenues_.begin(), avenues_.end(), [&](const Avenue& a) { return std::abs(lx - a.cx) <= a.halfW + 2; })) continue;
960 const int topY = seabedY_ - kLampHeight + 1;
961 for (int y = topY + 1; y <= seabedY_; ++y) {
962 if (Cell* c = cellAt(out, lx, y)) {
963 c->ch = U'|';
964 c->fg = backdrop(Rgb{96, 98, 108}, light);
965 c->defaultFg = style == Style::Mono;
966 }
967 }
968 if (Cell* c = cellAt(out, lx, topY)) {
969 c->ch = U'O';
970 c->fg = lerp(Rgb{150, 150, 150}, kLit, on);
971 c->defaultFg = style == Style::Mono;
972 }
