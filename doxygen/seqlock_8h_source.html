92class EventRing {

94
95 public:
97 void push(const T& event) {
98 const uint64_t n = written_.load(std::memory_order_relaxed);
99 Slot& slot = slots_[n % N];
100 slot.seq.store(2 * n + 1, std::memory_order_relaxed); // odd: being written
101 std::atomic_thread_fence(std::memory_order_release);
102 slot.data.store(event);
103 slot.seq.store(2 * n + 2, std::memory_order_release); // even: complete, and says which event it is
104 written_.store(n + 1, std::memory_order_release);
105 }

109 bool pop(T& event) {
110 for (;;) {
111 const uint64_t w = written_.load(std::memory_order_acquire);
112 if (read_ == w) return false;
113 Slot& slot = slots_[read_ % N];
114 const uint64_t before = slot.seq.load(std::memory_order_acquire);
115 if (before != 2 * read_ + 2) { // the writer lapped us and overwrote this event: it is lost (an older one cannot be here)
116 ++dropped_;
117 ++read_;
118 continue;
119 }
120 T copy;
121 slot.data.load(copy);
122 std::atomic_thread_fence(std::memory_order_acquire);
123 if (slot.seq.load(std::memory_order_relaxed) != before) continue; // overwritten while reading
124 event = copy;
125 ++read_;
126 return true;
127 }
128 }
