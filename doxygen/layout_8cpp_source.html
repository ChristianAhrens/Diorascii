51void unitVector(float azimuth, float elevation, float v[3]) {
52 const float a = azimuth * kPi / 180.0f, e = elevation * kPi / 180.0f;
53 v[0] = std::sin(a) * std::cos(e);
54 v[1] = std::cos(a) * std::cos(e);
55 v[2] = std::sin(e);
56}

60void canonicalPosition(Channel channel, float& azimuth, float& elevation) {
61 azimuth = 0;
62 elevation = 0;
63 switch (channel) {
64 case Channel::L: azimuth = -30; break;
65 case Channel::R: azimuth = 30; break;
66 case Channel::Lw: azimuth = -60; break;
67 case Channel::Rw: azimuth = 60; break;
68 case Channel::Ls: azimuth = -90; break;
69 case Channel::Rs: azimuth = 90; break;
70 case Channel::Lrs: azimuth = -135; break;
71 case Channel::Rrs: azimuth = 135; break;
72 case Channel::Ltf: azimuth = -45; elevation = 45; break;
73 case Channel::Rtf: azimuth = 45; elevation = 45; break;
74 case Channel::Ltm: azimuth = -90; elevation = 45; break;
75 case Channel::Rtm: azimuth = 90; elevation = 45; break;
76 case Channel::Ltr: azimuth = -135; elevation = 45; break;
77 case Channel::Rtr: azimuth = 135; elevation = 45; break;
78 default: break; // C straight ahead; LFE and None have no position
79 }
80}

82Channel nearestChannel(float azimuth, float elevation) {
83 if (!std::isfinite(azimuth)) azimuth = 0;
84 if (!std::isfinite(elevation)) elevation = 0;
85 float s[3];
86 unitVector(azimuth, elevation, s);
87 Channel best = Channel::C;
88 float bestDot = -2;
89 for (int i = 0; i < kMaxChannels; ++i) {
90 const Channel c = static_cast<Channel>(i);
91 if (c == Channel::LFE) continue;
92 float az, el, p[3];
93 canonicalPosition(c, az, el);
94 unitVector(az, el, p);
95 const float dot = s[0] * p[0] + s[1] * p[1] + s[2] * p[2];
96 if (dot > bestDot) { // strict: the earlier channel wins a tie
97 bestDot = dot;
98 best = c;
99 }
100 }
101 return best;
102}

106Layout Layout::named(LayoutId id) {
107 Layout out{Empty{}};
108 out.name_ = kNames[static_cast<int>(id)];
109 const uint32_t set = kSets[static_cast<int>(id)];
110 for (int i = 0; i < kMaxChannels; ++i) {
111 const Channel c = static_cast<Channel>(i);
112 if (!(set & bit(c))) continue;
113 Speaker& s = out.speakers_[out.count_];
114 s.channel = c;
115 canonicalPosition(c, s.azimuth, s.elevation);
116 if (isFiveOneFamily(id) && (c == Channel::Ls || c == Channel::Rs)) s.azimuth = c == Channel::Ls ? -110.0f : 110.0f;
117 if (c == Channel::LFE) out.lfe_ = out.count_;
118 ++out.count_;
119 }
120 return out;
121}

135Layout Layout::fromDevice(const DeviceChannel* channels, int count) {
136 Layout out{Empty{}};
137 out.count_ = std::clamp(count, 0, kMaxChannels);
138 out.lfe_ = -1;
139 out.name_ = "custom";
140 for (int i = 0; i < out.count_; ++i) {
141 Speaker& s = out.speakers_[i];
142 const DeviceChannel& d = channels[i];
143 if (d.lfe && out.lfe_ < 0) {
144 s = Speaker{Channel::LFE, 0, 0, 1};
145 out.lfe_ = i;
146 } else if (d.lfe) {
147 s = Speaker{Channel::None, 0, 0, 1}; // a second LFE channel stays silent
148 } else if (!d.known || !std::isfinite(d.azimuth) || !std::isfinite(d.elevation)) {
149 s = Speaker{Channel::None, 0, 0, 1};
150 } else {
151 s = Speaker{nearestChannel(d.azimuth, d.elevation), d.azimuth, d.elevation, 1};
152 }
153 }
154 return out;
155}

163bool Layout::stereoPair(int& left, int& right) const {
164 left = right = -1;
165 for (int i = 0; i < count_; ++i) {
166 if (speakers_[i].channel == Channel::L) left = i;
167 if (speakers_[i].channel == Channel::R) right = i;
168 }
169 if (left >= 0 && right >= 0) return true;
170 for (int i = 0; i < count_; ++i) { // no L/R: the extremes by azimuth
171 if (!speakers_[i].isDirectional()) continue;
172 if (left < 0 || speakers_[i].azimuth < speakers_[left].azimuth) left = i;
173 if (right < 0 || speakers_[i].azimuth > speakers_[right].azimuth) right = i;
174 }
175 return left >= 0;
176}
