37float cricket(double t, double burstHz, double pulseHz, double carrierHz) {
38 const double burst = std::fmod(t * burstHz, 1.0);
39 if (burst > 0.35) return 0.0f; // silent between bursts
40 const double pulse = std::fmod(t * pulseHz, 1.0);
41 const double env = pulse < 0.6 ? std::sin(pulse / 0.6 * kTwoPi / 2.0) : 0.0;
42 return static_cast<float>(env * std::sin(kTwoPi * carrierHz * t));
43}

46void Ambience::clap(double delay, double gain, bool strike) {
47 Thunder& th = *std::min_element(std::begin(thunder_), std::end(thunder_),
48 [](const Thunder& a, const Thunder& b) { return a.start < b.start; });
49 th.start = t_ + delay;
50 th.gain = gain;
51 th.strike = strike;
52 // The rumble of a strike overhead rolls in after the crack has cleared.
53 const double lag = strike ? 0.32 : 0.0;
54 th.swellAt[0] = lag;
55 th.swellAt[1] = lag + rng_.range(0.6, 1.4);
56 th.swellAt[2] = lag + rng_.range(1.6, 2.8);
57 th.swellGain[0] = 1.0;
58 th.swellGain[1] = rng_.range(0.4, 0.8);
59 th.swellGain[2] = rng_.range(0.2, 0.5);
60}

62void Ambience::thunderLevels(float crack[2], float rumble[2], float& snap, bool& strikeLive) const {
63 crack[0] = crack[1] = rumble[0] = rumble[1] = snap = 0;
64 strikeLive = false;
65 for (const Thunder& th : thunder_) {
66 const double t = t_ - th.start;
67 if (th.strike && th.gain > 0 && t >= -1.0 && t <= 9.0) strikeLive = true;
68 if (th.gain <= 0 || t < 0 || t > 9) continue;
69 const int part = th.strike ? 0 : 1; // a strike is heard where it struck, a far clap where the storm is
70 crack[part] += static_cast<float>(th.gain * std::exp(-t * 22.0)); // the sharp first crack
71 if (th.strike) { // two snaps, matching the two quick flashes of the lightning
72 snap += static_cast<float>(th.gain * std::exp(-t * 17.0));
73 if (t > 0.22) snap += static_cast<float>(0.6 * th.gain * std::exp(-(t - 0.22) * 20.0));
74 }
75 for (int i = 0; i < 3; ++i) { // rolling swells, each slowly fading
76 const double s = t - th.swellAt[i];
77 if (s > 0) rumble[part] += static_cast<float>(th.swellGain[i] * th.gain * (1 - std::exp(-s * 12.0)) * std::exp(-s * 0.7));
78 }
79 }

188void Ambience::route() {
189 // The directional speaker nearest the left and right of the room play chains 0 and 1, which carry
190 // the voices as well as the bed; every other directional speaker gets a bed chain of its own, in
191 // output order; the LFE channel plays the bass of all of them.
192 int8_t next[kMaxChannels];
193 int* speakerOf = speakerOf_; // the output channel of each chain
194 std::fill(next, next + kMaxChannels, static_cast<int8_t>(kSilent));
195 std::fill(speakerOf, speakerOf + kMaxChannels, -1);
196 int left, right, used = 2;
197 if (layout_.stereoPair(left, right)) {
198 if (left == right) {
199 next[left] = kBoth;
200 } else {
201 next[left] = 0;
202 next[right] = 1;
203 }
204 speakerOf[0] = left;
205 speakerOf[1] = right;
206 }
207 for (int c = 0; c < layout_.channels(); ++c) {
208 if (next[c] != kSilent) continue;
209 if (c == layout_.lfeIndex()) next[c] = kBass;
210 else if (layout_.speaker(c).isDirectional() && used < kMaxChannels) {
211 speakerOf[used] = c;
212 next[c] = static_cast<int8_t>(used++);
213 }
214 }
215 chains_ = used;
216
217 // The level of every chain in every scene: the rear a little quieter, the centre not to form a
218 // phantom image, the heights by how much of the scene's bed is overhead (none for the aquarium).
219 static constexpr float kHeightShare[kAmbientKinds] = {0.0f, 0.6f, 0.5f, 0.7f, 0.4f, 0.7f, 0.0f};
220 for (int k = 0; k < kAmbientKinds; ++k) {
221 liveMask_[k] = 0;
222 for (int ch = 0; ch < kMaxChannels; ++ch) {
223 float t = 0;
224 if (ch < 2) {
225 t = 1.0f; // the stereo pair is the synth as it always was
226 } else if (ch < chains_) {
227 const Speaker& sp = layout_.speaker(speakerOf[ch]);
228 if (sp.elevation > 1.0f) t = kHeightShare[k];
229 else if (std::fabs(sp.azimuth) > 100.0f) t = 0.85f;
230 else if (std::fabs(sp.azimuth) < 5.0f) t = 0.7f;
231 else t = 1.0f;
232 }
233 trim_[k][ch] = t;
234 if (k == static_cast<int>(AmbientKind::Forest) && ch >= 2 && ch < chains_) { // the rain falls from above: overhead most, in front least
235 const Speaker& rs = layout_.speaker(speakerOf[ch]);
236 rainShape_[ch] = rs.elevation > 1.0f ? 1.5f : std::fabs(rs.azimuth) < 50.0f ? 0.8f : 1.0f;
237 }
238 if (t > 0) liveMask_[k] |= 1u << ch;
239 }
240 }
241 stereoRef_ = layout_.channels() == 1 ? 1.0f : kStereoRef; // a single speaker plays the mean of the old stereo pair, so a placed sound is not doubled there
242 updateDrift();
243 for (Place& pl : place_) pl.v.setTarget(layout_, forLayout(pl.src), false); // placed sounds glide to their place in the new layout
244
245 for (int c = 0; c < kMaxChannels; ++c) {
246 routeTarget_[c] = next[c] == kSilent ? 0.0f : 1.0f;
247 if (next[c] != feed_[c]) routeGain_[c] = 0; // a channel with a new role fades in
248 feed_[c] = next[c];
249 }
250}

252void Ambience::updateDrift() {
253 for (int k = 0; k < kAmbientKinds; ++k) {
254 for (int ch = 0; ch < kMaxChannels; ++ch) {
255 // The stereo pair does not drift. The others swell and ebb at slightly different, very slow
256 // rates and phases, so the wind seems to move round the room.
257 const double wave = ch < 2 ? 0.0 : std::sin(kTwoPi * (0.045 + 0.0137 * ch) * t_ + 1.9 * ch);
258 bedGain_[k][ch] = trim_[k][ch] * (1.0f + driftDepth_[k] * static_cast<float>(wave));
259 }
260 }

263Source Ambience::stageSource(float x, float y, float spread, float lfeSend) {
264 Source s;
265 const float ax = std::fabs(x);
266 // The screen is a window of +-45 degrees; beyond it the sound moves round to the side (110 degrees at the limit).
267 const float az = ax <= 1.0f ? ax * 45.0f : 45.0f + (std::min(ax, kStageReach) - 1.0f) * (110.0f - 45.0f) / (kStageReach - 1.0f);
268 s.azimuth = x < 0 ? -az : az;
269 s.elevation = std::clamp(y, 0.0f, 1.0f) * 60.0f;
270 s.spread = spread;
271 s.lfeSend = lfeSend;
272 return s;
273}

358 out[i] = Source();
359 out[i].azimuth = azimuth + static_cast<float>(rng.range(-8.0, 8.0));
360 out[i].elevation = std::min(elevation + i * climb, 75.0f);
361 }

420 Bubble& b = *std::min_element(std::begin(bubbles_), std::end(bubbles_),
421 [](const Bubble& x, const Bubble& y) { return x.start < y.start; });
422 b.start = t_ + delay;
423 b.f0 = f0;
424 b.dur = dur;
425 b.amp = amp;
426 b.pan = static_cast<float>(rng_.range(0.2, 0.8));
427 Place& pl = place_[kBubble0 + static_cast<int>(&b - bubbles_)]; // each blip has a place of its own
428 pl.src = at ? *at : panSource(b.pan);
429 pl.v.setTarget(layout_, forLayout(pl.src), true);
430 pl.on = true;
431}

433void Ambience::moonSample(const AmbientParams& p, float out[kMaxChannels]) {
434 (void)p;
435 const double dt = 1.0 / rate_;
436 const int nch = layout_.channels();
437 const bool room3d = nch > 2;
438 const float mixMoon = mix_[static_cast<int>(AmbientKind::Moon)];
439 // No air, so nothing to hear but what could be felt through the ground: a faint, deep drone of
440 // stacked fifths that drifts slowly, with a hint of shimmer above it.
441 const double drift = 0.7 + 0.3 * std::sin(kTwoPi * 0.043 * t_);
442 const double shimmer = 0.5 + 0.5 * std::sin(kTwoPi * 0.11 * t_ + 0.8);
443 float pad[kMaxChannels];
444 for (int ch = 0; ch < 2; ++ch) {
445 const double o = ch ? 0.9 : 0.0;
446 pad[ch] = static_cast<float>(
447 drift * (0.5 * std::sin(kTwoPi * 55.0 * t_ + o) + 0.35 * std::sin(kTwoPi * 82.5 * t_ + 1.0 + o)) +
448 0.18 * std::sin(kTwoPi * 220.0 * t_ + 2.0 + o) * drift +
449 0.07 * shimmer * std::sin(kTwoPi * 330.7 * t_ + o) + 0.03 * (1 - shimmer) * std::sin(kTwoPi * 441.0 * t_ + o)) * 0.06f;
450 }
451 if (chains_ > 2) {
452 // The other speakers play the same chord with their own phase offsets. The sine and cosine of
453 // each partial are computed once, and every chain rotates them: sin(a + o) = sin a cos o + cos a sin o.
454 const double arg[5] = {kTwoPi * 55.0 * t_, kTwoPi * 82.5 * t_ + 1.0, kTwoPi * 220.0 * t_ + 2.0, kTwoPi * 330.7 * t_, kTwoPi * 441.0 * t_};
455 double sn[5], cs[5];
456 for (int k = 0; k < 5; ++k) {
457 sn[k] = std::sin(arg[k]);
458 cs[k] = std::cos(arg[k]);
459 }
460 for (int ch = 2; ch < chains_; ++ch) {
461 if (!live(AmbientKind::Moon, ch)) continue;
462 double s[5];
463 for (int k = 0; k < 5; ++k) s[k] = sn[k] * padCos_[ch] + cs[k] * padSin_[ch];
464 pad[ch] = static_cast<float>(drift * (0.5 * s[0] + 0.35 * s[1]) + 0.18 * s[2] * drift + 0.07 * shimmer * s[3] +
465 0.03 * (1 - shimmer) * s[4]) * 0.06f;
466 }
467 }
468
469 // A meteorite strike: a thud that thumps up out of the ground and falls away, a dull rumble,
470 // and then debris pattering down slowly in the low gravity.
471 float thud = 0, rumble = 0, patter = 0, click = 0;
472 bool striking = false; // a meteorite's sound is pending or still ringing
473 for (const Impact& im : impacts_) {
474 const double tau = t_ - im.start;
475 if (im.gain > 0 && tau >= -0.2 && tau <= 9) striking = true;
476 if (im.gain <= 0 || tau < 0 || tau > 9) continue;
477 const double sweep = 34.0 * tau + 70.0 * (1.0 - std::exp(-9.0 * tau)) / 9.0; // 104 Hz sinking to 34 Hz
478 thud += static_cast<float>(std::sin(kTwoPi * sweep) * (1.0 - std::exp(-tau * 400.0)) * std::exp(-tau * 3.2)) * im.gain;
479 rumble += static_cast<float>(im.gain * std::exp(-tau * 1.3) * (1.0 - std::exp(-tau * 30.0)));
480 if (tau > 0.5) {
481 const double density = 60.0 * std::exp(-(tau - 0.5) * 0.8); // clicks per second
482 if (rng_.uniform() < density * dt) click += static_cast<float>(rng_.range(0.4, 1.0) * (rng_.chance(0.5) ? 1 : -1)) * im.gain;
483 }
484 }
485 patter = click;
486 if (!striking) place_[kThud].on = place_[kDebris].on = false;
487 if (click != 0 && room3d && place_[kDebris].on) { // in a room each piece of debris lands somewhere round the impact
488 Source s = impactAt_;
489 s.azimuth += static_cast<float>(fxRng_.range(-35.0, 35.0));
490 s.elevation = static_cast<float>(fxRng_.range(0.0, 25.0));
491 s.spread = 0.0f;
492 place_[kDebris].v.setTarget(layout_, forLayout(s), true);
493 }
494 if (place_[kThud].on) place_[kThud].v.add(thud * 0.55f * stereoRef_, room_, nch); // thumps up out of the ground where it struck
495
496 // The alien: an eerie rising warble.
497 float zap = 0;
498 const double zt = t_ - zapStart_;
499 if (zt >= 0.9) place_[kZap].on = false;
500 if (zapStart_ > 0 && zt >= 0 && zt < 0.9) {
501 const double f = 350.0 + 1000.0 * (zt / 0.9) + 120.0 * std::sin(kTwoPi * 17.0 * zt);
502 zapPhase_ += kTwoPi * f * dt;
503 const double env = std::pow(std::sin(3.14159265 * zt / 0.9), 1.5);
504 zap = static_cast<float>((std::sin(zapPhase_) + 0.3 * std::sin(1.5 * zapPhase_)) * env) * 0.12f;
505 if (place_[kZap].on) place_[kZap].v.add(zap * stereoRef_, room_, nch); // from the alien
506 }
507
508 const float debrisF = 2.0f * std::sin(3.14159265f * std::min(900.0f, static_cast<float>(rate_) / 7.0f) / static_cast<float>(rate_));
509 // The drone and the impacts' rumble are the bed, one chain per speaker; the thud, the debris
510 // and the warble are voices, which only the stereo pair carries until they are placed.
511 // Debris: each click rings a short, dull resonance (one sound: it was the same in both ears).
512 debrisLow_[0] += debrisF * debrisBand_[0];
513 const float high = patter * 6.0f - debrisLow_[0] - 0.6f * debrisBand_[0];
514 debrisBand_[0] += debrisF * high;
515 if (place_[kDebris].on) place_[kDebris].v.add(debrisBand_[0] * 0.3f * stereoRef_, room_, nch);
516 for (int ch = 0; ch < chains_; ++ch) {
517 if (!live(AmbientKind::Moon, ch)) continue;
518 const float n = noise(chainRng(ch)) * noiseScale_;
519 moonLow_[ch][0] += (n - moonLow_[ch][0]) * lowMoon_;
520 moonLow_[ch][1] += (moonLow_[ch][0] - moonLow_[ch][1]) * lowMoon_;
521 out[ch] = pad[ch] + moonLow_[ch][1] * 14.0f * rumble * 0.35f; // the rumble rolls all round (a bed)
522 }
523 flushRoom(voiceOut_, mixMoon);
524}

526void Ambience::mountainSample(const AmbientParams& p, float out[kMaxChannels]) {
527 (void)p;
528 const double dt = 1.0 / rate_;
529 // Thin alpine wind. In a snowstorm it is much stronger and higher, with a whistle that glides
530 // with the gusts and the fine hiss of ice crystals on top.
531 const float cutoffHz = (300.0f + 360.0f * gust_) * (1.0f + 1.3f * rain_);
532 const float lowWind = 1.0f - std::exp(-6.2831853f * cutoffHz / static_cast<float>(rate_));
533 const float windGain = (0.35f + 0.8f * gust_) * (1.0f + 5.0f * rain_) * 0.075f;
534 const int nch = layout_.channels();
535 whistlePhase_ += kTwoPi * (620.0 + 340.0 * gust_ + 60.0 * std::sin(0.4 * t_)) * dt;
536 const float whistle = rain_ * rain_ * gust_ * 0.02f * static_cast<float>(std::sin(whistlePhase_));
537 if (rain_ < 1e-3f && !(p.rain > 0)) place_[kWhistle].on = false;
538 if (place_[kWhistle].on) place_[kWhistle].v.add(whistle * stereoRef_, room_, nch); // drifting round the room
539 const float hiss = rain_ * (0.1f + 0.2f * gust_);
540 // The stream: noise between two low-passes, quietly, that bubbles and gurgles: it is broken up by
541 // a few unrelated fast ripples so that it does not sound like steady brushing.
542 const float babble = std::clamp(0.35f + 0.65f * static_cast<float>(
543 std::sin(kTwoPi * 6.1 * t_ + 2.0 * std::sin(kTwoPi * 0.7 * t_)) *
544 std::sin(kTwoPi * 9.7 * t_ + 1.3 * std::sin(kTwoPi * 1.3 * t_ + 1.0))), 0.0f, 1.0f);
545 // An eagle's cry now and then, by day only and not in a snowstorm: a few shrill, falling notes.
546 const float day = (1.0f - night_) * (1.0f - std::min(1.0f, 2.0f * rain_));
547 if (nextCry_ == 0) nextCry_ = t_ + rng_.range(5.0, 15.0);
548 if (t_ >= nextCry_) {
549 cryStart_ = t_;
550 cryPan_ = static_cast<float>(rng_.range(0.2, 0.8));
551 cryPitch_ = static_cast<float>(rng_.range(0.9, 1.12));
552 nextCry_ = t_ + rng_.range(15.0, 35.0);
553 place_[kEagle].src = panSource(cryPan_); // the eagle is in the sky
554 place_[kEagle].src.elevation = 35.0f;
555 place_[kEagle].v.setTarget(layout_, forLayout(place_[kEagle].src), true);
556 }
557 float cry = 0;
558 const double ct = t_ - cryStart_;
559 if (ct >= 0 && ct < 1.3) {
560 const double syl = std::fmod(ct, 0.42), n = std::floor(ct / 0.42); // three notes, 0.42 s apart
561 if (n < 3 && syl < 0.3) {
562 const double x = syl / 0.3;
563 const double f = cryPitch_ * (3400.0 - 1300.0 * x) * (1.0 - 0.05 * n) + 60.0 * std::sin(kTwoPi * 38.0 * syl);
564 cryPhase_ += kTwoPi * f * dt;
565 const double env = std::sin(3.14159265 * x) * (1.0 - 0.18 * n);
566 cry = static_cast<float>((std::sin(cryPhase_) + 0.5 * std::sin(2 * cryPhase_ + 0.4) + 0.3 * noise(rng_)) * env) * 0.09f * day;
567 }
568 }
569 if (cry != 0) place_[kEagle].v.add(cry * 1.4f, room_, nch);
570 // A woodpecker drums on a tree now and then, as often as the eagle cries: a quick run of dry,
571 // woody knocks that dies away a little. Same rules: by day, not in a snowstorm.
572 if (nextPeck_ == 0) nextPeck_ = t_ + rng_.range(8.0, 20.0);
573 if (t_ >= nextPeck_) {
574 peckStart_ = t_;
575 peckKnocks_ = rng_.rangeInt(7, 13);
576 peckPan_ = static_cast<float>(rng_.range(0.15, 0.85));
577 peckPitch_ = static_cast<float>(rng_.range(0.85, 1.2));
578 nextPeck_ = t_ + rng_.range(15.0, 35.0);
579 place_[kPecker].src = panSource(peckPan_); // on a tree, low
580 place_[kPecker].src.elevation = 5.0f;
581 place_[kPecker].v.setTarget(layout_, forLayout(place_[kPecker].src), true);
582 }
583 float peck = 0;
584 const double pt = t_ - peckStart_;
585 constexpr double kKnockHz = 16.0;
586 if (pt >= 0 && pt < peckKnocks_ / kKnockHz) {
587 const int k = static_cast<int>(pt * kKnockHz);
588 const double lt = pt - k / kKnockHz; // time since this knock began
589 const double amp = 1.0 - 0.5 * k / peckKnocks_;
590 peck = static_cast<float>((std::sin(kTwoPi * 1150.0 * peckPitch_ * lt) * std::exp(-lt * 150.0) +
591 0.5 * std::sin(kTwoPi * 2300.0 * peckPitch_ * lt) * std::exp(-lt * 260.0) +
592 0.3 * noise(rng_) * std::exp(-lt * 500.0)) * amp) * 0.12f * day;
593 }
594 if (peck != 0) place_[kPecker].v.add(peck * 1.4f, room_, nch);
595 // An avalanche: when the slab breaks and while it tumbles, snow, rock and trees knock against
596 // each other: a burst of crackling at the start, then clicks and now and then a dull thump with
597 // a crack when something big goes down.
598 if (p.avalanche > 0.7f && !slideBroke_) { slideBroke_ = true; breakEnv_ = 1.0f; thumpEnv_ = 1.0f; thumpFreq_ = 60.0f; }
599 if (p.avalanche < 0.3f) slideBroke_ = false;
600 breakEnv_ -= breakEnv_ * static_cast<float>(dt / 1.1);
601 if (p.avalanche > 0.5f && rng_.chance(1.3 * p.avalanche * dt)) {
602 thumpEnv_ = static_cast<float>(rng_.range(0.4, 0.9));
603 thumpFreq_ = static_cast<float>(rng_.range(55, 120));
604 }
605 thumpEnv_ -= thumpEnv_ * static_cast<float>(dt * 7.0);
606 thumpPhase_ += kTwoPi * thumpFreq_ * dt;
607 const float thump = static_cast<float>(std::sin(thumpPhase_)) * thumpEnv_ * 0.35f;
608 const double clickRate = 220.0 * breakEnv_ + 18.0 * p.avalanche * p.avalanche + 260.0 * thumpEnv_;
609 float click = 0;
610 if (clickRate > 0 && rng_.uniform() < clickRate * dt) click = static_cast<float>(rng_.range(0.4, 1.0)) * (rng_.chance(0.5) ? 1.0f : -1.0f);
611 const float crackF = 2.0f * std::sin(3.14159265f * std::min(1800.0f, static_cast<float>(rate_) / 7.0f) / static_cast<float>(rate_));
612 // An avalanche: a deep rumble that swells as it gets going, and the roar and crackle of tumbling
613 // snow and rock on top of it.
614 const float slide = avalanche_;
615 const float roll = std::clamp(0.7f + 0.3f * roll_, 0.2f, 1.0f);
616 // An animal buried by the snow: a short squeal that sinks in pitch, heard through the snow
617 // (low-passed hard, so it is muted).
618 for (int i = 0; i < 3; ++i) { // each animal's squeal is low-passed on its own and placed where it was buried
619 Squeal& s = squealVoices_[i];
620 float squeal = 0;
621 const double tau = t_ - s.start;
622 if (!(!s.used || tau < 0 || tau > 0.6)) {
623 const double hz = s.pitch * (1150.0 - 450.0 * tau / 0.6) * (1.0 + 0.03 * std::sin(kTwoPi * 11.0 * tau));
624 s.phase += kTwoPi * hz * dt;
625 const double env = std::min(1.0, tau / 0.03) * std::exp(-tau / 0.25);
626 squeal = static_cast<float>(env * (std::sin(s.phase) + 0.5 * std::sin(2 * s.phase) + 0.3 * std::sin(3 * s.phase)));
627 }
628 squealLow_[i][0] += (squeal - squealLow_[i][0]) * lowSqueal_;
629 squealLow_[i][1] += (squealLow_[i][0] - squealLow_[i][1]) * lowSqueal_;
630 if (squeal != 0 || std::fabs(squealLow_[i][1]) > 1e-7f) place_[kSqueal0 + i].v.add(squealLow_[i][1] * 0.6f * stereoRef_, room_, nch);
631 }
632 // The avalanche's crackle and thump are voices at the slide (a place with a full bass send); its rumble
633 // and roar are noise, one chain per speaker, weighted by how near each speaker is to the slide.
634 if (avalanche_ < 1e-3f && !(p.avalanche > 0)) place_[kSlide].on = false;
635 crackLowAv_[0] += crackF * crackBandAv_[0];
636 crackBandAv_[0] += crackF * (click * 8.0f - crackLowAv_[0] - 0.5f * crackBandAv_[0]);
637 if (place_[kSlide].on) place_[kSlide].v.add((crackBandAv_[0] * 0.35f + thump) * stereoRef_, room_, nch);
638 const float mixMountains = mix_[static_cast<int>(AmbientKind::Mountains)];
639
640 // Wind, ice and the stream are the bed, one chain per speaker (the stream runs on the ground, so
641 // the overhead speakers do not play it). The avalanche's rumble and roar are noise placed at the slide.
642 // The first two chains always run, so that the draws of the shared generator do not depend on where
643 // anything is.
644 for (int ch = 0; ch < chains_; ++ch) {
645 const bool bed = live(AmbientKind::Mountains, ch);
646 const float wSlide = weight(kSlide, ch);
647 const bool slideOn = ch < 2 || wSlide >= 1e-3f;
648 if (!bed && !slideOn) continue;
649 const bool ground = speakerOf_[ch] < 0 || layout_.speaker(speakerOf_[ch]).elevation <= 1.0f;
650 Rng& r = chainRng(ch);
651 if (bed) {
652 const float n = noise(r) * noiseScale_;
653 mountLow_[ch][0] += (n - mountLow_[ch][0]) * lowWind;
654 mountLow_[ch][1] += (mountLow_[ch][0] - mountLow_[ch][1]) * lowWind;
655 const float n2 = noise(r) * noiseScale_;
656 iceLow_[ch] += (n2 - iceLow_[ch]) * lowIce_;
657 float streamOut = 0;
658 if (ch < 2 || ground) {
659 const float n3 = noise(r) * noiseScale_;
660 streamLow_[ch][0] += (n3 - streamLow_[ch][0]) * lowStreamTop_;
661 streamLow_[ch][1] += (n3 - streamLow_[ch][1]) * lowStreamBottom_;
662 streamOut = (streamLow_[ch][0] - streamLow_[ch][1]) * 0.085f * babble * (1.0f - 0.5f * rain_);
663 }
664 out[ch] = mountLow_[ch][1] * 9.0f * windGain + (n2 - iceLow_[ch]) * 0.6f * hiss + streamOut;
665 }
666 if (slideOn) {
667 const float n4 = noise(r) * noiseScale_;
668 slideLow_[ch][0] += (n4 - slideLow_[ch][0]) * lowSlide_;
669 slideLow_[ch][1] += (slideLow_[ch][0] - slideLow_[ch][1]) * lowSlide_;
670 const float n5 = noise(r) * noiseScale_;
671 slideMid_[ch] += (n5 - slideMid_[ch]) * lowSlideMid_;
672 voiceOut_[ch] += wSlide * (slideLow_[ch][1] * 14.0f * slide * slide * roll + slideMid_[ch] * 0.7f * slide * slide * slide * roll) * mixMountains;
673 }
674 }
675 flushRoom(voiceOut_, mixMountains);
676}

678void Ambience::desertSample(const AmbientParams& p, float out[kMaxChannels]) {
679 const double dt = 1.0 / rate_;
680 const int nch = layout_.channels();
681 const float mixDesert = mix_[static_cast<int>(AmbientKind::Desert)];
682 if (tornado_ < 1e-3f && !(p.tornado > 0)) place_[kTornado].on = false;
683 if (wormNear_ < 1e-3f && storm_ < 1e-3f && !(p.worm > 0) && !(p.storm > 0)) place_[kWorm].on = false;
684 // Dry wind: low-passed noise like the forest's, but steadier and a little higher. In a sandstorm
685 // it howls stronger, with the hiss of blown sand on top.
686 const float cutoffHz = (260.0f + 300.0f * gust_) * (1.0f + 0.7f * rain_);
687 const float lowWind = 1.0f - std::exp(-6.2831853f * cutoffHz / static_cast<float>(rate_));
688 // Calm weather is barely a breath; a sandstorm or a tornado brings the full wind.
689 const float calm = 0.22f + 0.78f * std::min(1.0f, rain_ + tornado_);
690 const float windGain = (0.45f + 0.8f * gust_) * (1.0f + 3.0f * rain_ + 0.8f * tornado_) *
691 (1.0f - 0.3f * night_) * calm * 0.085f;
692 const float sandLevel = std::max(rain_, tornado_ * 0.6f) * (0.55f + 0.45f * gust_);
693 // A tornado: a deep roar that turns, loud and quiet about three times a second.
694 tornadoPhase_ += kTwoPi * 2.8 * dt;
695 const float spin = 0.65f + 0.35f * static_cast<float>(std::sin(tornadoPhase_));
696 const float roar = tornado_ * std::sqrt(tornado_) * spin * 2.0f;
697 // Crickets at night, fewer than in the forest.
698 const float cricketA = cricket(t_, 2.1, 33, 4300), cricketB = 0.7f * cricket(t_ + 0.5, 1.7, 28, 4800);
699 const float chirps = night_ * 0.014f * (cricketA + cricketB);
700 const bool singing = nch > 2 && night_ > 1e-3f; // in a room each cricket is a voice at its own place
701 for (int i = 0; i < 2; ++i) {
702 Place& pl = place_[kDesertCricket0 + i];
703 if (pl.on != singing) {
704 pl.on = singing;
705 if (singing) pl.v.setTarget(layout_, forLayout(pl.src), true);
706 }
707 if (singing) pl.v.add(night_ * 0.014f * (i ? cricketB : cricketA) * stereoRef_, room_, nch);
708 }
709 // The sandworm below, as it nears: a growl in the ground.
710 const float growl = storm_ * storm_ * 0.5f * std::clamp(0.55f + 0.45f * roll_, 0.1f, 1.0f);
711 // The worm bellows now and then while it crosses: a very deep, trumpeting roar. The pitch
712 // swoops up and sinks again, the tone is rough (a fast tremble) and low-passed to stay deep.
713 if (wormNear_ > 0.05f) {
714 if (nextRoar_ == 0) nextRoar_ = t_ + rng_.range(0.3, 1.2);
715 if (t_ >= nextRoar_) {
716 roarStart_ = t_;
717 roarDur_ = rng_.range(2.0, 2.8);
718 nextRoar_ = t_ + roarDur_ + rng_.range(1.2, 3.5);
719 }
720 } else {
721 nextRoar_ = 0;
722 }
723 float bellow = 0;
724 const double rt = t_ - roarStart_;
725 if (rt >= 0 && rt < roarDur_) {
726 const double u = rt / roarDur_;
727 const double env = std::min(1.0, rt / 0.3) * std::min(1.0, (roarDur_ - rt) / 0.8);
728 const double hz = 34.0 + 46.0 * std::sin(3.14159265 * std::pow(u, 0.7)) + 3.0 * std::sin(kTwoPi * 5.5 * rt);
729 roarPhase_ += kTwoPi * hz * dt;
730 const double tone = std::tanh(2.4 * std::sin(roarPhase_) + 1.3 * std::sin(2 * roarPhase_ + 0.6) +
731 0.7 * std::sin(3 * roarPhase_ + 1.1));
732 bellow = static_cast<float>(tone * env * (1.0 + 0.4 * std::sin(kTwoPi * 23.0 * t_)));
733 }
734 roarLow_[0] += (bellow - roarLow_[0]) * lowRoar_;
735 roarLow_[1] += (roarLow_[0] - roarLow_[1]) * lowRoar_;
736 const float roarOut = roarLow_[1] * 0.45f * wormNear_;
737 if (place_[kWorm].on && std::fabs(roarOut) > 1e-9f) place_[kWorm].v.add(roarOut * stereoRef_, room_, nch); // the bellow comes from the worm (nothing to add between bellows)
738 // Crushing: chewing. Each thing it swallows is ground between a few soft bites, every one a
739 // short swell of low-passed noise (no sharp start, so it never sounds like a shot), over a faint
740 // low grumble.
741 float crunchEnv[4] = {}; // the chewing of each slot (placed where it was crushed)
742 for (int i = 0; i < 4; ++i) {
743 const Crunch& c = crunches_[i];
744 const double tau = t_ - c.start;
745 if (c.gain <= 0 || tau > 1.6) place_[kCrunch0 + i].on = false;
746 if (c.gain <= 0 || tau < 0 || tau > 1.6) continue;
747 for (int b = 0; b < c.bites; ++b) {
748 const double bt = tau - c.bite[b];
749 if (bt < 0 || bt > 0.09) continue;
750 const double swell = std::sin(3.14159265 * bt / 0.09);
751 crunchEnv[i] += static_cast<float>(c.gain * c.force[b] * swell * swell);
752 }
753 const float g = static_cast<float>(c.gain * std::sin(kTwoPi * 62.0 * tau) * std::min(1.0, tau / 0.12) * std::exp(-tau / 0.7));
754 if (place_[kCrunch0 + i].on) place_[kCrunch0 + i].v.add(g * 0.03f * stereoRef_, room_, nch);
755 }
756 const bool chewing = crunchEnv[0] > 0 || crunchEnv[1] > 0 || crunchEnv[2] > 0 || crunchEnv[3] > 0;
757
758 // Wind and sand are the bed, one chain per speaker. The tornado's roar, the worm's growl and the
759 // chewing are placed: noise per chain, weighted by how near each speaker is to the sound. The
760 // crickets are voices, which only the stereo pair carries until they are placed.
761 for (int ch = 0; ch < chains_; ++ch) {
762 if (!live(AmbientKind::Desert, ch)) continue;
763 Rng& r = chainRng(ch);
764 const float n = noise(r) * noiseScale_;
765 desertLow_[ch][0] += (n - desertLow_[ch][0]) * lowWind;
766 desertLow_[ch][1] += (desertLow_[ch][0] - desertLow_[ch][1]) * lowWind;
767 // (Past the stereo pair, whose generator must advance as it always did, a layer that is silent is not computed.)
768 if (ch < 2 || sandLevel > 0) {
769 const float n2 = noise(r) * noiseScale_;
770 sandLow_[ch][0] += (n2 - sandLow_[ch][0]) * lowSandCut_;
771 sandLow_[ch][1] += (n2 - sandLow_[ch][1]) * lowSandTop_;
772 }
773 if (ch < 2 || roar > 0) {
774 const float n3 = noise(r) * noiseScale_;
775 tornadoLow_[ch][0] += (n3 - tornadoLow_[ch][0]) * lowTornado_;
776 tornadoLow_[ch][1] += (tornadoLow_[ch][0] - tornadoLow_[ch][1]) * lowTornado_;
777 }
778 // The chewing: the slots' envelopes weighted for this chain. Only the stereo pair keeps its
779 // noise running while nothing chews (its generator must advance as it always did).
780 float chew = 0;
781 if (chewing)
782 for (int i = 0; i < 4; ++i)
783 if (crunchEnv[i] > 0) chew += weight(kCrunch0 + i, ch) * crunchEnv[i];
784 float crunchOut = 0;
785 if (ch < 2 || chewing) {
786 const float nc = noise(r) * noiseScale_;
787 crunchLow_[ch][0] += (nc - crunchLow_[ch][0]) * lowCrunch_;
788 crunchLow_[ch][1] += (crunchLow_[ch][0] - crunchLow_[ch][1]) * lowCrunch_;
789 // the grinding is the noise below ~500 Hz, with a little more of the bite's grit above it
790 crunchOut = (crunchLow_[ch][1] * 7.0f + (nc - crunchLow_[ch][0]) * 0.25f) * chew;
791 }
792 if (ch < 2 || growl > 0) {
793 const float n4 = noise(r) * noiseScale_;
794 deepLow_[ch][0] += (n4 - deepLow_[ch][0]) * lowDeep_;
795 deepLow_[ch][1] += (deepLow_[ch][0] - deepLow_[ch][1]) * lowDeep_;
796 }
797 float placed = crunchOut;
798 if (roar > 0) placed += weight(kTornado, ch) * tornadoLow_[ch][1] * 8.0f * roar;
799 if (growl > 0) placed += weight(kWorm, ch) * deepLow_[ch][1] * 20.0f * growl;
800 voiceOut_[ch] += placed * mixDesert;
801 out[ch] = desertLow_[ch][1] * 9.0f * windGain + (sandLow_[ch][1] - sandLow_[ch][0]) * 0.5f * sandLevel +
802 (ch < 2 && nch <= 2 ? chirps * (ch ? 0.9f : 1.0f) : 0.0f);
803 }
804 flushRoom(voiceOut_, mixDesert);
805}

807void Ambience::citySample(const AmbientParams& p, float out[kMaxChannels]) {
808 (void)p;
809 const double dt = 1.0 / rate_;
810 const float day = 1.0f - 0.55f * night_;
811 // Cars passing by now and then: an engine swell that sinks in pitch as it goes, panned across.
812 if (nextPass_ == 0) nextPass_ = t_ + rng_.range(0.3, 1.5);
813 if (t_ >= nextPass_) {
814 Pass& v = *std::min_element(std::begin(passes_), std::end(passes_), [](const Pass& a, const Pass& b) { return a.start < b.start; });
815 v.start = t_;
816 v.dur = rng_.range(1.4, 2.6);
817 v.pitch = static_cast<float>(rng_.range(80, 125));
818 v.pan = static_cast<float>(rng_.range(0.1, 0.9));
819 v.gain = static_cast<float>(rng_.range(0.5, 1.0));
820 nextPass_ = t_ + rng_.range(1.2, 4.5) * (1.0 + 2.0 * night_);
821 Place& pl = place_[kEngine0 + static_cast<int>(&v - passes_)]; // the car is placed where its pan says
822 pl.src = panSource(v.pan);
823 pl.v.setTarget(layout_, pl.src, true);
824 }
825 // Each engine is one mono sound, low-passed on its own and placed; the stereo pan is the spatializer's.
826 const int nch = layout_.channels();
827 for (int i = 0; i < 4; ++i) {
828 Pass& v = passes_[i];
829 float tone = 0;
830 const double tau = t_ - v.start;
831 if (!(v.gain <= 0 || tau < 0 || tau > v.dur)) {
832 const double u = tau / v.dur;
833 const double env = std::sin(3.14159265 * u) * std::sin(3.14159265 * u);
834 v.phase += kTwoPi * v.pitch * (1.3 - 0.55 * u) * dt; // the pitch sinks as it drives away
835 tone = static_cast<float>(std::tanh(1.8 * std::sin(v.phase) + 0.8 * std::sin(2 * v.phase + 0.4) + 0.5 * std::sin(3 * v.phase)) * env * v.gain);
836 }
837 engineLow_[i] += (tone - engineLow_[i]) * lowEngine_;
838 if (tone != 0 || std::fabs(engineLow_[i]) > 1e-7f) place_[kEngine0 + i].v.add(engineLow_[i] * 0.16f * day, room_, nch);
839 }
840 // Horns: now and then one or two honks, rarer at night.
841 if (nextHonk_ == 0) nextHonk_ = t_ + rng_.range(2.0, 6.0);
842 if (t_ >= nextHonk_) {
843 Honk& h = *std::min_element(std::begin(honks_), std::end(honks_), [](const Honk& a, const Honk& b) { return a.start < b.start; });
844 h.start = t_;
845 h.pitch = static_cast<float>(rng_.range(0.88, 1.18));
846 h.pan = static_cast<float>(rng_.range(0.1, 0.9));
847 h.gain = static_cast<float>(rng_.range(0.6, 1.0));
848 h.long_ = rng_.chance(0.2) ? 1.0f : 0.0f;
849 h.count = (h.long_ > 0 || rng_.chance(0.55)) ? 1 : 2;
850 nextHonk_ = t_ + rng_.range(5.0, 16.0) * (1.0 + 2.5 * night_);
851 place_[kHorn].src = panSource(h.pan); // honks are seconds apart: one place serves them all
852 place_[kHorn].v.setTarget(layout_, place_[kHorn].src, true);
853 }
854 float horn = 0;
855 for (Honk& h : honks_) {
856 const double tau = t_ - h.start;
857 if (h.gain <= 0 || tau < 0 || tau > 1.2) continue;
858 const double len = h.long_ > 0 ? 0.8 : 0.22, gap = 0.34;
859 for (int k = 0; k < h.count; ++k) {
860 const double lt = tau - k * gap;
861 if (lt < 0 || lt > len) continue;
862 const double env = std::min(1.0, lt / 0.01) * std::min(1.0, (len - lt) / 0.04);
863 h.phaseA += kTwoPi * 420.0 * h.pitch * dt;
864 h.phaseB += kTwoPi * 530.0 * h.pitch * dt; // two notes a third apart, like a car horn
865 const float s = static_cast<float>((std::tanh(3.0 * std::sin(h.phaseA)) + std::tanh(3.0 * std::sin(h.phaseB))) * 0.5 * env * h.gain);
866 horn += s;
867 }
868 }
869 hornLow_ += (horn - hornLow_) * lowHorn_;
870 if (horn != 0 || std::fabs(hornLow_) > 1e-7f) place_[kHorn].v.add(hornLow_ * 0.045f, room_, nch);
871 // The siren of an ambulance or a fire engine: two notes, the pitch gliding between them.
872 const float sirenTarget = std::fmod(t_, 1.0) < 0.5 ? 960.0f : 770.0f;
873 sirenFreq_ += (sirenTarget - sirenFreq_) * 0.0025f * 48000.0f / static_cast<float>(rate_);
874 sirenPhase_ += kTwoPi * sirenFreq_ * dt;
875 const float siren = static_cast<float>(std::tanh(1.6 * std::sin(sirenPhase_)) * 0.8 + 0.25 * std::sin(2 * sirenPhase_)) * sirenNear_ * std::sqrt(sirenNear_);
876 // The fire engine's siren is a slow wail: the pitch rises and falls smoothly over about three and a
877 // half seconds ("uuhh-wwiiww-hhuu"), with a little vibrato and a vowel-like roundness, where the
878 // ambulance's is a quick alternation of two notes.
879 const double sweep = 0.5 - 0.5 * std::cos(kTwoPi * t_ / 3.5);
880 const double wailHz = (430.0 + 520.0 * sweep) * (1.0 + 0.012 * std::sin(kTwoPi * 5.5 * t_));
881 wailPhase_ += kTwoPi * wailHz * dt;
882 const float wail = static_cast<float>(std::tanh(1.3 * std::sin(wailPhase_)) * 0.75 + 0.3 * std::sin(2 * wailPhase_ + 0.7) +
883 0.15 * std::sin(0.5 * wailPhase_)) *
884 wailNear_ * std::sqrt(wailNear_) * (0.8f + 0.2f * static_cast<float>(sweep));
885 // A train: a deep rumble and the clatter of wheels over the rail joints, and a horn as it arrives.
886 // The wheels beat out "klack-klack ... klack-klack": two quick knocks, then a pause, over and over
887 // (the two axles of a bogie crossing a rail joint). Each knock is a woody thunk, not a click.
888 // The wheels only knock while the train rolls, and slower as it slows down; standing at the
889 // platform it is silent, apart from the squeal of the brakes and the hiss of the air as it stops.
890 trainClock_ += dt * trainMove_ * (1.0 + 0.2 * trainNear_);
891 if (trainClock_ >= kKlackPeriod) trainClock_ -= kKlackPeriod;
892 float clack = 0, thunk = 0;
893 for (int k = 0; k < 2; ++k) {
894 const double tau = trainClock_ - k * 0.21;
895 if (tau < 0 || tau > 0.4) continue;
896 const double rise = 1.0 - std::exp(-tau / 0.004);
897 const double gain = k ? 0.8 : 1.0; // the second knock is a little softer
898 clack += static_cast<float>(gain * rise * std::exp(-tau / 0.05));
899 thunk += static_cast<float>(gain * rise * std::exp(-tau / 0.09) * std::sin(kTwoPi * (k ? 105.0 : 92.0) * tau));
900 }
901 const double squealHz = 1500.0 + 900.0 * trainMove_; // the pitch falls as the train slows
902 trainSquealPhase_ += kTwoPi * squealHz * dt;
903 const float squeal = static_cast<float>(std::tanh(1.5 * std::sin(trainSquealPhase_)) * 0.6 + 0.4 * std::sin(trainSquealPhase_ * 1.51 + 0.4)) *
904 trainBrake_ * (0.85f + 0.15f * static_cast<float>(std::sin(kTwoPi * 7.0 * t_))) * trainNear_; // a wavering squeal
905 if (trainNear_ > 0.2f && !trainHorned_) { trainHorned_ = true; trainHornStart_ = t_; }
906 if (trainNear_ < 0.03f) trainHorned_ = false;
907 float trainHorn = 0;
908 const double th = t_ - trainHornStart_;
909 if (th >= 0 && th < 1.4) {
910 const double env = std::min(1.0, th / 0.05) * std::min(1.0, (1.4 - th) / 0.25);
911 trainHornPhase_ += kTwoPi * 311.0 * dt;
912 trainHorn = static_cast<float>(env * (std::tanh(2.0 * std::sin(trainHornPhase_)) + 0.7 * std::tanh(2.0 * std::sin(trainHornPhase_ * 1.26)))) * 0.5f;
913 }
914 // A fire: a low roar and crackling.
915 const bool crackle = rng_.uniform() < std::min(1.0, 90.0 * fireNear_ * dt * 4.0);
916 const float spark = crackle ? static_cast<float>(rng_.range(0.4, 1.0)) * (rng_.chance(0.5) ? 1.0f : -1.0f) : 0.0f;
917 fireCrack_ += (spark - fireCrack_) * 0.35f;
918
919 // The sources the world has a place for sound from where they are: the sirens, the train's knocks and
920 // horn and the fire's sparks are mono sounds, placed at their emitters (a place that is off is silent).
921 // (Switched off once the sound has died away and is not wanted any more; a wanted one that is still
922 // swelling in stays on, so that it is placed from its first moment.)
923 if (sirenNear_ < 1e-3f && !(p.siren > 0)) place_[kSiren].on = false;
924 if (wailNear_ < 1e-3f && !(p.wail > 0)) place_[kWail].on = false;
925 if (trainNear_ < 1e-3f && !(p.train > 0)) place_[kTrain].on = false;
926 if (planeNear_ < 1e-3f && !(p.plane > 0)) place_[kPlane].on = false;
927 if (fireNear_ < 1e-3f && !(p.fire > 0)) place_[kFire].on = false;
928 if (place_[kSiren].on) place_[kSiren].v.add(siren * 0.15f * stereoRef_, room_, nch);
929 if (place_[kWail].on) place_[kWail].v.add(wail * 0.13f * stereoRef_, room_, nch);
930 if (place_[kTrain].on) place_[kTrain].v.add((thunk * 0.12f * trainNear_ * trainMove_ + trainHorn * 0.05f * trainNear_ + squeal * 0.08f) * stereoRef_, room_, nch);
931 if (place_[kFire].on) place_[kFire].v.add(fireCrack_ * 0.3f * fireNear_ * stereoRef_, room_, nch);
932
933 const float mixCity = mix_[static_cast<int>(AmbientKind::City)];
934 // Traffic is the bed, one chain per speaker. The train's rumble and clatter, the fire's roar and the
935 // jet are noise too, but they come from somewhere: each chain plays its own noise (so they stay wide
936 // and unlike one another) weighted by how near the speaker is to the source. The first two chains
937 // always run, so that the draws of the shared generator do not depend on where anything is.
938 for (int ch = 0; ch < chains_; ++ch) {
939 const bool bed = live(AmbientKind::City, ch);
940 const float wTrain = weight(kTrain, ch), wFire = weight(kFire, ch), wPlane = weight(kPlane, ch);
941 const bool train = ch < 2 || wTrain >= 1e-3f, fire = ch < 2 || wFire >= 1e-3f, plane = ch < 2 || wPlane >= 1e-3f;
942 if (!bed && !train && !fire && !plane) continue;
943 Rng& r = chainRng(ch);
944 if (bed) {
945 const float n = noise(r) * noiseScale_;
946 carLow_[ch][0] += (n - carLow_[ch][0]) * lowCar_;
947 carLow_[ch][1] += (carLow_[ch][0] - carLow_[ch][1]) * lowCar_;
948 const float n2 = noise(r) * noiseScale_;
949 carMid_[ch][0] += (n2 - carMid_[ch][0]) * lowCarMid_;
950 carMid_[ch][1] += (n2 - carMid_[ch][1]) * lowCarTop_;
951 out[ch] = carLow_[ch][1] * 9.0f * 0.075f * day + (carMid_[ch][0] - carMid_[ch][1]) * 0.45f * 0.04f * day;
952 }
953 float placed = 0;
954 if (train) {
955 const float n3 = noise(r) * noiseScale_;
956 trainLow_[ch][0] += (n3 - trainLow_[ch][0]) * lowTrain_;
957 trainLow_[ch][1] += (trainLow_[ch][0] - trainLow_[ch][1]) * lowTrain_;
958 const float n4 = noise(r) * noiseScale_;
959 trainBand_[ch][0] += (n4 - trainBand_[ch][0]) * lowTrainBand_;
960 trainBand_[ch][1] += (n4 - trainBand_[ch][1]) * lowKlack_;
961 trainHissLow_[ch] += (n4 - trainHissLow_[ch]) * 0.25f; // what the low-pass leaves of the noise is the hiss
962 placed += wTrain * ((trainLow_[ch][1] * 9.0f * 0.07f * trainNear_ * trainNear_ +
963 (trainBand_[ch][0] - trainBand_[ch][1]) * 1.6f * clack * trainNear_) * trainMove_ +
964 (n4 - trainHissLow_[ch]) * 0.9f * trainHiss_ * trainNear_);
965 }
966 if (fire) {
967 const float n5 = noise(r) * noiseScale_;
968 fireLow_[ch][0] += (n5 - fireLow_[ch][0]) * lowFire_;
969 fireLow_[ch][1] += (fireLow_[ch][0] - fireLow_[ch][1]) * lowFire_;
970 placed += wFire * fireLow_[ch][1] * 9.0f * 0.11f * fireNear_;
971 }
972 if (plane) {
973 const float n6 = noise(r) * noiseScale_;
974 planeLow_[ch][0] += (n6 - planeLow_[ch][0]) * lowJet_;
975 planeLow_[ch][1] += (planeLow_[ch][0] - planeLow_[ch][1]) * lowJet_;
976 placed += wPlane * planeLow_[ch][1] * 9.0f * 0.06f * planeNear_ * planeNear_;
977 }
978 voiceOut_[ch] += placed * mixCity;
979 }
980 flushRoom(voiceOut_, mixCity);
981}

983void Ambience::aquariumSample(const AmbientParams& p, float out[kMaxChannels]) {
984 const double dt = 1.0 / rate_;
985 const int nch = layout_.channels();
986 const bool room3d = nch > 2; // in a room the water is all around: bubbles rise at every side and the glints come from every speaker
987 const float mixAq = mix_[static_cast<int>(AmbientKind::Aquarium)];
988 if (storm_ < 1e-3f && !(p.storm > 0)) place_[kGrowl].on = false;
989 // Slow swells of the water, from the same irregular gusts that move the wind; night is a bit calmer.
990 const float swell = (0.55f + 0.9f * gust_) * (1.0f - 0.25f * night_);
991
992 // Random bubbles rise now and then, sometimes in a little cluster.
993 if (nextBubble_ == 0) nextBubble_ = t_ + 0.5;
994 if (t_ >= nextBubble_) {
995 const int n = rng_.chance(0.25) ? rng_.rangeInt(2, 3) : 1;
996 // In a room a cluster is a string of bubbles that rises through the water (see bubbleString).
997 Source string[3];
998 if (room3d) bubbleString(fxRng_, n, string);
999 for (int i = 0; i < n; ++i) {
1000 bubble(i * rng_.range(0.05, 0.22), static_cast<float>(rng_.range(350, 1100)),
1001 static_cast<float>(rng_.range(0.04, 0.10)), static_cast<float>(rng_.range(0.03, 0.07)), room3d ? &string[i] : nullptr);
1002 }
1003 nextBubble_ = t_ + rng_.range(0.25, 1.4) * (1.0 + 0.8 * night_);
1004 }
1005 for (int i = 0; i < 10; ++i) {
1006 const Bubble& b = bubbles_[i];
1007 const double tau = t_ - b.start;
1008 if (tau > b.dur) place_[kBubble0 + i].on = false;
1009 if (b.amp <= 0 || tau < 0 || tau > b.dur) continue;
1010 // A bubble sings a short rising note: it starts at f0 and climbs as it shrinks.
1011 const double phase = kTwoPi * b.f0 * (tau + 1.1 * tau * tau / b.dur);
1012 const float v = static_cast<float>(std::sin(phase) * (1.0 - std::exp(-tau * 900.0)) *
1013 std::exp(-tau / (b.dur * 0.38))) * b.amp;
1014 if (place_[kBubble0 + i].on) place_[kBubble0 + i].v.add(v, room_, nch); // the blip's own pan law, or its place in the room
1015 }
1016
1017 // A whale passing: a long, gliding moan that repeats while it is near.
1018 float song = 0;
1019 const float whaleLevel = whale_ * std::sqrt(whale_);
1020 if (whale_ > 0.03f) {
1021 if (nextSong_ == 0) nextSong_ = t_ + rng_.range(1.0, 3.0);
1022 const bool singing = t_ - song_.start <= song_.dur && song_.dur > 0;
1023 if (!singing && t_ >= nextSong_) {
1024 song_ = Song{t_, rng_.range(3.0, 5.0), static_cast<float>(rng_.range(70, 110)),
1025 static_cast<float>(rng_.range(0.3, 0.7)), 0};
1026 nextSong_ = t_ + song_.dur + rng_.range(3.0, 8.0) * (1.6 - whale_);
1027 // The phrase starts where the whale is (a caller that has no whale on the stage keeps the old random pan).
1028 const Emitter& whale = p.emitters[static_cast<int>(EmitterKey::Whale)];
1029 place_[kWhale].src = whale.level > 0 ? stageSource(whale.x, whale.y, 0.6f, 0.4f) : panSource(song_.pan);
1030 place_[kWhale].v.setTarget(layout_, forLayout(place_[kWhale].src), true);
1031 place_[kWhale].on = true;
1032 }
1033 } else {
1034 nextSong_ = 0;
1035 }
1036 if (t_ - song_.start <= song_.dur && song_.dur > 0) {
1037 const double tau = t_ - song_.start, x = tau / song_.dur;
1038 const double f = song_.f0 * (1.0 + 0.55 * std::sin(3.14159265 * x) + 0.3 * x +
1039 0.02 * std::sin(kTwoPi * 5.5 * tau));
1040 song_.phase += kTwoPi * f * dt;
1041 const double env = std::sin(3.14159265 * x);
1042 song = static_cast<float>((std::sin(song_.phase) + 0.35 * std::sin(2 * song_.phase + 0.5) +
1043 0.12 * std::sin(3 * song_.phase)) * env * env) * whaleLevel * 0.16f;
1044 if (place_[kWhale].on) place_[kWhale].v.add(song * 1.4f, room_, nch);
1045 } else {
1046 place_[kWhale].on = false;
1047 }
1048
1049 // The growl of something huge (the kraken) as it nears.
1050 const float growl = storm_ * storm_ * 0.5f * std::clamp(0.55f + 0.45f * roll_, 0.1f, 1.0f);
1051 // What its tentacles catch is gulped down: two soft glugs, each a tone that sinks in pitch and
1052 // wobbles a little, heard through the water (low-passed, so it is muffled).
1053 float gulp = 0;
1054 bool gulping = false; // a gulp is under way or about to start
1055 for (const Gulp& g : gulps_) {
1056 const double tau = t_ - g.start;
1057 if (g.used && tau <= 0.5) gulping = true;
1058 if (!g.used || tau < 0 || tau > 0.5) continue;
1059 for (int k = 0; k < 2; ++k) { // the second glug follows a moment later, lower
1060 const double lt = tau - k * 0.2;
1061 if (lt < 0 || lt > 0.17) continue;
1062 const double u = lt / 0.17;
1063 const double hz = g.pitch * (k ? 0.8 : 1.0) * (250.0 - 120.0 * u) * (1.0 + 0.06 * std::sin(kTwoPi * 24.0 * lt));
1064 gulpPhase_ += kTwoPi * hz * dt;
1065 const double env = std::sin(3.14159265 * u);
1066 gulp += static_cast<float>(env * env * (std::sin(gulpPhase_) + 0.45 * std::sin(2 * gulpPhase_ + 0.5)));
1067 }
1068 }
1069 gulpLow_[0] += (gulp - gulpLow_[0]) * lowGulp_;
1070 gulpLow_[1] += (gulpLow_[0] - gulpLow_[1]) * lowGulp_;
1071 if (place_[kGulp].on) {
1072 place_[kGulp].v.add(gulpLow_[1] * 0.45f * stereoRef_, room_, nch); // from the kraken's mouth
1073 if (!gulping && std::fabs(gulpLow_[1]) < 1e-6f && std::fabs(gulpLow_[0]) < 1e-6f) place_[kGulp].on = false;
1074 }
1075
1076 // Hum and flow are the bed, one chain per speaker, with (in a room) the glints of tiny bubbles in
1077 // every speaker. The kraken's growl is placed: noise per chain, weighted by how near each speaker
1078 // is to the kraken. Bubbles, song and gulps are voices, each with its place.
1079 for (int ch = 0; ch < chains_; ++ch) {
1080 if (!live(AmbientKind::Aquarium, ch)) continue;
1081 Rng& r = chainRng(ch);
1082 const float n = noise(r) * noiseScale_;
1083 waterLow_[ch][0] += (n - waterLow_[ch][0]) * lowHum_;
1084 waterLow_[ch][1] += (waterLow_[ch][0] - waterLow_[ch][1]) * lowHum_;
1085 const float n2 = noise(r) * noiseScale_;
1086 muffleLow_[ch] += (n2 - muffleLow_[ch]) * lowMuffle_;
1087 if (ch < 2 || growl > 0) { // (past the stereo pair, whose generator must advance as it always did, a silent layer is not computed)
1088 const float n3 = noise(r) * noiseScale_;
1089 deepLow_[ch][0] += (n3 - deepLow_[ch][0]) * lowDeep_;
1090 deepLow_[ch][1] += (deepLow_[ch][0] - deepLow_[ch][1]) * lowDeep_;
1091 }
1092 if (growl > 0) voiceOut_[ch] += weight(kGrowl, ch) * deepLow_[ch][1] * 20.0f * growl * mixAq;
1093 float glint = 0;
1094 if (room3d) {
1095 Micro& m = micro_[ch];
1096 if (m.env > 1e-4f) {
1097 m.phase += m.inc;
1098 m.inc *= 1.00012f; // it climbs as it shrinks, like the bubbles do
1099 glint = std::sin(m.phase) * m.env;
1100 m.env *= m.decay;
1101 } else if (microRng_[ch].uniform() < 1.5 * dt) { // about one tiny bubble a second and a half in each speaker
1102 m.env = static_cast<float>(microRng_[ch].range(0.008, 0.020)) * (1.0f - 0.25f * night_);
1103 m.inc = static_cast<float>(kTwoPi * microRng_[ch].range(1500.0, 5000.0) / rate_);
1104 m.decay = static_cast<float>(std::exp(-1.0 / (rate_ * microRng_[ch].range(0.008, 0.02))));
1105 m.phase = 0;
1106 }
1107 }
