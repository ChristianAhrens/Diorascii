90void World::blazeStep(Entity& e, double dt) {
91 if (blazeOrigin_ < 0) { // sets one of the front skyscrapers alight, away from the edges
92 std::vector<int> choices;
93 for (int minVisible : {6, 3, 1}) { // a building whose top shows above the front row; the side streets cover their own cells
94 for (size_t i = 0; i < city_.size(); ++i) {
95 const Building& b = city_[i];
96 const int mid = b.x + b.w / 2;
97 bool inStreet = false;
98 for (const Avenue& a : avenues_) inStreet |= b.x < a.cx + a.halfW + 2 && b.x + b.w > a.cx - a.halfW - 2;
99 int hidden = 0; // rows of it that the front row hides
100 if (b.layer < 2) {
101 for (const Building& f : city_) {
102 if (f.layer == 2 && f.x < b.x + b.w && f.x + f.w > b.x) hidden = std::max(hidden, f.h);
103 }
104 }
105 if (b.layer >= 1 && b.h - hidden >= minVisible && !inStreet && b.state == Building::State::Normal && mid > w_ / 8 && mid < w_ - w_ / 8) {
106 choices.push_back(static_cast<int>(i));
107 }
108 }
109 if (!choices.empty()) break;
110 }
111 if (choices.empty()) {
112 e.dead = true; // nothing to burn on this screen
113 return;
114 }
115 blazeOrigin_ = choices[static_cast<size_t>(rng_.rangeInt(0, static_cast<int>(choices.size()) - 1))];
116 blazeBuildings_ = {blazeOrigin_};
117 city_[static_cast<size_t>(blazeOrigin_)].state = Building::State::Burning;
118 city_[static_cast<size_t>(blazeOrigin_)].timer = 0;
119 // Sometimes the fire engine does not win: the fire spreads to 3-5 neighbours and they all burn down.
120 blazeFails_ = forcedOutcome_ >= 0 ? forcedOutcome_ == 1 : rng_.chance(kFireFails);
121 spreadLeft_ = blazeFails_ ? rng_.rangeInt(3, 5) : 0;
122 nextSpread_ = rng_.range(7.0, 10.0);
123 blazeClock_ = 0;
124 fitCordon();
125 if (auto truck = pack_->find("firetruck"); truck && countSprite("firetruck") == 0) spawnCreature(truck, false, true);
126 }
127 blazeClock_ += dt;
128 if (spreadLeft_ > 0 && blazeClock_ >= nextSpread_) { // it jumps to the nearest building that is not burning yet
129 const Building& origin = city_[static_cast<size_t>(blazeOrigin_)];
130 const double from = origin.x + origin.w / 2.0;
131 int best = -1;
132 double bestDist = 1e9;
133 for (size_t i = 0; i < city_.size(); ++i) {
134 const Building& b = city_[i];
135 bool inStreet = false;
136 for (const Avenue& a : avenues_) inStreet |= b.x < a.cx + a.halfW + 2 && b.x + b.w > a.cx - a.halfW - 2;
137 const double dist = std::fabs(b.x + b.w / 2.0 - from);
138 if (b.layer >= 1 && b.state == Building::State::Normal && !inStreet && b.x + b.w > 0 && b.x < w_ && dist < bestDist) {
139 best = static_cast<int>(i);
140 bestDist = dist;
141 }
142 }
143 if (best < 0) {
144 spreadLeft_ = 0;
145 } else {
146 city_[static_cast<size_t>(best)].state = Building::State::Burning;
147 city_[static_cast<size_t>(best)].timer = 0;
148 blazeBuildings_.push_back(best);
149 --spreadLeft_;
150 nextSpread_ += rng_.range(2.5, 4.0);
151 fitCordon();
152 if (cordon_) { // the barriers move out to the new edges
153 liftBarriers();
154 raiseBarriers();
155 }
156 }
157 }
158 const bool allDone = stepBuildings(dt) && spreadLeft_ == 0;
159 if (allDone) { // everything is standing again: the site is open
160 blazeBuildings_.clear();
161 blazeOrigin_ = -1;
162 cordon_ = false;
163 liftBarriers();
164 e.dead = true;
165 }
166}

168bool World::stepBuildings(double dt) {
169 double level = 0;
170 bool allDone = true;
171 for (int idx : blazeBuildings_) {
172 Building& b = city_[static_cast<size_t>(idx)];
173 if (b.state == Building::State::Normal) continue;
174 allDone = false;
175 b.timer += dt;
176 if (b.state == Building::State::Burning) {
177 if (!blazeFails_ && sprayTime_ > 0) b.timer += 0.7 * dt; // the water puts a fire it can beat out sooner
178 const double now = fireAt(b.timer);
179 level = std::max(level, now);
180 smokePuffs_ += 6.0 * now * dt; // smoke billows up from the roof
181 for (; smokePuffs_ >= 1; smokePuffs_ -= 1) {
182 spawnEffect("smoke", b.x + rng_.range(0, std::max(b.w - 3, 1)), seabedY_ - b.h - rng_.range(0, 3), 4,
183 rng_.range(0.5, 2.0), -rng_.range(1.2, 2.6));
184 }
185 if (b.timer >= kBurn) {
186 b.state = blazeFails_ ? Building::State::Ruin : Building::State::Scaffold;
187 b.ruined = blazeFails_;
188 b.timer = 0;
189 }
190 } else if (b.state == Building::State::Ruin) {
191 if (b.timer < 14.0 && rng_.chance(2.0 * (1.0 - b.timer / 14.0) * dt)) { // the ruin smoulders for a while
192 spawnEffect("smoke", b.x + rng_.range(0, std::max(b.w - 3, 1)), seabedY_ - std::max(2, b.h / 6) - rng_.range(0, 2), 4,
193 rng_.range(-0.5, 0.5), -rng_.range(0.8, 1.6));
194 }
195 if (b.timer >= kRuin && !b.hold) { // after a long while the rebuilding starts
196 b.state = Building::State::Scaffold;
197 b.timer = 0;
198 }
199 } else if (b.timer >= (b.ruined ? kRebuild : kScaffold)) { // renovated or rebuilt: new looks, maybe a new height
200 rollBuilding(b);
201 b.state = Building::State::Normal;
202 b.ruined = false;
203 b.timer = 0;
204 }
205 }
206 sprayTime_ = std::max(0.0, sprayTime_ - dt);
207 fireLevel_ = level;
208 return allDone;
209}

227 if (scene_->avenues <= 0 || seabedY_ < 12) return;
228 const int n = std::min(scene_->avenues, w_ / 55);
229 if (n <= 0) return;
230 const int halfW = 6, depth = std::clamp(seabedY_ / 6, 4, 7);
231 auto clear = [&](int c) { return c >= halfW + 4 && c <= w_ - halfW - 5; };
232 for (int i = 0; i < n; ++i) {

236 for (int sign : {1, -1}) {

238 bool apart = true;

250 static const char* const kinds[] = {"car_sedan", "car_hatch", "taxi", "truck"};
251 const int rows = std::max(h_ - seabedY_, 1);
252 for (size_t i = 0; i < avenues_.size(); ++i) {
253 avenueTimer_[i] -= dt;
254 if (avenueTimer_[i] > 0) continue;
255 avenueTimer_[i] = rng_.range(3.5, 9.0);
256 if (!rng_.chance(trafficLevel())) continue; // fewer cars come down the side streets when it is quiet
257 const Avenue& a = avenues_[i];
258 const std::string name = kinds[rng_.rangeInt(0, 3)];
259 Entity* e = spawnEffect(name, 0, 0, 24, 0, 0); // comes down the avenue from far away
260 if (!e) continue;
261 e->driveState = 3;
262 if (e->sprite->frameCount() >= 5) e->frame = 4; // seen from the front
263 e->driveAvenue = static_cast<int>(i);

269 e->driveY0 = seabedY_ - e->height() + feet + 1;
270 e->driveZ = 12 - feet;

272 e->y = (seabedY_ - a.depth + 1) - e->height() + 1; // its feet at the far end of the avenue

276void World::driveStep(Entity& e, double dt) {
277 auto smooth = [](double t) { t = std::clamp(t, 0.0, 1.0); return t * t * (3 - 2 * t); };
278 // Which way the car faces the camera: frame 0 side, 1 and 2 from behind (three-quarters, straight),
279 // 3 and 4 from the front. A sprite with fewer frames keeps its side view.
280 auto setView = [&](size_t frame) { if (e.sprite->frameCount() >= 5) e.frame = frame; };
281 if (e.driveAvenue == -2) { // decide once whether to turn off, and into which street
282 e.driveAvenue = -1;
283 if (e.driveState == 0 && !avenues_.empty() && rng_.chance(e.turnChance)) {
284 const double mid = e.x + e.width() / 2.0;
285 double best = 1e9;
286 for (size_t i = 0; i < avenues_.size(); ++i) {
287 const double ahead = (avenues_[i].cx - mid) * (e.vx > 0 ? 1 : -1);
288 if (ahead > 2 && ahead < best) { best = ahead; e.driveAvenue = static_cast<int>(i); }
289 }
290 }
291 }
292 switch (e.driveState) {
293 case 0: { // along the main road
294 e.x += e.vx * dt;
295 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) { e.dead = true; return; }
296 if (e.driveAvenue >= 0) {
297 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
298 const double mid = e.x + e.width() / 2.0;
299 if ((e.vx > 0 && mid >= a.cx - 3) || (e.vx < 0 && mid <= a.cx + 3)) {
300 e.driveState = 1;
301 e.driveT = 0;
302 e.driveX0 = e.x;
303 e.driveY0 = e.y;
304 e.driveZ = e.z;
305 e.z = 24; // further away than the traffic on the main road, and behind the viaduct
306 }
307 }
308 break;
309 }
310 case 1: { // swinging into the side street
311 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
312 e.driveT += dt;
313 const double k = smooth(e.driveT / 1.1);
314 e.x = e.driveX0 + (a.cx - e.width() / 2.0 - e.driveX0) * k;
315 e.y = e.driveY0 - 2.0 * k;
316 setView(k < 0.3 ? 0 : k < 0.7 ? 1 : 2); // side, then three-quarters from behind, then from behind
317 if (e.driveT >= 1.1) e.driveState = 2;
318 break;
319 }
320 case 2: { // up the avenue, smaller steps as it gets far away, until it is gone
321 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
322 const double feet = e.y + e.height() - 1;
323 const double progress = std::clamp((seabedY_ - 1 - feet) / a.depth, 0.0, 1.0);
324 setView(2);
325 e.y -= 3.6 * (1.0 - 0.6 * progress) * dt;
326 if (e.y + e.height() - 1 <= seabedY_ - a.depth + 1) e.dead = true;
327 break;
328 }
329 case 3: { // coming down the avenue, faster as it gets near
330 const Avenue& a = avenues_[static_cast<size_t>(e.driveAvenue)];
331 const double feet = e.y + e.height() - 1;
332 const double progress = std::clamp((seabedY_ - 1 - feet) / a.depth, 0.0, 1.0); // 1 = far away
333 e.y += (2.0 + 2.2 * (1.0 - progress)) * dt;
334 setView(e.y >= e.driveY0 - 4.0 ? 3 : 4); // facing the camera, then turning to the side
335 if (e.y >= e.driveY0 - 2.0) {
336 e.driveState = 4;
337 e.driveT = 0;
338 e.driveX0 = e.x;
339 }
340 break;
341 }
342 default: { // turning out onto the main road
343 e.driveT += dt;
344 const double k = smooth(e.driveT / 1.1);
345 setView(k < 0.45 ? 3 : 0);
346 e.vx = e.driveDir * e.driveSpeed * k;
347 e.x += e.vx * dt;
348 e.y += (e.driveY0 - e.y) * std::min(1.0, 4.0 * dt);
349 if (e.driveT >= 1.1) {
350 e.driveState = 0;
351 e.driveAvenue = -1;
352 e.vx = e.driveDir * e.driveSpeed;
353 e.y = e.anchorY = e.driveY0;
354 e.z = e.driveZ;
355 }
356 break;
357 }
358 }

361void World::paintAvenue(Frame& out, const Avenue& a, Style style, const Light& light) const {
362 const bool full = style == Style::Full;
363 const Rgb pavement = backdrop(scene_->groundA, light), asphalt = backdrop(scene_->groundB, light);
364 const Rgb haze = backdrop(scene_->skyLow, light);
365 for (int r = 0; r < a.depth; ++r) {
366 const int y = seabedY_ - 1 - r;
367 const double far = static_cast<double>(r) / a.depth;
368 const int hw = std::max(2, static_cast<int>(std::lround(a.halfW * (1.0 - 0.35 * far)))); // a little narrower further back
369 for (int x = a.cx - hw - 1; x <= a.cx + hw + 1; ++x) {
370 Cell* c = cellAt(out, x, y);
371 if (!c) continue;
372 const int dx = x - a.cx;
373 const bool road = std::abs(dx) <= hw;
374 c->ch = road ? ((dx == 0 && r % 2 == 1) ? U':' : U' ') : (dx < 0 ? U'/' : U'\\');
375 c->fg = road ? backdrop(Rgb{225, 200, 90}, light) : scale(pavement, 0.7);
376 c->bg = lerp(road ? asphalt : pavement, haze, 0.3 * far);
377 c->defaultBg = !full;
378 c->defaultFg = style == Style::Mono;
379 }
380 }

404void World::engineStep(Entity& e, double dt) {
405 const bool burning = blazeOrigin_ >= 0 && fireBurning();
406 if (e.driveState == 0 && e.driveSpeed == 0) e.driveSpeed = std::fabs(e.vx); // its cruising speed
407 if (e.driveState == 0) {
408 e.x += e.vx * dt; // on its way to the fire
409 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) { e.dead = true; return; }
410 if (burning) {
411 const Building& b = city_[static_cast<size_t>(blazeOrigin_)];
412 const double target = b.x + b.w / 2.0, mid = e.x + e.width() / 2.0;
413 if ((e.vx > 0 && mid >= target) || (e.vx < 0 && mid <= target)) { // arrived: park and cordon the site off
414 e.driveDir = e.vx > 0 ? 1 : -1;
415 e.vx = 0;
416 e.driveState = 1;
417 cordon_ = true;
418 fitCordon();
419 raiseBarriers();
420 }
421 }
422 } else if (e.driveState == 1) {
423 if (burning) { // fighting the fire: a jet of water arcs up to the facade
424 sprayTime_ = 0.3;
425 const double mine = e.x + e.width() / 2.0;
426 const Building* target = nullptr; // the burning building nearest to it
427 for (int idx : blazeBuildings_) {
428 const Building& cand = city_[static_cast<size_t>(idx)];
429 if (cand.state != Building::State::Burning) continue;
430 if (!target || std::fabs(cand.x + cand.w / 2.0 - mine) < std::fabs(target->x + target->w / 2.0 - mine)) target = &cand;
431 }
432 if (!target) return;
433 const Building& b = *target;
434 const double nx = e.x + e.width() / 2.0, ny = e.y + 1;
435 const double reachTime = 0.55;
436 const int drops = static_cast<int>(40 * dt) + (rng_.chance(40 * dt - static_cast<int>(40 * dt)) ? 1 : 0);
437 for (int i = 0; i < drops; ++i) {
438 const double tx = b.x + b.w / 2.0 + rng_.range(-b.w / 3.0, b.w / 3.0);
439 const double ty = seabedY_ - 1 - b.h * rng_.range(0.3, 0.8);
440 spawnEffect("water", nx, ny, 3, (tx - nx) / reachTime, (ty - ny) / reachTime);
441 }
442 } else { // the fire is out, put out or burnt out: drive on, siren off
443 e.driveState = 2;
444 e.vx = e.driveDir * e.driveSpeed;
445 e.tag.clear(); // it no longer sounds its siren
446 }
447 } else {
448 e.x += e.vx * dt;
449 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) e.dead = true;
450 }
451}

459void World::rollBuilding(Building& b) {
460 const int ground = seabedY_;
461 const int lo = std::max(3, static_cast<int>(scene_->skyline.first * ground));
462 const int hi = std::clamp(static_cast<int>(scene_->skyline.second * ground), lo, std::max(ground - 2, lo));
463 b.roof = rng_.rangeInt(0, 5);
464 b.antenna = b.roof == 1 ? rng_.rangeInt(2, 4) : (b.roof == 3 || b.roof == 4) ? 2 : 0;
465 // Heights: the far row is tall and pale, the middle row mixes low offices, mid-rise and towers,
466 // the front row keeps to low and mid-rise buildings.
467 const double span = hi - lo;
468 double share;
469 if (b.layer == 0) share = rng_.range(0.35, 1.0);
470 else if (b.layer == 1) {
471 const double r = rng_.uniform();
472 share = r < 0.3 ? rng_.range(0.0, 0.3) : r < 0.7 ? rng_.range(0.25, 0.6) : rng_.range(0.6, 1.0);
473 } else {
474 share = rng_.range(0.0, 0.55);
475 }
476 const int room = std::max(ground - 1 - b.antenna, 3);
477 b.h = std::clamp(lo + static_cast<int>(share * span), 3, room);
478 b.tint = rng_.rangeInt(0, kWallCount - 1);
479 b.facade = rng_.rangeInt(0, 3);
480 b.seed = static_cast<uint32_t>(rng_.rangeInt(1, 1 << 24));
481}

483void World::buildCity() {
484 city_.clear();
485 glimpses_.clear();
486 railY_ = -1;
487 if (seabedY_ < 8) return; // too small for a skyline
488 if (seabedY_ >= 16) railY_ = seabedY_ - 7;
489 // The front row keeps out of the side streets; this is the column where the way is free again.
490 auto gapEnd = [&](int x0, int x1) {
491 for (const Avenue& a : avenues_) {
492 if (x1 > a.cx - a.halfW - 1 && x0 < a.cx + a.halfW + 2) return a.cx + a.halfW + 2;
493 }
494 return -1;
495 };
496 for (int layer = 0; layer < 3; ++layer) {
497 int x = -rng_.rangeInt(0, 6);
498 while (x < w_) {
499 Building b;
500 b.layer = layer;
501 b.w = layer == 0 ? rng_.rangeInt(9, 20) : layer == 1 ? rng_.rangeInt(6, 15) : rng_.rangeInt(5, 12);
502 if (layer == 2) {
503 for (int end = gapEnd(x, x + b.w); end >= 0; end = gapEnd(x, x + b.w)) x = end;
504 if (x >= w_) break;
505 }
506 b.x = x;
507 rollBuilding(b);
508 city_.push_back(b);
509 // The far row overlaps so that there are no holes in the sky; the front row has gaps in places.
510 x += b.w + (layer == 0 ? rng_.rangeInt(-2, 2) : layer == 1 ? rng_.rangeInt(1, 4) : (rng_.chance(0.4) ? rng_.rangeInt(6, 12) : rng_.rangeInt(2, 6)));
511 }
512 }
513 // Where the front row has a gap, another side street can be seen running back behind it.
514 std::vector<bool> covered(static_cast<size_t>(w_), false);
515 for (const Building& b : city_) {
516 if (b.layer != 2) continue;
517 for (int x = std::max(b.x, 0); x < std::min(b.x + b.w, w_); ++x) covered[static_cast<size_t>(x)] = true;
518 }
519 const int most = std::max(1, w_ / 55);
520 for (int x = 0; x < w_ && static_cast<int>(glimpses_.size()) < most;) {
521 if (covered[static_cast<size_t>(x)]) { ++x; continue; }
522 int end = x;
523 while (end < w_ && !covered[static_cast<size_t>(end)]) ++end;
524 const int width = end - x, cx = (x + end) / 2;
525 bool near = false;
526 for (const Avenue& a : avenues_) near |= std::abs(a.cx - cx) < a.halfW + 8;
527 for (const Avenue& g : glimpses_) near |= std::abs(g.cx - cx) < 14;
528 if (width >= 6 && !near && x > 0 && end < w_) glimpses_.push_back({cx, std::min(3, width / 2 - 1), std::clamp(seabedY_ / 4, 4, 9)});
529 x = end + 1;
530 }

538void World::renderSkyline(Frame& out, Style style, const Light& light) const {
539 const int base = seabedY_ - 1; // the lowest sky row: where the buildings stand
540 const bool full = style == Style::Full;
541 // How many windows are lit: none by day, more and more as it gets dark.
542 const double dark = std::clamp((0.8 - light.sun) / 0.6, 0.0, 1.0);
543 const double lightsOn = 0.58 * dark;
544 const int bucket = static_cast<int>(time_ / 14.0); // a few windows switch on and off now and then
545
546 bool glimpsed = false;
547 for (const Building& b : city_) {
548 if (!glimpsed && b.layer == 2) { // streets seen between the front buildings are behind them
549 for (const Avenue& g : glimpses_) paintAvenue(out, g, style, light);
550 glimpsed = true;
551 }
552 Rgb wall = kWalls[b.tint];
553 if (b.layer == 0) wall = lerp(wall, scene_->skyLow, 0.45); // far away: pale
554 else if (b.layer == 1) wall = lerp(wall, scene_->skyLow, 0.18); // in the middle: a little hazy
555 const Rgb wallLit = backdrop(wall, light);
556 const Rgb glass = backdrop(lerp(Rgb{128, 176, 214}, wall, b.layer == 0 ? 0.5 : 0.3), light);
557 // A building that burns down falls in; one that is rebuilt from the rubble grows back up.
558 const int rubble = rubbleRows(b.h);
559 int shown = b.h;
560 if (b.state == Building::State::Ruin) shown = rubble + static_cast<int>((b.h - rubble) * std::max(0.0, 1.0 - b.timer / kCollapse));
561 else if (b.state == Building::State::Scaffold && b.ruined) shown = rubble + static_cast<int>((b.h - rubble) * std::min(1.0, b.timer / kRebuild));
562 const int top = base - shown + 1;
563 if (b.state == Building::State::Ruin) { // charred rubble with embers that fade
564 const double embers = std::max(0.0, 1.0 - b.timer / 24.0);
565 for (int y = top; y <= base; ++y) {
566 for (int x = b.x; x < b.x + b.w; ++x) {
567 Cell* c = cellAt(out, x, y);
568 if (!c) continue;
569 const uint32_t hv = hash2(x + static_cast<int>(b.seed % 97), y);
570 const bool jagged = y == top && hv % 3 == 0; // a ragged top edge
571 if (jagged) continue;
572 const bool glow = (hv >> 9) % 100 < 22 * embers;
573 c->ch = full ? (hv % 4 == 0 ? U'%' : U'#') : U'#';
574 c->fg = glow ? Rgb{255, 120, 40} : backdrop(Rgb{96, 80, 70}, light);
575 c->defaultFg = style == Style::Mono;
576 if (full) {
577 c->bg = glow ? lerp(Rgb{70, 30, 20}, Rgb{150, 50, 20}, ((hv >> 4) % 10) / 10.0) : backdrop(Rgb{40, 36, 34}, light);
578 c->defaultBg = false;
579 } else {
580 c->defaultBg = true;
581 }
582 }
583 }
584 continue;
585 }
586 for (int y = top; y <= base; ++y) {
587 // A stepped roof is narrower in its top two rows.
588 const int inset = (b.roof == 2 && y < top + 2) ? 2 : 0;
589 for (int x = b.x + inset; x < b.x + b.w - inset; ++x) {
590 Cell* c = cellAt(out, x, y);
591 if (!c) continue;
592 const int cx = x - b.x, cy = base - y; // column in the building, rows above the street
593 const bool edge = x == b.x + inset || x == b.x + b.w - 1 - inset;
594 const bool roofLine = y == top;
595 bool window = false;
596 if (!edge && !roofLine && cy >= 1 && y > top + 1) {
597 switch (b.facade) {
598 case 0: window = cx % 2 == 1 && cy % 2 == 1; break; // a grid of windows
599 case 1: window = cx % 3 != 0 && cy % 4 != 0; break; // a glass curtain wall
600 case 2: window = cy % 3 == 1; break; // ribbons of windows
601 default: window = cx % 2 == 0 && cy % 2 == 1; break; // small windows
602 }
603 }
604 bool lit = false;
605 if (window) {
606 const uint32_t h = hash2(x + static_cast<int>(b.seed % 997), y);
607 lit = (h % 1000) < lightsOn * 1000.0;
608 if (lightsOn > 0.05 && (hash2(x, y + bucket * 31) % 100) < 3) lit = !lit; // someone turns a light on or off
609 }
610 if (full) {
611 Rgb bg = wallLit;
612 if (x >= b.x + b.w - 2 - inset) bg = scale(bg, 0.84); // the far side in shade
613 if (roofLine) bg = scale(wallLit, 0.7);
614 if (window) bg = lit ? kLit : glass;
615 c->bg = bg;
616 c->ch = U' ';
617 c->defaultBg = false;
618 } else {
619 c->defaultBg = true;
620 c->ch = roofLine ? U'_' : edge ? U'|' : window ? (lit ? U'o' : U'.') : U' ';
621 c->fg = window && lit ? kLit : scale(wall, 1.1);
622 c->defaultFg = style == Style::Mono;
623 }
624 }
625 }
626 if (b.state == Building::State::Burning) {
627 const double level = fireAt(b.timer);
628 const int flameRows = std::max(2, static_cast<int>(b.h * 0.6 * level));
629 const int tick = static_cast<int>(time_ * 9);
630 for (int y = top - 2; y < top + flameRows; ++y) {
631 for (int x = b.x; x < b.x + b.w; ++x) {
632 Cell* c = cellAt(out, x, y);
633 if (!c || y > base) continue;
634 const uint32_t hv = hash2(x + tick * 13, y);
635 const bool above = y < top; // flames reaching over the roof
636 const double chance = above ? 0.55 * level * (1.0 - 0.4 * (top - y)) : (0.75 - 0.4 * (y - top) / flameRows) * level;
637 if ((hv % 100) >= chance * 100) continue;
638 const double heat = ((hv >> 8) % 100) / 100.0;
639 static const char32_t kFlame[] = {U'^', U'*', U'"', U'\'', U'('};
640 c->ch = kFlame[(hv >> 16) % 5];
641 c->fg = lerp(Rgb{255, 150, 40}, Rgb{255, 235, 110}, heat);
642 c->defaultFg = style == Style::Mono;
643 if (full && !above) {
644 c->bg = lerp(Rgb{130, 24, 10}, Rgb{250, 120, 30}, heat);
645 c->defaultBg = false;
646 }
647 }
648 }
649 if (full) { // the windows below glow red
650 for (int y = top + flameRows; y < top + static_cast<int>(b.h * 0.85); ++y) {
651 for (int x = b.x + 1; x < b.x + b.w - 1; ++x) {
652 Cell* c = cellAt(out, x, y);
653 if (c && (x - b.x) % 2 == 1 && (base - y) % 2 == 1 && (hash2(x, y + tick) % 3) == 0) c->bg = Rgb{235, 70, 35};
654 }
655 }
656 }
657 } else if (b.state == Building::State::Scaffold) {
658 const Rgb pole = backdrop(Rgb{168, 124, 72}, light);
659 for (int y = top; y <= base; ++y) {
660 for (int x = b.x; x < b.x + b.w; ++x) {
661 Cell* c = cellAt(out, x, y);
662 if (!c) continue;
663 if (full) c->bg = scale(c->bg, 0.7);
664 if (y % 3 == 0 || (x - b.x) % 4 == 0) {
665 c->ch = y % 3 == 0 ? U'=' : U'|';
666 c->fg = pole;
667 c->defaultFg = style == Style::Mono;
668 }
669 }
670 }
671 const int cx = b.x + b.w / 2; // a crane on the roof
672 for (int k = 1; k <= 3; ++k) {
673 if (Cell* c = cellAt(out, cx, top - k)) { c->ch = U'|'; c->fg = pole; c->defaultFg = style == Style::Mono; }
674 }
675 for (int k = -5; k <= 3; ++k) {
676 if (Cell* c = cellAt(out, cx + k, top - 3)) { c->ch = k == 3 ? U'v' : U'-'; c->fg = pole; c->defaultFg = style == Style::Mono; }
677 }
678 }
679 // What stands on the roof: an antenna with a red light that blinks at night, a crown, a water
680 // tower or a helipad.
681 if (b.roof == 1 && b.antenna > 0) {
682 const int ax = b.x + b.w / 2;
683 for (int k = 1; k <= b.antenna; ++k) {
684 Cell* c = cellAt(out, ax, top - k);
685 if (!c) continue;
686 const bool tip = k == b.antenna;
687 c->ch = tip ? U'*' : U'|';
688 const bool blink = std::sin(time_ * 3.0) > 0;
689 c->fg = tip ? (blink || light.sun > 0.5 ? Rgb{235, 60, 50} : Rgb{120, 40, 40}) : backdrop(Rgb{150, 152, 160}, light);
690 c->defaultFg = style == Style::Mono;
691 }
692 } else if (b.roof == 3 || b.roof == 4) {
693 const Rgb trim = scale(wallLit, 0.65);
694 auto block = [&](int x0, int x1, int y, char32_t ch) {
695 for (int x = x0; x < x1; ++x) {
696 Cell* c = cellAt(out, x, y);
697 if (!c) continue;
698 c->ch = full ? U' ' : ch;
699 c->bg = trim;
700 c->defaultBg = !full;
701 c->fg = scale(wall, 1.1);
702 c->defaultFg = style == Style::Mono;
703 }
704 };
705 if (b.roof == 3) { // a crown of two narrowing steps
706 block(b.x + 2, b.x + b.w - 2, top - 1, U'_');
707 block(b.x + b.w / 2 - 1, b.x + b.w / 2 + 1, top - 2, U'^');
708 } else { // a water tower on legs
709 const int tx = b.x + std::max(1, b.w / 2 - 2);
710 block(tx, tx + 4, top - 2, U'_');
711 block(tx, tx + 4, top - 1, U'#');
712 }
713 } else if (b.roof == 5 && b.w >= 6) { // a helipad
714 if (Cell* c = cellAt(out, b.x + b.w / 2, top)) {
715 c->ch = U'H';
716 c->fg = backdrop(Rgb{245, 245, 245}, light);
717 c->defaultFg = style == Style::Mono;
718 }
719 }
720 }

723void World::renderStreet(Frame& out, Style style, const Light& light) const {
724 const bool full = style == Style::Full;
725 const int rows = h_ - seabedY_;
726 const Rgb pavement = backdrop(scene_->groundA, light), asphalt = backdrop(scene_->groundB, light);
727
728 // The elevated railway: catenary masts and a wire above the train, a dark deck with two
729 // bright rails, a concrete beam under it, and heavy pillars down to the street.
730 if (railY_ >= 0) {
731 const Rgb steel = backdrop(Rgb{215, 218, 228}, light);
732 const Rgb deck = backdrop(Rgb{58, 60, 70}, light);
733 const Rgb beam = backdrop(Rgb{134, 132, 126}, light);
734 const Rgb mast = backdrop(Rgb{92, 94, 104}, light);
735 for (int x = 0; x < w_; ++x) {
736 Cell& rail = out.at(x, railY_); // the rails
737 rail.ch = U'=';
738 rail.fg = steel;
739 rail.bg = deck;
740 rail.defaultFg = style == Style::Mono;
741 rail.defaultBg = !full;
742 Cell& under = out.at(x, railY_ + 1); // the beam under the deck
743 under.ch = U'_';
744 under.fg = scale(beam, 0.7);
745 under.bg = beam;
746 under.defaultFg = style == Style::Mono;
747 under.defaultBg = !full;
748 if (railY_ - 5 >= 0 && x % 2 == 0) { // the wire the train draws its current from
749 Cell& wire = out.at(x, railY_ - 5);
750 wire.ch = U'-';
751 wire.fg = mast;
752 wire.defaultFg = style == Style::Mono;
753 }
754 if (x % 14 == 3) { // a mast
755 for (int y = std::max(railY_ - 5, 0); y < railY_; ++y) {
756 Cell& m = out.at(x, y);
757 m.ch = U'|';
758 m.fg = mast;
759 m.defaultFg = style == Style::Mono;
760 }
761 }
762 if (x % 12 < 2) { // a pillar, two columns wide
763 for (int y = railY_ + 2; y < seabedY_; ++y) {
764 Cell& p = out.at(x, y);
765 p.ch = full ? U' ' : U'#';
766 p.bg = scale(beam, y % 2 ? 0.9 : 1.0);
767 p.fg = scale(beam, 0.7);
768 p.defaultBg = !full;
769 p.defaultFg = style == Style::Mono;
770 }
771 }
772 }
773 }
774
775 // The street: a sidewalk behind, the road with its lane line, a sidewalk in front.
776 for (int g = 0; g < rows; ++g) {
777 const int y = seabedY_ + g;
778 const bool walk = g == 0 || g == rows - 1;
779 for (int x = 0; x < w_; ++x) {
780 Cell c;
781 const uint32_t h = hash2(x, y);
782 if (walk) {
783 c.bg = pavement;
784 c.ch = (h % 7 == 0) ? scene_->groundGlyphs.empty() ? U' ' : scene_->groundGlyphs[h % scene_->groundGlyphs.size()] : (x % 5 == 0 ? U'_' : U' ');
785 c.fg = scale(pavement, 0.75);
786 } else {
787 c.bg = asphalt;
788 c.ch = U' ';
789 c.fg = scale(asphalt, 1.5);
790 // The lane line: along the upper edge of the lower lane if the road is two rows, else down the middle.
791 const bool lane = rows <= 4 ? g == 2 : g == 1 + (rows - 2) / 2;
792 if (lane && (x % 6) < 3) {
793 c.ch = rows <= 4 ? U'_' : U'-';
794 c.fg = backdrop(Rgb{225, 200, 90}, light);
795 }
796 }
797 if (!full) c.defaultBg = true;
798 if (style == Style::Mono) c.defaultFg = true;
799 out.at(x, y) = c;
800 }
801 }
802 // Where a side street meets the main street: the sidewalk gives way to the road, and a zebra
803 // crossing runs across the side street.
804 for (const Avenue& a : avenues_) {
805 for (int x = a.cx - a.halfW - 1; x <= a.cx + a.halfW + 1; ++x) {
806 Cell* c = cellAt(out, x, seabedY_);
807 if (!c) continue;
808 const bool edge = std::abs(x - a.cx) > a.halfW;
809 c->ch = edge ? (x < a.cx ? U'/' : U'\\') : U' ';
810 c->fg = scale(pavement, 0.7);
811 c->bg = edge ? pavement : asphalt;
812 c->defaultBg = !full;
813 c->defaultFg = style == Style::Mono;
814 }
815 for (int x = a.cx - a.halfW; x <= a.cx + a.halfW; ++x) { // the zebra crossing, one row up
816 Cell* c = cellAt(out, x, seabedY_ - 1);
817 if (!c) continue;
818 const bool stripe = (x - a.cx) % 2 == 0;
819 c->ch = stripe ? U'=' : U' ';
820 c->fg = backdrop(Rgb{232, 232, 236}, light);
821 c->bg = stripe ? scale(backdrop(Rgb{210, 210, 216}, light), 0.55) : asphalt;
822 c->defaultBg = !full;
823 c->defaultFg = style == Style::Mono;
824 }
825 }
826 // Zebra crossings here and there.
827 for (int x = 4; x + 3 < w_; x += 29) {
828 const int at = x + static_cast<int>(hash2(x, 3) % 9);
829 for (int g = 1; g < rows - 1; ++g) {
830 for (int k = 0; k < 3; ++k) {
831 if (at + k >= w_) continue;
832 Cell& c = out.at(at + k, seabedY_ + g);
833 c.ch = U'=';
834 c.fg = backdrop(Rgb{210, 210, 215}, light);
835 c.defaultFg = style == Style::Mono;
836 }
837 }
838 }
839
840 // Street lamps along the back sidewalk, lit from dusk on, with a warm pool of light below.
841 const double on = std::clamp((0.55 - light.sun) / 0.3, 0.0, 1.0);
842 for (int lx = kLampFirst; lx < w_; lx += kLampGap) {
843 bool inStreet = false; // no lamp stands in the mouth of a side street
844 for (const Avenue& a : avenues_) inStreet |= std::abs(lx - a.cx) <= a.halfW + 2;
845 if (inStreet) continue;
846 const int topY = seabedY_ - kLampHeight + 1;
847 for (int y = topY + 1; y <= seabedY_; ++y) {
848 if (Cell* c = cellAt(out, lx, y)) {
849 c->ch = U'|';
850 c->fg = backdrop(Rgb{96, 98, 108}, light);
851 c->defaultFg = style == Style::Mono;
852 }
853 }
854 if (Cell* c = cellAt(out, lx, topY)) {
855 c->ch = U'O';
856 c->fg = lerp(Rgb{150, 150, 150}, kLit, on);
857 c->defaultFg = style == Style::Mono;
858 }
859 if (full && on > 0.02) {
860 for (int y = topY + 1; y < h_; ++y) {
861 for (int x = lx - 6; x <= lx + 6; ++x) {
862 Cell* c = cellAt(out, x, y);
863 if (!c) continue;
864 const double dx = (x - lx) / 6.0, dy = (y - topY) / static_cast<double>(std::max(h_ - topY, 1));
865 const double k = 1.0 - std::sqrt(dx * dx + dy * dy * 0.6);
866 if (k > 0) c->bg = lerp(c->bg, Rgb{255, 220, 140}, 0.4 * k * on);
867 }
868 }
869 }
870 }
