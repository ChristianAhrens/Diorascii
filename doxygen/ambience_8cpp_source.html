24float cricket(double t, double burstHz, double pulseHz, double carrierHz) {
25 const double burst = std::fmod(t * burstHz, 1.0);
26 if (burst > 0.35) return 0.0f; // silent between bursts
27 const double pulse = std::fmod(t * pulseHz, 1.0);
28 const double env = pulse < 0.6 ? std::sin(pulse / 0.6 * kTwoPi / 2.0) : 0.0;
29 return static_cast<float>(env * std::sin(kTwoPi * carrierHz * t));
30}

33void Ambience::clap(double delay, double gain, bool strike) {
34 Thunder& th = *std::min_element(std::begin(thunder_), std::end(thunder_),
35 [](const Thunder& a, const Thunder& b) { return a.start < b.start; });
36 th.start = t_ + delay;
37 th.gain = gain;
38 th.strike = strike;
39 // The rumble of a strike overhead rolls in after the crack has cleared.
40 const double lag = strike ? 0.32 : 0.0;
41 th.swellAt[0] = lag;
42 th.swellAt[1] = lag + rng_.range(0.6, 1.4);
43 th.swellAt[2] = lag + rng_.range(1.6, 2.8);
44 th.swellGain[0] = 1.0;
45 th.swellGain[1] = rng_.range(0.4, 0.8);
46 th.swellGain[2] = rng_.range(0.2, 0.5);
47}

49void Ambience::thunderLevels(float& crack, float& rumble, float& snap) const {
50 crack = rumble = snap = 0;
51 for (const Thunder& th : thunder_) {
52 const double t = t_ - th.start;
53 if (th.gain <= 0 || t < 0 || t > 9) continue;
54 crack += static_cast<float>(th.gain * std::exp(-t * 22.0)); // the sharp first crack
55 if (th.strike) { // two snaps, matching the two quick flashes of the lightning
56 snap += static_cast<float>(th.gain * std::exp(-t * 17.0));
57 if (t > 0.22) snap += static_cast<float>(0.6 * th.gain * std::exp(-(t - 0.22) * 20.0));
58 }
59 for (int i = 0; i < 3; ++i) { // rolling swells, each slowly fading
60 const double s = t - th.swellAt[i];
61 if (s > 0) rumble += static_cast<float>(th.swellGain[i] * th.gain * (1 - std::exp(-s * 12.0)) * std::exp(-s * 0.7));
62 }
63 }

147void Ambience::moonSample(const AmbientParams& p, float out[2]) {
148 (void)p;
149 const double dt = 1.0 / rate_;
150 // No air, so nothing to hear but what could be felt through the ground: a faint, deep drone of
151 // stacked fifths that drifts slowly, with a hint of shimmer above it.
152 const double drift = 0.7 + 0.3 * std::sin(kTwoPi * 0.043 * t_);
153 const double shimmer = 0.5 + 0.5 * std::sin(kTwoPi * 0.11 * t_ + 0.8);
154 float pad[2];
155 for (int ch = 0; ch < 2; ++ch) {
156 const double o = ch ? 0.9 : 0.0;
157 pad[ch] = static_cast<float>(
158 drift * (0.5 * std::sin(kTwoPi * 55.0 * t_ + o) + 0.35 * std::sin(kTwoPi * 82.5 * t_ + 1.0 + o)) +
159 0.18 * std::sin(kTwoPi * 220.0 * t_ + 2.0 + o) * drift +
160 0.07 * shimmer * std::sin(kTwoPi * 330.7 * t_ + o) + 0.03 * (1 - shimmer) * std::sin(kTwoPi * 441.0 * t_ + o)) * 0.06f;
161 }
162
163 // A meteorite strike: a thud that thumps up out of the ground and falls away, a dull rumble,
164 // and then debris pattering down slowly in the low gravity.
165 float thud = 0, rumble = 0, patter = 0, click = 0;
166 for (const Impact& im : impacts_) {
167 const double tau = t_ - im.start;
168 if (im.gain <= 0 || tau < 0 || tau > 9) continue;
169 const double sweep = 34.0 * tau + 70.0 * (1.0 - std::exp(-9.0 * tau)) / 9.0; // 104 Hz sinking to 34 Hz
170 thud += static_cast<float>(std::sin(kTwoPi * sweep) * (1.0 - std::exp(-tau * 400.0)) * std::exp(-tau * 3.2)) * im.gain;
171 rumble += static_cast<float>(im.gain * std::exp(-tau * 1.3) * (1.0 - std::exp(-tau * 30.0)));
172 if (tau > 0.5) {
173 const double density = 60.0 * std::exp(-(tau - 0.5) * 0.8); // clicks per second
174 if (rng_.uniform() < density * dt) click += static_cast<float>(rng_.range(0.4, 1.0) * (rng_.chance(0.5) ? 1 : -1)) * im.gain;
175 }
176 }
177 patter = click;
178
179 // The alien: an eerie rising warble.
180 float zap = 0;
181 const double zt = t_ - zapStart_;
182 if (zapStart_ > 0 && zt >= 0 && zt < 0.9) {
183 const double f = 350.0 + 1000.0 * (zt / 0.9) + 120.0 * std::sin(kTwoPi * 17.0 * zt);
184 zapPhase_ += kTwoPi * f * dt;
185 const double env = std::pow(std::sin(3.14159265 * zt / 0.9), 1.5);
186 zap = static_cast<float>((std::sin(zapPhase_) + 0.3 * std::sin(1.5 * zapPhase_)) * env) * 0.12f;
187 }
188
189 const float debrisF = 2.0f * std::sin(3.14159265f * std::min(900.0f, static_cast<float>(rate_) / 7.0f) / static_cast<float>(rate_));
190 for (int ch = 0; ch < 2; ++ch) {
191 const float n = noise(rng_) * noiseScale_;
192 moonLow_[ch][0] += (n - moonLow_[ch][0]) * lowMoon_;
193 moonLow_[ch][1] += (moonLow_[ch][0] - moonLow_[ch][1]) * lowMoon_;
194 // Debris: each click rings a short, dull resonance.
195 debrisLow_[ch] += debrisF * debrisBand_[ch];
196 const float high = patter * 6.0f - debrisLow_[ch] - 0.6f * debrisBand_[ch];
197 debrisBand_[ch] += debrisF * high;
198 out[ch] = pad[ch] + thud * 0.55f + moonLow_[ch][1] * 14.0f * rumble * 0.35f + debrisBand_[ch] * 0.3f + zap;
199 }
200}

202void Ambience::mountainSample(const AmbientParams& p, float out[2]) {
203 (void)p;
204 const double dt = 1.0 / rate_;
205 // Thin alpine wind. In a snowstorm it is much stronger and higher, with a whistle that glides
206 // with the gusts and the fine hiss of ice crystals on top.
207 const float cutoffHz = (300.0f + 360.0f * gust_) * (1.0f + 1.3f * rain_);
208 const float lowWind = 1.0f - std::exp(-6.2831853f * cutoffHz / static_cast<float>(rate_));
209 const float windGain = (0.35f + 0.8f * gust_) * (1.0f + 5.0f * rain_) * 0.075f;
210 whistlePhase_ += kTwoPi * (620.0 + 340.0 * gust_ + 60.0 * std::sin(0.4 * t_)) * dt;
211 const float whistle = rain_ * rain_ * gust_ * 0.02f * static_cast<float>(std::sin(whistlePhase_));
212 const float hiss = rain_ * (0.1f + 0.2f * gust_);
213 // The stream: noise between two low-passes, quietly, that bubbles and gurgles: it is broken up by
214 // a few unrelated fast ripples so that it does not sound like steady brushing.
215 const float babble = std::clamp(0.35f + 0.65f * static_cast<float>(
216 std::sin(kTwoPi * 6.1 * t_ + 2.0 * std::sin(kTwoPi * 0.7 * t_)) *
217 std::sin(kTwoPi * 9.7 * t_ + 1.3 * std::sin(kTwoPi * 1.3 * t_ + 1.0))), 0.0f, 1.0f);
218 // An eagle's cry now and then, by day only and not in a snowstorm: a few shrill, falling notes.
219 const float day = (1.0f - night_) * (1.0f - std::min(1.0f, 2.0f * rain_));
220 if (nextCry_ == 0) nextCry_ = t_ + rng_.range(5.0, 15.0);
221 if (t_ >= nextCry_) {
222 cryStart_ = t_;
223 cryPan_ = static_cast<float>(rng_.range(0.2, 0.8));
224 cryPitch_ = static_cast<float>(rng_.range(0.9, 1.12));
225 nextCry_ = t_ + rng_.range(15.0, 35.0);
226 }
227 float cry = 0;
228 const double ct = t_ - cryStart_;
229 if (ct >= 0 && ct < 1.3) {
230 const double syl = std::fmod(ct, 0.42), n = std::floor(ct / 0.42); // three notes, 0.42 s apart
231 if (n < 3 && syl < 0.3) {
232 const double x = syl / 0.3;
233 const double f = cryPitch_ * (3400.0 - 1300.0 * x) * (1.0 - 0.05 * n) + 60.0 * std::sin(kTwoPi * 38.0 * syl);
234 cryPhase_ += kTwoPi * f * dt;
235 const double env = std::sin(3.14159265 * x) * (1.0 - 0.18 * n);
236 cry = static_cast<float>((std::sin(cryPhase_) + 0.5 * std::sin(2 * cryPhase_ + 0.4) + 0.3 * noise(rng_)) * env) * 0.09f * day;
237 }
238 }
239 // A woodpecker drums on a tree now and then, as often as the eagle cries: a quick run of dry,
240 // woody knocks that dies away a little. Same rules: by day, not in a snowstorm.
241 if (nextPeck_ == 0) nextPeck_ = t_ + rng_.range(8.0, 20.0);
242 if (t_ >= nextPeck_) {
243 peckStart_ = t_;
244 peckKnocks_ = rng_.rangeInt(7, 13);
245 peckPan_ = static_cast<float>(rng_.range(0.15, 0.85));
246 peckPitch_ = static_cast<float>(rng_.range(0.85, 1.2));
247 nextPeck_ = t_ + rng_.range(15.0, 35.0);
248 }
249 float peck = 0;
250 const double pt = t_ - peckStart_;
251 constexpr double kKnockHz = 16.0;
252 if (pt >= 0 && pt < peckKnocks_ / kKnockHz) {
253 const int k = static_cast<int>(pt * kKnockHz);
254 const double lt = pt - k / kKnockHz; // time since this knock began
255 const double amp = 1.0 - 0.5 * k / peckKnocks_;
256 peck = static_cast<float>((std::sin(kTwoPi * 1150.0 * peckPitch_ * lt) * std::exp(-lt * 150.0) +
257 0.5 * std::sin(kTwoPi * 2300.0 * peckPitch_ * lt) * std::exp(-lt * 260.0) +
258 0.3 * noise(rng_) * std::exp(-lt * 500.0)) * amp) * 0.12f * day;
259 }
260 // An avalanche: when the slab breaks and while it tumbles, snow, rock and trees knock against
261 // each other: a burst of crackling at the start, then clicks and now and then a dull thump with
262 // a crack when something big goes down.
263 if (p.avalanche > 0.7f && !slideBroke_) { slideBroke_ = true; breakEnv_ = 1.0f; thumpEnv_ = 1.0f; thumpFreq_ = 60.0f; }
264 if (p.avalanche < 0.3f) slideBroke_ = false;
265 breakEnv_ -= breakEnv_ * static_cast<float>(dt / 1.1);
266 if (p.avalanche > 0.5f && rng_.chance(1.3 * p.avalanche * dt)) {
267 thumpEnv_ = static_cast<float>(rng_.range(0.4, 0.9));
268 thumpFreq_ = static_cast<float>(rng_.range(55, 120));
269 }
270 thumpEnv_ -= thumpEnv_ * static_cast<float>(dt * 7.0);
271 thumpPhase_ += kTwoPi * thumpFreq_ * dt;
272 const float thump = static_cast<float>(std::sin(thumpPhase_)) * thumpEnv_ * 0.35f;
273 const double clickRate = 220.0 * breakEnv_ + 18.0 * p.avalanche * p.avalanche + 260.0 * thumpEnv_;
274 float click = 0;
275 if (clickRate > 0 && rng_.uniform() < clickRate * dt) click = static_cast<float>(rng_.range(0.4, 1.0)) * (rng_.chance(0.5) ? 1.0f : -1.0f);
276 const float crackF = 2.0f * std::sin(3.14159265f * std::min(1800.0f, static_cast<float>(rate_) / 7.0f) / static_cast<float>(rate_));
277 // An avalanche: a deep rumble that swells as it gets going, and the roar and crackle of tumbling
278 // snow and rock on top of it.
279 const float slide = avalanche_;
280 const float roll = std::clamp(0.7f + 0.3f * roll_, 0.2f, 1.0f);
281 // An animal buried by the snow: a short squeal that sinks in pitch, heard through the snow
282 // (low-passed hard, so it is muted).
283 float squeal = 0;
284 for (Squeal& s : squealVoices_) {
285 const double tau = t_ - s.start;
286 if (!s.used || tau < 0 || tau > 0.6) continue;
287 const double hz = s.pitch * (1150.0 - 450.0 * tau / 0.6) * (1.0 + 0.03 * std::sin(kTwoPi * 11.0 * tau));
288 s.phase += kTwoPi * hz * dt;
289 const double env = std::min(1.0, tau / 0.03) * std::exp(-tau / 0.25);
290 squeal += static_cast<float>(env * (std::sin(s.phase) + 0.5 * std::sin(2 * s.phase) + 0.3 * std::sin(3 * s.phase)));
291 }
292 squealLow_[0] += (squeal - squealLow_[0]) * lowSqueal_;
293 squealLow_[1] += (squealLow_[0] - squealLow_[1]) * lowSqueal_;
294
295 for (int ch = 0; ch < 2; ++ch) {
296 const float n = noise(rng_) * noiseScale_;
297 mountLow_[ch][0] += (n - mountLow_[ch][0]) * lowWind;
298 mountLow_[ch][1] += (mountLow_[ch][0] - mountLow_[ch][1]) * lowWind;
299 const float n2 = noise(rng_) * noiseScale_;
300 iceLow_[ch] += (n2 - iceLow_[ch]) * lowIce_;
301 const float n3 = noise(rng_) * noiseScale_;
302 streamLow_[ch][0] += (n3 - streamLow_[ch][0]) * lowStreamTop_;
303 streamLow_[ch][1] += (n3 - streamLow_[ch][1]) * lowStreamBottom_;
304 const float n4 = noise(rng_) * noiseScale_;
305 slideLow_[ch][0] += (n4 - slideLow_[ch][0]) * lowSlide_;
306 slideLow_[ch][1] += (slideLow_[ch][0] - slideLow_[ch][1]) * lowSlide_;
307 const float n5 = noise(rng_) * noiseScale_;
308 slideMid_[ch] += (n5 - slideMid_[ch]) * lowSlideMid_;
309 crackLowAv_[ch] += crackF * crackBandAv_[ch];
310 const float highClick = click * 8.0f - crackLowAv_[ch] - 0.5f * crackBandAv_[ch];
311 crackBandAv_[ch] += crackF * highClick;
312 const float panCry = ch ? std::sin(cryPan_ * 1.5708f) : std::cos(cryPan_ * 1.5708f);
313 out[ch] = mountLow_[ch][1] * 9.0f * windGain + whistle * (ch ? 0.9f : 1.0f) + (n2 - iceLow_[ch]) * 0.6f * hiss +
314 (streamLow_[ch][0] - streamLow_[ch][1]) * 0.085f * babble * (1.0f - 0.5f * rain_) +
315 cry * panCry * 1.4f +
316 peck * (ch ? std::sin(peckPan_ * 1.5708f) : std::cos(peckPan_ * 1.5708f)) * 1.4f + crackBandAv_[ch] * 0.35f + thump + slideLow_[ch][1] * 14.0f * slide * slide * roll +
317 slideMid_[ch] * 0.7f * slide * slide * slide * roll + squealLow_[1] * 0.6f;
318 }
319}

321void Ambience::desertSample(const AmbientParams& p, float out[2]) {
322 (void)p;
323 const double dt = 1.0 / rate_;
324 // Dry wind: low-passed noise like the forest's, but steadier and a little higher. In a sandstorm
325 // it howls stronger, with the hiss of blown sand on top.
326 const float cutoffHz = (260.0f + 300.0f * gust_) * (1.0f + 0.7f * rain_);
327 const float lowWind = 1.0f - std::exp(-6.2831853f * cutoffHz / static_cast<float>(rate_));
328 // Calm weather is barely a breath; a sandstorm or a tornado brings the full wind.
329 const float calm = 0.22f + 0.78f * std::min(1.0f, rain_ + tornado_);
330 const float windGain = (0.45f + 0.8f * gust_) * (1.0f + 3.0f * rain_ + 0.8f * tornado_) *
331 (1.0f - 0.3f * night_) * calm * 0.085f;
332 const float sandLevel = std::max(rain_, tornado_ * 0.6f) * (0.55f + 0.45f * gust_);
333 // A tornado: a deep roar that turns, loud and quiet about three times a second.
334 tornadoPhase_ += kTwoPi * 2.8 * dt;
335 const float spin = 0.65f + 0.35f * static_cast<float>(std::sin(tornadoPhase_));
336 const float roar = tornado_ * std::sqrt(tornado_) * spin * 2.0f;
337 // Crickets at night, fewer than in the forest.
338 const float chirps = night_ * 0.014f * (cricket(t_, 2.1, 33, 4300) + 0.7f * cricket(t_ + 0.5, 1.7, 28, 4800));
339 // The sandworm below, as it nears: a growl in the ground.
340 const float growl = storm_ * storm_ * 0.5f * std::clamp(0.55f + 0.45f * roll_, 0.1f, 1.0f);
341 // The worm bellows now and then while it crosses: a very deep, trumpeting roar. The pitch
342 // swoops up and sinks again, the tone is rough (a fast tremble) and low-passed to stay deep.
343 if (wormNear_ > 0.05f) {
344 if (nextRoar_ == 0) nextRoar_ = t_ + rng_.range(0.3, 1.2);
345 if (t_ >= nextRoar_) {
346 roarStart_ = t_;
347 roarDur_ = rng_.range(2.0, 2.8);
348 nextRoar_ = t_ + roarDur_ + rng_.range(1.2, 3.5);
349 }
350 } else {
351 nextRoar_ = 0;
352 }
353 float bellow = 0;
354 const double rt = t_ - roarStart_;
355 if (rt >= 0 && rt < roarDur_) {
356 const double u = rt / roarDur_;
357 const double env = std::min(1.0, rt / 0.3) * std::min(1.0, (roarDur_ - rt) / 0.8);
358 const double hz = 34.0 + 46.0 * std::sin(3.14159265 * std::pow(u, 0.7)) + 3.0 * std::sin(kTwoPi * 5.5 * rt);
359 roarPhase_ += kTwoPi * hz * dt;
360 const double tone = std::tanh(2.4 * std::sin(roarPhase_) + 1.3 * std::sin(2 * roarPhase_ + 0.6) +
361 0.7 * std::sin(3 * roarPhase_ + 1.1));
362 bellow = static_cast<float>(tone * env * (1.0 + 0.4 * std::sin(kTwoPi * 23.0 * t_)));
363 }
364 roarLow_[0] += (bellow - roarLow_[0]) * lowRoar_;
365 roarLow_[1] += (roarLow_[0] - roarLow_[1]) * lowRoar_;
366 const float roarOut = roarLow_[1] * 0.45f * wormNear_;
367 // Crushing: chewing. Each thing it swallows is ground between a few soft bites, every one a
368 // short swell of low-passed noise (no sharp start, so it never sounds like a shot), over a faint
369 // low grumble.
370 float crunchEnv = 0, grumble = 0;
371 for (const Crunch& c : crunches_) {
372 const double tau = t_ - c.start;
373 if (c.gain <= 0 || tau < 0 || tau > 1.6) continue;
374 for (int b = 0; b < c.bites; ++b) {
375 const double bt = tau - c.bite[b];
376 if (bt < 0 || bt > 0.09) continue;
377 const double swell = std::sin(3.14159265 * bt / 0.09);
378 crunchEnv += static_cast<float>(c.gain * c.force[b] * swell * swell);
379 }
380 grumble += static_cast<float>(c.gain * std::sin(kTwoPi * 62.0 * tau) * std::min(1.0, tau / 0.12) * std::exp(-tau / 0.7));
381 }
382
383 for (int ch = 0; ch < 2; ++ch) {
384 const float n = noise(rng_) * noiseScale_;
385 desertLow_[ch][0] += (n - desertLow_[ch][0]) * lowWind;
386 desertLow_[ch][1] += (desertLow_[ch][0] - desertLow_[ch][1]) * lowWind;
387 const float n2 = noise(rng_) * noiseScale_;
388 sandLow_[ch][0] += (n2 - sandLow_[ch][0]) * lowSandCut_;
389 sandLow_[ch][1] += (n2 - sandLow_[ch][1]) * lowSandTop_;
390 const float n3 = noise(rng_) * noiseScale_;
391 tornadoLow_[ch][0] += (n3 - tornadoLow_[ch][0]) * lowTornado_;
392 tornadoLow_[ch][1] += (tornadoLow_[ch][0] - tornadoLow_[ch][1]) * lowTornado_;
393 const float nc = noise(rng_) * noiseScale_;
394 crunchLow_[ch][0] += (nc - crunchLow_[ch][0]) * lowCrunch_;
395 crunchLow_[ch][1] += (crunchLow_[ch][0] - crunchLow_[ch][1]) * lowCrunch_;
396 // the grinding is the noise below ~500 Hz, with a little more of the bite's grit above it
397 const float crunchOut = (crunchLow_[ch][1] * 7.0f + (nc - crunchLow_[ch][0]) * 0.25f) * crunchEnv;
398 const float n4 = noise(rng_) * noiseScale_;
399 deepLow_[ch][0] += (n4 - deepLow_[ch][0]) * lowDeep_;
400 deepLow_[ch][1] += (deepLow_[ch][0] - deepLow_[ch][1]) * lowDeep_;
401 out[ch] = desertLow_[ch][1] * 9.0f * windGain + (sandLow_[ch][1] - sandLow_[ch][0]) * 0.5f * sandLevel +
402 tornadoLow_[ch][1] * 8.0f * roar + deepLow_[ch][1] * 20.0f * growl +
403 chirps * (ch ? 0.9f : 1.0f) + roarOut + grumble * 0.03f + crunchOut;
404 }

407void Ambience::citySample(const AmbientParams& p, float out[2]) {
408 (void)p;
409 const double dt = 1.0 / rate_;
410 const float day = 1.0f - 0.55f * night_;
411 // Cars passing by now and then: an engine swell that sinks in pitch as it goes, panned across.
412 if (nextPass_ == 0) nextPass_ = t_ + rng_.range(0.3, 1.5);
413 if (t_ >= nextPass_) {
414 Pass& v = *std::min_element(std::begin(passes_), std::end(passes_), [](const Pass& a, const Pass& b) { return a.start < b.start; });
415 v.start = t_;
416 v.dur = rng_.range(1.4, 2.6);
417 v.pitch = static_cast<float>(rng_.range(80, 125));
418 v.pan = static_cast<float>(rng_.range(0.1, 0.9));
419 v.gain = static_cast<float>(rng_.range(0.5, 1.0));
420 nextPass_ = t_ + rng_.range(1.2, 4.5) * (1.0 + 2.0 * night_);
421 }
422 float engine[2] = {0, 0};
423 for (Pass& v : passes_) {
424 const double tau = t_ - v.start;
425 if (v.gain <= 0 || tau < 0 || tau > v.dur) continue;
426 const double u = tau / v.dur;
427 const double env = std::sin(3.14159265 * u) * std::sin(3.14159265 * u);
428 v.phase += kTwoPi * v.pitch * (1.3 - 0.55 * u) * dt; // the pitch sinks as it drives away
429 const float tone = static_cast<float>(std::tanh(1.8 * std::sin(v.phase) + 0.8 * std::sin(2 * v.phase + 0.4) + 0.5 * std::sin(3 * v.phase)) * env * v.gain);
430 engine[0] += tone * std::cos(v.pan * 1.5708f);
431 engine[1] += tone * std::sin(v.pan * 1.5708f);
432 }
433 // Horns: now and then one or two honks, rarer at night.
434 if (nextHonk_ == 0) nextHonk_ = t_ + rng_.range(2.0, 6.0);
435 if (t_ >= nextHonk_) {
436 Honk& h = *std::min_element(std::begin(honks_), std::end(honks_), [](const Honk& a, const Honk& b) { return a.start < b.start; });
437 h.start = t_;
438 h.pitch = static_cast<float>(rng_.range(0.88, 1.18));
439 h.pan = static_cast<float>(rng_.range(0.1, 0.9));
440 h.gain = static_cast<float>(rng_.range(0.6, 1.0));
441 h.long_ = rng_.chance(0.2) ? 1.0f : 0.0f;
442 h.count = (h.long_ > 0 || rng_.chance(0.55)) ? 1 : 2;
443 nextHonk_ = t_ + rng_.range(5.0, 16.0) * (1.0 + 2.5 * night_);
444 }
445 float horn[2] = {0, 0};
446 for (Honk& h : honks_) {
447 const double tau = t_ - h.start;
448 if (h.gain <= 0 || tau < 0 || tau > 1.2) continue;
449 const double len = h.long_ > 0 ? 0.8 : 0.22, gap = 0.34;
450 for (int k = 0; k < h.count; ++k) {
451 const double lt = tau - k * gap;
452 if (lt < 0 || lt > len) continue;
453 const double env = std::min(1.0, lt / 0.01) * std::min(1.0, (len - lt) / 0.04);
454 h.phaseA += kTwoPi * 420.0 * h.pitch * dt;
455 h.phaseB += kTwoPi * 530.0 * h.pitch * dt; // two notes a third apart, like a car horn
456 const float s = static_cast<float>((std::tanh(3.0 * std::sin(h.phaseA)) + std::tanh(3.0 * std::sin(h.phaseB))) * 0.5 * env * h.gain);
457 horn[0] += s * std::cos(h.pan * 1.5708f);
458 horn[1] += s * std::sin(h.pan * 1.5708f);
459 }
460 }
461 // The siren of an ambulance or a fire engine: two notes, the pitch gliding between them.
462 const float sirenTarget = std::fmod(t_, 1.0) < 0.5 ? 960.0f : 770.0f;
463 sirenFreq_ += (sirenTarget - sirenFreq_) * 0.0025f * 48000.0f / static_cast<float>(rate_);
464 sirenPhase_ += kTwoPi * sirenFreq_ * dt;
465 const float siren = static_cast<float>(std::tanh(1.6 * std::sin(sirenPhase_)) * 0.8 + 0.25 * std::sin(2 * sirenPhase_)) * sirenNear_ * std::sqrt(sirenNear_);
466 // The fire engine's siren is a slow wail: the pitch rises and falls smoothly over about three and a
467 // half seconds ("uuhh-wwiiww-hhuu"), with a little vibrato and a vowel-like roundness, where the
468 // ambulance's is a quick alternation of two notes.
469 const double sweep = 0.5 - 0.5 * std::cos(kTwoPi * t_ / 3.5);
470 const double wailHz = (430.0 + 520.0 * sweep) * (1.0 + 0.012 * std::sin(kTwoPi * 5.5 * t_));
471 wailPhase_ += kTwoPi * wailHz * dt;
472 const float wail = static_cast<float>(std::tanh(1.3 * std::sin(wailPhase_)) * 0.75 + 0.3 * std::sin(2 * wailPhase_ + 0.7) +
473 0.15 * std::sin(0.5 * wailPhase_)) *
474 wailNear_ * std::sqrt(wailNear_) * (0.8f + 0.2f * static_cast<float>(sweep));
475 // A train: a deep rumble and the clatter of wheels over the rail joints, and a horn as it arrives.
476 // The wheels beat out "klack-klack ... klack-klack": two quick knocks, then a pause, over and over
477 // (the two axles of a bogie crossing a rail joint). Each knock is a woody thunk, not a click.
478 trainClock_ += dt * (1.0 + 0.2 * trainNear_);
479 if (trainClock_ >= kKlackPeriod) trainClock_ -= kKlackPeriod;
480 float clack = 0, thunk = 0;
481 for (int k = 0; k < 2; ++k) {
482 const double tau = trainClock_ - k * 0.21;
483 if (tau < 0 || tau > 0.4) continue;
484 const double rise = 1.0 - std::exp(-tau / 0.004);
485 const double gain = k ? 0.8 : 1.0; // the second knock is a little softer
486 clack += static_cast<float>(gain * rise * std::exp(-tau / 0.05));
487 thunk += static_cast<float>(gain * rise * std::exp(-tau / 0.09) * std::sin(kTwoPi * (k ? 105.0 : 92.0) * tau));
488 }
489 if (trainNear_ > 0.2f && !trainHorned_) { trainHorned_ = true; trainHornStart_ = t_; }
490 if (trainNear_ < 0.03f) trainHorned_ = false;
491 float trainHorn = 0;
492 const double th = t_ - trainHornStart_;
493 if (th >= 0 && th < 1.4) {
494 const double env = std::min(1.0, th / 0.05) * std::min(1.0, (1.4 - th) / 0.25);
495 trainHornPhase_ += kTwoPi * 311.0 * dt;
496 trainHorn = static_cast<float>(env * (std::tanh(2.0 * std::sin(trainHornPhase_)) + 0.7 * std::tanh(2.0 * std::sin(trainHornPhase_ * 1.26)))) * 0.5f;
497 }
498 // A fire: a low roar and crackling.
499 const bool crackle = rng_.uniform() < std::min(1.0, 90.0 * fireNear_ * dt * 4.0);
500 const float spark = crackle ? static_cast<float>(rng_.range(0.4, 1.0)) * (rng_.chance(0.5) ? 1.0f : -1.0f) : 0.0f;
501
502 for (int ch = 0; ch < 2; ++ch) {
503 const float n = noise(rng_) * noiseScale_;
504 carLow_[ch][0] += (n - carLow_[ch][0]) * lowCar_;
505 carLow_[ch][1] += (carLow_[ch][0] - carLow_[ch][1]) * lowCar_;
506 const float n2 = noise(rng_) * noiseScale_;
507 carMid_[ch][0] += (n2 - carMid_[ch][0]) * lowCarMid_;
508 carMid_[ch][1] += (n2 - carMid_[ch][1]) * lowCarTop_;
509 engineLow_[ch] += (engine[ch] - engineLow_[ch]) * lowEngine_;
510 const float n3 = noise(rng_) * noiseScale_;
511 trainLow_[ch][0] += (n3 - trainLow_[ch][0]) * lowTrain_;
512 trainLow_[ch][1] += (trainLow_[ch][0] - trainLow_[ch][1]) * lowTrain_;
513 const float n4 = noise(rng_) * noiseScale_;
514 trainBand_[ch][0] += (n4 - trainBand_[ch][0]) * lowTrainBand_;
515 trainBand_[ch][1] += (n4 - trainBand_[ch][1]) * lowKlack_;
516 const float n5 = noise(rng_) * noiseScale_;
517 fireLow_[ch][0] += (n5 - fireLow_[ch][0]) * lowFire_;
518 fireLow_[ch][1] += (fireLow_[ch][0] - fireLow_[ch][1]) * lowFire_;
519 fireCrack_[ch] += (spark * (ch ? -1.0f : 1.0f) - fireCrack_[ch]) * 0.35f;
520 const float n6 = noise(rng_) * noiseScale_;
521 planeLow_[ch][0] += (n6 - planeLow_[ch][0]) * lowJet_;
522 planeLow_[ch][1] += (planeLow_[ch][0] - planeLow_[ch][1]) * lowJet_;
523 hornLow_[ch] += ((ch ? horn[1] : horn[0]) - hornLow_[ch]) * lowHorn_;
524 const float hornOut = hornLow_[ch];
525 const float traffic = carLow_[ch][1] * 9.0f * 0.075f * day + (carMid_[ch][0] - carMid_[ch][1]) * 0.45f * 0.04f * day;
526 out[ch] = traffic + engineLow_[ch] * 0.16f * day + hornOut * 0.045f + siren * 0.15f + wail * 0.13f +
527 trainLow_[ch][1] * 9.0f * 0.07f * trainNear_ * trainNear_ +
528 ((trainBand_[ch][0] - trainBand_[ch][1]) * 1.6f * clack + thunk * 0.12f) * trainNear_ + trainHorn * 0.05f * trainNear_ +
529 fireLow_[ch][1] * 9.0f * 0.11f * fireNear_ + fireCrack_[ch] * 0.3f * fireNear_ +
530 planeLow_[ch][1] * 9.0f * 0.06f * planeNear_ * planeNear_;
531 }
532}

534void Ambience::aquariumSample(const AmbientParams& p, float out[2]) {
535 (void)p;
536 const double dt = 1.0 / rate_;
537 // Slow swells of the water, from the same irregular gusts that move the wind; night is a bit calmer.
538 const float swell = (0.55f + 0.9f * gust_) * (1.0f - 0.25f * night_);
539
540 // Random bubbles rise now and then, sometimes in a little cluster.
541 if (nextBubble_ == 0) nextBubble_ = t_ + 0.5;
542 if (t_ >= nextBubble_) {
543 const int n = rng_.chance(0.25) ? rng_.rangeInt(2, 3) : 1;
544 for (int i = 0; i < n; ++i) {
545 bubble(i * rng_.range(0.05, 0.22), static_cast<float>(rng_.range(350, 1100)),
546 static_cast<float>(rng_.range(0.04, 0.10)), static_cast<float>(rng_.range(0.03, 0.07)));
547 }
548 nextBubble_ = t_ + rng_.range(0.25, 1.4) * (1.0 + 0.8 * night_);
549 }
550 float bub[2] = {0, 0};
551 for (const Bubble& b : bubbles_) {
552 const double tau = t_ - b.start;
553 if (b.amp <= 0 || tau < 0 || tau > b.dur) continue;
554 // A bubble sings a short rising note: it starts at f0 and climbs as it shrinks.
555 const double phase = kTwoPi * b.f0 * (tau + 1.1 * tau * tau / b.dur);
556 const float v = static_cast<float>(std::sin(phase) * (1.0 - std::exp(-tau * 900.0)) *
557 std::exp(-tau / (b.dur * 0.38))) * b.amp;
558 bub[0] += v * std::cos(b.pan * 1.5708f);
559 bub[1] += v * std::sin(b.pan * 1.5708f);
560 }
561
562 // A whale passing: a long, gliding moan that repeats while it is near.
563 float song = 0;
564 const float whaleLevel = whale_ * std::sqrt(whale_);
565 if (whale_ > 0.03f) {
566 if (nextSong_ == 0) nextSong_ = t_ + rng_.range(1.0, 3.0);
567 const bool singing = t_ - song_.start <= song_.dur && song_.dur > 0;
568 if (!singing && t_ >= nextSong_) {
569 song_ = Song{t_, rng_.range(3.0, 5.0), static_cast<float>(rng_.range(70, 110)),
570 static_cast<float>(rng_.range(0.3, 0.7)), 0};
571 nextSong_ = t_ + song_.dur + rng_.range(3.0, 8.0) * (1.6 - whale_);
572 }
573 } else {
574 nextSong_ = 0;
575 }
576 if (t_ - song_.start <= song_.dur && song_.dur > 0) {
577 const double tau = t_ - song_.start, x = tau / song_.dur;
578 const double f = song_.f0 * (1.0 + 0.55 * std::sin(3.14159265 * x) + 0.3 * x +
579 0.02 * std::sin(kTwoPi * 5.5 * tau));
580 song_.phase += kTwoPi * f * dt;
581 const double env = std::sin(3.14159265 * x);
582 song = static_cast<float>((std::sin(song_.phase) + 0.35 * std::sin(2 * song_.phase + 0.5) +
583 0.12 * std::sin(3 * song_.phase)) * env * env) * whaleLevel * 0.16f;
584 }
585
586 // The growl of something huge (the kraken) as it nears.
587 const float growl = storm_ * storm_ * 0.5f * std::clamp(0.55f + 0.45f * roll_, 0.1f, 1.0f);
588 // What its tentacles catch is gulped down: two soft glugs, each a tone that sinks in pitch and
589 // wobbles a little, heard through the water (low-passed, so it is muffled).
590 float gulp = 0;
591 for (const Gulp& g : gulps_) {
592 const double tau = t_ - g.start;
593 if (!g.used || tau < 0 || tau > 0.5) continue;
594 for (int k = 0; k < 2; ++k) { // the second glug follows a moment later, lower
595 const double lt = tau - k * 0.2;
596 if (lt < 0 || lt > 0.17) continue;
597 const double u = lt / 0.17;
598 const double hz = g.pitch * (k ? 0.8 : 1.0) * (250.0 - 120.0 * u) * (1.0 + 0.06 * std::sin(kTwoPi * 24.0 * lt));
599 gulpPhase_ += kTwoPi * hz * dt;
600 const double env = std::sin(3.14159265 * u);
601 gulp += static_cast<float>(env * env * (std::sin(gulpPhase_) + 0.45 * std::sin(2 * gulpPhase_ + 0.5)));
602 }
603 }
604 gulpLow_[0] += (gulp - gulpLow_[0]) * lowGulp_;
605 gulpLow_[1] += (gulpLow_[0] - gulpLow_[1]) * lowGulp_;
606
607 for (int ch = 0; ch < 2; ++ch) {
608 const float n = noise(rng_) * noiseScale_;
609 waterLow_[ch][0] += (n - waterLow_[ch][0]) * lowHum_;
610 waterLow_[ch][1] += (waterLow_[ch][0] - waterLow_[ch][1]) * lowHum_;
611 const float n2 = noise(rng_) * noiseScale_;
612 muffleLow_[ch] += (n2 - muffleLow_[ch]) * lowMuffle_;
613 const float n3 = noise(rng_) * noiseScale_;
614 deepLow_[ch][0] += (n3 - deepLow_[ch][0]) * lowDeep_;
615 deepLow_[ch][1] += (deepLow_[ch][0] - deepLow_[ch][1]) * lowDeep_;
616 const float panSong = ch ? std::sin(song_.pan * 1.5708f) : std::cos(song_.pan * 1.5708f);
617 out[ch] = waterLow_[ch][1] * 9.0f * 0.11f * swell + muffleLow_[ch] * 0.05f * swell +
618 deepLow_[ch][1] * 20.0f * growl + bub[ch] + song * panSong * 1.4f + gulpLow_[1] * 0.45f;
619 }
620}

626void Ambience::menaceSample(const AmbientParams& p, float out[2]) {
627 out[0] = out[1] = 0;
628 if (menace_ < 1e-3f) return;
629 const double dt = 1.0 / rate_;
630 const float level = menace_;
631 for (int ch = 0; ch < 2; ++ch) {
632 const float n = noise(rng_) * noiseScale_;
633 menaceLow_[ch][0] += (n - menaceLow_[ch][0]) * lowThump_;
634 menaceLow_[ch][1] += (menaceLow_[ch][0] - menaceLow_[ch][1]) * lowThump_;
635 out[ch] = 0;
636 switch (p.kind) {
637 case AmbientKind::Aquarium: { // poison gas: a hiss that wells up, and glugging bubbles
638 menaceHigh_[ch] += (n - menaceHigh_[ch]) * lowHiss_;
639 out[ch] = (n - menaceHigh_[ch]) * 0.3f * level * (0.7f + 0.3f * static_cast<float>(std::sin(kTwoPi * 0.7 * t_ + ch)));
640 break;
641 }
642 case AmbientKind::Forest: { // a flood: the deep roar of rushing water and the hiss of its surface
643 menaceHigh_[ch] += (n - menaceHigh_[ch]) * lowFloodTop_;
644 const float swell = 0.75f + 0.25f * static_cast<float>(std::sin(kTwoPi * 0.31 * t_ + 1.7 * ch));
645 out[ch] = (menaceLow_[ch][1] * 3.2f + (n - menaceHigh_[ch]) * 0.06f + menaceHigh_[ch] * 0.28f) * 0.55f * level * swell;
646 break;
647 }
648 case AmbientKind::Desert: { // a helicopter: the chop of the rotor, over a whine
649 const double chop = std::pow(0.5 + 0.5 * std::sin(kTwoPi * 11.5 * t_), 2.2);
650 menaceHigh_[ch] += (n - menaceHigh_[ch]) * lowChop_;
651 menacePhase_[1] += kTwoPi * 390.0 * dt * 0.5;
652 out[ch] = (menaceHigh_[ch] * static_cast<float>(chop) * 1.9f + menaceLow_[ch][1] * 1.5f * static_cast<float>(chop) +
653 0.012f * static_cast<float>(std::sin(menacePhase_[1]))) * 0.7f * level;
654 break;
655 }
656 case AmbientKind::Moon: { // the big meteor: a rumble in the ground that swells as it nears
657 out[ch] = menaceLow_[ch][1] * 4.5f * level * level;
658 break;
659 }
660 default: break;
661 }
662 }
663 if (p.kind == AmbientKind::Aquarium && rng_.uniform() < 5.5 * level * dt) { // bubbles of gas
664 bubble(0, static_cast<float>(rng_.range(230, 560)), 0.12f, static_cast<float>(rng_.range(0.05, 0.11)) * level);
665 }
666 if (p.kind == AmbientKind::City && flames_ > 1e-3f) {
667 // The streams of fire: one steady, rushing roar (broad noise over a deep body), only unevenly
668 // swelling and with a few faint crackles. Nothing in it beats in time.
669 flameWander_ += (noise(rng_) - flameWander_) * static_cast<float>(1.0 - std::exp(-kTwoPi * 2.0 * dt));
670 const float swell = 0.88f + 0.12f * std::clamp(flameWander_ * 45.0f, -1.0f, 1.0f);
671 for (int ch = 0; ch < 2; ++ch) {
672 const float n = noise(rng_) * noiseScale_;
673 flameHi_[ch] += (n - flameHi_[ch]) * lowFlameTop_;
674 flameLo_[ch] += (n - flameLo_[ch]) * lowFlameBottom_;
675 const float band = flameHi_[ch] - flameLo_[ch]; // the rush
676 const float body = flameLo_[ch]; // the deep roar under it
677 float crackle = 0;
678 if (rng_.uniform() < 14.0 * dt) crackle = noise(rng_) * 0.18f;
679 out[ch] += (band * 3.2f + body * 1.8f + crackle) * swell * 0.7f * flames_;
680 }
681 }
682 if (p.kind != AmbientKind::Mountains && p.kind != AmbientKind::City) return;
683 // A creature: heavy steps and a roar now and then. The monsters of the city are bigger and slower.
684 const bool big = p.kind == AmbientKind::City;
685 const double period = big ? 0.95 : 0.55;
686 const double ph = std::fmod(t_, period);
687 const float thump = static_cast<float>(std::exp(-ph / (big ? 0.2 : 0.11)) * std::sin(kTwoPi * (big ? 44.0 : 62.0) * ph));
688 out[0] += thump * (big ? 0.36f : 0.38f) * level + menaceLow_[0][1] * (big ? 1.3f : 1.2f) * level * static_cast<float>(std::exp(-ph / 0.25));
689 out[1] += thump * (big ? 0.36f : 0.38f) * level + menaceLow_[1][1] * (big ? 1.3f : 1.2f) * level * static_cast<float>(std::exp(-ph / 0.25));
690 if (nextMenaceRoar_ == 0) nextMenaceRoar_ = t_ + rng_.range(0.5, 1.5);
691 if (t_ >= nextMenaceRoar_) {
692 menaceRoarStart_ = t_;
693 menaceRoarDur_ = rng_.range(1.3, 2.0);
694 menaceRoarVoice_ = big && wingedNear_ > 0.5f ? (menaceRoarVoice_ + 1) % 2 : 0; // the screech needs the winged one alive
695 nextMenaceRoar_ = t_ + menaceRoarDur_ + rng_.range(2.0, 4.5);
696 }
697 const double rt = t_ - menaceRoarStart_;
698 if (rt >= 0 && rt < menaceRoarDur_) {
699 const double u = rt / menaceRoarDur_;
700 const double env = std::min(1.0, rt / 0.15) * std::min(1.0, (menaceRoarDur_ - rt) / 0.5);
701 // The yeti and the lizard bellow low; the winged monster screeches high.
702 const double hz = menaceRoarVoice_ == 1 ? 520.0 + 380.0 * std::sin(3.14159265 * u) : (big ? 58.0 : 96.0) + 50.0 * std::sin(3.14159265 * std::pow(u, 0.8));
703 menacePhase_[0] += kTwoPi * hz * dt;
704 const double tone = std::tanh(2.2 * std::sin(menacePhase_[0]) + 1.1 * std::sin(2 * menacePhase_[0] + 0.5) + 0.5 * std::sin(3 * menacePhase_[0] + 1.0));
705 const float roar = static_cast<float>(tone * env * (1.0 + 0.35 * std::sin(kTwoPi * 19.0 * t_))) * level * (menaceRoarVoice_ == 1 ? 0.09f * wingedNear_ : 0.14f);
706 out[0] += roar;
707 out[1] += roar;
708 }

711void Ambience::render(const AmbientParams& p, float* out, int frames) {

907 }

914 }

923 }
