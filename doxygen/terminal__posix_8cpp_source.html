45void writeAll(const char* data, size_t n) {
46 while (n > 0) {
47 const ssize_t w = ::write(STDOUT_FILENO, data, n);
48 if (w < 0) {
49 if (errno == EINTR) continue;
50 return;
51 }
52 data += w;
53 n -= static_cast<size_t>(w);
54 }
55}

81bool Terminal::init() {
82 if (!isatty(STDIN_FILENO) || !isatty(STDOUT_FILENO)) return false;
83 if (tcgetattr(STDIN_FILENO, &impl_->saved) != 0) return false;
84
85 termios raw = impl_->saved;
86 raw.c_lflag &= ~static_cast<tcflag_t>(ICANON | ECHO); // keep ISIG so Ctrl-C still signals
87 raw.c_iflag &= ~static_cast<tcflag_t>(IXON | ICRNL);
88 raw.c_cc[VMIN] = 0;
89 raw.c_cc[VTIME] = 0;
90 if (tcsetattr(STDIN_FILENO, TCSAFLUSH, &raw) != 0) return false;
91
92 struct sigaction sa {};
93 sigemptyset(&sa.sa_mask);
94 sa.sa_handler = onWinch;
95 sigaction(SIGWINCH, &sa, &impl_->oldWinch);
96 sa.sa_handler = onQuit;
97 sigaction(SIGINT, &sa, &impl_->oldInt);
98 sigaction(SIGTERM, &sa, &impl_->oldTerm);
99
100 impl_->active = true;
101 static const char kEnter[] = "\x1b[?1049h\x1b[?25l\x1b[2J";
102 writeAll(kEnter, sizeof(kEnter) - 1);
103
104 const char* ct = std::getenv("COLORTERM");
105 if (ct && (std::strstr(ct, "truecolor") || std::strstr(ct, "24bit"))) {
106 colorMode_ = ColorMode::TrueColor;
107 }
108 if (onLinuxConsole()) colorMode_ = ColorMode::Console; // 16 colors whatever COLORTERM claims
109 return true;
110}

126std::vector<Action> Terminal::poll(int timeoutMs) {
127 std::vector<Action> actions;
128 pollfd pfd{STDIN_FILENO, POLLIN, 0};
129 if (::poll(&pfd, 1, timeoutMs) <= 0) return actions;
130
131 char buf[64];
132 const ssize_t n = ::read(STDIN_FILENO, buf, sizeof(buf));
133 for (ssize_t i = 0; i < n; ++i) {
134 switch (buf[i]) {
135 case 'q': case 'Q': case 3: actions.push_back(Action::Quit); break;
136 case 'p': case 'P': actions.push_back(Action::Pause); break;
137 case 'f': case 'F': actions.push_back(Action::Interact); break;
138 case 'i': case 'I': actions.push_back(Action::Info); break;
139 case 'r': case 'R': actions.push_back(Action::Redraw); break;
140 case 'm': case 'M': actions.push_back(Action::Menu); break;
141 case 'c': case 'C': actions.push_back(Action::Style); break;
142 case 't': case 'T': actions.push_back(Action::Time); break;
143 case 'x': actions.push_back(Action::Rare); break;
144 case 'X': actions.push_back(Action::RareOther); break;
145 case 'a': case 'A': actions.push_back(Action::Audio); break;
146 case ' ': actions.push_back(Action::Pause); break;
147 case '\r': case '\n': actions.push_back(Action::Select); break;
148 case 0x1b:
149 if (i + 2 < n && buf[i + 1] == '[') {
150 switch (buf[i + 2]) {
151 case 'A': actions.push_back(Action::Up); break;
152 case 'B': actions.push_back(Action::Down); break;
153 case 'C': actions.push_back(Action::Right); break;
154 case 'D': actions.push_back(Action::Left); break;
155 default: break;
156 }
157 i += 2;
158 } else {
159 actions.push_back(Action::Back);
160 }
161 break;
162 default: break;
163 }
164 }
165 return actions;
166}
