51Session::Session(const AssetPack& pack, std::vector<std::shared_ptr<const SceneDef>> scenes,
52 const Config& config, int width, int height)
53 : pack_(pack),
54 scenes_(std::move(scenes)),
55 sceneIdx_(config.scene % scenes_.size()),
56 w_(width),
57 h_(height),
58 seed_(config.seed),
59 style_(config.style),
60 timeMode_(config.realtime ? World::TimeMode::Realtime : World::TimeMode::Cycle),
61 carriedPhase_(config.hour >= 0 ? config.hour / 24.0 : -1),
62 rng_(config.seed ^ 0x5eed5eedULL),
63 menu_("DIORASCII"),
64 info_(config.info),
65 immersive_(isStereo(config.audioLayout) ? LayoutChoice{} : config.audioLayout) {
66 std::vector<std::string> titles;
67 titles.reserve(scenes_.size());
68 std::transform(scenes_.begin(), scenes_.end(), std::back_inserter(titles), [](const auto& sc) { return sc->title; });
69 menu_.addChoice("scene", "Scene", titles, sceneIdx_);
70 menu_.addChoice("style", "Style",
71 {styleName(Style::Full), styleName(Style::Terminal), styleName(Style::Mono)},
72 static_cast<size_t>(style_));
73 menu_.addChoice("mode", "Mode", {"User", "Cycle"}, 0);
74 menu_.addChoice("time", "Time of day", timeOptions(scenes_[sceneIdx_]->dayNight),
75 scenes_[sceneIdx_]->dayNight ? static_cast<size_t>(timeMode_) : 0);
76 menu_.addChoice("info", "Info line", {"off", "on"}, info_ ? 1 : 0);
77 menu_.addChoice("audio", "Sound", {"off", "stereo", "immersive"}, 0);
78 menu_.addButton("action", scenes_[sceneIdx_]->action.label);
79 menu_.addButton("rare", "Rare event");
80 menu_.addButton("rare2", "Other rare event");
81 menu_.addButton("resume", "Resume");
82 if (config.quitEntry) menu_.addButton("quit", "Quit");
83 updateFooter();
84 rebuild();
85 toast(scenes_[sceneIdx_]->title, 2.0);
86 if (config.sound) setSound(isStereo(config.audioLayout) ? SoundMode::Stereo : SoundMode::Immersive);
87 if (config.cycle) setCycle(true);
88}

100void Session::flush() {
101 if (!rebuildPending_) return;
102 // The time of day survives scene switches, restarts and resizes.
103 if (world_->hasDayNight()) carriedPhase_ = world_->phase();
104 endFade();
105 if (fadePending_) { // the old world stays on screen and in the ears while the new one comes in
106 outgoing_ = std::move(world_);
107 fadeLeft_ = kFadeSeconds;
108 }
109 fadePending_ = false;
110 rebuild();
111}

186void Session::setCycle(bool on) {
187 if (on == cycle_) return;
188 cycle_ = on;
189 flush();
190 if (on) { // the clock runs by itself; what the user chose is kept for later
191 userTimeMode_ = timeMode_;
192 timeMode_ = World::TimeMode::Cycle;
193 } else {
194 timeMode_ = userTimeMode_;
195 }
196 world_->setTimeMode(timeMode_);
197 syncClock();
198 menu_.setValue("mode", on ? 1 : 0);
199 menu_.setValue("time", static_cast<size_t>(timeMode_));
200 markProgress();
201 toast(on ? "Cycle mode" : "User mode", 1.5);
202}

204bool Session::sceneDone() const {
205 const SceneDef& sc = *scenes_[sceneIdx_];
206 const bool day = !sc.dayNight || world_->daysElapsed() - baseDays_ >= 1.0;
207 const bool rare = sc.rare.empty() || (world_->raresStarted() > baseRares_ && !world_->rareActive());
208 const bool act = sc.action.kind == SceneDef::Action::Kind::None || actions_ > 0;
209 return day && rare && act;
210}

212void Session::stepCycle(double dt) {
213 actionTimer_ -= dt;
214 if (actionTimer_ <= 0) { // the F key, pressed by nobody
215 actionTimer_ = rng_.range(10.0, 90.0);
216 runAction();
217 }
218 // A rare event that has not come yet is waited for: the scene stays on, day after day.
219 if (sceneDone()) {
220 ++seed_; // a scene that comes round again looks different
221 fadePending_ = true;
222 switchScene(sceneIdx_ + 1);
223 }

237void Session::handle(Action a) {
238 if (menu_.handle(a)) { // the open menu consumes navigation
239 applyMenuEvents();
240 return;
241 }
242 switch (a) {
243 case Action::Quit: quit_ = true; break;
244 case Action::Pause: paused_ = !paused_; break;
245 case Action::Info: info_ = !info_; menu_.setValue("info", info_); break;
246 case Action::Interact: setCycle(false); flush(); runAction(); break;
247 case Action::Rare: setCycle(false); flush(); world_->triggerRare(); break;
248 case Action::RareOther: setCycle(false); flush(); world_->triggerRare(RarePick::Other); break;
249 case Action::Mode: setCycle(!cycle_); break;
250 case Action::Audio: { // off, stereo, immersive, and round
251 static const char* const names[] = {"off", "stereo", "immersive"};
252 const int next = (static_cast<int>(mode_) + 1) % 3;
253 setSound(static_cast<SoundMode>(next));
254 toast(std::string("Sound: ") + names[next], 1.2);
255 break;
256 }
257 case Action::Redraw: rebuildPending_ = true; ++seed_; break;
258 case Action::Style:
259 style_ = static_cast<Style>((static_cast<int>(style_) + 1) % 3);
260 menu_.setValue("style", static_cast<size_t>(style_));
261 repaint_ = true;
262 break;
263 case Action::Time:
264 if (scenes_[sceneIdx_]->dayNight) {
265 if (cycle_) { // choosing the time by hand ends the cycle mode; the clock steps on from "Cycle"
266 setCycle(false);
267 timeMode_ = World::TimeMode::Cycle;
268 }
269 flush();
270 timeMode_ = static_cast<World::TimeMode>((static_cast<int>(timeMode_) + 1) % World::kTimeModes);
271 world_->setTimeMode(timeMode_);
272 syncClock();
273 menu_.setValue("time", static_cast<size_t>(timeMode_));
274 toast(timeOptions(true)[static_cast<size_t>(timeMode_)], 1.5);
275 }
276 break;
277 case Action::Left: switchScene(sceneIdx_ + scenes_.size() - 1); break;
278 case Action::Right: switchScene(sceneIdx_ + 1); break;
279 case Action::Menu:
280 case Action::Select:
281 case Action::Back: menu_.open(); break;
282 default: break;
283 }
284}

286void Session::applyMenuEvents() {
287 for (const auto& ev : menu_.takeEvents()) {
288 if (ev.id == "scene") switchScene(ev.value);
289 else if (ev.id == "style") { style_ = static_cast<Style>(ev.value); repaint_ = true; }
290 else if (ev.id == "mode") setCycle(ev.value == 1);
291 else if (ev.id == "time") {
292 setCycle(false); // the chosen time of day replaces the automatic one
293 flush();
294 timeMode_ = static_cast<World::TimeMode>(ev.value);
295 world_->setTimeMode(timeMode_);
296 syncClock();
297 menu_.setValue("time", ev.value); // leaving the cycle mode reset the entry
298 }
299 else if (ev.id == "info") info_ = ev.value == 1;
300 else if (ev.id == "audio") setSound(static_cast<SoundMode>(ev.value));
301 else if (ev.id == "action") { setCycle(false); flush(); runAction(); }
302 else if (ev.id == "rare") { setCycle(false); flush(); world_->triggerRare(); }
303 else if (ev.id == "rare2") { setCycle(false); flush(); world_->triggerRare(RarePick::Other); }
304 else if (ev.id == "resume") menu_.close();
305 else if (ev.id == "quit") quit_ = true;
306 }
307}

349AmbientParams Session::ambient() const {
350 AmbientParams p = ambientFor(*world_);
351 if (outgoing_) { // out with the old scene's sound, then in with the new one's
352 const double progress = 1.0 - fadeLeft_ / kFadeSeconds;
353 if (progress < 0.5) {
354 p = ambientFor(*outgoing_);
355 p.volume = static_cast<float>(1.0 - 2.0 * progress);
356 } else {
357 p.volume = static_cast<float>(2.0 * progress - 1.0);
358 }
359 }
360 if (paused_) p.volume = 0;
361 return p;
362}
