108void onAudio(ma_device* device, void* output, const void*, ma_uint32 frames) {
109 auto* impl = static_cast<AudioOutput::Impl*>(device->pUserData);
110 impl->shared.load(impl->current);
111 SoundEvent e;
112 while (impl->events.pop(e)) impl->ambience.event(e);
113 const int layoutChannels = impl->layout.channels();
114 auto* out = static_cast<float*>(output);
115 if (impl->deviceChannels == layoutChannels) {
116 impl->ambience.render(impl->current, out, static_cast<int>(frames));
117 return;
118 }
119 // The device has more channels than the synth can write: the other channels stay silent.
120 while (frames > 0) {
121 const int n = static_cast<int>(std::min<ma_uint32>(frames, kScratchFrames));
122 impl->ambience.render(impl->current, impl->scratch, n);
123 padChannels(impl->scratch, layoutChannels, out, impl->deviceChannels, n);
124 out += static_cast<size_t>(n) * static_cast<size_t>(impl->deviceChannels);
125 frames -= static_cast<ma_uint32>(n);
126 }
127}

136bool openDevice(AudioOutput::Impl& impl, int channels, const DevicePort* ports) {
137 ma_device_config cfg = ma_device_config_init(ma_device_type_playback);
138 cfg.playback.format = ma_format_f32;
139 cfg.playback.channels = static_cast<ma_uint32>(channels);
140 cfg.sampleRate = 48000;
141 cfg.dataCallback = onAudio;
142 cfg.notificationCallback = onNotification;
143 cfg.pUserData = &impl;
144 ma_channel map[MA_MAX_CHANNELS] = {};
145 if (ports) {
146 for (int i = 0; i < channels; ++i) map[i] = channelOf(ports[i]);
147 cfg.playback.pChannelMap = map;
148 }
149 return ma_device_init(nullptr, &cfg, &impl.device) == MA_SUCCESS;
150}

153DeviceReport reportOf(const AudioOutput::Impl& impl) {
154 DeviceReport r;
155 r.channels = static_cast<int>(impl.device.playback.channels);
156 r.hasMap = true;
157 for (int i = 0; i < r.channels && i < kMaxChannels; ++i) r.position[i] = positionOf(impl.device.playback.channelMap[i]);
158 return r;

165bool AudioOutput::start(const LayoutChoice& choice) {
166 if (impl_->open) return true;
167 Impl& impl = *impl_;
168 impl.choice = choice;
169 impl.routeChanged = false;
170
171 // Sound libraries such as ALSA print their complaints to stderr, which would scribble over
172 // the scene. Silence stderr while the device opens.
173#ifndef _WIN32
174 const int savedErr = dup(STDERR_FILENO);
175 const int devNull = open("/dev/null", O_WRONLY);
176 if (devNull >= 0) dup2(devNull, STDERR_FILENO);
177#endif
178 // First open with what the device has (0 channels): that is what it reports. Never infer the device
179 // from a successful open with a count: a sound server accepts any count and remaps.
180 bool ok = openDevice(impl, 0, nullptr);
181 if (ok) {
182 Negotiation n = negotiate(choice, reportOf(impl));
183 if (n.reopen) { // a layout was asked for: open with its own channel map (the device always reports a map, so automatic never gets here)
184 const DeviceReport native = reportOf(impl);
185 ma_device_uninit(&impl.device);
186 LayoutChoice step = choice;
187 for (;;) {
188 n = negotiate(step, native);
189 DevicePort ports[kMaxChannels];
190 portsForLayout(n.layout, ports);
191 ok = openDevice(impl, n.layout.channels(), ports);
192 if (ok) break;
193 LayoutId lower;
194 if (!lowerLayout(step.id, lower)) break; // stereo did not open: nothing is left to try
195 step.id = lower;
196 }
197 if (!ok) ok = openDevice(impl, 0, nullptr); // last resort: whatever the device has
198 if (ok && impl.device.playback.channels != static_cast<ma_uint32>(n.layout.channels())) {
199 n = negotiate(LayoutChoice{}, reportOf(impl)); // it did not take the map: render to what it says it has
200 }
201 }
202 if (ok) {
203 impl.layout = n.layout;
204 impl.deviceChannels = static_cast<int>(impl.device.playback.channels);
205 impl.downmixed = n.downmixed;
206 impl.ambience.setLayout(impl.layout); // (the device is not started yet: nothing renders)
207 ok = ma_device_start(&impl.device) == MA_SUCCESS;
208 if (!ok) ma_device_uninit(&impl.device);
209 }
210 }
211#ifndef _WIN32
212 if (savedErr >= 0) { dup2(savedErr, STDERR_FILENO); close(savedErr); }
213 if (devNull >= 0) close(devNull);
214#endif
215 impl.open = ok;
216 return ok;
217}
