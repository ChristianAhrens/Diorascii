59void World::leakStep(Entity& e, double dt) {
60 if (!toxin_.active) { // first step: cruising at a third of the depth of the sea, the trail begins
61 e.y = surfaceY_ + std::max(1.0, (seabedY_ - surfaceY_ - e.height()) * 0.3);
62 e.anchorY = e.y;
63 toxin_ = Toxin{};
64 toxin_.active = true;
65 const double cy = e.y + e.height() / 2.0;
66 toxin_.reach = std::max(8.0, std::max(cy - surfaceY_, seabedY_ - cy) * kCellAspect + 3);
67 }
68 e.x += e.vx * dt;
69 e.y = e.anchorY + 0.6 * std::sin(e.age * 0.8);
70 // Poison leaks from the stern: a puff every column or so, bubbles rise from the hull.
71 const double stern = e.movingRight() ? e.x : e.x + e.width() - 1;
72 const double cy = e.y + e.height() / 2.0;
73 if (stern >= 0 && stern < w_ && (toxin_.puffs.empty() || std::fabs(toxin_.puffs.back().x - stern) >= 1.0)) {
74 toxin_.puffs.push_back({stern, cy, toxin_.age});
75 }
76 if (rng_.chance(6.0 * dt)) {
77 spawnEffect("toxinpuff", stern + rng_.range(-1, 1), cy + rng_.range(-2, 2), e.z - 1, rng_.range(-1, 1), -rng_.range(1, 3));
78 }
79 if ((e.vx > 0 && e.x > w_) || (e.vx < 0 && e.x + e.width() < 0)) { // gone: the trail stays
80 toxin_.sourceGone = true;
81 e.dead = true;
82 }

89void World::stepToxin(double dt) {
90 if (!toxin_.active) {
91 toxinLevel_ = 0;
92 return;
93 }
94 toxin_.age += dt;
95 if (toxin_.sourceGone) toxin_.gone += dt;
96 toxin_.strength = toxin_.gone < kToxinLinger ? std::min(1.0, toxin_.age / 2.0)
97 : std::max(0.0, 1.0 - (toxin_.gone - kToxinLinger) / kToxinFade);
98 double widest = 0;
99 for (const auto& p : toxin_.puffs) widest = std::max(widest, puffRadius(p));
100 toxinLevel_ = toxin_.strength * (0.3 + 0.7 * widest / toxin_.reach);
101 if (toxin_.gone >= kToxinLinger + kToxinFade) {
102 toxin_.active = false;
103 toxin_.puffs.clear();
104 return;
105 }
106 // Everything alive that a puff has reached dies; what would come back waits until it has thinned.
107 for (auto& o : entities_) {
108 if (o->dead || o->isRare || !o->life) continue;
109 const double ox = o->x + o->width() / 2.0, oy = o->y + o->height() / 2.0;
110 for (const auto& p : toxin_.puffs) {
111 const double r = puffRadius(p);
112 if (std::fabs(ox - p.x) > r) continue;
113 const double dy = (oy - p.y) * kCellAspect;
114 if ((ox - p.x) * (ox - p.x) + dy * dy <= r * r) {
115 claim(*o, "toxinpuff");
116 break;
117 }
118 }
119 }
120 const double remaining = (toxin_.sourceGone ? kToxinLinger + kToxinFade - toxin_.gone : 30.0 + kToxinFade);
121 for (auto& r : restock_) {
122 if (r.left < remaining) r.left = remaining + rng_.range(0.0, 20.0);
123 }

126void World::renderToxin(Frame& out, Style style, const Light& light) const {
127 if (!toxin_.active) return;
128 // One puff per column (the latest), so that a cell only looks at the columns around it.
129 std::vector<const Toxin::Puff*> byColumn(static_cast<size_t>(w_), nullptr);
130 for (const auto& p : toxin_.puffs) {
131 const int col = static_cast<int>(p.x);
132 if (col >= 0 && col < w_) byColumn[static_cast<size_t>(col)] = &p;
133 }
134 const int span = static_cast<int>(toxin_.reach);
135 for (int y = surfaceY_; y < seabedY_; ++y) {
136 for (int x = 0; x < w_; ++x) {
137 double depth = -1e9; // how far inside the nearest edge of the cloud the cell lies
138 for (int c = std::max(0, x - span); c <= std::min(w_ - 1, x + span); ++c) {
139 const Toxin::Puff* p = byColumn[static_cast<size_t>(c)];
140 if (!p) continue;
141 const double dx = x - p->x, dy = (y - p->y) * kCellAspect;
142 depth = std::max(depth, puffRadius(*p) - std::sqrt(dx * dx + dy * dy));
143 }
144 if (depth <= 0) continue;
145 // Thick inside, a ragged edge, and slowly rolling.
146 const double edge = std::clamp(depth / 5.0, 0.0, 1.0);
147 const double roll = 0.7 + 0.3 * std::sin(time_ * 1.3 + hash2(x / 3, y / 2) * 0.01 + x * 0.21 + y * 0.5);
148 const double density = edge * roll * toxin_.strength;
149 if (density < 0.05) continue;
150 Cell& c = out.at(x, y);
151 if (style == Style::Full) {
152 c.bg = lerp(c.bg, shade(kToxinColor, light), 0.62 * density);
153 c.fg = lerp(c.fg, shade(kToxinColor, light), 0.5 * density);
154 }
155 if (c.ch == U' ' && hash2(x, y + static_cast<int>(time_ * 2)) % 9 == 0) {
156 c.ch = hash2(x, y) % 2 ? U'o' : U'.';
157 c.fg = shade(scale(kToxinColor, 1.15), light);
158 c.defaultFg = style == Style::Mono;
159 }
160 }
161 }

178double World::floodDepth(int x) const {
179 if (!flood_.active) return 0;
180 const double dist = flood_.dir > 0 ? x : w_ - 1 - x;
181 const double rise = std::clamp((flood_.age * kFloodSpeed - dist) / kFloodRise, 0.0, 1.0);
182 return flood_.depthMax * rise * (1.0 - flood_.recede);

185void World::stepFlood(double dt) {
186 for (float& m : mud_) m = std::max(0.0f, m - static_cast<float>(dt / kMudDry));
187 if (!flood_.active) {
188 floodLevel_ = 0;
189 return;
190 }
191 flood_.age += dt;
192 const double covered = (w_ + kFloodRise) / kFloodSpeed; // when the water has reached the far side
193 flood_.recede = std::clamp((flood_.age - covered - kFloodHold) / kFloodRecede, 0.0, 1.0);
194 if (flood_.recede >= 1.0) {
195 flood_.active = false;
196 return;
197 }
198 const double front = flood_.age * kFloodSpeed;
199 floodLevel_ = std::min(1.0, 0.25 + front / (w_ + kFloodRise)) * (1.0 - flood_.recede);
200 for (int x = 0; x < w_; ++x) {
201 if (floodDepth(x) > 0.5) mud_[static_cast<size_t>(x)] = 1.0f;
202 }
203 // Whatever the water reaches at its feet is torn off: trees, bushes, animals on the bank and on the river.
204 for (auto& o : entities_) {
205 if (o->dead || o->isRare || !o->destructible || !(o->life || o->grounded)) continue;
206 const int column = std::clamp(static_cast<int>(o->x + o->width() / 2.0), 0, w_ - 1);
207 const double feet = o->y + o->height() - 1;
208 if (floodDepth(column) < h_ - feet) continue;
209 const bool plant = !o->life;
210 claim(*o, "splash");
211 if (plant) { // a trunk goes down the flood
212 spawnEffect("log", o->x + o->width() / 2.0, h_ - flood_.depthMax + rng_.range(0, 1), 3, flood_.dir * kFloodSpeed * 0.75, 0);
213 }
214 }
215 // Branches and logs ride the front.
216 if (flood_.recede == 0 && front < w_ && rng_.chance(9 * dt)) {
217 const double x = flood_.dir > 0 ? front - rng_.range(0, kFloodRise) : w_ - 1 - front + rng_.range(0, kFloodRise);
218 spawnEffect("log", x, h_ - flood_.depthMax + rng_.range(0, 2), 3, flood_.dir * kFloodSpeed * 0.8, 0);
219 }
220 // What was torn off is not replanted for a good while: the land stays devastated.
221 const double remaining = covered + kFloodHold + kFloodRecede - flood_.age;
222 for (auto& r : restock_) {
223 if (r.left < remaining + kReplantMin) r.left = remaining + rng_.range(kReplantMin, kReplantMax);
224 }

227void World::renderFlood(Frame& out, Style style, const Light& light) const {
228 if (!flood_.active) return;
229 const Rgb deep = backdrop({104, 78, 46}, light), shallow = backdrop({150, 116, 70}, light);
230 const double front = flood_.age * kFloodSpeed;
231 for (int x = 0; x < w_; ++x) {
232 const int depth = static_cast<int>(std::floor(floodDepth(x)));
233 if (depth < 1) continue;
234 // Just behind the front the water is a sparse tumble of waves; it fills in to the solid flood.
235 const double behind = front - (flood_.dir > 0 ? x : w_ - 1 - x);
236 const double solid = std::clamp(behind / kWaveLength, 0.0, 1.0);
237 for (int y = h_ - depth; y < h_; ++y) {
238 if (y < 0) continue;
239 Cell& c = out.at(x, y);
240 const bool top = y == h_ - depth;
241 const double wave = std::sin(x * 0.7 - time_ * 5.0 * flood_.dir + y * 1.3) + std::sin(x * 0.31 - time_ * 3.0);
242 const bool filled = solid >= 1.0 || (hash2(x * 5 + y, static_cast<int>(time_ * 6)) % 100) < solid * 100.0 - 5.0;
243 if (!filled) { // a gap in the waves: only a splash of foam on the land behind
244 if (hash2(x, y + static_cast<int>(time_ * 8)) % 3 == 0) {
245 c.ch = wave > 0.4 ? U'~' : wave > -0.4 ? U'-' : U'\'';
246 c.fg = shade(scale(shallow, 1.6), light);
247 c.defaultFg = style == Style::Mono;
248 }
249 continue;
250 }
251 c.bg = lerp(shallow, deep, std::min(1.0, 0.25 + 0.15 * (y - (h_ - depth))));
252 c.fg = shade(scale(shallow, 1.5), light);
253 c.ch = top ? (wave > 0.2 ? U'~' : U'-') : wave > 1.0 ? U'~' : wave > 0.2 ? U'-' : (hash2(x, y) % 6 == 0 ? U'.' : U' ');
254 c.bold = top;
255 c.defaultBg = style != Style::Full;
256 c.defaultFg = style == Style::Mono;
257 }
258 }

261void World::poachStep(Entity& e, double dt) {
262 if (e.cruise <= 0) e.cruise = std::max(std::fabs(e.vx), 8.0); // the speed the sprite was given
263 const double cx = e.x + e.width() / 2.0, cy = e.y + e.height() / 2.0;
264 const Entity* target = nullptr;
265 double best = 1e9;
266 for (const auto& o : entities_) {
267 if (o->dead || o->isRare || !o->life || isAsleep(*o)) continue;
268 const double d = std::fabs(o->x + o->width() / 2.0 - cx) + 2.0 * std::fabs(o->y + o->height() / 2.0 - cy);
269 if (d < best) {
270 best = d;
271 target = o.get();
272 }
273 }
274 e.idle = target ? 0 : e.idle + dt;
275 if (e.age <= kPoachMax) { // what has been shot does not come back while the poachers are about
276 for (auto& r : restock_) r.left = std::max(r.left, 8.0);
277 }
278 const bool leaving = e.age > kPoachMax || e.idle > kPoachIdle;
279 double wantX = e.vx, wantY = e.y;
280 if (leaving) { // off to the nearer edge, a little higher
281 wantX = (cx < w_ / 2.0 ? -1 : 1) * e.cruise * 1.3;
282 wantY = e.y - 2;
283 } else if (target) {
284 const double tx = target->x + target->width() / 2.0;
285 wantX = std::clamp((tx - cx) * 1.2, -e.cruise, e.cruise);
286 // Hovers a few rows above its target, never lower than the ground.
287 wantY = std::clamp(target->y - e.height() - 3.0, 1.0, std::max(1.0, seabedY_ - e.height() - 2.0));
288 }
289 e.vx += std::clamp(wantX - e.vx, -14.0 * dt, 14.0 * dt);
290 e.x += e.vx * dt;
291 e.y += std::clamp(wantY - e.y, -3.0 * dt, 3.0 * dt);
292 if (std::fabs(e.vx) > 1.0) e.mirrored = (e.vx > 0) != e.facesRight;
293 if (!leaving) {
294 lash(e);
295 } else if (e.x > w_ || e.x + e.width() < 0) {
296 e.dead = true;
297 }

300void World::maulStep(Entity& e, double dt) {
301 if (e.cruise <= 0) { // the first step: it comes over the highest ridge
302 e.cruise = std::max(std::fabs(e.vx), 4.0);
303 int peak = w_ / 2;
304 for (int x = 0; x < w_; ++x) {
305 if (ridgeRow(x) < ridgeRow(peak)) peak = x;
306 }
307 e.x = peak - e.width() / 2.0;
308 e.vx = 0.01;
309 e.behindRidge = true;
310 }
311 if (e.age < kClimb) { // it climbs over the ridge from the back: first hidden by the mountain, then rising above it
312 const int peak = static_cast<int>(e.x + e.width() / 2.0);
313 e.fps = 3.0;
314 const double foot = ridgeRow(peak) + 1 + e.height() * (1.0 - e.age / kClimb);
315 e.y = foot - e.height() + 1;
316 e.z = kTitanZ;
317 return;
318 }
319 e.behindRidge = false;
320 const double cx = e.x + e.width() / 2.0;
321 const Entity* target = nullptr;
322 double best = 1e9;
323 for (const auto& o : entities_) {
324 if (o->dead || o->isRare || !o->life || !o->grounded || isAsleep(*o)) continue;
325 const double d = std::fabs(o->x + o->width() / 2.0 - cx) + 3.0 * std::fabs(o->y + o->height() - (e.y + e.height())); // near it, and on its level
326 if (d < best) {
327 best = d;
328 target = o.get();
329 }
330 }
331 e.idle = target ? 0 : e.idle + dt;
332 if (e.age <= kPoachMax) {
333 for (auto& r : restock_) r.left = std::max(r.left, 8.0);
334 }
335 const bool leaving = e.age > kPoachMax || e.idle > kPoachIdle;
336 double want = 0;
337 if (leaving) { // back over the mountain, off to the nearer side
338 want = (cx < w_ / 2.0 ? -1 : 1) * e.cruise;
339 } else if (target) {
340 const double gap = target->x + target->width() / 2.0 - cx;
341 // It has to get to the animal itself: it walks until the animal is under its body (and then kills it).
342 want = std::fabs(gap) > 2.0 ? (gap > 0 ? 1 : -1) * e.cruise : 0.0;
343 if (std::fabs(gap) > 1) e.vx = (gap > 0 ? 1 : -1) * std::max(std::fabs(e.vx), 0.01); // faces its target
344 }
345 const double dir = e.vx > 0 ? 1 : -1;
346 e.vx = want != 0 ? want : dir * 0.01;
347 e.x += want * dt;
348 e.mirrored = (e.vx > 0) != e.facesRight;
349 // Its feet go where its prey stands (the mountain is a face, with the meadow at the bottom): down
350 // the slope into the valley. Never above the skyline, and not into the stream. Leaving, it climbs.
351 const int column = static_cast<int>(e.x + e.width() / 2.0);
352 const double lowest = static_cast<double>(riverRows_ > 0 ? riverTopY_ - 1 : h_ - 1);
353 double foot = ridgeRow(column) + 2.0;
354 if (!leaving && target) foot = std::clamp(target->y + target->height() - 1.0, ridgeRow(column) + 1.0, lowest);
355 const double wantY = foot - e.height() + 1;
356 e.y += std::clamp(wantY - e.y, -7.0 * dt, 7.0 * dt);
357 e.anchorY = e.y;
358 e.z = depthFor(e.y + e.height());
359 // Walking shakes the ground: the legs move only while it travels.
360 e.fps = want != 0 ? 3.0 : 0.0;
361 if (!leaving) { // whatever is under its body when it gets there is grabbed and killed
362 for (auto& o : entities_) {
363 if (o->dead || o->isRare || !o->life || !o->grounded || isAsleep(*o)) continue;
364 const double mid = o->x + o->width() / 2.0;
365 if (mid >= e.x + 4 && mid <= e.x + e.width() - 4 && std::fabs((o->y + o->height()) - (e.y + e.height() - 1)) <= 3.5) {
366 claim(*o, "snowpuff");
367 spawnEffect("snowpuff", mid - 1, o->y, e.z - 1, rng_.range(-2, 2), -rng_.range(0.5, 2));
368 }
369 }
370 } else if (e.x > w_ || e.x + e.width() < 0) {
371 e.dead = true;
372 }

375void World::clashStep(Entity& e, double dt) {
376 Entity* other = nullptr; // the winged monster
377 for (auto& o : entities_) {
378 if (!o->dead && o->sprite->name() == "winged_titan") other = o.get();
379 }
380 for (auto& o : incoming_) {
381 if (o->sprite->name() == "winged_titan") other = o.get();
382 }
383 Clash& c = clash_;
384 if (c.started && c.phase < 3) { // everyone on the streets runs from the monsters, and nothing new comes while they are about
385 for (auto& o : entities_) {
386 if (o->dead || o->fleeing || o->isRare || !o->grounded || std::fabs(o->vx) < 0.2 || o->driveState != 0 ||
387 o->sprite->name() == "train") {
388 continue;
389 }
390 const double cx = o->x + o->width() / 2.0;
391 o->fleeing = true;
392 o->vx = (cx < c.meetX ? -1.0 : 1.0) * std::max(14.0, std::fabs(o->vx) * 3.0);
393 o->mirrored = (o->vx > 0) != o->facesRight;
394 }
395 for (auto& r : restock_) r.left = std::max(r.left, 6.0);
396 }
397 if (!c.started) { // both are called: the lizard walks in from one side, the winged one flies in from the other
398 c = Clash{};
399 auto flyer = pack_->find("winged_titan");
400 if (!flyer || h_ < 28 || w_ < 110) { // no room for them on this screen
401 e.dead = true;
402 return;
403 }
404 c.started = true;
405 c.dir = rng_.chance(0.5) ? 1 : -1;
406 c.meetX = w_ * rng_.range(0.4, 0.6);
407 spawnCreature(flyer, false, true);
408 if (!incoming_.empty()) incoming_.back()->isRare = true;
409 c.nextBlow = rng_.range(1.0, 2.0);
410 e.fps = 0;
411 e.x = c.dir > 0 ? -e.width() : w_;
412 return;
413 }
414 if (!other && c.phase < 2) { // the other one has not joined the world yet
415 return;
416 }
417 c.t += dt;
418 const bool whole = stepBuildings(dt); // ruins smoulder and are rebuilt as time goes by
419 const double footY = seabedY_ + 3; // the row below the feet: they stand in the road
420 const double ew = e.width();
421 e.z = kTitanZ;
422 e.y = footY - e.height();
423 e.mirrored = c.dir < 0;
424 // The fire spreads out from the middle of the fight as soon as it begins: every building catches
425 // when the front reaches it, burns for a few seconds and comes down. Once the fight is over the
426 // rebuilding spreads out from the middle in the same way.
427 auto spreadFire = [&](double step) {
428 if (c.order.empty() && c.nextIgnite == 0) {
429 for (size_t i = 0; i < city_.size(); ++i) {
430 const double d = std::fabs(city_[i].x + city_[i].w / 2.0 - c.meetX);
431 if (d <= w_ * kSpreadReach) c.order.push_back({d, static_cast<int>(i)});
432 }
433 std::sort(c.order.begin(), c.order.end());
434 blazeFails_ = true;
435 }
436 c.spread += step;
437 while (c.nextIgnite < c.order.size() && c.order[c.nextIgnite].first <= c.spread * kSpreadSpeed) {
438 Building& b = city_[static_cast<size_t>(c.order[c.nextIgnite].second)];
439 if (b.state == Building::State::Normal) {
440 b.state = Building::State::Burning;
441 b.timer = 0;
442 b.hold = true;
443 if (std::find(blazeBuildings_.begin(), blazeBuildings_.end(), c.order[c.nextIgnite].second) == blazeBuildings_.end()) {
444 blazeBuildings_.push_back(c.order[c.nextIgnite].second);
445 }
446 }
447 ++c.nextIgnite;
448 }
449 for (const auto& [dist, idx] : c.order) {
450 Building& b = city_[static_cast<size_t>(idx)];
451 if (b.state == Building::State::Burning && b.hold && b.timer >= kClashBurn) { // it comes down
452 b.state = Building::State::Normal;
453 ruinBuilding(idx);
454 b.hold = true;
455 } else if (b.state == Building::State::Ruin && b.hold && c.rebuildT0 >= 0 && b.timer >= kMinRuin &&
456 (c.t - c.rebuildT0) * kSpreadSpeed >= dist) {
457 b.hold = false; // the rebuilding has reached it
458 }
459 }
460 };
461 // Where each stands once the fight is on: fronts apart by kTitanGap, the lizard on its own side.
462 const double lizardStop = c.dir > 0 ? c.meetX - kTitanGap / 2.0 - ew : c.meetX + kTitanGap / 2.0;
463 if (c.phase == 0) { // arriving
464 const double toGo = (lizardStop - e.x) * c.dir;
465 if (toGo > 0.3) {
466 e.x += c.dir * std::min(kTitanSpeed * dt, toGo);
467 e.frame = static_cast<size_t>(std::max(0.0, std::floor(c.t * 2.0))) % 2;
468 } else {
469 e.frame = 0;
470 }
471 bool landed = false;
472 if (other) { // the winged one comes in over the roofs and lands where the lizard stops
473 const double ow = other->width();
474 const double stop = c.dir > 0 ? c.meetX + kTitanGap / 2.0 : c.meetX - kTitanGap / 2.0 - ow;
475 if (!c.placed) {
476 other->x = c.dir > 0 ? w_ : -ow;
477 c.placed = true;
478 }
479 other->mirrored = c.dir < 0;
480 const double left = (stop - other->x) * -c.dir; // distance still to fly
481 if (left > 0.5) {
482 other->x -= c.dir * std::min(15.0 * dt, left);
483 other->y = std::max(1.0, footY - other->height() - 12 + 0.0);
484 other->frame = static_cast<size_t>(std::floor(c.t * 6.0)) % 2; // wings beating
485 } else { // it settles on the street
486 other->y = std::min(footY - other->height(), other->y + 7.0 * dt);
487 other->frame = 2;
488 landed = other->y >= footY - other->height() - 0.01;
489 }
490 other->z = kTitanZ;
491 other->fps = 0;
492 }
493 if (toGo <= 0.3 && landed) {
494 c.phase = 1;
495 c.fightT = 0;
496 }
497 return;
498 }
499 if (c.phase == 1 && other) { // the fight
500 c.fightT += dt;
501 spreadFire(dt);
502 other->y = footY - other->height();
503 other->z = kTitanZ;
504 // The blows come in turns: the lizard spits blue fire at its rival, the rival spits violet fire
505 // back, and now and then both do at once.
506 if (c.fightT >= c.nextBlow && c.fightT < kFightTime) {
507 c.nextBlow = c.fightT + rng_.range(kBlowMin, kBlowMax);
508 const int kind = c.blows % 3; // 0 the lizard, 1 the winged one, 2 both
509 auto spit = [&](const Entity& from, const Entity& to, Rgb color, double ox) {
510 Tongue t;
511 t.attacker = from.id;
512 t.victim = to.id;
513 t.color = color;
514 t.harmless = true;
515 t.fire = true;
516 t.reach = 0.45;
517 t.dur = kBreathTime;
518 t.ox = ox;
519 tongues_.push_back(t);
520 };
521 if (kind != 1) {
522 c.breath = kBreathTime;
523 spit(e, *other, kBreathColor, c.dir > 0 ? ew - 1 : 0);
524 }
525 if (kind != 0) {
526 c.strike = kBreathTime;
527 spit(*other, e, kVioletColor, c.dir > 0 ? 0 : other->width() - 1);
528 }
529 ++c.blows;
530 }
531 e.frame = c.breath > 0 ? 2 : 0;
532 other->frame = c.strike > 0 ? 4 : 2;
533 c.breath = std::max(0.0, c.breath - dt);
534 c.strike = std::max(0.0, c.strike - dt);
535 // Each blow shakes the one that is hit.
536 const double shake = (c.breath > 0 || c.strike > 0) ? std::sin(c.fightT * 40.0) * 0.5 : 0.0;
537 const double otherStop = c.dir > 0 ? c.meetX + kTitanGap / 2.0 : c.meetX - kTitanGap / 2.0 - other->width();
538 other->x = otherStop + (c.breath > 0 ? shake : 0.0);
539 e.x = lizardStop + (c.strike > 0 ? shake : 0.0);
540 if (c.breath > 0 || c.strike > 0) {
541 spawnEffect("smoke", c.meetX + rng_.range(-4, 4), footY - rng_.range(4, 14), 4, rng_.range(-4, 4), -rng_.range(1, 3));
542 }
543 if (c.fightT >= kFightTime && c.breath <= 0 && c.strike <= 0) { // the winged one falls
544 c.phase = 2;
545 c.endT = 0;
546 for (int i = 0; i < 14; ++i) {
547 spawnEffect("smoke", other->x + rng_.range(0, other->width()), other->y + rng_.range(0, other->height()), 4,
548 rng_.range(-5, 5), -rng_.range(1, 4));
549 }
550 other->dead = true;
551 c.rebuildT0 = c.t; // the fight is over: the rebuilding starts in the middle
552 }
553 return;
554 }
555 if (c.phase == 2) {
556 spreadFire(dt); // the lizard roars, then walks on across the screen
557 c.endT += dt;
558 if (c.endT < 3.0) {
559 e.frame = 2;
560 } else {
561 e.x += c.dir * kTitanSpeed * dt;
562 e.frame = static_cast<size_t>(std::floor(c.endT * 2.0)) % 2;
563 }
564 if ((c.dir > 0 && e.x > w_) || (c.dir < 0 && e.x + ew < 0)) {
565 c.phase = 3;
566 e.x = -1000; // out of the picture; the entity stays until the city is whole again
567 }
568 return;
569 }
570 // Only the ruins are left: they are rebuilt, one after the other.
571 spreadFire(dt);
572 if (c.phase == 3 && whole && c.nextIgnite >= c.order.size()) {
573 blazeBuildings_.clear();
574 e.dead = true;
575 }
576}

578void World::asteroidStep(Entity& e, double dt) {
579 constexpr double kFlight = 9.0; // seconds from far away to the strike
580 (void)dt;
581 if (!asteroid_.started) {
582 asteroid_.started = true;
583 asteroid_.xt = w_ * rng_.range(0.4, 0.6);
584 asteroid_.x0 = asteroid_.xt + w_ * rng_.range(-0.35, 0.35);
585 }
586 const double p = std::clamp(e.age / kFlight, 0.0, 1.0);
587 const double cx = asteroid_.x0 + (asteroid_.xt - asteroid_.x0) * p;
588 const double cy = 1.0 + (seabedY_ - e.height() / 2.0 - 1.0) * std::pow(p, 1.7);
589 e.x = cx - e.width() / 2.0;
590 e.y = cy - e.height() / 2.0;
591 // Far away it is a speck; the frames grow from there.
592 const size_t frames = e.sprite->frameCount();
593 e.frame = std::min(frames - 1, static_cast<size_t>(std::pow(p, 1.25) * static_cast<double>(frames)));
594 asteroidLevel_ = p;
595 if (p >= 1.0) {
596 asteroidLevel_ = 0;
597 bigImpactAt(cx);
598 asteroid_ = Asteroid{};
599 e.dead = true;
600 }

603void World::bigImpactAt(double cx) {
604 impacts_ += 1;
605 const int bank = (riverRows_ > 0 ? riverTopY_ : h_) - seabedY_;
606 const int maxDepth = std::clamp(bank - 1, 1, 3); // always leave some ground
607 for (int x = 0; x < w_; ++x) { // a crater across the whole width: deepest where it hit
608 const double t = std::fabs(x - cx) / std::max(w_ / 2.0, 1.0);
609 crater_[static_cast<size_t>(x)] = std::max(crater_[static_cast<size_t>(x)], std::max(1.0, maxDepth * (1.0 - 0.6 * t * t)));
610 }
611 auto wreck = [&](Entity& o) { // the shock wrecks everything standing on the ground
612 if (o.dead || !o.grounded || !o.destructible || o.isRare) return;
613 claim(o, ""); // what stood here is replaced later, as after the other catastrophes
614 ++destroyed_;
615 for (int i = 0; i < 3; ++i) {
616 spawnEffect("debris", o.x + rng_.range(0, o.width()), o.y + rng_.range(0, o.height()), 1, rng_.range(-6, 6), -rng_.range(2, 7));
617 }
618 };
619 for (auto& o : entities_) wreck(*o);
620 for (size_t i = 0; i < incoming_.size(); ++i) wreck(*incoming_[i]);
621 for (auto& r : restock_) r.left = std::max(r.left, rng_.range(12.0, 40.0)); // nothing is replaced for a while
622 flash(1.6);
623 if (auto bang = pack_->find("flash")) {
624 for (int x = 0; x < w_; x += std::max(bang->width(), 1)) {
625 spawnEffect("flash", x, seabedY_ - bang->height(), 1, 0, 0);
626 }
627 }
628 for (int x = 0; x < w_; x += 2) { // a wave of dust runs out from the middle to both edges
629 const double away = x < cx ? -1.0 : 1.0;
630 spawnEffect("dust", x, seabedY_ - 1, 1, away * (4.0 + std::fabs(x - cx) * 0.18) + rng_.range(-1, 1), -rng_.range(0.5, 3.5));
631 }
632 for (int i = 0; i < 70; ++i) { // and rock is thrown high into the black sky
633 spawnEffect("debris", cx + rng_.range(-w_ / 8.0, w_ / 8.0), seabedY_ - 1, 1, rng_.range(-14, 14), -rng_.range(3, 12));
634 }
