47Session::Session(const AssetPack& pack, std::vector<std::shared_ptr<const SceneDef>> scenes,
48 const Config& config, int width, int height)
49 : pack_(pack),
50 scenes_(std::move(scenes)),
51 sceneIdx_(config.scene % scenes_.size()),
52 w_(width),
53 h_(height),
54 seed_(config.seed),
55 style_(config.style),
56 timeMode_(config.realtime ? World::TimeMode::Realtime : World::TimeMode::Cycle),
57 carriedPhase_(config.hour >= 0 ? config.hour / 24.0 : -1),
58 menu_("DIORASCII"),
59 info_(config.info) {
60 std::vector<std::string> titles;
61 for (const auto& sc : scenes_) titles.push_back(sc->title);
62 menu_.addChoice("scene", "Scene", titles, sceneIdx_);
63 menu_.addChoice("style", "Style",
64 {styleName(Style::Full), styleName(Style::Terminal), styleName(Style::Mono)},
65 static_cast<size_t>(style_));
66 menu_.addChoice("time", "Time of day", timeOptions(scenes_[sceneIdx_]->dayNight),
67 scenes_[sceneIdx_]->dayNight ? static_cast<size_t>(timeMode_) : 0);
68 menu_.addChoice("info", "Info line", {"off", "on"}, info_ ? 1 : 0);
69 menu_.addChoice("audio", "Sound", {"off", "on"}, 0);
70 menu_.addButton("action", scenes_[sceneIdx_]->action.label);
71 menu_.addButton("rare", "Rare event");
72 menu_.addButton("rare2", "Other rare event");
73 menu_.addButton("resume", "Resume");
74 if (config.quitEntry) menu_.addButton("quit", "Quit");
75 menu_.setFooter({"diorascii " DIORASCII_VERSION, "Copyright (c) 2026 Christian Ahrens"});
76 rebuild();
77 toast(scenes_[sceneIdx_]->title, 2.0);
78 if (config.sound) setSound(true);
79}

131void Session::handle(Action a) {
132 if (menu_.handle(a)) { // the open menu consumes navigation
133 applyMenuEvents();
134 return;
135 }
136 switch (a) {
137 case Action::Quit: quit_ = true; break;
138 case Action::Pause: paused_ = !paused_; break;
139 case Action::Info: info_ = !info_; menu_.setValue("info", info_); break;
140 case Action::Interact: flush(); runAction(); break;
141 case Action::Rare: flush(); world_->triggerRare(); break;
142 case Action::RareOther: flush(); world_->triggerRare(RarePick::Other); break;
143 case Action::Audio: setSound(!sound_); break;
144 case Action::Redraw: rebuildPending_ = true; ++seed_; break;
145 case Action::Style:
146 style_ = static_cast<Style>((static_cast<int>(style_) + 1) % 3);
147 menu_.setValue("style", static_cast<size_t>(style_));
148 repaint_ = true;
149 break;
150 case Action::Time:
151 if (scenes_[sceneIdx_]->dayNight) {
152 flush();
153 timeMode_ = static_cast<World::TimeMode>((static_cast<int>(timeMode_) + 1) % World::kTimeModes);
154 world_->setTimeMode(timeMode_);
155 syncClock();
156 menu_.setValue("time", static_cast<size_t>(timeMode_));
157 toast(timeOptions(true)[static_cast<size_t>(timeMode_)], 1.5);
158 }
159 break;
160 case Action::Left: switchScene(sceneIdx_ + scenes_.size() - 1); break;
161 case Action::Right: switchScene(sceneIdx_ + 1); break;
162 case Action::Menu:
163 case Action::Select:
164 case Action::Back: menu_.open(); break;
165 default: break;
166 }
167}

169void Session::applyMenuEvents() {
170 for (const auto& ev : menu_.takeEvents()) {
171 if (ev.id == "scene") switchScene(ev.value);
172 else if (ev.id == "style") { style_ = static_cast<Style>(ev.value); repaint_ = true; }
173 else if (ev.id == "time") {
174 flush();
175 timeMode_ = static_cast<World::TimeMode>(ev.value);
176 world_->setTimeMode(timeMode_);
177 syncClock();
178 }
179 else if (ev.id == "info") info_ = ev.value == 1;
180 else if (ev.id == "audio") setSound(ev.value == 1);
181 else if (ev.id == "action") { flush(); runAction(); }
182 else if (ev.id == "rare") { flush(); world_->triggerRare(); }
183 else if (ev.id == "rare2") { flush(); world_->triggerRare(RarePick::Other); }
184 else if (ev.id == "resume") menu_.close();
185 else if (ev.id == "quit") quit_ = true;
186 }
187}
