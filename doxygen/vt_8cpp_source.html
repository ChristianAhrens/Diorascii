12int rgbTo256(Rgb c) {
13 auto level = [](int v) { return v < 48 ? 0 : v < 115 ? 1 : (v - 35) / 40; };
14 const int r = level(c.r), g = level(c.g), b = level(c.b);
15 static const int steps[] = {0, 95, 135, 175, 215, 255};
16 const int cubeDist = (c.r - steps[r]) * (c.r - steps[r]) + (c.g - steps[g]) * (c.g - steps[g]) +
17 (c.b - steps[b]) * (c.b - steps[b]);
18
19 const int avg = (c.r + c.g + c.b) / 3;
20 const int gi = avg < 8 ? 0 : avg > 238 ? 23 : (avg - 8) / 10;
21 const int gv = 8 + gi * 10;
22 const int grayDist = (c.r - gv) * (c.r - gv) + (c.g - gv) * (c.g - gv) + (c.b - gv) * (c.b - gv);
23
24 return grayDist < cubeDist ? 232 + gi : 16 + 36 * r + 6 * g + b;
25}

27int rgbToAnsi16(Rgb c) {
28 // xterm's default palette; index 0 (black) is deliberately left out.
29 static const Rgb pal[16] = {{0, 0, 0}, {205, 0, 0}, {0, 205, 0}, {205, 205, 0},
30 {0, 0, 238}, {205, 0, 205}, {0, 205, 205}, {229, 229, 229},
31 {127, 127, 127}, {255, 0, 0}, {0, 255, 0}, {255, 255, 0},
32 {92, 92, 255}, {255, 0, 255}, {0, 255, 255}, {255, 255, 255}};
33 int best = 8, bestD = 1 << 30;
34 for (int i = 1; i < 16; ++i) {
35 const int dr = c.r - pal[i].r, dg = c.g - pal[i].g, db = c.b - pal[i].b;
36 const int d = dr * dr + dg * dg + db * db;
37 if (d < bestD) { bestD = d; best = i; }
38 }
39 return best;
40}

50int dist2(Rgb a, Rgb b) {
51 const int dr = a.r - b.r, dg = a.g - b.g, db = a.b - b.b;
52 return dr * dr + dg * dg + db * db;

56int consoleNearest(Rgb c, int first, int last) {
57 int best = first, bestD = 1 << 30;
58 for (int i = first; i < last; ++i) {
59 const int d = dist2(c, kConsolePal[i]);
60 if (d < bestD) { bestD = d; best = i; }
61 }
62 return best;
63}

68Cell toConsoleCell(Cell c) {
69 if (!c.defaultFg) c.fg = kConsolePal[consoleNearest(c.fg, 1, 16)];
70 if (!c.defaultBg) {
71 // Backdrops are scaled down a little so daylight skies don't glare.
72 c.bg = kConsolePal[consoleNearest(c.reverse ? c.bg : scale(c.bg, 0.8), 0, 8)];
73 }
74 return c;

78std::string colorParam(bool fg, Rgb c, ColorMode mode) {
79 switch (mode) {
80 case ColorMode::TrueColor:
81 return std::string(fg ? "38;2;" : "48;2;") + std::to_string(c.r) + ';' + std::to_string(c.g) +
82 ';' + std::to_string(c.b);
83 case ColorMode::Ansi256:
84 return std::string(fg ? "38;5;" : "48;5;") + std::to_string(rgbTo256(c));
85 case ColorMode::Console: {
86 const int i = consoleNearest(c, fg ? 1 : 0, fg ? 16 : 8);
87 return std::to_string((fg ? (i < 8 ? 30 : 90) : 40) + (i & 7));
88 }
89 case ColorMode::Ansi16: {
90 const int i = rgbToAnsi16(c);
91 const int base = fg ? (i < 8 ? 30 : 90) : (i < 8 ? 40 : 100);
92 return std::to_string(base + (i & 7));
93 }
94 }
95 return "";
96}

109std::string encodeFrame(const Frame* prev, const Frame& cur, ColorMode mode) {
110 std::string out;
111 const bool full = !prev || prev->width() != cur.width() || prev->height() != cur.height();
112 if (full) out += "\x1b[0m\x1b[2J";
113
114 bool haveStyle = false;
115 Cell style;
116 for (int y = 0; y < cur.height(); ++y) {
117 int nextX = -1; // x the cursor sits at after the last write on this row
118 for (int x = 0; x < cur.width(); ++x) {
119 if (!full && prev->at(x, y) == cur.at(x, y)) continue;
120 const Cell c = mode == ColorMode::Console ? toConsoleCell(cur.at(x, y)) : cur.at(x, y);
121
122 if (x != nextX) {
123 out += "\x1b[" + std::to_string(y + 1) + ';' + std::to_string(x + 1) + 'H';
124 }
125 std::string params;
126 auto add = [&params](const std::string& p) {
127 if (!params.empty()) params += ';';
128 params += p;
129 };
130 if (!haveStyle || style.bold != c.bold) add(c.bold ? "1" : "22");
131 if (!haveStyle || style.reverse != c.reverse) add(c.reverse ? "7" : "27");
132 if (!haveStyle || !sameFg(style, c)) add(c.defaultFg ? "39" : colorParam(true, c.fg, mode));
133 if (!haveStyle || !sameBg(style, c)) add(c.defaultBg ? "49" : colorParam(false, c.bg, mode));
134 if (!params.empty()) out += "\x1b[" + params + 'm';
135 style = c;
136 haveStyle = true;
137
138 appendUtf8(out, c.ch);
139 nextX = x + 1;
140 }
141 }
142 if (haveStyle) out += "\x1b[0m";
143 return out;
144}
