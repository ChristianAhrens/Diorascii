33std::shared_ptr<const SceneDef> SceneDef::parse(std::string_view text, std::string& err) {
34 auto sc = std::make_shared<SceneDef>();
35 Props p;
36 enum class Sec { Header, Population, Decor, Events, Rare } sec = Sec::Header;
37
38 size_t pos = 0;
39 int lineNo = 0;
40 while (pos <= text.size()) {
41 size_t nl = text.find('\n', pos);
42 if (nl == std::string_view::npos) nl = text.size();
43 const std::string line = trim(text.substr(pos, nl - pos));
44 pos = nl + 1;
45 ++lineNo;
46 if (line.empty() || line[0] == '#') continue;
47
48 if (line.substr(0, 3) == "---") {
49 const std::string kind = lower(trim(line.substr(3)));
50 if (kind == "population") sec = Sec::Population;
51 else if (kind == "decor") sec = Sec::Decor;
52 else if (kind == "events") sec = Sec::Events;
53 else if (kind == "rare") sec = Sec::Rare;
54 else { err = at(lineNo, "unknown section '" + kind + "'"); return nullptr; }
55 continue;
56 }
57
58 if (sec == Sec::Header) {
59 const size_t colon = line.find(':');
60 if (colon == std::string::npos) { err = at(lineNo, "expected 'key: value'"); return nullptr; }
61 p.set(lower(trim(line.substr(0, colon))), trim(line.substr(colon + 1)));
62 continue;
63 }
64
65 const auto t = splitWs(line);
66 if (sec == Sec::Population) {
67 // <sprite> per-cells N | per-cols N | count N
68 if (t.size() != 3) { err = at(lineNo, "expected '<sprite> per-cells|per-cols|count <n>'"); return nullptr; }
69 Population pop;
70 pop.sprite = t[0];
71 pop.n = std::atof(t[2].c_str());
72 if (t[1] == "per-cells") pop.kind = Population::Kind::PerCells;
73 else if (t[1] == "per-cols") pop.kind = Population::Kind::PerCols;
74 else if (t[1] == "count") pop.kind = Population::Kind::Count;
75 else { err = at(lineNo, "unknown population mode '" + t[1] + "'"); return nullptr; }
76 if (pop.n <= 0) { err = at(lineNo, "population number must be > 0"); return nullptr; }
77 sc->population.push_back(pop);
78 } else if (sec == Sec::Decor) {
79 // <sprite> left|right|center|random <margin> [y <row>]
80 if (t.size() != 3 && t.size() != 5) { err = at(lineNo, "expected '<sprite> <align> <margin> [y <row>]'"); return nullptr; }
81 Decor d;
82 d.sprite = t[0];
83 d.align = t[1];
84 if (d.align != "left" && d.align != "right" && d.align != "center" && d.align != "random") {
85 err = at(lineNo, "unknown alignment '" + d.align + "'");
86 return nullptr;
87 }
88 d.margin = std::atoi(t[2].c_str());
89 if (t.size() == 5) {
90 if (t[3] != "y") { err = at(lineNo, "expected 'y <row>'"); return nullptr; }
91 d.hasY = true;
92 d.y = std::atoi(t[4].c_str());
93 }
94 sc->decor.push_back(d);
95 } else {
96 // <sprite> <weight>
97 if (t.size() != 2) { err = at(lineNo, "expected '<sprite> <weight>'"); return nullptr; }
98 (sec == Sec::Rare ? sc->rare : sc->events).push_back({t[0], std::atof(t[1].c_str())});
99 }
100 }
101
102 sc->name = p.str("name");
103 if (sc->name.empty()) { err = "missing 'name:' header"; return nullptr; }
104 sc->title = p.str("title", sc->name);
105 sc->order = static_cast<int>(p.num("order", 100));
106 sc->hasMedium = p.str("medium", "yes") != "no";
107 sc->skyFraction = p.num("sky-fraction", sc->skyFraction);
108 const auto rows = p.range("sky-rows", {1, 5});
109 sc->skyRowsMin = static_cast<int>(rows.first);
110 sc->skyRowsMax = static_cast<int>(rows.second);
111 sc->groundRows = static_cast<int>(p.num("ground-rows", 2));
112 sc->starDensity = static_cast<int>(p.num("stars", 0));
113 sc->gravity = p.num("gravity", sc->gravity);
114 sc->dustSettle = std::max(p.num("dust-settle", sc->dustSettle), 1.0);
115 sc->mountains = p.str("mountains", "no") == "yes";
116 sc->peakHeight = std::clamp(p.num("peak-height", sc->peakHeight), 0.3, 1.0);
117 sc->treeline = std::clamp(p.num("treeline", sc->treeline), 0.1, 0.9);
118 sc->snowline = std::clamp(p.num("snowline", sc->snowline), sc->treeline + 0.05, 0.95);
119 sc->riverRows = std::max(static_cast<int>(p.num("river-rows", 0)), 0);
120 sc->riverFlow = p.num("river-flow", sc->riverFlow);
121 sc->weatherDuration = std::max(p.num("weather-duration", sc->weatherDuration), 1.0);
122 if (p.has("overcast-tint")) {
123 const auto t = p.list("overcast-tint");
124 if (t.size() != 2 || !parseHexColor(t[0], sc->overcastTint)) {
125 err = "'overcast-tint' needs a '#rrggbb' color and a strength from 0 to 1";
126 return nullptr;
127 }
128 sc->overcastTintAmount = std::clamp(std::atof(t[1].c_str()), 0.0, 1.0);
129 }
130 sc->dayNight = p.str("day-night", "no") == "yes";
131 sc->airless = p.str("airless", "no") == "yes";
132 sc->city = p.str("city", "no") == "yes";
133 sc->avenues = std::clamp(static_cast<int>(p.num("avenues", 0)), 0, 6);
134 sc->skyline = p.range("skyline", sc->skyline);
135 sc->dayLength = std::max(p.num("day-length", sc->dayLength), 1.0);
136 sc->startHour = std::clamp(p.num("start-hour", sc->startHour), 0.0, 24.0);
137 sc->sunSprite = p.str("sun-sprite");
138 sc->moonSprite = p.str("moon-sprite");
139 if (p.has("action")) {
140 const auto t = p.list("action");
141 if (t.size() == 1 && t[0] == "none") {
142 sc->action.kind = Action::Kind::None;
143 } else if ((t.size() == 2 && (t[0] == "drop" || t[0] == "impact" || t[0] == "weather" || t[0] == "summon")) ||
144 (t.size() == 3 && t[0] == "weather")) {
145 sc->action.kind = t[0] == "drop" ? Action::Kind::Drop
146 : t[0] == "impact" ? Action::Kind::Impact
147 : t[0] == "summon" ? Action::Kind::Summon
148 : Action::Kind::Weather;
149 sc->action.sprite = t[1];
150 if (t.size() == 3) sc->action.extra = t[2];
151 } else {
152 err = "'action' must be 'none', 'drop <sprite>', 'impact <sprite>', 'summon <sprite>' or 'weather <sprite> [<funnel>]'";
153 return nullptr;
154 }
155 }
156 sc->action.label = p.str("action-label", sc->action.kind == Action::Kind::None ? "No action" : "Action");
157 sc->eventFirst = p.range("event-first", sc->eventFirst);
158 sc->eventInterval = p.range("event-interval", sc->eventInterval);
159 sc->rareFirst = p.range("rare-first", sc->rareFirst);
160 sc->rareInterval = p.range("rare-interval", sc->rareInterval);
161 if (p.has("ground-glyphs")) sc->groundGlyphs = decodeUtf8(p.str("ground-glyphs"));
162 if (p.has("forest-glyphs")) sc->forestGlyphs = decodeUtf8(p.str("forest-glyphs"));
163 if (p.has("rock-glyphs")) sc->rockGlyphs = decodeUtf8(p.str("rock-glyphs"));
164 if (p.has("snow-glyphs")) sc->snowGlyphs = decodeUtf8(p.str("snow-glyphs"));
165 if (p.has("surface-glyphs")) {
166 sc->surfaceGlyphs = decodeUtf8(p.str("surface-glyphs"));
167 if (!sc->surfaceGlyphs.empty() && sc->surfaceGlyphs.size() != 3) {
168 err = "'surface-glyphs' needs exactly 3 characters (crest, mid, trough) or nothing";
169 return nullptr;
170 }
171 }
172
173 Rgb sky[2] = {sc->skyTop, sc->skyLow}, med[2] = {sc->mediumTop, sc->mediumDeep},
174 gnd[2] = {sc->groundA, sc->groundB}, sur[1] = {sc->surfaceFg},
175 riv[2] = {sc->riverTop, sc->riverDeep}, forest[2] = {sc->forestA, sc->forestB},
176 rock[2] = {sc->rockA, sc->rockB}, snow[2] = {sc->snowA, sc->snowB};
177 if (!colors(p, "forest-colors", forest, 2) || !colors(p, "rock-colors", rock, 2) ||
178 !colors(p, "snow-colors", snow, 2)) {
179 err = "'forest-colors', 'rock-colors' and 'snow-colors' need two '#rrggbb' values";
180 return nullptr;
181 }
182 sc->forestA = forest[0]; sc->forestB = forest[1];
183 sc->rockA = rock[0]; sc->rockB = rock[1];
184 sc->snowA = snow[0]; sc->snowB = snow[1];
185 if (!colors(p, "sky", sky, 2) || !colors(p, "medium-colors", med, 2) ||
186 !colors(p, "ground-colors", gnd, 2) || !colors(p, "surface-color", sur, 1) ||
187 !colors(p, "river-colors", riv, 2)) {
188 err = "color headers need '#rrggbb' values (sky, medium-colors, ground-colors, river-colors: two; surface-color: one)";
189 return nullptr;
190 }
191 sc->skyTop = sky[0]; sc->skyLow = sky[1];
192 sc->mediumTop = med[0]; sc->mediumDeep = med[1];
193 sc->groundA = gnd[0]; sc->groundB = gnd[1];
194 sc->surfaceFg = sur[0];
195 sc->riverTop = riv[0]; sc->riverDeep = riv[1];
196 return sc;
197}
